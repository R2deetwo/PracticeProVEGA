   102→// ─── File Upload Helper ──────────────────────────────────────────────────────
   103→async function uploadFilesToConvex(files: File[]): Promise<string[]> {
   104→  const storageIds: string[] = [];
   105→  for (const file of files) {
   106→    const postUrl = await fetch('/api/storage/upload-url');
   107→    // Fallback: use the Convex action for upload URL
   108→    // We'll use the Convex generateUploadUrl mutation via the hook instead
   109→    // For now, we'll use the standard Convex file upload flow
   110→    const { url, storageId } = await (async () => {
   111→      // Use Convex's built-in file upload
   112→      const uploadUrlRes = await fetch(`${window.location.origin}/api/storage/upload-url`);
   113→      if (uploadUrlRes.ok) {
   114→        const { url: u, storageId: s } = await uploadUrlRes.json();
   115→        return { url: u, storageId: s };
   116→      }
   117→      throw new Error('Failed to get upload URL');
   118→    })();
   119→
   120→    const uploadRes = await fetch(url, {
   121→      method: 'POST',
   122→      body: file,
   123→    });
   124→    if (uploadRes.ok) {
   125→      const { storageId: sid } = await uploadRes.json();
   126→      storageIds.push(sid);
   127→    }
   128→  }
   129→  return storageIds;
   130→}

   305→// ─── Financial Ledger Tab ────────────────────────────────────────────────────
   306→const LedgerTab: React.FC<{ tenantInfo: any }> = ({ tenantInfo }) => {
   307→  const { currentUser } = useAuth();
   308→  const firmId = currentUser?.firmId || '';
   309→  const userId = currentUser?.id || '';
   310→
   311→  const resolvedTenantId = tenantInfo?.tenantId || userId;
   312→
   313→  // Fetch ledger entries from Convex using the resolved tenant ID
   314→  const ledgerEntries = useQuery(
   315→    api.portals.getTenantLedger,
   316→    firmId && resolvedTenantId ? { firmId, tenantId: resolvedTenantId } : 'skip'
   317→  );
   318→
   319→  // Fetch service charges from Convex to compute current SC/MV
   320→  const serviceCharges = useQuery(
   321→    api.sentry.getServiceChargesByFirm,
   322→    firmId ? { firmId } : 'skip'
   323→  );
   324→
   325→  const isLoading = ledgerEntries === undefined || serviceCharges === undefined;
   326→
   327→  // Compute tenant-specific service charges
   328→  const tenantServiceCharges = useMemo(() => {
   329→    if (!serviceCharges || !resolvedTenantId) return [];
   330→    return serviceCharges.filter((sc: any) => sc.tenantId === resolvedTenantId);
   331→  }, [serviceCharges, resolvedTenantId]);
   332→
   333→  // Summary calculations
   334→  const currentMonthSC = useMemo(() => {
   335→    return tenantServiceCharges
   336→      .filter((sc: any) => !sc.isMinimumVend)
   337→      .reduce((sum: number, sc: any) => sum + (sc.outstandingBalance ?? sc.amount), 0);
   338→  }, [tenantServiceCharges]);
   339→
   340→  const currentMonthMV = useMemo(() => {
   341→    return tenantServiceCharges
   342→      .filter((sc: any) => sc.isMinimumVend)
   343→      .reduce((sum: number, sc: any) => sum + (sc.outstandingBalance ?? sc.amount), 0);
   344→  }, [tenantServiceCharges]);
   345→
   346→  const outstandingBalance = useMemo(() => {
   347→    if (!ledgerEntries) return 0;
   348→    return ledgerEntries
   349→      .filter((e: any) => e.status === 'pending' || e.status === 'defaulted')
   350→      .reduce((sum: number, e: any) => sum + e.amount, 0);
   351→  }, [ledgerEntries]);
   352→
   353→  const scPaid = tenantServiceCharges.filter((sc: any) => !sc.isMinimumVend && sc.serviceChargeStatus === 'PAID_FULLY').length > 0;
   354→  const mvStatus = tenantServiceCharges.filter((sc: any) => sc.isMinimumVend).length > 0
   355→    ? (tenantServiceCharges.filter((sc: any) => sc.isMinimumVend && sc.serviceChargeStatus === 'PAID_FULLY').length > 0 ? 'paid' : 'pending')
   356→    : null;
   357→
   358→  if (isLoading) {
   359→    return (
   360→      <div>
   361→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 mb-6">
   362→          {[1, 2, 3].map(i => (
   363→            <div key={i} className="bg-white dark:bg-zinc-800 rounded-xl p-4 border border-slate-200 dark:border-zinc-700 animate-pulse">
   364→              <div className="h-3 bg-slate-200 dark:bg-zinc-700 rounded w-24 mb-2" />
   365→              <div className="h-7 bg-slate-200 dark:bg-zinc-700 rounded w-32 mb-1" />
   366→              <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-16" />
   367→            </div>
   368→          ))}
   369→        </div>
   370→        <div className="bg-white dark:bg-zinc-800 rounded-xl border border-slate-200 dark:border-zinc-700 p-6 animate-pulse">
   371→          <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-full mb-3" />
   372→          <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-3/4 mb-3" />
   373→          <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-1/2" />
   374→        </div>
   375→      </div>
   376→    );
   377→  }
   378→
   379→  const hasLedgerData = ledgerEntries && ledgerEntries.length > 0;
   380→  const hasServiceCharges = tenantServiceCharges.length > 0;
   381→
   382→  return (
   383→    <div>
   384→      <h3 className="text-lg font-bold text-slate-900 dark:text-white mb-1">Payment Ledger</h3>
   385→      <p className="text-sm text-slate-500 dark:text-zinc-400 mb-6">
   386→        View your rent, Service Charge (SC), and Minimum Vend (MV) obligations and payment status.
   387→      </p>
   388→
   389→      {/* Summary Cards */}
   390→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 mb-6">
   391→        <div className="bg-white dark:bg-zinc-800 rounded-xl p-4 border border-slate-200 dark:border-zinc-700">
   392→          <p className="text-[10px] font-bold uppercase tracking-widest text-slate-400 dark:text-zinc-500 mb-1">Current Month SC</p>
   393→          <p className="text-2xl font-bold text-emerald-600 dark:text-emerald-400">
   394→            {formatNaira(currentMonthSC)}
   395→          </p>
   396→          {hasServiceCharges && !currentMonthSC ? (
   397→            <span className="inline-flex items-center gap-1 mt-1 text-[10px] font-bold text-emerald-600 dark:text-emerald-400 bg-emerald-50 dark:bg-emerald-900/20 px-2 py-0.5 rounded-full">
   398→              <CheckIcon className="w-3 h-3" /> Paid
   399→            </span>
   400→          ) : currentMonthSC > 0 ? (
   401→            <span className="inline-flex items-center gap-1 mt-1 text-[10px] font-bold text-amber-600 dark:text-amber-400 bg-amber-50 dark:bg-amber-900/20 px-2 py-0.5 rounded-full">
   402→              <ExclamationTriangleIcon className="w-3 h-3" /> Due
   403→            </span>
   404→          ) : null}
   405→        </div>
   406→        <div className="bg-white dark:bg-zinc-800 rounded-xl p-4 border border-slate-200 dark:border-zinc-700">
   407→          <p className="text-[10px] font-bold uppercase tracking-widest text-slate-400 dark:text-zinc-500 mb-1">Current Month MV</p>
   408→          <p className="text-2xl font-bold text-amber-600 dark:text-amber-400">
   409→            {currentMonthMV > 0 ? formatNaira(currentMonthMV) : '—'}
