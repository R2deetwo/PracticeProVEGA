   537→                return;
   538→            }
   539→            if (isDataLoaded || forceEntry) {
   540→                setLoadingMessage("Ready");
   541→                const t = setTimeout(() => {
   542→                    if (!currentUser.firmId) {
   543→                        setFlowState('setup');
   544→                    } else {
   545→                        setFlowState('app');
   546→                        if (appMode === AppMode.Solo) {
   547→                            setIsToolkitOpen(true);
   548→                        }
   549→                    }
   550→                    setHasInitialized(true);
   551→                    if (!hasInitialized && (modal === 'login' || modal === 'signup')) {
   552→                        closeModal();
   553→                    }
   554→                }, 500);
   555→                return () => clearTimeout(t);
   556→            } else {
   557→                setLoadingMessage("Finalizing Data Synchronization...");
   558→            }
   559→        }
   560→    }, [currentUser, visualsComplete, isDataLoaded, forceEntry, modal, closeModal]);
   561→
   562→    useEffect(() => {
   563→        if (flowState === 'app' && currentUser) {
   564→            if (forceEntry && !isDataLoaded) return;
   565→            // Don't auto-start the tour for portal users (Client/Tenant)
   566→            const isPortal = currentUser.role === UserRole.Client || currentUser.role === UserRole.Tenant;
   567→            if (isPortal) return;
   568→            const hasCompletedTour = localStorage.getItem('practicepro_tour_completed');
   569→            if (hasCompletedTour !== 'true') setTimeout(() => startTour(), 500);
   570→        }
   571→    }, [flowState, currentUser, startTour]);
   572→
   573→    const isPortalUserRole = currentUser?.role === UserRole.Client || currentUser?.role === UserRole.Tenant;
   574→
   575→    const renderAppContent = () => {
   576→        // ── Portal login routes ──
   577→        // If user is already authenticated as a portal user, redirect them to their dashboard
   578→        // instead of showing the login page again. This handles the case where a portal user
   579→        // navigates to the login page while already logged in.
   580→        if (location.pathname === '/portal/client/login') {
   581→            if (currentUser && currentUser.role === UserRole.Client) {
   582→                // Already logged in as client — redirect to portal dashboard via React Router
   583→                navigate('/', { replace: true });
   584→                return null;
   585→            }
   586→            return <ClientPortalLogin />;
   587→        }
   588→        if (location.pathname === '/portal/tenant/login') {
   589→            if (currentUser && currentUser.role === UserRole.Tenant) {
   590→                // Already logged in as tenant — redirect to portal dashboard via React Router
   591→                navigate('/', { replace: true });
   592→                return null;
   593→            }
   594→            return <TenantPortalLogin />;
   595→        }
   596→        if (location.pathname === '/setup-password') return <SetupPassword />;
   597→
   598→        if (!currentUser && !isLoadingSession) {
   599→            // Check if a portal user session is being restored — if so, show a loading
   600→            // state instead of the LandingPage to prevent the jarring flash
   601→            const hasRememberedPortal = typeof sessionStorage !== 'undefined' && sessionStorage.getItem('practicepro_portal_type');
   602→            if (hasRememberedPortal && !portalRememberTimedOut) {
   603→                // Session might still be loading (e.g. slow Convex query). Show a minimal
   604→                // portal-themed loading screen rather than the public LandingPage.
   605→                // Auto-timeout after 10s: if the session truly isn't valid, clear the
   606→                // remembered portal type and fall through to the LandingPage.
   607→                setTimeout(() => setPortalRememberTimedOut(true), 10000);
   608→                return (
   609→                    <div className="min-h-screen bg-gradient-to-br from-slate-900 via-slate-950 to-slate-900 flex items-center justify-center">
   610→                        <div className="text-center">
   611→                            <div className="w-8 h-8 border-2 border-emerald-500 border-t-transparent rounded-full animate-spin mx-auto mb-4" />
   612→                            <p className="text-slate-400 text-sm">Loading your portal...</p>
   613→                        </div>
   614→                    </div>
   615→                );
   616→            }
   617→            if (view === 'termsOfService') return <TermsOfService onBack={() => navigateTo('dashboard')} activeProduct={product === 'property' ? 'atrium' : product === 'legal' ? 'vega' : undefined} />;
   618→            if (view === 'privacyPolicy') return <PrivacyPolicy onBack={() => navigateTo('dashboard')} />;
   619→            if (view === 'dataProcessingAgreement') return <DataProcessingAgreement onBack={() => navigateTo('dashboard')} />;
   620→            if (view === 'cookiePolicy') return <CookiePolicy onBack={() => navigateTo('dashboard')} />;
   621→            return <LandingPage onDemo={(product) => openModal('leadCapture', null, { demoProduct: product })} />;
   622→        }
   623→
   624→        // New User Flow: Go straight to setup if no firm exists
   625→        if (currentUser && !currentUser.firmId) {
   626→            return <OnboardingWizard onComplete={() => setFlowState('app')} />;
   627→        }
   628→
   629→        // Only block on SplashScreen if the user has a saved session being restored.
   630→        // New/unauthenticated visitors must NEVER see the splash — they go straight to the Landing Page.
   631→        if (isLoadingSession) {
   632→            return null; // Top-level SplashScreen handles the branding
   633→        }
   634→
   635→        if (currentUser?.role === 'Pending') {
   636→            return (
   637→                <div className="flex flex-col items-center justify-center min-h-screen bg-slate-50 dark:bg-zinc-950 w-full p-8 text-center font-sans">
   638→                    <div className="max-w-md w-full bg-white dark:bg-zinc-900 p-10 rounded-[32px] shadow-2xl shadow-slate-200 dark:shadow-black/50 border border-slate-100 dark:border-zinc-800 flex flex-col items-center animate-in fade-in zoom-in duration-500">
   639→                        <div className="w-20 h-20 bg-amber-50 dark:bg-amber-900/20 rounded-full flex items-center justify-center mb-8 ring-8 ring-amber-50/50 dark:ring-amber-900/10">
   640→                            <OfficeBuildingIcon className="w-10 h-10 text-amber-600 dark:text-amber-400" />
   641→                        </div>
   642→                        <h2 className="text-3xl font-black text-slate-900 dark:text-white mb-3 tracking-tight">Access Pending</h2>
   643→                        <p className="text-slate-500 dark:text-zinc-400 mb-10 leading-relaxed text-lg">
   644→                            Your request to join <span className="font-bold text-slate-900 dark:text-white">{coreState?.firmDetails?.name || 'this workspace'}</span> is awaiting approval from {product === 'atrium' ? 'the portfolio manager' : 'a firm administrator'}.
   645→                        </p>
   646→                        
   647→                        <div className="flex flex-col gap-4 w-full">
   648→                            <button 
   649→                                onClick={() => window.location.reload()} 
   650→                                className="w-full py-4 bg-green-600 hover:bg-green-700 text-white rounded-2xl font-black text-sm uppercase tracking-widest transition-all shadow-xl shadow-green-600/20 active:scale-[0.98]"
   651→                            >
   652→                                Check Approval Status
   653→                            </button>
   654→                            <button 
   655→                                onClick={() => logout()} 
   656→                                className="w-full py-3 text-slate-400 dark:text-zinc-500 hover:text-slate-900 dark:hover:text-white font-bold text-sm transition-colors"
   657→                            >
   658→                                Sign Out & Switch Account
   659→                            </button>
   660→                        </div>
   661→
   662→                        <div className="mt-10 pt-8 border-t border-slate-50 dark:border-zinc-800 w-full">
   663→                            <p className="text-[10px] font-bold text-slate-400 dark:text-zinc-600 uppercase tracking-widest mb-1">Account Reference</p>
   664→                            <p className="text-xs font-mono text-slate-300 dark:text-zinc-700">{currentUser.id}</p>
   665→                        </div>
   666→                    </div>
   667→                </div>
   668→            );
   669→        }
   670→
   671→        return (
   672→            <>
   673→                {(flowState === 'app' || flowState === 'splash') && (
   674→                    <>
   675→                        <MainContent
   676→                            onToggleToolkit={() => setIsToolkitOpen(prev => !prev)}
   677→                            isToolkitOpen={isToolkitOpen}
   678→                            onCloseToolkit={() => setIsToolkitOpen(false)}
   679→                            onEnableDevMode={() => setIsToolkitOpen(true)}
   680→                            flowState={flowState}
   681→                        />
   682→                        {import.meta.env.DEV && <FloatingTestControls isOpen={isToolkitOpen} onClose={() => setIsToolkitOpen(false)} />}
   683→                    </>
   684→                )}
   685→            </>
   686→        );
   687→    };
   688→
   689→    useEffect(() => {
   690→        const timer = setTimeout(() => {
   691→            setHasInitialSplashFinished(true);
   692→        }, 12000);
   693→        return () => clearTimeout(timer);
   694→    }, []);
   695→
   696→    useEffect(() => {
   697→        if (splashAnimationComplete && isDataLoaded) {
   698→            setHasInitialSplashFinished(true);
   699→        }
   700→    }, [splashAnimationComplete, isDataLoaded]);
   701→
   702→    return (
   703→        <div className={`app-container font-sans text-base ${theme} h-[100dvh] bg-[rgb(var(--bg-main))] text-[rgb(var(--text-main))]`}>
   704→            <SplashScreen 
   705→                isVisible={!hasInitialSplashFinished && (hasSavedSession || flowState === 'splash')} 
   706→                statusMessage={loadingMessage} 
   707→                onReset={handleReset} 
   708→                onForceEnter={() => { setForceEntry(true); setHasInitialSplashFinished(true); }} 
   709→                onComplete={() => setSplashAnimationComplete(true)}
   710→                product={(!isDataLoaded || !currentUser?.firmId) ? 'practicepro' : (product === 'property' ? 'atrium' : 'vega')} 
   711→            />
   712→            {isSessionLocked ? (
   713→                <LockScreen onUnlock={() => setIsSessionLocked(false)} />
   714→            ) : (
   715→                renderAppContent()
   716→            )}
   717→            {((!currentUser) || (currentUser && flowState !== 'splash')) && (
   718→                <>
   719→                    <ModalManager />
   720→                    <ToastContainer />
   721→                </>
   722→            )}
   723→            {/* Onboarding Tour only shown for non-portal users; portal users get a simpler experience */}
   724→            {currentUser && currentUser.role !== UserRole.Client && currentUser.role !== UserRole.Tenant && <OnboardingTour />}
   725→            {/* What's New only for admin/firm users, not portal users */}
   726→            {flowState === 'app' && currentUser && currentUser.role !== UserRole.Client && currentUser.role !== UserRole.Tenant && <WhatsNew />}
   727→            <CookieConsent />
   728→        </div>
   729→    );
   730→};
   731→
   732→export default App;
