    91→    91→/** Generate a crypto-random invite token (12 chars, URL-safe) */
    92→    92→function generateToken(): string {
    93→    93→  const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
    94→    94→  const arr = new Uint8Array(12);
    95→    95→  // @ts-ignore — crypto available in Convex runtime
    96→    96→  crypto.getRandomValues(arr);
    97→    97→  return Array.from(arr, b => chars[b % chars.length]).join("");
    98→    98→}

   317→   317→/**
   318→   318→ * Internal mutation — only used by createPortalInvite action to write the DB record.
   319→   319→ * Not exported for direct frontend use.
   320→   320→ */
   321→   321→export const insertInviteRecord = mutation({
   322→   322→  args: {
   323→   323→    firmId: v.string(),
   324→   324→    inviterId: v.string(),
   325→   325→    inviteeEmail: v.optional(v.string()),
   326→   326→    inviteeName: v.optional(v.string()),
   327→   327→    inviteePhone: v.optional(v.string()),
   328→   328→    portalType: v.union(v.literal("client"), v.literal("resident")),
   329→   329→    relatedId: v.optional(v.string()),
   330→   330→    token: v.optional(v.string()),
   331→   331→    channel: v.optional(v.string()),
   332→   332→    message: v.optional(v.string()),
   333→   333→    expiresAt: v.number(),
   334→   334→  },
   335→   335→  handler: async (ctx, args) => {
   336→   336→    const now = Date.now();
   337→   337→
   338→   338→    // Clean up any existing invites for the same email + firm + portal type.
   339→   339→    // This prevents "invitation already accepted" errors when re-inviting
   340→   340→    // after a previous invite was deleted via the simple deletePortalInvite
   341→   341→    // (which doesn't clean up). We mark old invites as "superseded" rather
   342→   342→    // than deleting them, so there's an audit trail.
   343→   343→    const email = (args.inviteeEmail || "").toLowerCase().trim();
   344→   344→    if (email) {
   345→   345→      const existingInvites = await ctx.db
   346→   346→        .query("portal_invites")
   347→   347→        .withIndex("by_email", (q) => q.eq("inviteeEmail", email))
   348→   348→        .collect();
   349→   349→
   350→   350→      for (const inv of existingInvites) {
   351→   351→        // Only supersede invites for the same firm and portal type
   352→   352→        if (inv.firmId === args.firmId && inv.portalType === args.portalType) {
   353→   353→          if (inv.status === "pending" || inv.status === "accepted") {
   354→   354→            await ctx.db.patch(inv._id, {
   355→   355→              status: "superseded",
   356→   356→              updatedAt: now,
   357→   357→            });
   358→   358→          }
   359→   359→        }
   360→   360→      }
   361→   361→    }
   362→   362→
   363→   363→    return await ctx.db.insert("portal_invites", {
   364→   364→      firmId: args.firmId,
   365→   365→      inviterId: args.inviterId,
   366→   366→      inviteeEmail: args.inviteeEmail,
   367→   367→      inviteeName: args.inviteeName,
   368→   368→      inviteePhone: args.inviteePhone,
   369→   369→      portalType: args.portalType,
   370→   370→      relatedId: args.relatedId,
   371→   371→      token: args.token ?? undefined,
   372→   372→      channel: args.channel,
   373→   373→      message: args.message,
   374→   374→      status: "pending",
   375→   375→      expiresAt: args.expiresAt,
   376→   376→      createdAt: now,
   377→   377→      updatedAt: now,
   378→   378→    });
   379→   379→  },
   380→   380→});

   382→   382→/**
   383→   383→ * resendPortalInvite — action that updates the existing record's expiry + token,
   384→   384→ * then re-sends via the stored channel.
   385→   385→ */
   386→   386→export const resendPortalInvite = action({
   387→   387→  args: {
   388→   388→    inviteId: v.id("portal_invites"),
   389→   389→  },
   390→   390→  handler: async (ctx, args) => {
   391→   391→    const existing = await ctx.runQuery(api.portals.getPortalInviteById, { inviteId: args.inviteId });
   392→   392→    if (!existing) throw new Error("Invitation not found");
   393→   393→    if (existing.status === "revoked") throw new Error("Cannot resend a revoked invitation");

   562→   562→/**
   563→   563→ * deletePortalInviteAndCleanup — Permanently deletes a portal invite AND resets
   564→   564→ * the associated portal user account so the same email can be re-invited cleanly.
   565→   565→ *
   566→   566→ * This prevents the "invitation already accepted" error when an admin:
   567→   567→ * 1. Deletes a portal access record
   568→   568→ * 2. Creates a new invite for the same email
   569→   569→ *
   570→   570→ * Steps:
   571→   571→ * 1. Delete the portal_invites record
   572→   572→ * 2. Find the user with the matching email + Client/Tenant role
   573→   573→ * 3. Reset their role, verification, and password so they're treated as a brand-new invitee
   574→   574→ */
   575→   575→export const deletePortalInviteAndCleanup = mutation({
   576→   576→  args: {
   577→   577→    inviteId: v.id("portal_invites"),
   578→   578→    inviteeEmail: v.optional(v.string()),
   579→   579→  },
   580→   580→  handler: async (ctx, args) => {
   581→   581→    // 1. Delete the specified invite record
   582→   582→    await ctx.db.delete(args.inviteId);
   583→   583→
   584→   584→    // 2. Also delete ANY other invite records for the same email
   585→   585→    //    to prevent residual "already accepted" errors on re-invite
   586→   586→    const email = (args.inviteeEmail || "").toLowerCase().trim();
   587→   587→    if (!email) return;
   588→   588→
   589→   589→    const otherInvites = await ctx.db
   590→   590→      .query("portal_invites")
   591→   591→      .withIndex("by_email", (q) => q.eq("inviteeEmail", email))
   592→   592→      .collect();
   593→   593→
   594→   594→    for (const inv of otherInvites) {
   595→   595→      if (String(inv._id) === String(args.inviteId)) continue; // Already deleted above
   596→   596→      try { await ctx.db.delete(inv._id); } catch (e) { /* ignore */ }
   597→   597→    }
   598→   598→
   599→   599→    // 3. Find and reset the associated portal user
   600→   600→    const existingUser = await ctx.db
   601→   601→      .query("users")
   602→   602→      .withIndex("by_token", (q) => q.eq("tokenIdentifier", email))
   603→   603→      .first();
   604→   604→
   605→   605→    if (!existingUser) return;
   606→   606→
   607→   607→    // Only reset if the user has a Client or Tenant role (portal user)
   608→   608→    const role = (existingUser as any).role;
   609→   609→    if (role === "Client" || role === "Tenant" || role === "Portal User") {
   610→   610→      // Reset the user so they can be re-invited cleanly
   611→   611→      // Set isVerified to false so they can't log in with old credentials
   612→   612→      // Remove the password so setup-password creates a fresh one
   613→   613→      // Clear firmId so it will be set fresh on the next invite acceptance
   614→   614→      await ctx.db.patch(existingUser._id, {
   615→   615→        isVerified: false,
   616→   616→        emailVerified: false,
   617→   617→        password: undefined,
   618→   618→        role: "Pending",
   619→   619→        firmId: undefined,
   620→   620→      } as any);
   621→   621→    }
   622→   622→  },
   623→   623→});

   674→   674→ * verifyInviteToken — validates a portal invite token and returns
   675→   675→ * the invite details + whether the user already has an account.
   676→   676→ * Used by the /setup-password page to determine what to show.
   677→   677→ *
   678→   678→ * ROBUSTNESS FIX: If the invite is "accepted" but the user's role is
   679→   679→ * "Pending" (meaning their access was deleted via deletePortalInviteAndCleanup
   680→   680→ * and they need to re-setup), we allow them to proceed. This prevents the
   681→   681→ * "invitation already accepted" dead-end when a user is re-invited after
   682→   682→ * their portal access was removed.
   683→   683→ */
   684→   684→export const verifyInviteToken = query({
   685→   685→  args: { token: v.string() },
   686→   686→  handler: async (ctx, args) => {
   687→   687→    const results = await ctx.db
   688→   688→      .query("portal_invites")
   689→   689→      .withIndex("by_token", (q) => q.eq("token", args.token))
   690→   690→      .collect();
   691→   691→    const invite = results[0] || null;
   692→   692→
   693→   693→    if (!invite) {
   694→   694→      return { valid: false, reason: "not_found" } as const;
   695→   695→    }
   696→   696→
   697→   697→    // Check expiry (note: we don't patch status here since this is a query/read-only)
   698→   698→    if (invite.expiresAt < Date.now()) {
   699→   699→      return { valid: false, reason: "expired" } as const;
   700→   700→    }
   701→   701→
   702→   702→    if (invite.status === "revoked") {
   703→   703→      return { valid: false, reason: "revoked" } as const;
   704→   704→    }
   705→   705→
   706→   706→    // Check if the user exists and their current state
   707→   707→    const email = invite.inviteeEmail?.toLowerCase().trim();
   708→   708→    let existingUser: any = null;
   709→   709→    if (email) {
   710→   710→      existingUser = await ctx.db
   711→   711→        .query("users")
   712→   712→        .withIndex("by_token", (q) => q.eq("tokenIdentifier", email))
   713→   713→        .first();
   714→   714→      // Case-insensitive fallback
   715→   715→      if (!existingUser) {
   716→   716→        existingUser = await ctx.db
   717→   717→          .query("users")
   718→   718→          .withIndex("by_token", (q) => q.eq("tokenIdentifier", email.toLowerCase()))
   719→   719→          .first();
   720→   720→      }
   721→   721→    }
   722→   722→
   723→   723→    if (invite.status === "accepted") {
   724→   724→      // ROBUSTNESS: If the user's role is Pending (their access was reset by
   725→   725→      // deletePortalInviteAndCleanup), allow them to re-accept. This invite
   726→   726→      // might be a stale record from before the cleanup ran. The user needs
   727→   727→      // to set up their password again, so we should let them through.
   728→   728→      const userRole = existingUser?.role;
   729→   729→      const isUserUnverified = existingUser && !existingUser.isVerified;
   730→   730→      if (userRole === "Pending" || isUserUnverified) {
   731→   731→        // User was reset — allow re-accepting this invite
   732→   732→        // Fall through to return valid: true
   733→   733→      } else {
   734→   734→        return { valid: false, reason: "already_accepted" } as const;
   735→   735→      }
   736→   736→    }
   737→   737→
   738→   738→    // Token is valid and pending — check if user already has an account
   739→   739→    return {
   740→   740→      valid: true,
   741→   741→      invite: {
   742→   742→        _id: invite._id,
   743→   743→        inviteeEmail: invite.inviteeEmail,

   748→   748→        expiresAt: invite.expiresAt,
   749→   749→      },
   750→   750→      hasAccount: !!existingUser,
   751→   751→      hasPassword: !!(existingUser as any)?.password,
   752→   752→    } as const;
   753→   753→  },

   798→   798→ * 4. Marks the invite as accepted
   799→   799→ * 5. Clears the token to prevent replay attacks
   800→   800→ */
   801→   801→export const setupPortalPassword = action({
   802→   802→  args: {
   803→   803→    token: v.string(),
   804→   804→    password: v.string(),
   805→   805→    name: v.optional(v.string()),
   806→   806→  },
   807→   807→  handler: async (ctx, args): Promise<{ success: boolean; message?: string; email?: string }> => {
   808→   808→    // 1. Validate the invite
   809→   809→    const invite = await ctx.runQuery(api.portals.getInviteByToken, { token: args.token });
   810→   810→    if (!invite) return { success: false, message: "Invalid invitation link." };
   811→   811→    if (invite.status === "revoked") return { success: false, message: "This invitation has been revoked." };
   812→   812→    if (invite.status === "expired" || invite.expiresAt < Date.now()) {
   813→   813→      await ctx.runMutation(api.portals.updateInviteRecord, {
   814→   814→        inviteId: invite._id,
   815→   815→        updates: { status: "expired", updatedAt: Date.now() },
   816→   816→      });
   817→   817→      return { success: false, message: "This invitation has expired. Please request a new one." };
   818→   818→    }
   819→   819→
   820→   820→    // ROBUSTNESS: If the invite was already accepted but the user's account was
   821→   821→    // reset (role=Pending, isVerified=false), allow them to re-accept. This can
   822→   822→    // happen when deletePortalInviteAndCleanup resets the user but a stale invite
   823→   823→    // record still exists with status "accepted" and a valid token.
   824→   824→    if (invite.status === "accepted") {
   825→   825→      const emailCheck = (invite.inviteeEmail || "").toLowerCase().trim();
   826→   826→      const existingUserCheck: any = await ctx.runQuery(api.myFunctions.getUser, { tokenIdentifier: emailCheck });
   827→   827→      const wasReset = existingUserCheck?.role === "Pending" || (existingUserCheck && !existingUserCheck.isVerified);
   828→   828→      if (!wasReset) {
   829→   829→        return { success: false, message: "This invitation has already been used." };
   830→   830→      }
   831→   831→      // User was reset — allow re-accepting this invite (fall through)
   832→   832→    }
   833→   833→
   834→   834→    const email = (invite.inviteeEmail || "").toLowerCase().trim();
   835→   835→    if (!email) return { success: false, message: "This invitation has no email address associated." };
   836→   836→
   837→   837→    // 1.5. For resident invites, resolve the canonical tenant name from the
   838→   838→    // property/tenancy record. The property record is the source of truth for
   839→   839→    // a tenant's name — NOT the invite. This ensures the user's name always
   840→   840→    // matches the property record.
   841→   841→    let canonicalTenantName: string | null = null;
   842→   842→    if (invite.portalType === "resident" && invite.relatedId) {
   843→   843→      try {
   844→   844→        const parts = invite.relatedId.split("_");
   845→   845→        const propertyCustomId = parts[0];
   846→   846→        const unitId = parts.length > 1 ? parts.slice(1).join("_") : null;
   847→   847→
   848→   848→        const property: any = await ctx.runQuery(internal.portals.findPropertyByCustomId, {
   849→   849→          customId: propertyCustomId,
   850→   850→        });
   851→   851→
   852→   852→        if (property) {
   853→   853→          const units = property.units || [];
   854→   854→          if (units.length > 0 && unitId) {
   855→   855→            const unit = units.find((u: any) =>
   856→   856→              u.id === unitId || u.unitName === unitId || u.id === invite.relatedId
   857→   857→            );
   858→   858→            if (unit?.tenantName) {
   859→   859→              canonicalTenantName = unit.tenantName;
   860→   860→            }
   861→   861→          } else if (units.length > 0) {
   862→   862→            if (units[0]?.tenantName) {
   863→   863→              canonicalTenantName = units[0].tenantName;
   864→   864→            }
   865→   865→          } else {
   866→   866→            const propTenantName = (property as any).rentalDetails?.tenantName || (property as any).tenantName;
   867→   867→            if (propTenantName) {
   868→   868→              canonicalTenantName = propTenantName;
   869→   869→            }
   870→   870→          }
   871→   871→        }
   872→   872→      } catch (e) {
   873→   873→        // Non-blocking: use invite name as fallback
   874→   874→      }
   875→   875→    }
   876→   876→
   877→   877→    // 2. Hash the password
   878→   878→    const hashedPassword = await ctx.runAction(internal.authUtils.hashPassword, { password: args.password });
   879→   879→
   880→   880→    // 3. Check if user exists
   881→   881→    const existingUser: any = await ctx.runQuery(api.myFunctions.getUser, { tokenIdentifier: email });
   882→   882→
   883→   883→    // Track the portal user's Convex _id so we can link them to their property/unit
   884→   884→    let portalUserDocId: string | null = null;
   885→   885→
   886→   886→    if (existingUser) {
   887→   887→      // Update existing user: set password + mark verified
   888→   888→      // Also upgrade their role to Client/Tenant if they were a "Portal User"
   889→   889→      // or had no role assigned, so the frontend can route them correctly.
   890→   890→      const portalRole = invite.portalType === "client" ? "Client" : "Tenant";
   891→   891→      const needsRoleUpdate = !existingUser.role
   892→   892→        || existingUser.role === "Portal User"
   893→   893→        || existingUser.role === "Pending";
   894→   894→      // Ensure firmId is always set — portal users need it for the ProductContext
   895→   895→      // to derive the correct product and for data loading to work properly.
   896→   896→      // When an existing user is re-invited after portal access deletion, their
   897→   897→      // firmId might have been cleared or might never have been set.
   898→   898→      const needsFirmId = !existingUser.firmId || existingUser.firmId !== invite.firmId;
   899→   899→      portalUserDocId = String(existingUser._id);
   900→   900→      await ctx.runMutation(internal.myFunctions.updateUserSecurityFields, {
   901→   901→        userId: existingUser._id,
   902→   902→        fields: {
   903→   903→          password: hashedPassword,
   904→   904→          isVerified: true,
   905→   905→          emailVerified: true,
   906→   906→          verificationCode: null,
   907→   907→          // For tenant users, always sync name with the property record (source of truth)
   908→   908→          ...(canonicalTenantName ? { name: canonicalTenantName } : (args.name && !existingUser.name ? { name: args.name } : {})),
   909→   909→          ...(needsRoleUpdate ? { role: portalRole } : {}),
   910→   910→          ...(needsFirmId ? { firmId: invite.firmId } : {}),
   911→   911→          // Online presence privacy: property/resident portals default to hidden
   912→   912→          portalPresenceHidden: invite.portalType === "resident" ? true : (existingUser as any).portalPresenceHidden ?? false,
   913→   913→        },
   914→   914→      });
   915→   915→    } else {
   916→   916→      // Create new user record for this portal user
   917→   917→      // Use the proper UserRole (Client / Tenant) so the frontend can route
   918→   918→      // them to the correct portal view automatically.
   919→   919→      const portalProduct = invite.portalType === "client" ? "legal" : "property";
   920→   920→      const portalRole = invite.portalType === "client" ? "Client" : "Tenant";
   921→   921→      const userName = canonicalTenantName || args.name || invite.inviteeName || email.split("@")[0];
   922→   922→      const newUserId = await ctx.runMutation(internal.myFunctions.createUser, {
   923→   923→        tokenIdentifier: email,
   924→   924→        name: userName,
   925→   925→        email: email,
   926→   926→        password: hashedPassword,
   927→   927→        role: portalRole,
   928→   928→        product: portalProduct,
   929→   929→        onboardingCompleted: false,
   930→   930→        isVerified: true,
   931→   931→        emailVerified: true,
   932→   932→        firmId: invite.firmId,
   933→   933→        avatarUrl: `https://ui-avatars.com/api/?name=${encodeURIComponent(userName)}&background=random`,
   934→   934→        // Online presence privacy: property/resident portals default to hidden
   935→   935→        // (portal users can't see if the manager is online). Legal/client portals
   936→   936→        // default to visible (clients can see if their lawyer is online).
   937→   937→        portalPresenceHidden: invite.portalType === "resident",
   938→   938→      });
   939→   939→      portalUserDocId = String(newUserId);
   940→   940→    }
   941→   941→
   942→   942→    // 3.5. Link portal user to property/unit if relatedId was provided
   943→   943→    // This is critical for getTenantInfo to find the tenant's property assignment.
   944→   944→    // When setupPortalPassword creates a new user, the property's unit record
   945→   945→    // may still point to a stale contact ID or email instead of the new Convex _id.
   946→   946→    if (invite.relatedId && invite.portalType === "resident" && portalUserDocId) {
   947→   947→      try {
   948→   948→        // Parse the relatedId format:
   949→   949→        // "propertyId_unitId" for multi-unit properties (composite key)
   950→   950→        // just "propertyId" for single properties
   951→   951→        const parts = invite.relatedId.split("_");
   952→   952→        const propertyCustomId = parts[0];
   953→   953→        const unitId = parts.length > 1 ? parts.slice(1).join("_") : null;
   954→   954→
   955→   955→        // Find the property using internal query (actions cannot use ctx.db directly)
   956→   956→        const property: any = await ctx.runQuery(internal.portals.findPropertyByCustomId, {
   957→   957→          customId: propertyCustomId,
   958→   958→        });
   959→   959→
   960→   960→        if (property) {
   961→   961→          const units = property.units || [];
   962→   962→          // Use the canonical tenant name from the property record (source of truth)
   963→   963→          // Fall back to the portal user's name or invite data if property lookup failed
   964→   964→          const tenantName = canonicalTenantName || invite.inviteeName || email.split("@")[0];
   965→   965→
   966→   966→          if (units.length > 0 && unitId) {
   967→   967→            // Multi-unit property: update the specific unit
   968→   968→            let unitFound = false;
   969→   969→            const updatedUnits = units.map((u: any) => {
   970→   970→              if (u.id === unitId || u.unitName === unitId || u.id === invite.relatedId) {
   971→   971→                unitFound = true;
   972→   972→                return {
   973→   973→                  ...u,
   974→   974→                  currentTenantId: portalUserDocId,
   975→   975→                  tenantId: portalUserDocId,
   976→   976→                  tenantEmail: email,
   977→   977→                  tenantName: tenantName,
   978→   978→                };
   979→   979→              }
   980→   980→              return u;
   981→   981→            });
   982→   982→            if (unitFound) {
   983→   983→              await ctx.runMutation(internal.portals.linkPortalUserToProperty, {
   984→   984→                propertyId: property._id,
   985→   985→                updates: { units: updatedUnits },
   986→   986→              });
   987→   987→            }
   988→   988→          } else {
   989→   989→            // Single property (no units or no unitId): update property-level tenant
   990→   990→            await ctx.runMutation(internal.portals.linkPortalUserToProperty, {
   991→   991→              propertyId: property._id,
   992→   992→              updates: {
   993→   993→                currentTenantId: portalUserDocId,
   994→   994→                tenantId: portalUserDocId,
   995→   995→              },
   996→   996→            });
   997→   997→          }
   998→   998→        }
   999→   999→      } catch (linkErr) {
  1000→  1000→        // Non-blocking: if linking fails, the user can still access the portal.
  1001→  1001→        // getTenantInfo will try to find them by email as a fallback.
  1002→  1002→        console.warn("Portal user-property linking failed:", linkErr);
  1003→  1003→      }
  1004→  1004→    }
  1005→  1005→
  1006→  1006→    // 4. Mark invite as accepted (including terms acceptance timestamp)
  1007→  1007→    await ctx.runMutation(api.portals.updateInviteRecord, {
  1008→  1008→      inviteId: invite._id,
  1009→  1009→      updates: {
  1010→  1010→        status: "accepted",
  1011→  1011→        acceptedAt: Date.now(),
  1012→  1012→        termsAcceptedAt: Date.now(),
  1013→  1013→        updatedAt: Date.now(),
  1014→  1014→        // Clear token to prevent replay attacks — the invite is now consumed
  1015→  1015→        token: undefined,
  1016→  1016→      },
  1017→  1017→    });
  1018→  1018→
  1019→  1019→    return { success: true, email };
  1020→  1020→  },
  1021→  1021→});

  1084→  1084→/**
  1085→  1085→ * getTenantInfo — Resolves the tenant's property/unit assignments using their
  1086→  1086→ * user ID and email. This is critical because the tenantId stored in
  1087→  1087→ * properties/units may be the user's Convex _id OR their email, depending
  1088→  1088→ * on how the invite was created. This query normalizes both cases.
  1089→  1089→ *
  1090→  1090→ * Returns: { tenantId, properties, units } where tenantId is the canonical
  1091→  1091→ * ID that should be used for all other portal queries.
  1092→  1092→ */
  1093→  1093→export const getTenantInfo = query({
  1094→  1094→  args: { firmId: v.string(), userId: v.string(), email: v.string() },
  1095→  1095→  handler: async (ctx, args) => {
  1096→  1096→    const properties = await ctx.db
  1097→  1097→      .query("properties")
  1098→  1098→      .withIndex("by_firm", (q) => q.eq("firmId", args.firmId))
  1099→  1099→      .collect();
  1100→  1100→
  1101→  1101→    // Search for tenant in properties using both userId and email
  1102→  1102→    const tenantProperties: any[] = [];
  1103→  1103→    const tenantUnits: any[] = [];
  1104→  1104→    let resolvedTenantId = args.userId;
  1105→  1105→
  1106→  1106→    for (const prop of properties) {
  1107→  1107→      // Check property-level tenant (legacy single-tenant model)
  1108→  1108→      const propTenantId = (prop as any).currentTenantId || (prop as any).tenantId;
  1109→  1109→      if (propTenantId === args.userId || propTenantId === args.email ||
  1110→  1110→          String(propTenantId).toLowerCase() === args.email.toLowerCase()) {
  1111→  1111→        tenantProperties.push({
  1112→  1112→          id: String(prop._id),
  1113→  1113→          name: (prop as any).name || prop.address || 'Unnamed Property',
  1114→  1114→          address: prop.address,
  1115→  1115→          tenantName: (prop as any).rentalDetails?.tenantName || (prop as any).tenantName || null,
  1116→  1116→        });
  1117→  1117→        if (propTenantId && propTenantId !== args.userId) {
  1118→  1118→          resolvedTenantId = propTenantId;
  1119→  1119→        }
  1120→  1120→      }
  1121→  1121→
  1122→  1122→      // Check unit-level tenants (multi-unit model)
  1123→  1123→      const units = (prop as any).units || [];
  1124→  1124→      for (const unit of units) {
  1125→  1125→        const unitTenantId = unit.currentTenantId || unit.tenantId;
  1126→  1126→        const unitTenantEmail = (unit.tenantEmail || '').toLowerCase();
  1127→  1127→        const matchesUserId = unitTenantId === args.userId;
  1128→  1128→        const matchesEmail = unitTenantId === args.email ||
  1129→  1129→            String(unitTenantId).toLowerCase() === args.email.toLowerCase() ||
  1130→  1130→            (unitTenantEmail && unitTenantEmail === args.email.toLowerCase());
  1131→  1131→
  1132→  1132→        if (matchesUserId || matchesEmail) {
  1133→  1133→          tenantUnits.push({
  1134→  1134→            id: unit.id || unit._id,
  1135→  1135→            name: unit.name || unit.unitName || unit.label,
  1136→  1136→            unitName: unit.name || unit.unitName || unit.label,
  1137→  1137→            propertyId: String(prop._id),
  1138→  1138→            propertyName: (prop as any).name || prop.address || 'Unnamed Property',
  1139→  1139→            propertyAddress: prop.address,
  1140→  1140→            amenities: unit.amenities || [],
  1141→  1141→            tenantName: unit.tenantName || null,
  1142→  1142→          });
  1143→  1143→          if (unitTenantId && unitTenantId !== args.userId) {
  1144→  1144→            resolvedTenantId = unitTenantId;
  1145→  1145→          }
  1146→  1146→        }
  1147→  1147→      }
  1148→  1148→    }
  1149→  1149→
  1150→  1150→    // Determine primary property/unit for this tenant (first match)
  1151→  1151→    const primaryUnit = tenantUnits.length > 0 ? tenantUnits[0] : null;
  1152→  1152→    const primaryProperty = primaryUnit
  1153→  1153→      ? { id: primaryUnit.propertyId, name: primaryUnit.propertyName, address: primaryUnit.propertyAddress }
  1154→  1154→      : tenantProperties.length > 0 ? tenantProperties[0] : null;
  1155→  1155→
  1156→  1156→    // Resolve the canonical tenant name from the property/tenancy record
  1157→  1157→    // This is the source of truth — used by the portal to display the tenant's name
  1158→  1158→    const resolvedTenantName = primaryUnit?.tenantName
  1159→  1159→      || tenantProperties.find(p => p.tenantName)?.tenantName
  1160→  1160→      || null;
  1161→  1161→
  1162→  1162→    return {
  1163→  1163→      tenantId: resolvedTenantId,
  1164→  1164→      properties: tenantProperties,
  1165→  1165→      units: tenantUnits,
  1166→  1166→      // Convenience fields for maintenance ticket creation
  1167→  1167→      primaryPropertyId: primaryProperty?.id || null,
  1168→  1168→      primaryUnitId: primaryUnit?.id || null,
  1169→  1169→      primaryPropertyName: primaryProperty?.name || null,
  1170→  1170→      primaryUnitName: primaryUnit?.unitName || null,
  1171→  1171→      primaryPropertyAddress: primaryProperty?.address || null,
  1172→  1172→      // Canonical tenant name from the property record (source of truth)
  1173→  1173→      tenantName: resolvedTenantName,
  1174→  1174→    };
