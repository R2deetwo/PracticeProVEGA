     1→/**
     2→ * PortalAccessSettings — Manage portal invitations for clients and/or residents
     3→ *
     4→ * Product-aware layout:
     5→ * - Komplete / Unified: Shows BOTH Client Portal and Residents' Portal sections
     6→ * - Vega (legal-only): Shows Client Portal only
     7→ * - Atrium (property-only): Shows Residents' Portal only
     8→ *
     9→ * Features:
    10→ * - Send invitations via Email, WhatsApp, or Both
    11→ * - Auto-populate name/phone from linked matter (client) or property/unit (resident)
    12→ * - Magic-link tokens embedded in invite URLs for setup-password flow
    13→ * - Resend, revoke, and copy invite links with real token
    14→ * - Track invitation status (pending / accepted / expired / revoked)
    15→ * - Delete with confirmation + cleanup of associated portal user
    16→ * - Preview portal as a specific client/resident
    17→ * - Grouped property/unit dropdown (address headers + selectable units)
    18→ */
    19→import React, { useState, useMemo, useCallback } from 'react';

   129→const PortalTypeBadge: React.FC<{ portalType: string }> = ({ portalType }) => {

   133→        <OfficeBuildingIcon className="w-2.5 h-2.5" /> Resident
   134→      </span>
   135→    );
   136→  }
   137→  return (
   138→    <span className="inline-flex items-center gap-1 px-1.5 py-0.5 rounded text-[9px] font-bold bg-violet-50 dark:bg-violet-900/20 text-violet-700 dark:text-violet-400">

   144→// ─── Invite Form ─────────────────────────────────────────────────────────────
   145→const InviteForm: React.FC<{
   146→  firmId: string;
   147→  inviterId: string;
   148→  portalType: 'client' | 'resident';
   149→  onSent: () => void;
   150→  onCancel: () => void;
   151→}> = ({ firmId, inviterId, portalType, onSent, onCancel }) => {
   152→  const { addToast } = useUI();
   153→  const sendInvite = useAction(api.portals.createPortalInvite);
   154→  const { coreState } = useCoreState();
   155→  const { matterState } = useMatterState();
   156→
   157→  const isProperty = portalType === 'resident';
   158→
   159→  const [email, setEmail] = useState('');
   160→  const [name, setName] = useState('');
   161→  const [phone, setPhone] = useState('');
   162→  const [relatedId, setRelatedId] = useState('');
   163→  const [channel, setChannel] = useState<'email' | 'whatsapp' | 'both'>('email');
   164→  const [isSending, setIsSending] = useState(false);
   165→
   166→  // Build list of matters (client portal) or properties (resident portal) to link invite to
   167→  const { groups: propertyGroups, flatUnits } = usePropertyGroups(coreState.properties || []);
   168→  const relatedItems = useMemo(() => {
   169→    if (isProperty) {
   170→      return flatUnits.map(u => ({
   171→        id: u.id,
   172→        label: u.label,
   173→        tenantName: u.tenantName || '',
   174→        tenantPhone: u.tenantPhone || '',
   175→        tenantEmail: u.tenantEmail || '',
   176→      }));
   177→    } else {
   178→      return (matterState.matters || []).map((m: any) => ({
   179→        id: m.id,
   180→        label: m.title || m.name || `Matter ${String(m.id || '').slice(-5)}`,
   181→        clientName: m.clientName || '',
   182→        clientEmail: m.clientEmail || '',
   183→        clientPhone: m.clientPhone || '',
   184→      }));
   185→    }
   186→  }, [isProperty, flatUnits, matterState.matters]);
   187→
   188→  // Auto-populate when user selects a matter/property
   189→  const handleRelatedChange = useCallback((selectedId: string) => {
   190→    setRelatedId(selectedId);
   191→    if (!selectedId) return;
   192→    const item = relatedItems.find((r: any) => r.id === selectedId);
   193→    if (!item) return;
   194→    if (isProperty) {
   195→      // Auto-fill from property tenant data — always override if tenant data exists
   196→      const propItem = item as { id: string; label: string; tenantName: string; tenantPhone: string; tenantEmail: string };
   197→      if (propItem.tenantName) setName(propItem.tenantName);
   198→      if (propItem.tenantPhone) setPhone(propItem.tenantPhone);
   199→      if (propItem.tenantEmail) setEmail(propItem.tenantEmail);
   200→    } else {
   201→      // Auto-fill from matter client data
   202→      const matterItem = item as { id: any; label: any; clientName: any; clientEmail: any; clientPhone: any };
   203→      if (matterItem.clientName) setName(matterItem.clientName);
   204→      if (matterItem.clientEmail) setEmail(matterItem.clientEmail);
   205→      if (matterItem.clientPhone) setPhone(matterItem.clientPhone);
   206→    }
   207→  }, [relatedItems, isProperty]);
   208→
   209→  const handleSubmit = async () => {
   210→    if (!email.trim() && channel !== 'whatsapp') {
   211→      addToast('Please enter an email address', { type: 'error' });
   212→      return;
   213→    }
   214→    if ((channel === 'whatsapp' || channel === 'both') && !phone.trim()) {
   215→      addToast('Please enter a phone number for WhatsApp delivery', { type: 'error' });
   216→      return;
   217→    }
   218→    setIsSending(true);
   219→    try {
   220→      const result = await sendInvite({
   221→        firmId,
   222→        inviterId,
   223→        inviteeEmail: email.trim().toLowerCase() || undefined,
   224→        inviteeName: name.trim() || undefined,
   225→        inviteePhone: phone.trim() || undefined,
   226→        portalType,
   227→        relatedId: relatedId || undefined,
   228→        channel,
   229→      });
   230→      // Build feedback message based on result
   231→      const parts: string[] = [];
   232→      if (result.emailSent) parts.push('email delivered');
   233→      else if (result.emailSimulated) parts.push('email simulated (Brevo API key not configured)');
   234→      if (result.whatsappSent) parts.push('WhatsApp delivered');
   235→      else if (result.whatsappSimulated) parts.push('WhatsApp simulated');
   236→      else if (result.whatsappSkipped && (channel === 'whatsapp' || channel === 'both')) parts.push('WhatsApp skipped (no phone number)');
   237→
   238→      const feedback = parts.length > 0 ? parts.join(', ') : 'invitation created';
   239→      addToast(`Invitation sent — ${feedback}`, { type: 'success' });
   240→      onSent();
   241→    } catch (err: any) {
   242→      addToast(err.message || 'Failed to send invitation', { type: 'error' });
   243→    } finally {
   244→      setIsSending(false);
   245→    }
   246→  };
   247→
   248→  const needsPhone = channel === 'whatsapp' || channel === 'both';
   249→
   250→  const portalLabel = isProperty ? 'Resident' : 'Client';
   251→  const portalDescription = isProperty
   252→    ? 'Invite a resident to access their payment ledger, maintenance tickets, and receipts.'
   253→    : 'Invite a client to access their matters, documents, messages, and billing.';
   254→
   255→  return (
   256→    <div className="bg-white dark:bg-zinc-800 rounded-xl border border-slate-200 dark:border-zinc-700 p-4 sm:p-6">
   257→      <div className="flex items-center justify-between mb-4">
   258→        <div>
   259→          <h4 className="text-base font-bold text-slate-900 dark:text-white flex items-center gap-2">
   260→            {isProperty ? <OfficeBuildingIcon className="w-4 h-4 text-sky-500" /> : <UsersIcon className="w-4 h-4 text-violet-500" />}
   261→            Invite {portalLabel} to Portal
   262→          </h4>
   263→          <p className="text-xs text-slate-500 dark:text-zinc-400 mt-0.5">{portalDescription}</p>
   264→        </div>
   265→        <button onClick={onCancel} className="text-slate-400 hover:text-slate-600 dark:hover:text-zinc-200 transition-colors">
   266→          <XIcon className="w-5 h-5" />
   267→        </button>
   268→      </div>
   269→
   270→      <div className="space-y-4">
   271→        {/* Channel Picker */}
   272→        <div>
   273→          <label className="block text-xs font-bold text-slate-500 dark:text-zinc-400 mb-2 uppercase tracking-wider">
   274→            Send Via
   275→          </label>
   276→          <div className="flex gap-2">
   277→            {[
   278→              { value: 'email' as const, label: 'Email', icon: MailIcon },
   279→              { value: 'whatsapp' as const, label: 'WhatsApp', icon: DeviceMobileIcon },
   280→              { value: 'both' as const, label: 'Both', icon: SendIcon },
   281→            ].map(opt => (
   282→              <button
   283→                key={opt.value}
   284→                onClick={() => setChannel(opt.value)}
   285→                className={`flex-1 flex items-center justify-center gap-1.5 px-3 py-2 rounded-lg text-xs font-bold border transition-colors ${
   286→                  channel === opt.value
   287→                    ? 'bg-primary-50 dark:bg-primary-900/20 border-primary-300 dark:border-primary-700 text-primary-700 dark:text-primary-400'
   288→                    : 'bg-slate-50 dark:bg-zinc-900 border-slate-200 dark:border-zinc-600 text-slate-500 dark:text-zinc-400 hover:bg-slate-100 dark:hover:bg-zinc-800'
   289→                }`}
   290→              >
   291→                <opt.icon className="w-3.5 h-3.5" />
   292→                {opt.label}
   293→              </button>
   294→            ))}
   295→          </div>
   296→        </div>
   297→
   298→        {/* Email — required unless WhatsApp-only */}
   299→        {channel !== 'whatsapp' && (
   300→          <div>
   301→            <label className="block text-xs font-bold text-slate-500 dark:text-zinc-400 mb-1 uppercase tracking-wider">
   302→              Email Address <span className="text-rose-500">*</span>
   303→            </label>
   304→            <div className="relative">
   305→              <MailIcon className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   306→              <input
   307→                type="email"
   308→                value={email}
   309→                onChange={e => setEmail(e.target.value)}
   310→                placeholder={isProperty ? 'resident@example.com' : 'client@example.com'}
   311→                className="w-full pl-9 pr-3 py-2 rounded-lg border border-slate-200 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-800 dark:text-zinc-200 focus:ring-2 focus:ring-primary-500 focus:border-transparent outline-none"
   312→              />
   313→            </div>
   314→          </div>
   315→        )}
   316→
   317→        {/* Phone — required for WhatsApp/Both */}
   318→        {needsPhone && (
   319→          <div>
   320→            <label className="block text-xs font-bold text-slate-500 dark:text-zinc-400 mb-1 uppercase tracking-wider">
   321→              Phone / WhatsApp <span className="text-rose-500">*</span>
   322→            </label>
   323→            <div className="relative">
   324→              <DeviceMobileIcon className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   325→              <input
   326→                type="tel"
   327→                value={phone}
   328→                onChange={e => setPhone(e.target.value)}
   329→                placeholder="e.g., +2348012345678"
   330→                className="w-full pl-9 pr-3 py-2 rounded-lg border border-slate-200 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-800 dark:text-zinc-200 focus:ring-2 focus:ring-primary-500 focus:border-transparent outline-none"
   331→              />
   332→            </div>
   333→          </div>
   334→        )}
   335→
   336→        {/* Name — always shown but optional */}
   337→        <div>
   338→          <label className="block text-xs font-bold text-slate-500 dark:text-zinc-400 mb-1 uppercase tracking-wider">
   339→            Full Name <span className="text-slate-400">(optional)</span>
   340→          </label>
   341→          <input
   342→            type="text"
   343→            value={name}
   344→            onChange={e => setName(e.target.value)}
   345→            placeholder={isProperty ? 'e.g., Chidi Okafor' : 'e.g., Adebayo & Associates'}
   346→            className="w-full px-3 py-2 rounded-lg border border-slate-200 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-800 dark:text-zinc-200 focus:ring-2 focus:ring-primary-500 focus:border-transparent outline-none"
   347→          />
   348→        </div>
   349→
   350→        {/* Link to matter/property — auto-fills name/phone/email */}
   351→        <div>
   352→          <label className="block text-xs font-bold text-slate-500 dark:text-zinc-400 mb-1 uppercase tracking-wider">
   353→            Link to {isProperty ? 'Property' : 'Matter'} <span className="text-slate-400">(optional — auto-fills details)</span>
   354→          </label>
   355→          {isProperty && flatUnits.length === 0 ? (
   356→            <div className="w-full px-3 py-2.5 rounded-lg border border-dashed border-slate-300 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-400 dark:text-zinc-500 italic">
   357→              No properties on file yet. Add properties to link them here.
   358→            </div>
   359→          ) : !isProperty && (matterState.matters || []).length === 0 ? (
   360→            <div className="w-full px-3 py-2.5 rounded-lg border border-dashed border-slate-300 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-400 dark:text-zinc-500 italic">
   361→              No matters on file yet. Create a matter to link it here.
   362→            </div>
   363→          ) : isProperty ? (
   364→            /* Grouped property/unit dropdown: address as non-selectable header, units below */
   365→            <>
   366→              <select
   367→                value={relatedId}
   368→                onChange={e => handleRelatedChange(e.target.value)}
   369→                className="w-full px-3 py-2 rounded-lg border border-slate-200 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-800 dark:text-zinc-200 focus:ring-2 focus:ring-primary-500 focus:border-transparent outline-none"
   370→              >
   371→                <option value="">Select a unit to auto-fill details</option>
   372→                {propertyGroups.map((group: PropertyGroup) => (
   373→                  <optgroup key={group.addressKey} label={group.shortAddress}>
   374→                    {group.units.map(u => (
   375→                      <option key={u.id} value={u.id}>
   376→                        {u.unitName ? `${u.unitName}` : u.shortAddress}{u.tenantName ? ` — ${u.tenantName}` : ''}
   377→                      </option>
   378→                    ))}
   379→                  </optgroup>
   380→                ))}
   381→              </select>
   382→              {relatedId && (
   383→                <p className="text-[10px] text-primary-600 dark:text-primary-400 mt-1 font-medium">
   384→                  Name, email, and phone will be auto-filled from the selected tenant record.
   385→                </p>
   386→              )}
   387→            </>
   388→          ) : (
   389→            <>
   390→              <select
   391→                value={relatedId}
   392→                onChange={e => handleRelatedChange(e.target.value)}
   393→                className="w-full px-3 py-2 rounded-lg border border-slate-200 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-800 dark:text-zinc-200 focus:ring-2 focus:ring-primary-500 focus:border-transparent outline-none"
   394→              >
   395→                <option value="">Select a matter to auto-fill details</option>
   396→                {relatedItems.map((item: any) => (
   397→                  <option key={item.id} value={item.id}>{item.label}</option>
   398→                ))}
   399→              </select>
   400→              {relatedId && (
   401→                <p className="text-[10px] text-primary-600 dark:text-primary-400 mt-1 font-medium">
   402→                  Name, email, and phone will be auto-filled from the selected client record.
   403→                </p>
   404→              )}
   405→            </>
   406→          )}
   407→        </div>
   408→
   409→        {/* Actions */}
   410→        <div className="flex items-center gap-3 pt-2 flex-wrap">
   411→          <button
   412→            onClick={onCancel}
   413→            className="px-4 py-2 text-sm font-medium text-slate-600 dark:text-zinc-300 hover:text-slate-800 dark:hover:text-zinc-100 transition-colors"
   414→          >
   415→            Cancel
   416→          </button>
   417→          <button
   418→            onClick={handleSubmit}
   419→            disabled={isSending || (!email.trim() && channel !== 'whatsapp') || (needsPhone && !phone.trim())}
   420→            className="inline-flex items-center gap-2 px-5 py-2 bg-primary-600 text-white rounded-lg font-semibold text-sm hover:bg-primary-700 transition-colors disabled:opacity-50 disabled:cursor-not-allowed"
   421→          >
   422→            {isSending ? (
   423→              <>
   424→                <div className="w-4 h-4 border-2 border-white border-t-transparent rounded-full animate-spin" />
   425→                Sending…
   426→              </>
   427→            ) : (
   428→              <>
   429→                <SendIcon className="w-4 h-4" />
   430→                Send via {channel === 'email' ? 'Email' : channel === 'whatsapp' ? 'WhatsApp' : 'Email & WhatsApp'}
   431→              </>
