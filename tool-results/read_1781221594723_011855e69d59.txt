     1→/**

    12→ * - Magic-link tokens embedded in invite URLs for setup-password flow
    13→ * - Resend, revoke, and copy invite links with real token
    14→ * - Track invitation status (pending / accepted / expired / revoked)
    15→ * - Delete with confirmation + cleanup of associated portal user
    16→ * - Preview portal as a specific client/resident

   144→// ─── Invite Form ─────────────────────────────────────────────────────────────
   145→const InviteForm: React.FC<{
   146→  firmId: string;
   147→  inviterId: string;
   148→  portalType: 'client' | 'resident';
   149→  onSent: () => void;
   150→  onCancel: () => void;
   151→}> = ({ firmId, inviterId, portalType, onSent, onCancel }) => {
   152→  const { addToast } = useUI();
   153→  const sendInvite = useAction(api.portals.createPortalInvite);
   154→  const { coreState } = useCoreState();
   155→  const { matterState } = useMatterState();
   156→
   157→  const isProperty = portalType === 'resident';
   158→
   159→  const [email, setEmail] = useState('');
   160→  const [name, setName] = useState('');
   161→  const [phone, setPhone] = useState('');
   162→  const [relatedId, setRelatedId] = useState('');
   163→  const [channel, setChannel] = useState<'email' | 'whatsapp' | 'both'>('email');
   164→  const [isSending, setIsSending] = useState(false);
   165→  const [isNameAutoFilled, setIsNameAutoFilled] = useState(false);
   166→
   167→  // Build list of matters (client portal) or properties (resident portal) to link invite to
   168→  const { groups: propertyGroups, flatUnits } = usePropertyGroups(coreState.properties || []);
   169→  const relatedItems = useMemo(() => {
   170→    if (isProperty) {
   171→      return flatUnits.map(u => ({
   172→        id: u.id,
   173→        label: u.label,
   174→        tenantName: u.tenantName || '',
   175→        tenantPhone: u.tenantPhone || '',
   176→        tenantEmail: u.tenantEmail || '',
   177→      }));
   178→    } else {
   179→      return (matterState.matters || []).map((m: any) => ({
   180→        id: m.id,
   181→        label: m.title || m.name || `Matter ${String(m.id || '').slice(-5)}`,
   182→        clientName: m.clientName || '',
   183→        clientEmail: m.clientEmail || '',
   184→        clientPhone: m.clientPhone || '',
   185→      }));
   186→    }
   187→  }, [isProperty, flatUnits, matterState.matters]);
   188→
   189→  // Auto-populate when user selects a matter/property
   190→  const handleRelatedChange = useCallback((selectedId: string) => {
   191→    setRelatedId(selectedId);
   192→    if (!selectedId) {
   193→      setIsNameAutoFilled(false);
   194→      return;
   195→    }
   196→    const item = relatedItems.find((r: any) => r.id === selectedId);
   197→    if (!item) return;
   198→    if (isProperty) {
   199→      // Auto-fill from property tenant data — always override if tenant data exists
   200→      // For resident invites, the tenant name comes from the property record (source of truth)
   201→      const propItem = item as { id: string; label: string; tenantName: string; tenantPhone: string; tenantEmail: string };
   202→      if (propItem.tenantName) {
   203→        setName(propItem.tenantName);
   204→        setIsNameAutoFilled(true);
   205→      } else {
   206→        setIsNameAutoFilled(false);
   207→      }
   208→      if (propItem.tenantPhone) setPhone(propItem.tenantPhone);
   209→      if (propItem.tenantEmail) setEmail(propItem.tenantEmail);
   210→    } else {
   211→      // Auto-fill from matter client data
   212→      const matterItem = item as { id: any; label: any; clientName: any; clientEmail: any; clientPhone: any };
   213→      if (matterItem.clientName) setName(matterItem.clientName);
   214→      if (matterItem.clientEmail) setEmail(matterItem.clientEmail);
   215→      if (matterItem.clientPhone) setPhone(matterItem.clientPhone);
   216→    }
   217→  }, [relatedItems, isProperty]);
   218→
   219→  const handleSubmit = async () => {
   220→    if (!email.trim() && channel !== 'whatsapp') {
   221→      addToast('Please enter an email address', { type: 'error' });
   222→      return;
   223→    }
   224→    if ((channel === 'whatsapp' || channel === 'both') && !phone.trim()) {
   225→      addToast('Please enter a phone number for WhatsApp delivery', { type: 'error' });
   226→      return;
   227→    }
   228→    setIsSending(true);
   229→    try {
   230→      const result = await sendInvite({
   231→        firmId,
   232→        inviterId,
   233→        inviteeEmail: email.trim().toLowerCase() || undefined,
   234→        inviteeName: name.trim() || undefined,
   235→        inviteePhone: phone.trim() || undefined,
   236→        portalType,
   237→        relatedId: relatedId || undefined,
   238→        channel,
   239→      });
   240→      // Build feedback message based on result
   241→      const parts: string[] = [];
   242→      if (result.emailSent) parts.push('email delivered');
   243→      else if (result.emailSimulated) parts.push('email simulated (Brevo API key not configured)');
   244→      if (result.whatsappSent) parts.push('WhatsApp delivered');
   245→      else if (result.whatsappSimulated) parts.push('WhatsApp simulated');
   246→      else if (result.whatsappSkipped && (channel === 'whatsapp' || channel === 'both')) parts.push('WhatsApp skipped (no phone number)');
   247→
   248→      const feedback = parts.length > 0 ? parts.join(', ') : 'invitation created';
   249→      addToast(`Invitation sent — ${feedback}`, { type: 'success' });
   250→      onSent();
   251→    } catch (err: any) {
   252→      addToast(err.message || 'Failed to send invitation', { type: 'error' });
   253→    } finally {
   254→      setIsSending(false);
   255→    }
   256→  };
   257→
   258→  const needsPhone = channel === 'whatsapp' || channel === 'both';
   259→
   260→  const portalLabel = isProperty ? 'Resident' : 'Client';
   261→  const portalDescription = isProperty
   262→    ? 'Invite a resident to access their payment ledger, maintenance tickets, and receipts.'
   263→    : 'Invite a client to access their matters, documents, messages, and billing.';
   264→
   265→  return (
   266→    <div className="bg-white dark:bg-zinc-800 rounded-xl border border-slate-200 dark:border-zinc-700 p-4 sm:p-6">
   267→      <div className="flex items-center justify-between mb-4">
   268→        <div>
   269→          <h4 className="text-base font-bold text-slate-900 dark:text-white flex items-center gap-2">
   270→            {isProperty ? <OfficeBuildingIcon className="w-4 h-4 text-sky-500" /> : <UsersIcon className="w-4 h-4 text-violet-500" />}
   271→            Invite {portalLabel} to Portal
   272→          </h4>
   273→          <p className="text-xs text-slate-500 dark:text-zinc-400 mt-0.5">{portalDescription}</p>
   274→        </div>
   275→        <button onClick={onCancel} className="text-slate-400 hover:text-slate-600 dark:hover:text-zinc-200 transition-colors">
   276→          <XIcon className="w-5 h-5" />
   277→        </button>
   278→      </div>
   279→
   280→      <div className="space-y-4">
   281→        {/* Channel Picker */}
   282→        <div>
   283→          <label className="block text-xs font-bold text-slate-500 dark:text-zinc-400 mb-2 uppercase tracking-wider">
   284→            Send Via
   285→          </label>
   286→          <div className="flex gap-2">
   287→            {[
   288→              { value: 'email' as const, label: 'Email', icon: MailIcon },
   289→              { value: 'whatsapp' as const, label: 'WhatsApp', icon: DeviceMobileIcon },
   290→              { value: 'both' as const, label: 'Both', icon: SendIcon },
   291→            ].map(opt => (
   292→              <button
   293→                key={opt.value}
   294→                onClick={() => setChannel(opt.value)}
   295→                className={`flex-1 flex items-center justify-center gap-1.5 px-3 py-2 rounded-lg text-xs font-bold border transition-colors ${
   296→                  channel === opt.value
   297→                    ? 'bg-primary-50 dark:bg-primary-900/20 border-primary-300 dark:border-primary-700 text-primary-700 dark:text-primary-400'
   298→                    : 'bg-slate-50 dark:bg-zinc-900 border-slate-200 dark:border-zinc-600 text-slate-500 dark:text-zinc-400 hover:bg-slate-100 dark:hover:bg-zinc-800'
   299→                }`}
   300→              >
   301→                <opt.icon className="w-3.5 h-3.5" />
   302→                {opt.label}
   303→              </button>
   304→            ))}
   305→          </div>
   306→        </div>
   307→
   308→        {/* Email — required unless WhatsApp-only */}
   309→        {channel !== 'whatsapp' && (
   310→          <div>
   311→            <label className="block text-xs font-bold text-slate-500 dark:text-zinc-400 mb-1 uppercase tracking-wider">
   312→              Email Address <span className="text-rose-500">*</span>
   313→            </label>
   314→            <div className="relative">
   315→              <MailIcon className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   316→              <input
   317→                type="email"
   318→                value={email}
   319→                onChange={e => setEmail(e.target.value)}
   320→                placeholder={isProperty ? 'resident@example.com' : 'client@example.com'}
   321→                className="w-full pl-9 pr-3 py-2 rounded-lg border border-slate-200 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-800 dark:text-zinc-200 focus:ring-2 focus:ring-primary-500 focus:border-transparent outline-none"
   322→              />
   323→            </div>
   324→          </div>
   325→        )}
   326→
   327→        {/* Phone — required for WhatsApp/Both */}
   328→        {needsPhone && (
   329→          <div>
   330→            <label className="block text-xs font-bold text-slate-500 dark:text-zinc-400 mb-1 uppercase tracking-wider">
   331→              Phone / WhatsApp <span className="text-rose-500">*</span>
   332→            </label>
   333→            <div className="relative">
   334→              <DeviceMobileIcon className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   335→              <input
   336→                type="tel"
   337→                value={phone}
   338→                onChange={e => setPhone(e.target.value)}
   339→                placeholder="e.g., +2348012345678"
   340→                className="w-full pl-9 pr-3 py-2 rounded-lg border border-slate-200 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-800 dark:text-zinc-200 focus:ring-2 focus:ring-primary-500 focus:border-transparent outline-none"
   341→              />
   342→            </div>
   343→          </div>
   344→        )}
   345→
   346→        {/* Name — auto-filled from tenant record for resident invites, optional for client invites */}
   347→        <div>
   348→          <label className="block text-xs font-bold text-slate-500 dark:text-zinc-400 mb-1 uppercase tracking-wider">
   349→            Full Name{' '}
   350→            {isProperty && isNameAutoFilled ? (
   351→              <span className="text-emerald-600 dark:text-emerald-400 font-normal normal-case tracking-normal">· from tenant record</span>
   352→            ) : (
   353→              <span className="text-slate-400">(optional)</span>
   354→            )}
   355→          </label>
   356→          <input
   357→            type="text"
   358→            value={name}
   359→            onChange={e => { setName(e.target.value); if (isProperty && isNameAutoFilled) setIsNameAutoFilled(false); }}
   360→            readOnly={isProperty && isNameAutoFilled}
   361→            placeholder={isProperty ? 'e.g., Chidi Okafor' : 'e.g., Adebayo & Associates'}
   362→            className={`w-full px-3 py-2 rounded-lg border text-sm outline-none ${
   363→              isProperty && isNameAutoFilled
   364→                ? 'border-emerald-300 dark:border-emerald-700 bg-emerald-50/50 dark:bg-emerald-900/10 text-slate-800 dark:text-zinc-200 cursor-default'
   365→                : 'border-slate-200 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-slate-800 dark:text-zinc-200 focus:ring-2 focus:ring-primary-500 focus:border-transparent'
   366→            }`}
   367→          />
   368→          {isProperty && isNameAutoFilled && (
   369→            <p className="text-[10px] text-emerald-600 dark:text-emerald-400 mt-1 font-medium">
   370→              This name is from the property's tenant record and will be used as the official name on the portal.
   371→              Edit only if the tenant record is incorrect.
   372→            </p>
   373→          )}
   374→        </div>
   375→
   376→        {/* Link to matter/property — auto-fills name/phone/email */}
   377→        <div>
   378→          <label className="block text-xs font-bold text-slate-500 dark:text-zinc-400 mb-1 uppercase tracking-wider">
   379→            Link to {isProperty ? 'Property' : 'Matter'} <span className="text-slate-400">(optional — auto-fills details)</span>
   380→          </label>
   381→          {isProperty && flatUnits.length === 0 ? (
   382→            <div className="w-full px-3 py-2.5 rounded-lg border border-dashed border-slate-300 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-400 dark:text-zinc-500 italic">
   383→              No properties on file yet. Add properties to link them here.
   384→            </div>
   385→          ) : !isProperty && (matterState.matters || []).length === 0 ? (
   386→            <div className="w-full px-3 py-2.5 rounded-lg border border-dashed border-slate-300 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-400 dark:text-zinc-500 italic">
   387→              No matters on file yet. Create a matter to link it here.
   388→            </div>
   389→          ) : isProperty ? (
   390→            /* Grouped property/unit dropdown: address as non-selectable header, units below */
   391→            <>
   392→              <select
   393→                value={relatedId}
   394→                onChange={e => handleRelatedChange(e.target.value)}
   395→                className="w-full px-3 py-2 rounded-lg border border-slate-200 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-800 dark:text-zinc-200 focus:ring-2 focus:ring-primary-500 focus:border-transparent outline-none"
   396→              >
   397→                <option value="">Select a unit to auto-fill details</option>
   398→                {propertyGroups.map((group: PropertyGroup) => (
   399→                  <optgroup key={group.addressKey} label={group.shortAddress}>
   400→                    {group.units.map(u => (
   401→                      <option key={u.id} value={u.id}>
   402→                        {u.unitName ? `${u.unitName}` : u.shortAddress}{u.tenantName ? ` — ${u.tenantName}` : ''}
   403→                      </option>
   404→                    ))}
   405→                  </optgroup>
   406→                ))}
   407→              </select>
   408→              {relatedId && (
   409→                <p className="text-[10px] text-primary-600 dark:text-primary-400 mt-1 font-medium">
   410→                  Name, email, and phone will be auto-filled from the selected tenant record.
   411→                </p>
   412→              )}
   413→            </>
   414→          ) : (
   415→            <>
   416→              <select
   417→                value={relatedId}
   418→                onChange={e => handleRelatedChange(e.target.value)}
   419→                className="w-full px-3 py-2 rounded-lg border border-slate-200 dark:border-zinc-600 bg-slate-50 dark:bg-zinc-900 text-sm text-slate-800 dark:text-zinc-200 focus:ring-2 focus:ring-primary-500 focus:border-transparent outline-none"
   420→              >
   421→                <option value="">Select a matter to auto-fill details</option>
   422→                {relatedItems.map((item: any) => (
   423→                  <option key={item.id} value={item.id}>{item.label}</option>
   424→                ))}
   425→              </select>
   426→              {relatedId && (
   427→                <p className="text-[10px] text-primary-600 dark:text-primary-400 mt-1 font-medium">
