    33→export const getMaintenanceTicketsByTenant = query({
    34→  args: { tenantId: v.string() },
    35→  handler: async (ctx, args) => {
    36→    const tickets = await ctx.db
    37→      .query("maintenance_tickets")
    38→      .withIndex("by_tenant", (q) => q.eq("tenantId", args.tenantId))
    39→      .order("desc")
    40→      .collect();
    41→
    42→    // Also try matching by email in case the tenantId stored is the user's email
    43→    // This handles the case where the invite was created with the email as the tenant ID
    44→    if (tickets.length === 0 && args.tenantId.includes('@')) {
    45→      // The tenantId looks like an email, also search by the user's Convex _id
    46→      const user: any = await ctx.db
    47→        .query("users")
    48→        .withIndex("by_token", (q) => q.eq("tokenIdentifier", args.tenantId.toLowerCase()))
    49→        .first();
    50→      if (user) {
    51→        const byUserId = await ctx.db
    52→          .query("maintenance_tickets")
    53→          .withIndex("by_tenant", (q) => q.eq("tenantId", String(user._id)))
    54→          .order("desc")
    55→          .collect();
    56→        return byUserId;
    57→      }
    58→    }
    59→
    60→    return tickets;
    61→  },
    62→});

    91→/** Generate a crypto-random invite token (12 chars, URL-safe) */
    92→function generateToken(): string {
    93→  const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
    94→  const arr = new Uint8Array(12);
    95→  // @ts-ignore — crypto available in Convex runtime
    96→  crypto.getRandomValues(arr);
    97→  return Array.from(arr, b => chars[b % chars.length]).join("");
    98→}

   275→/**
   276→ * Internal mutation — only used by createPortalInvite action to write the DB record.
   277→ * Not exported for direct frontend use.
   278→ */
   279→export const insertInviteRecord = mutation({
   280→  args: {
   281→    firmId: v.string(),
   282→    inviterId: v.string(),
   283→    inviteeEmail: v.optional(v.string()),
   284→    inviteeName: v.optional(v.string()),
   285→    inviteePhone: v.optional(v.string()),
   286→    portalType: v.union(v.literal("client"), v.literal("resident")),
   287→    relatedId: v.optional(v.string()),
   288→    token: v.optional(v.string()),
   289→    channel: v.optional(v.string()),
   290→    message: v.optional(v.string()),
   291→    expiresAt: v.number(),
   292→  },
   293→  handler: async (ctx, args) => {
   294→    const now = Date.now();
   295→
   296→    // Clean up any existing invites for the same email + firm + portal type.
   297→    // This prevents "invitation already accepted" errors when re-inviting
   298→    // after a previous invite was deleted via the simple deletePortalInvite
   299→    // (which doesn't clean up). We mark old invites as "superseded" rather
   300→    // than deleting them, so there's an audit trail.
   301→    const email = (args.inviteeEmail || "").toLowerCase().trim();
   302→    if (email) {
   303→      const existingInvites = await ctx.db
   304→        .query("portal_invites")
   305→        .withIndex("by_email", (q) => q.eq("inviteeEmail", email))
   306→        .collect();
   307→
   308→      for (const inv of existingInvites) {
   309→        // Only supersede invites for the same firm and portal type
   310→        if (inv.firmId === args.firmId && inv.portalType === args.portalType) {
   311→          if (inv.status === "pending" || inv.status === "accepted") {
   312→            await ctx.db.patch(inv._id, {
   313→              status: "superseded",
   314→              updatedAt: now,
   315→            });
   316→          }
   317→        }
   318→      }
   319→    }
   320→
   321→    return await ctx.db.insert("portal_invites", {
   322→      firmId: args.firmId,
   323→      inviterId: args.inviterId,
   324→      inviteeEmail: args.inviteeEmail,
   325→      inviteeName: args.inviteeName,
   326→      inviteePhone: args.inviteePhone,
   327→      portalType: args.portalType,
   328→      relatedId: args.relatedId,
   329→      token: args.token ?? undefined,
   330→      channel: args.channel,
   331→      message: args.message,
   332→      status: "pending",
   333→      expiresAt: args.expiresAt,
   334→      createdAt: now,
   335→      updatedAt: now,
   336→    });
   337→  },
   338→});

   520→/**
   521→ * deletePortalInviteAndCleanup — Permanently deletes a portal invite AND resets
   522→ * the associated portal user account so the same email can be re-invited cleanly.
   523→ *
   524→ * This prevents the "invitation already accepted" error when an admin:
   525→ * 1. Deletes a portal access record
   526→ * 2. Creates a new invite for the same email
   527→ *
   528→ * Steps:
   529→ * 1. Delete the portal_invites record
   530→ * 2. Find the user with the matching email + Client/Tenant role
   531→ * 3. Reset their role, verification, and password so they're treated as a brand-new invitee
   532→ */
   533→export const deletePortalInviteAndCleanup = mutation({
   534→  args: {
   535→    inviteId: v.id("portal_invites"),
   536→    inviteeEmail: v.optional(v.string()),
   537→  },
   538→  handler: async (ctx, args) => {
   539→    // 1. Delete the specified invite record
   540→    await ctx.db.delete(args.inviteId);
   541→
   542→    // 2. Also delete ANY other invite records for the same email
   543→    //    to prevent residual "already accepted" errors on re-invite
   544→    const email = (args.inviteeEmail || "").toLowerCase().trim();
   545→    if (!email) return;
   546→
   547→    const otherInvites = await ctx.db
   548→      .query("portal_invites")
   549→      .withIndex("by_email", (q) => q.eq("inviteeEmail", email))
   550→      .collect();
   551→
   552→    for (const inv of otherInvites) {
   553→      if (String(inv._id) === String(args.inviteId)) continue; // Already deleted above
   554→      try { await ctx.db.delete(inv._id); } catch (e) { /* ignore */ }
   555→    }
   556→
   557→    // 3. Find and reset the associated portal user
   558→    const existingUser = await ctx.db
   559→      .query("users")
   560→      .withIndex("by_token", (q) => q.eq("tokenIdentifier", email))
   561→      .first();
   562→
   563→    if (!existingUser) return;
   564→
   565→    // Only reset if the user has a Client or Tenant role (portal user)
   566→    const role = (existingUser as any).role;
   567→    if (role === "Client" || role === "Tenant" || role === "Portal User") {
   568→      // Reset the user so they can be re-invited cleanly
   569→      // Set isVerified to false so they can't log in with old credentials
   570→      // Remove the password so setup-password creates a fresh one
   571→      // Clear firmId so it will be set fresh on the next invite acceptance
   572→      await ctx.db.patch(existingUser._id, {
   573→        isVerified: false,
   574→        emailVerified: false,
   575→        password: undefined,
   576→        role: "Pending",
   577→        firmId: undefined,
   578→      } as any);
   579→    }
   580→  },
   581→});

   630→/**
   631→ * verifyInviteToken — validates a portal invite token and returns
   632→ * the invite details + whether the user already has an account.
   633→ * Used by the /setup-password page to determine what to show.
   634→ *
   635→ * ROBUSTNESS FIX: If the invite is "accepted" but the user's role is
   636→ * "Pending" (meaning their access was deleted via deletePortalInviteAndCleanup
   637→ * and they need to re-setup), we allow them to proceed. This prevents the
   638→ * "invitation already accepted" dead-end when a user is re-invited after
   639→ * their portal access was removed.
   640→ */
   641→export const verifyInviteToken = query({
   642→  args: { token: v.string() },
   643→  handler: async (ctx, args) => {
   644→    const results = await ctx.db
   645→      .query("portal_invites")
   646→      .withIndex("by_token", (q) => q.eq("token", args.token))
   647→      .collect();
   648→    const invite = results[0] || null;
   649→
   650→    if (!invite) {
   651→      return { valid: false, reason: "not_found" } as const;
   652→    }
   653→
   654→    // Check expiry (note: we don't patch status here since this is a query/read-only)
   655→    if (invite.expiresAt < Date.now()) {
   656→      return { valid: false, reason: "expired" } as const;
   657→    }
   658→
   659→    if (invite.status === "revoked") {
   660→      return { valid: false, reason: "revoked" } as const;
   661→    }
   662→
   663→    // Check if the user exists and their current state
   664→    const email = invite.inviteeEmail?.toLowerCase().trim();
   665→    let existingUser: any = null;
   666→    if (email) {
   667→      existingUser = await ctx.db
   668→        .query("users")
   669→        .withIndex("by_token", (q) => q.eq("tokenIdentifier", email))
   670→        .first();
   671→      // Case-insensitive fallback
   672→      if (!existingUser) {
   673→        existingUser = await ctx.db
   674→          .query("users")
   675→          .withIndex("by_token", (q) => q.eq("tokenIdentifier", email.toLowerCase()))
   676→          .first();
   677→      }
   678→    }
   679→
   680→    if (invite.status === "accepted") {
   681→      // ROBUSTNESS: If the user's role is Pending (their access was reset by
   682→      // deletePortalInviteAndCleanup), allow them to re-accept. This invite
   683→      // might be a stale record from before the cleanup ran. The user needs
   684→      // to set up their password again, so we should let them through.
   685→      const userRole = existingUser?.role;
   686→      const isUserUnverified = existingUser && !existingUser.isVerified;
   687→      if (userRole === "Pending" || isUserUnverified) {
   688→        // User was reset — allow re-accepting this invite
   689→        // Fall through to return valid: true
   690→      } else {
   691→        return { valid: false, reason: "already_accepted" } as const;
   692→      }
   693→    }
   694→
   695→    // Token is valid and pending — check if user already has an account
   696→    return {
   697→      valid: true,
   698→      invite: {
   699→        _id: invite._id,
   700→        inviteeEmail: invite.inviteeEmail,
   701→        inviteeName: invite.inviteeName,
   702→        portalType: invite.portalType,
   703→        firmId: invite.firmId,
   704→        relatedId: invite.relatedId,
   705→        expiresAt: invite.expiresAt,
   706→      },
   707→      hasAccount: !!existingUser,
   708→      hasPassword: !!(existingUser as any)?.password,
   709→    } as const;
   710→  },
   711→});

   748→// ─── Setup Portal Password (Invite Acceptance) ──────────────────────────
   749→
   750→/**
   751→ * setupPortalPassword — ACTION that completes the invite flow:
   752→ * 1. Validates the invite token (must be pending + unexpired)
   753→ * 2. If user exists → sets their password + marks verified
   754→ * 3. If user doesn't exist → creates a new user record with the password
   755→ * 4. Marks the invite as accepted
   756→ * 5. Clears the token to prevent replay attacks
   757→ */
   758→export const setupPortalPassword = action({
   759→  args: {
   760→    token: v.string(),
   761→    password: v.string(),
   762→    name: v.optional(v.string()),
   763→  },
   764→  handler: async (ctx, args): Promise<{ success: boolean; message?: string; email?: string }> => {
   765→    // 1. Validate the invite
   766→    const invite = await ctx.runQuery(api.portals.getInviteByToken, { token: args.token });
   767→    if (!invite) return { success: false, message: "Invalid invitation link." };
   768→    if (invite.status === "revoked") return { success: false, message: "This invitation has been revoked." };
   769→    if (invite.status === "expired" || invite.expiresAt < Date.now()) {
   770→      await ctx.runMutation(api.portals.updateInviteRecord, {
   771→        inviteId: invite._id,
   772→        updates: { status: "expired", updatedAt: Date.now() },
   773→      });
   774→      return { success: false, message: "This invitation has expired. Please request a new one." };
   775→    }
   776→
   777→    // ROBUSTNESS: If the invite was already accepted but the user's account was
   778→    // reset (role=Pending, isVerified=false), allow them to re-accept. This can
   779→    // happen when deletePortalInviteAndCleanup resets the user but a stale invite
   780→    // record still exists with status "accepted" and a valid token.
   781→    if (invite.status === "accepted") {
   782→      const emailCheck = (invite.inviteeEmail || "").toLowerCase().trim();
   783→      const existingUserCheck: any = await ctx.runQuery(api.myFunctions.getUser, { tokenIdentifier: emailCheck });
   784→      const wasReset = existingUserCheck?.role === "Pending" || (existingUserCheck && !existingUserCheck.isVerified);
   785→      if (!wasReset) {
   786→        return { success: false, message: "This invitation has already been used." };
   787→      }
   788→      // User was reset — allow re-accepting this invite (fall through)
   789→    }
   790→
   791→    const email = (invite.inviteeEmail || "").toLowerCase().trim();
   792→    if (!email) return { success: false, message: "This invitation has no email address associated." };
   793→
   794→    // 2. Hash the password
   795→    const hashedPassword = await ctx.runAction(internal.authUtils.hashPassword, { password: args.password });
   796→
   797→    // 3. Check if user exists
   798→    const existingUser: any = await ctx.runQuery(api.myFunctions.getUser, { tokenIdentifier: email });
   799→
   800→    // Track the portal user's Convex _id so we can link them to their property/unit
   801→    let portalUserDocId: string | null = null;
   802→
   803→    if (existingUser) {
   804→      // Update existing user: set password + mark verified
   805→      // Also upgrade their role to Client/Tenant if they were a "Portal User"
   806→      // or had no role assigned, so the frontend can route them correctly.
   807→      const portalRole = invite.portalType === "client" ? "Client" : "Tenant";
   808→      const needsRoleUpdate = !existingUser.role
   809→        || existingUser.role === "Portal User"
   810→        || existingUser.role === "Pending";
   811→      // Ensure firmId is always set — portal users need it for the ProductContext
   812→      // to derive the correct product and for data loading to work properly.
   813→      // When an existing user is re-invited after portal access deletion, their
   814→      // firmId might have been cleared or might never have been set.
   815→      const needsFirmId = !existingUser.firmId || existingUser.firmId !== invite.firmId;
   816→      portalUserDocId = String(existingUser._id);
   817→      await ctx.runMutation(internal.myFunctions.updateUserSecurityFields, {
   818→        userId: existingUser._id,
   819→        fields: {
   820→          password: hashedPassword,
   821→          isVerified: true,
   822→          emailVerified: true,
   823→          verificationCode: null,
   824→          ...(args.name && !existingUser.name ? { name: args.name } : {}),
   825→          ...(needsRoleUpdate ? { role: portalRole } : {}),
   826→          ...(needsFirmId ? { firmId: invite.firmId } : {}),
   827→          // Online presence privacy: property/resident portals default to hidden
   828→          portalPresenceHidden: invite.portalType === "resident" ? true : (existingUser as any).portalPresenceHidden ?? false,
   829→        },
   830→      });
   831→    } else {
   832→      // Create new user record for this portal user
   833→      // Use the proper UserRole (Client / Tenant) so the frontend can route
   834→      // them to the correct portal view automatically.
   835→      const portalProduct = invite.portalType === "client" ? "legal" : "property";
   836→      const portalRole = invite.portalType === "client" ? "Client" : "Tenant";
   837→      const userName = args.name || invite.inviteeName || email.split("@")[0];
   838→      const newUserId = await ctx.runMutation(internal.myFunctions.createUser, {
   839→        tokenIdentifier: email,
   840→        name: userName,
   841→        email: email,
   842→        password: hashedPassword,
   843→        role: portalRole,
   844→        product: portalProduct,
   845→        onboardingCompleted: false,
   846→        isVerified: true,
   847→        emailVerified: true,
   848→        firmId: invite.firmId,
   849→        avatarUrl: `https://ui-avatars.com/api/?name=${encodeURIComponent(userName)}&background=random`,
   850→        // Online presence privacy: property/resident portals default to hidden
   851→        // (portal users can't see if the manager is online). Legal/client portals
   852→        // default to visible (clients can see if their lawyer is online).
   853→        portalPresenceHidden: invite.portalType === "resident",
   854→      });
   855→      portalUserDocId = String(newUserId);
   856→    }
   857→
   858→    // 3.5. Link portal user to property/unit if relatedId was provided
   859→    // This is critical for getTenantInfo to find the tenant's property assignment.
   860→    // When setupPortalPassword creates a new user, the property's unit record
   861→    // may still point to a stale contact ID or email instead of the new Convex _id.
   862→    if (invite.relatedId && invite.portalType === "resident" && portalUserDocId) {
   863→      try {
   864→        // Parse the relatedId format:
   865→        // "propertyId_unitId" for multi-unit properties (composite key)
   866→        // just "propertyId" for single properties
   867→        const parts = invite.relatedId.split("_");
   868→        const propertyCustomId = parts[0];
   869→        const unitId = parts.length > 1 ? parts.slice(1).join("_") : null;
   870→
   871→        // Find the property using internal query (actions cannot use ctx.db directly)
   872→        const property: any = await ctx.runQuery(internal.portals.findPropertyByCustomId, {
   873→          customId: propertyCustomId,
   874→        });
   875→
   876→        if (property) {
   877→          const units = property.units || [];
   878→          // Look up the user for their name (best-effort; use invite data as fallback)
   879→          const portalUser: any = await ctx.runQuery(api.myFunctions.getUser, { tokenIdentifier: email });
   880→          const tenantName = portalUser?.name || invite.inviteeName || email.split("@")[0];
   881→
   882→          if (units.length > 0 && unitId) {
   883→            // Multi-unit property: update the specific unit
   884→            let unitFound = false;
   885→            const updatedUnits = units.map((u: any) => {
   886→              if (u.id === unitId || u.unitName === unitId || u.id === invite.relatedId) {
   887→                unitFound = true;
   888→                return {
   889→                  ...u,
   890→                  currentTenantId: portalUserDocId,
   891→                  tenantId: portalUserDocId,
   892→                  tenantEmail: email,
   893→                  tenantName: tenantName,
   894→                };
   895→              }
   896→              return u;
   897→            });
   898→            if (unitFound) {
   899→              await ctx.runMutation(internal.portals.linkPortalUserToProperty, {
   900→                propertyId: property._id,
   901→                updates: { units: updatedUnits },
   902→              });
   903→            }
   904→          } else {
   905→            // Single property (no units or no unitId): update property-level tenant
   906→            await ctx.runMutation(internal.portals.linkPortalUserToProperty, {
   907→              propertyId: property._id,
   908→              updates: {
   909→                currentTenantId: portalUserDocId,
   910→                tenantId: portalUserDocId,
   911→              },
   912→            });
   913→          }
   914→        }
   915→      } catch (linkErr) {
   916→        // Non-blocking: if linking fails, the user can still access the portal.
   917→        // getTenantInfo will try to find them by email as a fallback.
   918→        console.warn("Portal user-property linking failed:", linkErr);
   919→      }
   920→    }
   921→
   922→    // 4. Mark invite as accepted
   923→    await ctx.runMutation(api.portals.updateInviteRecord, {
   924→      inviteId: invite._id,
   925→      updates: {
   926→        status: "accepted",
   927→        acceptedAt: Date.now(),
   928→        updatedAt: Date.now(),
   929→        // Clear token to prevent replay attacks — the invite is now consumed
   930→        token: undefined,
   931→      },
   932→    });
   933→
   934→    return { success: true, email };
   935→  },

   999→/**
  1000→ * getTenantInfo — Resolves the tenant's property/unit assignments using their
  1001→ * user ID and email. This is critical because the tenantId stored in
  1002→ * properties/units may be the user's Convex _id OR their email, depending
  1003→ * on how the invite was created. This query normalizes both cases.
  1004→ *
  1005→ * Returns: { tenantId, properties, units } where tenantId is the canonical
  1006→ * ID that should be used for all other portal queries.
  1007→ */
  1008→export const getTenantInfo = query({
  1009→  args: { firmId: v.string(), userId: v.string(), email: v.string() },
  1010→  handler: async (ctx, args) => {
  1011→    const properties = await ctx.db
  1012→      .query("properties")
  1013→      .withIndex("by_firm", (q) => q.eq("firmId", args.firmId))
  1014→      .collect();
  1015→
  1016→    // Search for tenant in properties using both userId and email
  1017→    const tenantProperties: any[] = [];
  1018→    const tenantUnits: any[] = [];
  1019→    let resolvedTenantId = args.userId;
  1020→
  1021→    for (const prop of properties) {
  1022→      // Check property-level tenant (legacy single-tenant model)
  1023→      const propTenantId = (prop as any).currentTenantId || (prop as any).tenantId;
  1024→      if (propTenantId === args.userId || propTenantId === args.email ||
  1025→          String(propTenantId).toLowerCase() === args.email.toLowerCase()) {
  1026→        tenantProperties.push({
  1027→          id: String(prop._id),
  1028→          name: (prop as any).name || prop.address || 'Unnamed Property',
  1029→          address: prop.address,
  1030→        });
  1031→        if (propTenantId && propTenantId !== args.userId) {
  1032→          resolvedTenantId = propTenantId;
  1033→        }
  1034→      }
  1035→
  1036→      // Check unit-level tenants (multi-unit model)
  1037→      const units = (prop as any).units || [];
  1038→      for (const unit of units) {
  1039→        const unitTenantId = unit.currentTenantId || unit.tenantId;
  1040→        const unitTenantEmail = (unit.tenantEmail || '').toLowerCase();
  1041→        const matchesUserId = unitTenantId === args.userId;
  1042→        const matchesEmail = unitTenantId === args.email ||
  1043→            String(unitTenantId).toLowerCase() === args.email.toLowerCase() ||
  1044→            (unitTenantEmail && unitTenantEmail === args.email.toLowerCase());
  1045→
  1046→        if (matchesUserId || matchesEmail) {
  1047→          tenantUnits.push({
  1048→            id: unit.id || unit._id,
  1049→            name: unit.name || unit.unitName || unit.label,
  1050→            unitName: unit.name || unit.unitName || unit.label,
  1051→            propertyId: String(prop._id),
  1052→            propertyName: (prop as any).name || prop.address || 'Unnamed Property',
  1053→            propertyAddress: prop.address,
  1054→            amenities: unit.amenities || [],
  1055→          });
  1056→          if (unitTenantId && unitTenantId !== args.userId) {
  1057→            resolvedTenantId = unitTenantId;
  1058→          }
  1059→        }
  1060→      }
  1061→    }
  1062→
  1063→    // Determine primary property/unit for this tenant (first match)
  1064→    const primaryUnit = tenantUnits.length > 0 ? tenantUnits[0] : null;
  1065→    const primaryProperty = primaryUnit
  1066→      ? { id: primaryUnit.propertyId, name: primaryUnit.propertyName, address: primaryUnit.propertyAddress }
  1067→      : tenantProperties.length > 0 ? tenantProperties[0] : null;
  1068→
  1069→    return {
  1070→      tenantId: resolvedTenantId,
  1071→      properties: tenantProperties,
  1072→      units: tenantUnits,
  1073→      // Convenience fields for maintenance ticket creation
  1074→      primaryPropertyId: primaryProperty?.id || null,
  1075→      primaryUnitId: primaryUnit?.id || null,
  1076→      primaryPropertyName: primaryProperty?.name || null,
  1077→      primaryUnitName: primaryUnit?.unitName || null,
  1078→      primaryPropertyAddress: primaryProperty?.address || null,
  1079→    };
  1080→  },
