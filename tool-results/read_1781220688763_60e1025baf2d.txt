    37→      .query("maintenance_tickets")
    38→      .withIndex("by_tenant", (q) => q.eq("tenantId", args.tenantId))
    39→      .order("desc")

    41→
    42→    // Also try matching by email in case the tenantId stored is the user's email
    43→    // This handles the case where the invite was created with the email as the tenant ID
    44→    if (tickets.length === 0 && args.tenantId.includes('@')) {
    45→      // The tenantId looks like an email, also search by the user's Convex _id

    47→        .query("users")
    48→        .withIndex("by_token", (q) => q.eq("tokenIdentifier", args.tenantId.toLowerCase()))

    52→          .query("maintenance_tickets")
    53→          .withIndex("by_tenant", (q) => q.eq("tenantId", String(user._id)))
    54→          .order("desc")
    55→          .collect();
    56→        return byUserId;
    57→      }
    58→    }
    59→
    60→    return tickets;

    91→/** Generate a crypto-random invite token (12 chars, URL-safe) */
    92→function generateToken(): string {
    93→  const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
    94→  const arr = new Uint8Array(12);
    95→  // @ts-ignore — crypto available in Convex runtime
    96→  crypto.getRandomValues(arr);
    97→  return Array.from(arr, b => chars[b % chars.length]).join("");
    98→}

   317→/**
   318→ * Internal mutation — only used by createPortalInvite action to write the DB record.
   319→ * Not exported for direct frontend use.
   320→ */
   321→export const insertInviteRecord = mutation({
   322→  args: {
   323→    firmId: v.string(),
   324→    inviterId: v.string(),
   325→    inviteeEmail: v.optional(v.string()),
   326→    inviteeName: v.optional(v.string()),
   327→    inviteePhone: v.optional(v.string()),
   328→    portalType: v.union(v.literal("client"), v.literal("resident")),
   329→    relatedId: v.optional(v.string()),
   330→    token: v.optional(v.string()),
   331→    channel: v.optional(v.string()),
   332→    message: v.optional(v.string()),
   333→    expiresAt: v.number(),
   334→  },
   335→  handler: async (ctx, args) => {
   336→    const now = Date.now();
   337→
   338→    // Clean up any existing invites for the same email + firm + portal type.
   339→    // This prevents "invitation already accepted" errors when re-inviting
   340→    // after a previous invite was deleted via the simple deletePortalInvite
   341→    // (which doesn't clean up). We mark old invites as "superseded" rather
   342→    // than deleting them, so there's an audit trail.
   343→    const email = (args.inviteeEmail || "").toLowerCase().trim();
   344→    if (email) {
   345→      const existingInvites = await ctx.db
   346→        .query("portal_invites")
   347→        .withIndex("by_email", (q) => q.eq("inviteeEmail", email))
   348→        .collect();
   349→
   350→      for (const inv of existingInvites) {
   351→        // Only supersede invites for the same firm and portal type
   352→        if (inv.firmId === args.firmId && inv.portalType === args.portalType) {
   353→          if (inv.status === "pending" || inv.status === "accepted") {
   354→            await ctx.db.patch(inv._id, {
   355→              status: "superseded",
   356→              updatedAt: now,
   357→            });
   358→          }
   359→        }
   360→      }
   361→    }
   362→
   363→    return await ctx.db.insert("portal_invites", {
   364→      firmId: args.firmId,
   365→      inviterId: args.inviterId,
   366→      inviteeEmail: args.inviteeEmail,
   367→      inviteeName: args.inviteeName,
   368→      inviteePhone: args.inviteePhone,
   369→      portalType: args.portalType,
   370→      relatedId: args.relatedId,
   371→      token: args.token ?? undefined,
   372→      channel: args.channel,
   373→      message: args.message,
   374→      status: "pending",
   375→      expiresAt: args.expiresAt,
   376→      createdAt: now,
   377→      updatedAt: now,
   378→    });
   379→  },
   380→});

   562→/**
   563→ * deletePortalInviteAndCleanup — Permanently deletes a portal invite AND resets
   564→ * the associated portal user account so the same email can be re-invited cleanly.
   565→ *
   566→ * This prevents the "invitation already accepted" error when an admin:
   567→ * 1. Deletes a portal access record
   568→ * 2. Creates a new invite for the same email
   569→ *
   570→ * Steps:
   571→ * 1. Delete the portal_invites record
   572→ * 2. Find the user with the matching email + Client/Tenant role
   573→ * 3. Reset their role, verification, and password so they're treated as a brand-new invitee
   574→ */
   575→export const deletePortalInviteAndCleanup = mutation({
   576→  args: {
   577→    inviteId: v.id("portal_invites"),
   578→    inviteeEmail: v.optional(v.string()),
   579→  },
   580→  handler: async (ctx, args) => {
   581→    // 1. Delete the specified invite record
   582→    await ctx.db.delete(args.inviteId);
   583→
   584→    // 2. Also delete ANY other invite records for the same email
   585→    //    to prevent residual "already accepted" errors on re-invite
   586→    const email = (args.inviteeEmail || "").toLowerCase().trim();
   587→    if (!email) return;
   588→
   589→    const otherInvites = await ctx.db
   590→      .query("portal_invites")
   591→      .withIndex("by_email", (q) => q.eq("inviteeEmail", email))
   592→      .collect();
   593→
   594→    for (const inv of otherInvites) {
   595→      if (String(inv._id) === String(args.inviteId)) continue; // Already deleted above
   596→      try { await ctx.db.delete(inv._id); } catch (e) { /* ignore */ }
   597→    }
   598→
   599→    // 3. Find and reset the associated portal user
   600→    const existingUser = await ctx.db
   601→      .query("users")
   602→      .withIndex("by_token", (q) => q.eq("tokenIdentifier", email))
   603→      .first();
   604→
   605→    if (!existingUser) return;
   606→
   607→    // Only reset if the user has a Client or Tenant role (portal user)
   608→    const role = (existingUser as any).role;
   609→    if (role === "Client" || role === "Tenant" || role === "Portal User") {
   610→      // Reset the user so they can be re-invited cleanly
   611→      // Set isVerified to false so they can't log in with old credentials
   612→      // Remove the password so setup-password creates a fresh one
   613→      // Clear firmId so it will be set fresh on the next invite acceptance
   614→      await ctx.db.patch(existingUser._id, {
   615→        isVerified: false,
   616→        emailVerified: false,
   617→        password: undefined,
   618→        role: "Pending",
   619→        firmId: undefined,
   620→      } as any);
   621→    }
   622→  },
   623→});

   673→/**
   674→ * verifyInviteToken — validates a portal invite token and returns
   675→ * the invite details + whether the user already has an account.
   676→ * Used by the /setup-password page to determine what to show.
   677→ *
   678→ * ROBUSTNESS FIX: If the invite is "accepted" but the user's role is
   679→ * "Pending" (meaning their access was deleted via deletePortalInviteAndCleanup
   680→ * and they need to re-setup), we allow them to proceed. This prevents the
   681→ * "invitation already accepted" dead-end when a user is re-invited after
   682→ * their portal access was removed.
   683→ */
   684→export const verifyInviteToken = query({
   685→  args: { token: v.string() },
   686→  handler: async (ctx, args) => {
   687→    const results = await ctx.db
   688→      .query("portal_invites")
   689→      .withIndex("by_token", (q) => q.eq("token", args.token))
   690→      .collect();
   691→    const invite = results[0] || null;
   692→
   693→    if (!invite) {
   694→      return { valid: false, reason: "not_found" } as const;
   695→    }
   696→
   697→    // Check expiry (note: we don't patch status here since this is a query/read-only)
   698→    if (invite.expiresAt < Date.now()) {
   699→      return { valid: false, reason: "expired" } as const;
   700→    }
   701→
   702→    if (invite.status === "revoked") {
   703→      return { valid: false, reason: "revoked" } as const;
   704→    }
   705→
   706→    // Check if the user exists and their current state
   707→    const email = invite.inviteeEmail?.toLowerCase().trim();
   708→    let existingUser: any = null;
   709→    if (email) {
   710→      existingUser = await ctx.db
   711→        .query("users")
   712→        .withIndex("by_token", (q) => q.eq("tokenIdentifier", email))
   713→        .first();
   714→      // Case-insensitive fallback
   715→      if (!existingUser) {
   716→        existingUser = await ctx.db
   717→          .query("users")
   718→          .withIndex("by_token", (q) => q.eq("tokenIdentifier", email.toLowerCase()))
   719→          .first();
   720→      }
   721→    }
   722→
   723→    if (invite.status === "accepted") {
   724→      // ROBUSTNESS: If the user's role is Pending (their access was reset by
   725→      // deletePortalInviteAndCleanup), allow them to re-accept. This invite
   726→      // might be a stale record from before the cleanup ran. The user needs
   727→      // to set up their password again, so we should let them through.
   728→      const userRole = existingUser?.role;
   729→      const isUserUnverified = existingUser && !existingUser.isVerified;
   730→      if (userRole === "Pending" || isUserUnverified) {
   731→        // User was reset — allow re-accepting this invite
   732→        // Fall through to return valid: true
   733→      } else {
   734→        return { valid: false, reason: "already_accepted" } as const;
   735→      }
   736→    }
   737→
   738→    // Token is valid and pending — check if user already has an account
   739→    return {
   740→      valid: true,
   741→      invite: {
   742→        _id: invite._id,
   743→        inviteeEmail: invite.inviteeEmail,
   744→        inviteeName: invite.inviteeName,
   745→        portalType: invite.portalType,
   746→        firmId: invite.firmId,
   747→        relatedId: invite.relatedId,
   748→        expiresAt: invite.expiresAt,
   749→      },
   750→      hasAccount: !!existingUser,
   751→      hasPassword: !!(existingUser as any)?.password,
   752→    } as const;
   753→  },
   754→});

   791→// ─── Setup Portal Password (Invite Acceptance) ──────────────────────────
   792→
   793→/**
   794→ * setupPortalPassword — ACTION that completes the invite flow:
   795→ * 1. Validates the invite token (must be pending + unexpired)
   796→ * 2. If user exists → sets their password + marks verified
   797→ * 3. If user doesn't exist → creates a new user record with the password
   798→ * 4. Marks the invite as accepted
   799→ * 5. Clears the token to prevent replay attacks
   800→ */
   801→export const setupPortalPassword = action({
   802→  args: {
   803→    token: v.string(),
   804→    password: v.string(),
   805→    name: v.optional(v.string()),
   806→  },
   807→  handler: async (ctx, args): Promise<{ success: boolean; message?: string; email?: string }> => {
   808→    // 1. Validate the invite
   809→    const invite = await ctx.runQuery(api.portals.getInviteByToken, { token: args.token });
   810→    if (!invite) return { success: false, message: "Invalid invitation link." };
   811→    if (invite.status === "revoked") return { success: false, message: "This invitation has been revoked." };
   812→    if (invite.status === "expired" || invite.expiresAt < Date.now()) {
   813→      await ctx.runMutation(api.portals.updateInviteRecord, {
   814→        inviteId: invite._id,
   815→        updates: { status: "expired", updatedAt: Date.now() },
   816→      });
   817→      return { success: false, message: "This invitation has expired. Please request a new one." };
   818→    }
   819→
   820→    // ROBUSTNESS: If the invite was already accepted but the user's account was
   821→    // reset (role=Pending, isVerified=false), allow them to re-accept. This can
   822→    // happen when deletePortalInviteAndCleanup resets the user but a stale invite
   823→    // record still exists with status "accepted" and a valid token.
   824→    if (invite.status === "accepted") {
   825→      const emailCheck = (invite.inviteeEmail || "").toLowerCase().trim();
   826→      const existingUserCheck: any = await ctx.runQuery(api.myFunctions.getUser, { tokenIdentifier: emailCheck });
   827→      const wasReset = existingUserCheck?.role === "Pending" || (existingUserCheck && !existingUserCheck.isVerified);
   828→      if (!wasReset) {
   829→        return { success: false, message: "This invitation has already been used." };
   830→      }
   831→      // User was reset — allow re-accepting this invite (fall through)
   832→    }
   833→
   834→    const email = (invite.inviteeEmail || "").toLowerCase().trim();
   835→    if (!email) return { success: false, message: "This invitation has no email address associated." };
   836→
   837→    // 1.5. For resident invites, resolve the canonical tenant name from the
   838→    // property/tenancy record. The property record is the source of truth for
   839→    // a tenant's name — NOT the invite. This ensures the user's name always
   840→    // matches the property record.
   841→    let canonicalTenantName: string | null = null;
   842→    if (invite.portalType === "resident" && invite.relatedId) {
   843→      try {
   844→        const parts = invite.relatedId.split("_");
   845→        const propertyCustomId = parts[0];
   846→        const unitId = parts.length > 1 ? parts.slice(1).join("_") : null;
   847→
   848→        const property: any = await ctx.runQuery(internal.portals.findPropertyByCustomId, {
   849→          customId: propertyCustomId,
   850→        });
   851→
   852→        if (property) {
   853→          const units = property.units || [];
   854→          if (units.length > 0 && unitId) {
   855→            const unit = units.find((u: any) =>
   856→              u.id === unitId || u.unitName === unitId || u.id === invite.relatedId
   857→            );
   858→            if (unit?.tenantName) {
   859→              canonicalTenantName = unit.tenantName;
   860→            }
   861→          } else if (units.length > 0) {
   862→            if (units[0]?.tenantName) {
   863→              canonicalTenantName = units[0].tenantName;
   864→            }
   865→          } else {
   866→            const propTenantName = (property as any).rentalDetails?.tenantName || (property as any).tenantName;
   867→            if (propTenantName) {
   868→              canonicalTenantName = propTenantName;
   869→            }
   870→          }
   871→        }
   872→      } catch (e) {
   873→        // Non-blocking: use invite name as fallback
   874→      }
   875→    }
   876→
   877→    // 2. Hash the password
   878→    const hashedPassword = await ctx.runAction(internal.authUtils.hashPassword, { password: args.password });
   879→
   880→    // 3. Check if user exists
   881→    const existingUser: any = await ctx.runQuery(api.myFunctions.getUser, { tokenIdentifier: email });
   882→
   883→    // Track the portal user's Convex _id so we can link them to their property/unit
   884→    let portalUserDocId: string | null = null;
   885→
   886→    if (existingUser) {
   887→      // Update existing user: set password + mark verified
   888→      // Also upgrade their role to Client/Tenant if they were a "Portal User"
   889→      // or had no role assigned, so the frontend can route them correctly.
   890→      const portalRole = invite.portalType === "client" ? "Client" : "Tenant";
   891→      const needsRoleUpdate = !existingUser.role
   892→        || existingUser.role === "Portal User"
   893→        || existingUser.role === "Pending";
   894→      // Ensure firmId is always set — portal users need it for the ProductContext
   895→      // to derive the correct product and for data loading to work properly.
   896→      // When an existing user is re-invited after portal access deletion, their
   897→      // firmId might have been cleared or might never have been set.
   898→      const needsFirmId = !existingUser.firmId || existingUser.firmId !== invite.firmId;
   899→      portalUserDocId = String(existingUser._id);
   900→      await ctx.runMutation(internal.myFunctions.updateUserSecurityFields, {
   901→        userId: existingUser._id,
   902→        fields: {
   903→          password: hashedPassword,
   904→          isVerified: true,
   905→          emailVerified: true,
   906→          verificationCode: null,
   907→          // For tenant users, always sync name with the property record (source of truth)
   908→          ...(canonicalTenantName ? { name: canonicalTenantName } : (args.name && !existingUser.name ? { name: args.name } : {})),
   909→          ...(needsRoleUpdate ? { role: portalRole } : {}),
   910→          ...(needsFirmId ? { firmId: invite.firmId } : {}),
   911→          // Online presence privacy: property/resident portals default to hidden
   912→          portalPresenceHidden: invite.portalType === "resident" ? true : (existingUser as any).portalPresenceHidden ?? false,
   913→        },
   914→      });
   915→    } else {
   916→      // Create new user record for this portal user
   917→      // Use the proper UserRole (Client / Tenant) so the frontend can route
   918→      // them to the correct portal view automatically.
   919→      const portalProduct = invite.portalType === "client" ? "legal" : "property";
   920→      const portalRole = invite.portalType === "client" ? "Client" : "Tenant";
   921→      const userName = canonicalTenantName || args.name || invite.inviteeName || email.split("@")[0];
   922→      const newUserId = await ctx.runMutation(internal.myFunctions.createUser, {
   923→        tokenIdentifier: email,
   924→        name: userName,
   925→        email: email,
   926→        password: hashedPassword,
   927→        role: portalRole,
   928→        product: portalProduct,
   929→        onboardingCompleted: false,
   930→        isVerified: true,
   931→        emailVerified: true,
   932→        firmId: invite.firmId,
   933→        avatarUrl: `https://ui-avatars.com/api/?name=${encodeURIComponent(userName)}&background=random`,
   934→        // Online presence privacy: property/resident portals default to hidden
   935→        // (portal users can't see if the manager is online). Legal/client portals
   936→        // default to visible (clients can see if their lawyer is online).
   937→        portalPresenceHidden: invite.portalType === "resident",
   938→      });
   939→      portalUserDocId = String(newUserId);
   940→    }
   941→
   942→    // 3.5. Link portal user to property/unit if relatedId was provided
   943→    // This is critical for getTenantInfo to find the tenant's property assignment.
   944→    // When setupPortalPassword creates a new user, the property's unit record
   945→    // may still point to a stale contact ID or email instead of the new Convex _id.
   946→    if (invite.relatedId && invite.portalType === "resident" && portalUserDocId) {
   947→      try {
   948→        // Parse the relatedId format:
   949→        // "propertyId_unitId" for multi-unit properties (composite key)
   950→        // just "propertyId" for single properties
   951→        const parts = invite.relatedId.split("_");
   952→        const propertyCustomId = parts[0];
   953→        const unitId = parts.length > 1 ? parts.slice(1).join("_") : null;
   954→
   955→        // Find the property using internal query (actions cannot use ctx.db directly)
   956→        const property: any = await ctx.runQuery(internal.portals.findPropertyByCustomId, {
   957→          customId: propertyCustomId,
   958→        });
   959→
   960→        if (property) {
   961→          const units = property.units || [];
   962→          // Use the canonical tenant name from the property record (source of truth)
   963→          // Fall back to the portal user's name or invite data if property lookup failed
   964→          const tenantName = canonicalTenantName || invite.inviteeName || email.split("@")[0];
   965→
   966→          if (units.length > 0 && unitId) {
   967→            // Multi-unit property: update the specific unit
   968→            let unitFound = false;
   969→            const updatedUnits = units.map((u: any) => {
   970→              if (u.id === unitId || u.unitName === unitId || u.id === invite.relatedId) {
   971→                unitFound = true;
   972→                return {
   973→                  ...u,
   974→                  currentTenantId: portalUserDocId,
   975→                  tenantId: portalUserDocId,
   976→                  tenantEmail: email,
   977→                  tenantName: tenantName,
   978→                };
   979→              }
   980→              return u;
   981→            });
   982→            if (unitFound) {
   983→              await ctx.runMutation(internal.portals.linkPortalUserToProperty, {
   984→                propertyId: property._id,
   985→                updates: { units: updatedUnits },
   986→              });
   987→            }
   988→          } else {
   989→            // Single property (no units or no unitId): update property-level tenant
   990→            await ctx.runMutation(internal.portals.linkPortalUserToProperty, {
   991→              propertyId: property._id,
   992→              updates: {
   993→                currentTenantId: portalUserDocId,
   994→                tenantId: portalUserDocId,
   995→              },
   996→            });
   997→          }
   998→        }
   999→      } catch (linkErr) {
  1000→        // Non-blocking: if linking fails, the user can still access the portal.
  1001→        // getTenantInfo will try to find them by email as a fallback.
  1002→        console.warn("Portal user-property linking failed:", linkErr);
  1003→      }
  1004→    }
  1005→
  1006→    // 4. Mark invite as accepted (including terms acceptance timestamp)
  1007→    await ctx.runMutation(api.portals.updateInviteRecord, {
  1008→      inviteId: invite._id,
  1009→      updates: {
  1010→        status: "accepted",
  1011→        acceptedAt: Date.now(),
  1012→        termsAcceptedAt: Date.now(),
  1013→        updatedAt: Date.now(),
  1014→        // Clear token to prevent replay attacks — the invite is now consumed
  1015→        token: undefined,
  1016→      },
  1017→    });
  1018→
  1019→    return { success: true, email };
  1020→  },
  1021→});

  1084→/**
  1085→ * getTenantInfo — Resolves the tenant's property/unit assignments using their
  1086→ * user ID and email. This is critical because the tenantId stored in
  1087→ * properties/units may be the user's Convex _id OR their email, depending
  1088→ * on how the invite was created. This query normalizes both cases.
  1089→ *
  1090→ * Returns: { tenantId, properties, units } where tenantId is the canonical
  1091→ * ID that should be used for all other portal queries.
  1092→ */
  1093→export const getTenantInfo = query({
  1094→  args: { firmId: v.string(), userId: v.string(), email: v.string() },
  1095→  handler: async (ctx, args) => {
  1096→    const properties = await ctx.db
  1097→      .query("properties")
  1098→      .withIndex("by_firm", (q) => q.eq("firmId", args.firmId))
  1099→      .collect();
  1100→
  1101→    // Search for tenant in properties using both userId and email
  1102→    const tenantProperties: any[] = [];
  1103→    const tenantUnits: any[] = [];
  1104→    let resolvedTenantId = args.userId;
  1105→
  1106→    for (const prop of properties) {
  1107→      // Check property-level tenant (legacy single-tenant model)
  1108→      const propTenantId = (prop as any).currentTenantId || (prop as any).tenantId;
  1109→      if (propTenantId === args.userId || propTenantId === args.email ||
  1110→          String(propTenantId).toLowerCase() === args.email.toLowerCase()) {
  1111→        tenantProperties.push({
  1112→          id: String(prop._id),
  1113→          name: (prop as any).name || prop.address || 'Unnamed Property',
  1114→          address: prop.address,
  1115→          tenantName: (prop as any).rentalDetails?.tenantName || (prop as any).tenantName || null,
  1116→        });
  1117→        if (propTenantId && propTenantId !== args.userId) {
  1118→          resolvedTenantId = propTenantId;
  1119→        }
  1120→      }
  1121→
  1122→      // Check unit-level tenants (multi-unit model)
  1123→      const units = (prop as any).units || [];
  1124→      for (const unit of units) {
  1125→        const unitTenantId = unit.currentTenantId || unit.tenantId;
  1126→        const unitTenantEmail = (unit.tenantEmail || '').toLowerCase();
  1127→        const matchesUserId = unitTenantId === args.userId;
  1128→        const matchesEmail = unitTenantId === args.email ||
  1129→            String(unitTenantId).toLowerCase() === args.email.toLowerCase() ||
  1130→            (unitTenantEmail && unitTenantEmail === args.email.toLowerCase());
  1131→
  1132→        if (matchesUserId || matchesEmail) {
  1133→          tenantUnits.push({
  1134→            id: unit.id || unit._id,
  1135→            name: unit.name || unit.unitName || unit.label,
  1136→            unitName: unit.name || unit.unitName || unit.label,
  1137→            propertyId: String(prop._id),
  1138→            propertyName: (prop as any).name || prop.address || 'Unnamed Property',
  1139→            propertyAddress: prop.address,
  1140→            amenities: unit.amenities || [],
  1141→            tenantName: unit.tenantName || null,
  1142→          });
  1143→          if (unitTenantId && unitTenantId !== args.userId) {
  1144→            resolvedTenantId = unitTenantId;
  1145→          }
  1146→        }
  1147→      }
  1148→    }
  1149→
  1150→    // Determine primary property/unit for this tenant (first match)
  1151→    const primaryUnit = tenantUnits.length > 0 ? tenantUnits[0] : null;
  1152→    const primaryProperty = primaryUnit
  1153→      ? { id: primaryUnit.propertyId, name: primaryUnit.propertyName, address: primaryUnit.propertyAddress }
  1154→      : tenantProperties.length > 0 ? tenantProperties[0] : null;
  1155→
  1156→    // Resolve the canonical tenant name from the property/tenancy record
  1157→    // This is the source of truth — used by the portal to display the tenant's name
  1158→    const resolvedTenantName = primaryUnit?.tenantName
  1159→      || tenantProperties.find(p => p.tenantName)?.tenantName
  1160→      || null;
  1161→
  1162→    return {
  1163→      tenantId: resolvedTenantId,
  1164→      properties: tenantProperties,
  1165→      units: tenantUnits,
  1166→      // Convenience fields for maintenance ticket creation
  1167→      primaryPropertyId: primaryProperty?.id || null,
  1168→      primaryUnitId: primaryUnit?.id || null,
  1169→      primaryPropertyName: primaryProperty?.name || null,
  1170→      primaryUnitName: primaryUnit?.unitName || null,
  1171→      primaryPropertyAddress: primaryProperty?.address || null,
  1172→      // Canonical tenant name from the property record (source of truth)
  1173→      tenantName: resolvedTenantName,
  1174→    };
  1175→  },
  1176→});

  1450→ *
  1451→ * Also searches by email as a fallback — the contact's userId field might
  1452→ * not always match the user's Convex _id (e.g. if the contact was created
  1453→ * before the portal user account existed).
  1454→ */
  1455→export const getClientContactByUserId = query({
  1456→  args: { firmId: v.string(), userId: v.string() },
  1457→  handler: async (ctx, args) => {
  1458→    const contacts = await ctx.db
  1459→      .query("contacts")
  1460→      .withIndex("by_firm", (q) => q.eq("firmId", args.firmId))
  1461→      .collect();
  1462→
  1463→    // First try matching by userId field
  1464→    let contact = contacts.find(c => c.userId === args.userId) || null;
  1465→
  1466→    // If not found, try to match by looking up the user's email and matching
  1467→    // the contact's email field. This handles the case where the contact was
  1468→    // created before the portal user account existed.
  1469→    if (!contact) {
  1470→      // Try to look up the user by their Convex _id to get their email
  1471→      let user: any = null;
  1472→      try { user = await ctx.db.get(args.userId as any); } catch {}
  1473→
  1474→      // If that didn't work, the userId might be an email-based tokenIdentifier
  1475→      if (!user) {
  1476→        user = await ctx.db
  1477→          .query("users")
  1478→          .withIndex("by_token", (q) => q.eq("tokenIdentifier", args.userId.toString()))
  1479→          .first();
  1480→      }
  1481→
  1482→      const userEmail = user?.email?.toLowerCase();
  1483→      if (userEmail) {
  1484→        contact = contacts.find(c =>
  1485→          (c as any).email?.toLowerCase() === userEmail
  1486→        ) || null;
  1487→      }
  1488→
  1489→      // Also try matching contact by userId against the user's _id as a string
  1490→      if (!contact && user) {
  1491→        const userDocId = String(user._id);
  1492→        contact = contacts.find(c => c.userId === userDocId) || null;
  1493→      }
  1494→    }
  1495→
  1496→    return contact;
  1497→  },
  1498→});

  1500→/**
  1501→ * getClientMattersByUserId — Returns the matters for a portal client,
  1502→ * looking up the contact by userId first. This avoids the ClientDashboard
  1503→ * needing to depend on matterState (which is empty for portal users).
  1504→ *
  1505→ * Also searches by email as a fallback for the same reason as
  1506→ * getClientContactByUserId.
  1507→ */
  1508→export const getClientMattersByUserId = query({
  1509→  args: { firmId: v.string(), userId: v.string() },
  1510→  handler: async (ctx, args) => {
  1511→    // Find the contact for this user
  1512→    const contacts = await ctx.db
  1513→      .query("contacts")
  1514→      .withIndex("by_firm", (q) => q.eq("firmId", args.firmId))
  1515→      .collect();
  1516→
  1517→    let clientContact = contacts.find(c => c.userId === args.userId);
  1518→
  1519→    // Fallback: search by email
  1520→    if (!clientContact) {
  1521→      let user: any = null;
  1522→      try { user = await ctx.db.get(args.userId as any); } catch {}
  1523→      if (!user) {
  1524→        user = await ctx.db
  1525→          .query("users")
  1526→          .withIndex("by_token", (q) => q.eq("tokenIdentifier", args.userId.toString()))
  1527→          .first();
  1528→      }
  1529→      const userEmail = user?.email?.toLowerCase();
  1530→      if (userEmail) {
  1531→        clientContact = contacts.find(c =>
  1532→          (c as any).email?.toLowerCase() === userEmail
  1533→        );
  1534→      }
  1535→      if (!clientContact && user) {
  1536→        clientContact = contacts.find(c => c.userId === String(user._id));
  1537→      }
  1538→    }
  1539→
  1540→    if (!clientContact) return [];
  1541→
  1542→    // Find matters for this contact
  1543→    const matters = await ctx.db
  1544→      .query("matters")
  1545→      .withIndex("by_firm", (q) => q.eq("firmId", args.firmId))
  1546→      .collect();
  1547→
  1548→    return matters
  1549→      .filter(m => m.clientId === String(clientContact._id))
  1550→      .map(m => ({
  1551→        id: String(m._id),
  1552→        title: m.title,
  1553→        suitNumber: m.suitNumber,
  1554→        referenceNumber: m.referenceNumber,
  1555→        stage: m.stage,
  1556→        status: m.status,
  1557→        type: m.type,
  1558→        nextAdjournedDate: m.nextAdjournedDate,
  1559→        stageLastUpdated: m.stageLastUpdated,
  1560→        assignedUsers: m.assignedUsers,
  1561→        clientId: m.clientId,
  1562→      }));
  1563→  },
  1564→});
