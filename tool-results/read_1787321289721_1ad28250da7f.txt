    24→// ─── AccordionSection (MODULE-LEVEL — outside MatterForm) ──────────
    25→// CRITICAL: This component MUST be defined outside the component render
    26→// function. When defined inside (as a closure), React treats it as a new
    27→// component type on every render, causing all children to unmount/remount
    28→// on every keystroke — which causes input focus loss.
    29→// By defining it at module level with React.memo, the component identity
    30→// is stable across re-renders, preserving DOM focus.
    31→// (Mirrors the pattern used in PropertyForm.tsx and SmartMatterModal.tsx.)
    32→interface AccordionSectionProps {
    33→    id: string;
    34→    title: string;
    35→    subtitle: string;
    36→    icon: React.ReactNode;
    37→    iconBg: string;
    38→    children: React.ReactNode;
    39→    isOpen: boolean;
    40→    onToggle: (id: string) => void;
    41→    badge?: React.ReactNode;
    42→    /** Optional right-side accessory rendered inline with the chevron (e.g. a toggle switch). */
    43→    accessory?: React.ReactNode;
    44→    /** When true, clicking the header does NOT toggle — used when the header has its own toggle (e.g. litigation). */
    45→    disableHeaderToggle?: boolean;
    46→}
    47→const AccordionSectionInner: React.FC<AccordionSectionProps> = ({ id, title, subtitle, icon, iconBg, children, isOpen, onToggle, badge, accessory, disableHeaderToggle }) => {
    48→    const headerRef = useRef<HTMLButtonElement>(null);
    49→    return (
    50→        <div
    51→            className={`rounded-lg border shadow-sm overflow-hidden ${isOpen ? 'bg-white dark:bg-zinc-800 border-slate-200 dark:border-zinc-700' : 'bg-slate-50/50 dark:bg-zinc-800/30 border-slate-100 dark:border-zinc-700/50'}`}
    52→            style={{ willChange: 'height', contain: 'layout style' }}
    53→        >
    54→            <div
    55→                className="w-full flex items-center gap-4 p-3 sm:p-4 hover:bg-slate-100/50 dark:hover:bg-zinc-700/30 transition-colors"
    56→            >
    57→                <button
    58→                    ref={headerRef}
    59→                    type="button"
    60→                    tabIndex={0}
    61→                    onClick={() => {
    62→                        if (disableHeaderToggle) return;
    63→                        onToggle(id);
    64→                        if (!isOpen) {
    65→                            setTimeout(() => {
    66→                                headerRef.current?.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
    67→                            }, 50);
    68→                        }
    69→                    }}
    70→                    onKeyDown={(e) => {
    71→                        if (disableHeaderToggle) return;
    72→                        if (e.key === 'Enter' || e.key === ' ') {
    73→                            e.preventDefault();
    74→                            onToggle(id);
    75→                        }
    76→                    }}
    77→                    className={`flex items-center gap-4 flex-1 min-w-0 text-left focus:outline-none ${disableHeaderToggle ? 'cursor-default' : 'focus:ring-2 focus:ring-primary-500/30'}`}
    78→                >
    79→                    <div className={`p-1.5 ${iconBg} text-white rounded-lg shadow-sm flex-shrink-0`}>
    80→                        {icon}
    81→                    </div>
    82→                    <div className="flex-1 min-w-0">
    83→                        <p className="text-2xs font-bold text-slate-600/70 dark:text-zinc-400 uppercase tracking-widest leading-none mb-0.5">{subtitle}</p>
    84→                        <h3 className="text-base font-black text-slate-800 dark:text-white tracking-tight">{title}</h3>
    85→                    </div>
    86→                    {badge}
    87→                </button>
    88→                {accessory}
    89→                {!disableHeaderToggle && (
    90→                    <svg className={`w-4 h-4 text-slate-400 transition-transform duration-200 flex-shrink-0 ${isOpen ? 'rotate-180' : ''}`} fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}>
    91→                        <path strokeLinecap="round" strokeLinejoin="round" d="M19 9l-7 7-7-7" />
    92→                    </svg>
    93→                )}
    94→            </div>
    95→            {isOpen && (
    96→                <div className="p-3 sm:p-4 pt-0 space-y-2 sm:space-y-3">
    97→                    {children}
    98→                </div>
    99→            )}
   100→        </div>
   101→    );
   102→};
