     1→     1→
     2→     2→import React, { useState, useMemo } from 'react';
     3→     3→import { useMutation } from "convex/react";
     4→     4→import { api } from "../../../convex/_generated/api";
     5→     5→import { Property, Contact, InvoiceStatus, InvoiceLineItem, BankAccount, RentPayment } from '../../types';
     6→     6→import { useCoreState } from '../../contexts/CoreContext';
     7→     7→import { useDataActions } from '../../contexts/DataContext';
     8→     8→import { useUI } from '../../contexts/UIContext';
     9→     9→import { useMatterState } from '../../contexts/MatterContext';
    10→    10→import { useFinanceState } from '../../contexts/FinanceContext';

    22→    22→import NairaSymbol from '../NairaSymbol';
    23→    23→import { formatNaira, formatDateWithOrdinal } from '../../utils/formatting';

    25→    25→import { generateReceiptPdf } from '../../services/reportGenerator';
    26→    26→
    27→    27→interface CollectRentModalProps {
    28→    28→  property: Property;
    29→    29→  onClose: () => void;
    30→    30→}
    31→    31→
    32→    32→/**
    33→    33→ * CollectRentModal — Accounting Principles
    34→    34→ * 
    35→    35→ * 1. RECEIPTS ARE FULL: The receipt issued to the tenant always reflects the full
    36→    36→ *  amount paid. No deductions are shown on the receipt. This is the standard
    37→    37→ *  practice for proper accounting and tax compliance.
    38→    38→ * 
    39→    39→ * 2. MANAGEMENT FEES ARE INVOICED: The agent/manager's fee is a separate obligation.
    40→    40→ *  When rent is collected, an invoice is auto-generated for the management fee,
    41→    41→ *  tied back to the rent receipt. The landlord owes this to the agent — it is NOT
    42→    42→ *  deducted from source on the receipt.
    43→    43→ * 
    44→    44→ * 3. FLEXIBLE ACCOUNTING: The "Net to Client" view is shown as an informational
    45→    45→ *  breakdown (what the landlord keeps after paying the agent), not as a receipt line.
    46→    46→ *  This accommodates different arrangements:
    47→    47→ *  - Landlord collects in full, pays agent separately
    48→    48→ *  - Agent collects and remits net (internal tracking only)
    49→    49→ * 
    50→    50→ * 4. LEDGER INTEGRITY: The ledger records the full rent amount. The management fee
    51→    51→ *  invoice creates a separate payable entry. Both are tied via a shared transactionRef.
    52→    52→ */
    53→    53→const CollectRentModal: React.FC<CollectRentModalProps> = ({ property, onClose }) => {
    54→    54→  const { coreState } = useCoreState();
    55→    55→  const { matterState } = useMatterState();
    56→    56→  const { financeState } = useFinanceState();
    57→    57→  const { updateItem, handleGenerateInvoice, logActivity, addItem } = useDataActions();
    58→    58→  const { addToast, modalContext } = useUI();
    59→    59→  const addLedgerEntry = useMutation(api.sentry.addLedgerEntry);
    60→    60→
    61→    61→  // Unit-specific overrides from context (if opened from a specific unit)
    62→    62→  const overrideUnitId = modalContext?.unitId;
    63→    63→  const overrideUnitName = modalContext?.unitName;
    64→    64→  const overrideTenantName = modalContext?.tenantName;
    65→    65→  const overrideRentAmount = modalContext?.rentAmount;
    66→    66→
    67→    67→  // Find the owner/contact
    68→    68→  const owner = matterState.contacts.find(c => 
    69→    69→    c.id === property.contactId || (c.properties || []).some(p => p.id === property.id)
    70→    70→  );
    71→    71→
    72→    72→  // Get last payment
    73→    73→  const sortedHistory = [...(property.rentPaymentHistory || [])].sort((a, b) => {
    74→    74→    const dateA = a.paidDate ? new Date(a.paidDate).getTime() : 0;
    75→    75→    const dateB = b.paidDate ? new Date(b.paidDate).getTime() : 0;
    76→    76→    return dateB - dateA;
    77→    77→  });
    78→    78→  const lastPayment = sortedHistory[0];
    79→    79→
    80→    80→  // FIX: removed property.value and lastPayment?.amount from the fallback
    81→    81→  // chain. Falling back to property.value (the property valuation) is
    82→    82→  // semantically wrong — if rent was never configured but the property is
    83→    83→  // valued at ₦50M, the rent-collection modal would open pre-filled with
    84→    84→  // ₦50M. lastPayment?.amount conflates "what was paid last time" with
    85→    85→  // "what is owed this time." Now falls back to 0 if no rent is configured.
    86→    86→  const initialAmount = overrideRentAmount || property.rentalDetails?.rentAmount || 0;
    87→    87→  const [amountValue, setAmountValue] = useState(initialAmount);
    88→    88→  const [displayAmount, setDisplayAmount] = useState(initialAmount.toLocaleString());
    89→    89→  const [paymentDate, setPaymentDate] = useState(new Date().toISOString().split('T')[0]);
    90→    90→  const [paymentMethod, setPaymentMethod] = useState('Bank Transfer');
    91→    91→  const [periodStart, setPeriodStart] = useState(new Date().toISOString().split('T')[0]);
    92→    92→  const [periodEnd, setPeriodEnd] = useState('');
    93→    93→  const [isProcessing, setIsProcessing] = useState(false);
    94→    94→
    95→    95→  const handleAmountChange = (val: string) => {
    96→    96→    // Remove commas and non-numeric chars except decimal
    97→    97→    const numeric = val.replace(/,/g, '').replace(/[^0-9.]/g, '');
    98→    98→    const num = parseFloat(numeric) || 0;
    99→    99→    setAmountValue(num);
   100→   100→    
   101→   101→    // Format with commas for display
   102→   102→    if (numeric === '') {
   103→   103→      setDisplayAmount('');
   104→   104→    } else {
   105→   105→      setDisplayAmount(num.toLocaleString());
   106→   106→    }
   107→   107→  };
   108→   108→
   109→   109→  // Fee Calculation — informational only, NOT deducted from receipt
   110→   110→  const feePercentage = property.managementFeePercentage || 0;
   111→   111→  const feeAmount = useMemo(() => (amountValue * feePercentage) / 100, [amountValue, feePercentage]);
   112→   112→  const tenantName = overrideTenantName || property.rentalDetails?.tenantName;
   113→   113→  // Build display label — allow wrapping instead of truncating
   114→   114→  const unitDisplayLabel = overrideUnitName ? `${property.address} - ${overrideUnitName}` : property.address;
   115→   115→
   116→   116→  const handleCollect = async () => {
   117→   117→    if (!owner) {
   118→   118→      addToast("Owner not found for this property.", { type: 'error' });
   119→   119→      return;
   120→   120→    }
   121→   121→
   122→   122→    setIsProcessing(true);
   123→   123→    try {
   124→   124→      // Generate a shared transaction reference to link receipt, invoice, and ledger
   125→   125→      const transactionRef = `TXN-${Date.now().toString().slice(-8)}`;
   126→   126→
   127→   127→      // 1. Update Property Rent History — receipt records FULL amount
   128→   128→      const newPayment = {
   129→   129→        id: uuidv4(),
   130→   130→        dueDate: paymentDate,
   131→   131→        paidDate: paymentDate,
   132→   132→        amount: amountValue, // FULL amount — receipt is always for what was paid
   133→   133→        status: 'paid' as const,
   134→   134→        paymentMethod,
   135→   135→        receiptNumber: `REC-${Date.now().toString().slice(-6)}-${Math.floor(1000 + Math.random() * 9000)}`,
   136→   136→        periodStart,
   137→   137→        periodEnd,
   138→   138→        transactionRef,
   139→   139→      };
   140→   140→
   141→   141→      const updatedHistory = [...(property.rentPaymentHistory || []), newPayment];
   142→   142→      
   143→   143→      // Update standalone property record
   144→   144→      await updateItem('properties', { id: property.id, rentPaymentHistory: updatedHistory, status: 'Occupied' }, 'Property Payment');
   145→   145→
   146→   146→      // Legacy support: also update contact if it has the nested property
   147→   147→      if (owner.properties?.some(p => p.id === property.id)) {
   148→   148→        const updatedProperties = owner.properties.map(p => 
   149→   149→          p.id === property.id ? { ...p, rentPaymentHistory: updatedHistory, status: 'Occupied' as const } : p
   150→   150→        );
   151→   151→        await updateItem('contacts', { id: owner.id, properties: updatedProperties }, 'Property Payment');
   152→   152→      }
   153→   153→
   154→   154→      // 2. Generate Invoice for Management Fee (if applicable)
   155→   155→      //  This is a SEPARATE invoice the landlord owes the agent — not deducted from the receipt.
   156→   156→      //  It is tied back to the rent transaction via transactionRef.
   157→   157→      if (feeAmount > 0 && feePercentage > 0) {
   158→   158→        const linkedMatter = matterState.matters.find(m => m.clientId === owner.id && m.title.includes(property.address));
   159→   159→
   160→   160→        const feeItem: InvoiceLineItem = {
   161→   161→          id: uuidv4(),
   162→   162→          description: `Management Fee — ${unitDisplayLabel} (${new Date(paymentDate).toLocaleString('default', { month: 'long', year: 'numeric' })})`,
   163→   163→          hours: 1,
   164→   164→          rate: feeAmount,
   165→   165→          total: feeAmount
   166→   166→        };
   167→   167→
   168→   168→        const defaultAccount: BankAccount = coreState.firmDetails?.bankAccounts?.[0] || {
   169→   169→          id: '1',
   170→   170→          bankName: 'PracticePro Default',
   171→   171→          accountName: coreState.firmDetails?.name || 'Firm',
   172→   172→          accountNumber: '0000000000'
   173→   173→        };
   174→   174→
   175→   175→        // Generate the management fee invoice — NOT marked as paid (landlord pays agent separately)
   176→   176→        await handleGenerateInvoice(
   177→   177→          linkedMatter || { id: property.id, title: `Property Management — ${property.address}`, clientId: owner.id } as any,
   178→   178→          [feeItem],
   179→   179→          { issueDate: paymentDate, dueDate: paymentDate }, // Due same day, but not auto-paid
   180→   180→          [], [],
   181→   181→          defaultAccount,
   182→   182→          { applicable: false }
   183→   183→        );
   184→   184→      }
   185→   185→
   186→   186→      // 3. Record in Atrium Ledger — full rent received
   187→   187→      const targetUnitId = overrideUnitId || property.id;
   188→   188→      const tenantContact = matterState.contacts.find(c => 
   189→   189→        (c.email && c.email === property.rentalDetails?.tenantEmail) || 
   190→   190→        (c.phone && c.phone === property.rentalDetails?.tenantPhone) ||
   191→   191→        (c.name && c.name === property.rentalDetails?.tenantName)
   192→   192→      );
   193→   193→
   194→   194→      if (targetUnitId) {
   195→   195→        await addLedgerEntry({
   196→   196→          firmId: property.firmId || coreState.firmDetails?.id || '',
   197→   197→          propertyId: property.id,
   198→   198→          unitId: targetUnitId,
   199→   199→          tenantId: tenantContact?.id || 'tenant-legacy', 
   200→   200→          amount: amountValue, // Full rent amount
   201→   201→          type: 'rent',
   202→   202→          status: 'cleared',
   203→   203→          description: `Rent collection for ${unitDisplayLabel}`,
   204→   204→          period: `${periodStart} to ${periodEnd}`,
   205→   205→          channel: transactionRef,
   206→   206→        });
   207→   207→
   208→   208→        // If there's a management fee, record it as a separate payable entry
   209→   209→        if (feeAmount > 0 && feePercentage > 0) {
   210→   210→          await addLedgerEntry({
   211→   211→            firmId: property.firmId || coreState.firmDetails?.id || '',
   212→   212→            propertyId: property.id,
   213→   213→            unitId: targetUnitId,
   214→   214→            tenantId: owner.id, // Payable TO the firm, FROM the landlord
   215→   215→            amount: feeAmount,
   216→   216→            type: 'management_fee',
   217→   217→            status: 'pending', // Not yet paid — landlord owes this
   218→   218→            description: `Management fee (${feePercentage}%) for ${unitDisplayLabel}`,
   219→   219→            period: `${periodStart} to ${periodEnd}`,
   220→   220→            channel: transactionRef, // Linked to the same transaction
   221→   221→          });
   222→   222→        }
   223→   223→      }
   224→   224→
   225→   225→      logActivity(
   226→   226→        `Collected rent for ${unitDisplayLabel}${feeAmount > 0 ? `. Mgmt fee invoice: ${formatNaira(feeAmount)}` : ''}`,
   227→   227→        'Contact',
   228→   228→        property.id,
   229→   229→        unitDisplayLabel
   230→   230→      );
   231→   231→
   232→   232→      // 4. Automatically trigger receipt generation — FULL amount, no deductions
   233→   233→      handleDownloadTenantReceipt(true);
   234→   234→
   235→   235→      // 5. DOCUMENT CROSS-PERSISTENCE — auto-save a document record
   236→   236→      // linking the receipt to both the property and any linked legal
   237→   237→      // matter. Previously, receipts were only downloadable as PDFs but
   238→   238→      // never persisted to the documents table, so they didn't appear
   239→   239→      // in the matter's Documents tab. Now they do.
   240→   240→      try {
   241→   241→        const linkedMatter = matterState.matters.find(m =>
   242→   242→          m.specialtyData?.realEstate?.propertyId === property.id ||
   243→   243→          m.title.includes(property.address.split(',')[0])
   244→   244→        );
   245→   245→        const receiptDocRecord = {
   246→   246→          firmId: property.firmId || coreState.firmDetails?.id || '',
   247→   247→          title: `Rent Receipt — ${unitDisplayLabel} (${new Date(paymentDate).toLocaleDateString('en-GB')})`,
   248→   248→          matter: linkedMatter ? { id: linkedMatter.id, title: linkedMatter.title } : undefined,
   249→   249→          matterId: linkedMatter?.id,
   250→   250→          propertyId: property.id,
   251→   251→          categoryId: 'receipts',
   252→   252→          dateFiled: paymentDate,
   253→   253→          assignedUsers: [],
   254→   254→          file: {
   255→   255→            name: `Receipt_${receiptNumber || `REC-${Date.now()}`}.pdf`,
   256→   256→            type: 'application/pdf',
   257→   257→            size: 0,
   258→   258→            dataUrl: '', // PDF is generated client-side; record serves as index
   259→   259→          },
   260→   260→          source: 'generated' as const,
   261→   261→          uploadedBy: 'system',
   262→   262→          createdAt: new Date().toISOString(),
   263→   263→        };
   264→   264→        await addItem('documents', receiptDocRecord, 'Rent Receipt');
   265→   265→      } catch (docError: any) {
   266→   266→        console.error('[CollectRentModal] Failed to persist receipt document record:', docError);
   267→   267→        // Non-blocking — receipt PDF was still generated and downloaded
   268→   268→      }
   269→   269→
   270→   270→      addToast(`Rent receipt issued for ${formatNaira(amountValue)}${feeAmount > 0 ? `. Management fee invoice of ${formatNaira(feeAmount)} generated.` : ''}`, { type: 'success' });
   271→   271→      onClose();
   272→   272→    } catch (error: any) {
   273→   273→      console.error("Failed to collect rent:", error);
   274→   274→      addToast(`Failed to process rent collection: ${error.message || 'Unknown error'}`, { type: 'error' });
   275→   275→    } finally {
   276→   276→      setIsProcessing(false);
   277→   277→    }
   278→   278→  };
   279→   279→
   280→   280→  const handleDownloadTenantReceipt = (isSilentOrPayment: boolean | RentPayment = false) => {
   281→   281→    const isSilent = typeof isSilentOrPayment === 'boolean' ? isSilentOrPayment : false;
   282→   282→    const payment = typeof isSilentOrPayment === 'object' ? isSilentOrPayment : null;
   283→   283→
   284→   284→    try {
   285→   285→
   286→   286→      
   287→   287→      // Receipt ALWAYS reflects the FULL amount paid — no deductions
   288→   288→      const receiptAmount = payment ? payment.amount : amountValue;
   289→   289→      const receiptDate = payment ? (payment.paidDate || payment.dueDate) : paymentDate;
   290→   290→      const receiptNumber = payment?.receiptNumber || generateReceiptNumber({
   291→   291→        firmName: coreState.firmDetails?.name,
   292→   292→        users: coreState.users || [],
   293→   293→        existingInvoiceCount: financeState.invoices?.length || 0,
   294→   294→      });
   295→   295→
   296→   296→      // Find actual tenant contact for the receipt
   297→   297→      const tenantContact = matterState.contacts.find(c => 
   298→   298→        (c.email && c.email === property.rentalDetails?.tenantEmail) || 
   299→   299→        (c.phone && c.phone === property.rentalDetails?.tenantPhone) ||
   300→   300→        (c.name && c.name === property.rentalDetails?.tenantName)
   301→   301→      );
   302→   302→
   303→   303→      const mockInvoice: any = {
   304→   304→        id: uuidv4(),
   305→   305→        invoiceNumber: receiptNumber,
   306→   306→        client: tenantContact || { id: 'tenant-legacy', name: tenantName || 'The Tenant' },
   307→   307→        matter: { id: property.id, title: `Rent Payment: ${unitDisplayLabel}` },
   308→   308→        lineItems: [{
   309→   309→          id: uuidv4(),
   310→   310→          description: `Rent for ${unitDisplayLabel}`,
   311→   311→          hours: 1,
   312→   312→          rate: receiptAmount,
   313→   313→          total: receiptAmount // FULL amount — receipt reflects what was actually paid
   314→   314→        }],
   315→   315→        status: InvoiceStatus.Paid,
   316→   316→        issueDate: receiptDate,
   317→   317→        dueDate: receiptDate,
   318→   318→        paidDate: receiptDate,
   319→   319→        paymentDetails: coreState.firmDetails?.bankAccounts?.[0] || { 
   320→   320→          bankName: 'PracticePro Default', 
   321→   321→          accountNumber: '0000000000',
   322→   322→          accountName: coreState.firmDetails?.name || 'Firm'
   323→   323→        },
   324→   324→        subTotal: receiptAmount,
   325→   325→        taxAmount: 0,
   326→   326→        total_amount: receiptAmount
   327→   327→      };
   328→   328→
   329→   329→      const pStart = payment ? payment.periodStart : periodStart;
   330→   330→      const pEnd = payment ? payment.periodEnd : periodEnd;
   331→   331→      
   332→   332→      const tenancyPeriod = (pStart && pEnd)
   333→   333→        ? `${formatDateWithOrdinal(pStart)} to ${formatDateWithOrdinal(pEnd)}`
   334→   334→        : property.rentalDetails?.leaseStart && property.rentalDetails?.leaseEnd 
   335→   335→          ? `${formatDateWithOrdinal(property.rentalDetails.leaseStart)} to ${formatDateWithOrdinal(property.rentalDetails.leaseEnd)}`
   336→   336→          : undefined;
   337→   337→
   338→   338→      generateReceiptPdf(mockInvoice, coreState.firmDetails, (tenantContact || { name: tenantName || 'The Tenant' }) as any, { tenancyPeriod });
   339→   339→      if (!isSilent) addToast("Tenant receipt generated.", { type: 'success' });
   340→   340→    } catch (error: any) {
   341→   341→      console.error("[CollectRentModal] Receipt generation failed:", error);
   342→   342→      if (!isSilent) addToast("Failed to generate receipt PDF.", { type: 'error' });
   343→   343→    }
   344→   344→  };
   345→   345→
   346→   346→  return (
   347→   347→    <div className="p-1 sm:p-4">
   348→   348→      <div className="mb-6 flex items-start gap-4">
   349→   349→        <div className="p-3 bg-green-100 dark:bg-green-900/40 rounded-2xl text-green-600 dark:text-green-400">
   350→   350→          <Receipt className="w-6 h-6" />
   351→   351→        </div>
   352→   352→        <div className="flex-1 min-w-0">
   353→   353→          <h3 className="text-xl font-bold text-slate-800 dark:text-zinc-100 tracking-tight">Issue Rent Receipt</h3>
   354→   354→          <p className="text-sm text-slate-500">Record rent collection for:</p>
   355→   355→          <p className="text-sm font-bold text-slate-700 dark:text-zinc-300 break-words">{unitDisplayLabel}</p>
   356→   356→        </div>
   357→   357→        {lastPayment && (
   358→   358→          <button
   359→   359→            onClick={() => handleDownloadTenantReceipt(lastPayment)}
   360→   360→            className="flex-shrink-0 px-3 py-2 text-xs font-bold bg-slate-100 dark:bg-zinc-800 hover:bg-slate-200 dark:hover:bg-zinc-700 text-slate-700 dark:text-zinc-300 rounded-lg flex items-center gap-2 border border-slate-200 dark:border-zinc-700"
   361→   361→            title={lastPayment.paidDate ? `Last paid: ${new Date(lastPayment.paidDate).toLocaleDateString('en-GB')} for ${formatNaira(lastPayment.amount)}` : undefined}
   362→   362→          >
   363→   363→            <DownloadIcon className="w-4 h-4" /> Download Last Receipt
   364→   364→          </button>
   365→   365→        )}
   366→   366→      </div>
   367→   367→
   368→   368→      <div className="grid grid-cols-1 md:grid-cols-2 gap-6 mb-8">
   369→   369→        {/* Left: Input Form */}
   370→   370→        <div className="space-y-4">
   371→   371→          <div className="p-4 bg-slate-50 dark:bg-zinc-900 rounded-2xl border border-slate-100 dark:border-zinc-800 space-y-4">
   372→   372→            <div className="space-y-1.5">
   373→   373→              <label className="text-2xs font-black text-slate-400 uppercase tracking-widest ml-1">Rent Amount</label>
   374→   374→              <div className="relative">
   375→   375→                <span className="absolute left-4 top-1/2 -translate-y-1/2 text-slate-400 font-bold"><NairaSymbol /></span>
   376→   376→                <input autoComplete="off" data-lpignore="true" 
   377→   377→                  type="text"
   378→   378→                  value={displayAmount}
   379→   379→                  onChange={e => handleAmountChange(e.target.value)}
   380→   380→                  className="w-full pl-10 pr-4 py-3 bg-white dark:bg-zinc-900 border border-slate-200 dark:border-zinc-700 rounded-lg font-bold text-slate-800 dark:text-zinc-100 focus:ring-2 focus:ring-primary-500/20 focus:border-primary-500 transition-all"
   381→   381→                />
   382→   382→              </div>
   383→   383→            </div>
   384→   384→
   385→   385→            <div className="grid grid-cols-2 gap-4">
   386→   386→              <div className="space-y-1.5">
   387→   387→                <label className="text-2xs font-black text-slate-400 uppercase tracking-widest ml-1">Date Paid</label>
   388→   388→                <div className="relative">
   389→   389→                  <CalendarIcon className="absolute left-4 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   390→   390→                  <input autoComplete="off" data-lpignore="true"
   391→   391→                    type="date"
   392→   392→                    value={paymentDate}
   393→   393→                    onChange={e => {
   394→   394→                      setPaymentDate(e.target.value);
   395→   395→                      if (!periodStart) setPeriodStart(e.target.value);
   396→   396→                    }}
   397→   397→                    className="w-full pl-11 pr-3 py-3 bg-white dark:bg-zinc-900 border border-slate-200 dark:border-zinc-700 rounded-lg text-sm font-medium text-slate-700 dark:text-zinc-300 focus:ring-2 focus:ring-primary-500/20 focus:border-primary-500 transition-all"
   398→   398→                  />
   399→   399→                </div>
   400→   400→                {/* Formatted date display — shows full "15 Sep 2027" format
   401→   401→                    below the native date input so the user can always read
   402→   402→                    the date clearly, even when the input is narrow. */}
   403→   403→                {paymentDate && (
