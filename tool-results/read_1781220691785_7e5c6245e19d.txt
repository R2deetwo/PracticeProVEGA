    57→    57→export function buildMessage(
    58→    58→  type: AutomationMessageType, 
    59→    59→  unitLabel: string, 
    60→    60→  tenantName?: string, 
    61→    61→  amount?: number, 
    62→    62→  customText?: string, 
    63→    63→  customTemplates?: Record<string, string>,
    64→    64→  extraData?: {
    65→    65→    serviceCharge?: number;
    66→    66→    legalFee?: number;
    67→    67→    agencyFee?: number;
    68→    68→    cautionDeposit?: number;
    69→    69→    dueDate?: string;
    70→    70→    firmName?: string;
    71→    71→  }
    72→    72→): string {
    73→    73→  if (customText) return customText;
    74→    74→
    75→    75→  const name = tenantName || 'Tenant';
    76→    76→  const addr = unitLabel;
    77→    77→  const baseRent = amount || 0;
    78→    78→  
    79→    79→  const sc = extraData?.serviceCharge || 0;
    80→    80→  const lf = extraData?.legalFee || 0;
    81→    81→  const af = extraData?.agencyFee || 0;
    82→    82→  const cd = extraData?.cautionDeposit || 0;
    83→    83→  const totalPayable = baseRent + sc + lf + af + cd;
    84→    84→  
    85→    85→  const amtStr = `₦${baseRent.toLocaleString('en-NG')}`;
    86→    86→  const totalStr = `₦${totalPayable.toLocaleString('en-NG')}`;
    87→    87→
    88→    88→  let message = customTemplates?.[type];
    89→    89→  
    90→    90→  if (!message) {
    91→    91→    switch (type) {
    92→    92→      case 'rent_reminder': 
    93→    93→        message = `OFFICIAL DEMAND NOTICE\n\nDear {{TENANT_NAME}},\n\nThis is a formal reminder that your payment for {{PROPERTY_ADDRESS}} is due on or before {{DUE_DATE}}.\n\nFinancial Breakdown:\n- Rent: {{AMOUNT}}\n{{SERVICE_CHARGE_LINE}}{{CAUTION_DEPOSIT_LINE}}{{FEES_LINE}}\nTotal Payable: {{TOTAL_PAYABLE}}\n\nPlease ensure payment is made promptly. Thank you.\n\n— {{FIRM_NAME}}`; 
    94→    94→        break;
    95→    95→      case 'late_notice': 
    96→    96→        message = `URGENT: OVERDUE PAYMENT\n\nDear {{TENANT_NAME}},\n\nYour payment of {{TOTAL_PAYABLE}} for {{PROPERTY_ADDRESS}} is now OVERDUE.\n\nPlease make payment immediately to avoid late penalties or access restrictions. Contact management at {{FIRM_NAME}} to confirm your payment.\n\n— {{FIRM_NAME}}`; 
    97→    97→        break;
    98→    98→      case 'payment_receipt': 
    99→    99→        message = `PAYMENT RECEIPT\n\nDear {{TENANT_NAME}},\n\nWe confirm receipt of {{TOTAL_PAYABLE}} for your unit at {{PROPERTY_ADDRESS}}.\n\nThank you for your prompt payment.\n\n— {{FIRM_NAME}}`; 
   100→   100→        break;
   101→   101→      case 'service_charge_alert': 
   102→   102→        message = `SERVICE CHARGE ALERT\n\nDear {{TENANT_NAME}},\n\nYour service charge of ₦${sc.toLocaleString('en-NG')} for {{PROPERTY_ADDRESS}} is outstanding. Please settle this at your earliest convenience to avoid access restrictions.\n\nTotal Payable: {{TOTAL_PAYABLE}}`; 
   103→   103→        break;
   104→   104→      case 'access_restriction': 
   105→   105→        message = `NOTICE OF ACCESS RESTRICTION\n\nAccess to {{PROPERTY_ADDRESS}} has been restricted due to non-payment of outstanding charges totaling {{TOTAL_PAYABLE}}. Please contact {{FIRM_NAME}} immediately to resolve.`; 
   106→   106→        break;
   107→   107→      case 'penalty_notice': 
   108→   108→        message = `PENALTY NOTICE\n\nDear {{TENANT_NAME}},\n\nA late payment penalty has been applied to your account for {{PROPERTY_ADDRESS}}.\n\nRevised Total Payable: {{TOTAL_PAYABLE}}\n\nPlease settle the outstanding balance urgently. — {{FIRM_NAME}}`; 
   109→   109→        break;
   110→   110→      case 'lease_renewal': 
   111→   111→        message = `LEASE RENEWAL\n\nDear {{TENANT_NAME}},\n\nYour lease for {{PROPERTY_ADDRESS}} is expiring soon. We invite you to renew your tenancy agreement. Please contact {{FIRM_NAME}} to discuss renewal terms.`; 
   112→   112→        break;
   113→   113→      case 'welcome_note': 
   114→   114→        message = `Welcome to your new home at {{PROPERTY_ADDRESS}}, {{TENANT_NAME}}! We are excited to have you. Please find the tenant handbook in your portal. — {{FIRM_NAME}}`; 
   115→   115→        break;
   116→   116→      case 'promotion': 
   117→   117→        message = `Hello {{TENANT_NAME}}, we have a special offer for our residents! Get 10% off professional cleaning services this month. Use code: CLEAN10. — {{FIRM_NAME}}`; 
   118→   118→        break;
   119→   119→      case 'vendor_update': 
   120→   120→        message = `Dear {{TENANT_NAME}}, we have partnered with new verified maintenance vendors to serve you better. You can now request plumbing and electrical repairs directly from the app. — {{FIRM_NAME}}`; 
   121→   121→        break;
   122→   122→      case 'general_announcement': 
   123→   123→        message = `Attention residents of {{PROPERTY_ADDRESS}}: Routine maintenance will be carried out on the central generators this Saturday. Expect intermittent power supply between 10am and 2pm. — {{FIRM_NAME}}`; 
   124→   124→        break;
   125→   125→      case 'maintenance_update': 
   126→   126→        message = `Update on your maintenance request for {{PROPERTY_ADDRESS}}: The vendor has confirmed your appointment for tomorrow. Please ensure someone is available to grant access. — {{FIRM_NAME}}`; 
   127→   127→        break;
   128→   128→      default: 
   129→   129→        message = ''; 
   130→   130→        break;
   131→   131→    }
   132→   132→  }
   133→   133→
   134→   134→  if (!message) return '';
   135→   135→
   136→   136→  const scLine = sc > 0 ? `- Service Charge: ₦${sc.toLocaleString('en-NG')}\n` : '';
   137→   137→  const cdLine = cd > 0 ? `- Caution Deposit: ₦${cd.toLocaleString('en-NG')}\n` : '';
   138→   138→  const feesLine = (lf + af) > 0 ? `- Legal/Agency Fees: ₦${(lf + af).toLocaleString('en-NG')}\n` : '';
   139→   139→
   140→   140→  let result = message
   141→   141→    .replace(/\{\{TENANT_NAME\}\}/g, name)
   142→   142→    .replace(/\{\{AMOUNT\}\}/g, amtStr)
   143→   143→    .replace(/\{\{TOTAL_PAYABLE\}\}/g, totalStr)
   144→   144→    .replace(/\{\{SERVICE_CHARGE_LINE\}\}/g, scLine)
   145→   145→    .replace(/\{\{CAUTION_DEPOSIT_LINE\}\}/g, cdLine)
   146→   146→    .replace(/\{\{FEES_LINE\}\}/g, feesLine)
   147→   147→    .replace(/\{\{FIRM_NAME\}\}/g, extraData?.firmName || 'Management');
   148→   148→
   149→   149→  // Handle Unit Context smartly
   150→   150→  if (addr === 'General' || addr === 'All Residents') {
   151→   151→      result = result.replace(/ for \{\{PROPERTY_ADDRESS\}\}/g, '');
   152→   152→      result = result.replace(/ at \{\{PROPERTY_ADDRESS\}\}/g, '');
   153→   153→      result = result.replace(/\{\{PROPERTY_ADDRESS\}\}/g, 'your unit');
   154→   154→  } else {
   155→   155→      result = result.replace(/\{\{PROPERTY_ADDRESS\}\}/g, addr);
   156→   156→  }
   157→   157→  
   158→   158→  if (extraData) {
   159→   159→    result = result.replace(/\{\{SERVICE_CHARGE\}\}/g, `₦${sc.toLocaleString('en-NG')}`);
   160→   160→    result = result.replace(/\{\{LEGAL_FEE\}\}/g, `₦${lf.toLocaleString('en-NG')}`);
   161→   161→    result = result.replace(/\{\{AGENCY_FEE\}\}/g, `₦${af.toLocaleString('en-NG')}`);
   162→   162→    result = result.replace(/\{\{CAUTION_DEPOSIT\}\}/g, `₦${cd.toLocaleString('en-NG')}`);
   163→   163→    result = result.replace(/\{\{DUE_DATE\}\}/g, extraData.dueDate || 'the due date');
   164→   164→  }
   165→   165→  
   166→   166→  return result;
   167→   167→}
