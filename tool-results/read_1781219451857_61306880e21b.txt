   113→// ─── Clipboard HTML Cleaner ──────────────────────────────────────────────────
   114→// Strips background colours, dark-mode artifacts, and DraftPro-specific
   115→// attributes from copied HTML so that pasting into Word / Google Docs
   116→// produces clean text with bold / italic / underline preserved and
   117→// NO dark-rectangle background.
   118→const CLEAN_PRESERVE_STYLES = new Set([
   119→    'font-weight', 'font-style', 'font-family', 'font-size',
   120→    'text-decoration', 'text-decoration-line', 'text-align',
   121→    'text-indent', 'line-height', 'list-style-type',
   122→    'margin-left', 'margin-right', 'text-transform',
   123→]);
   124→
   125→function cleanForClipboard(html: string): string {
   126→    const doc = new DOMParser().parseFromString(html, 'text/html');
   127→
   128→    // 1. Convert DraftPro placeholder / context atoms to readable plain text
   129→    doc.querySelectorAll('span[data-type="legal-placeholder"]').forEach(el => {
   130→        const label = el.getAttribute('data-label') || 'PLACEHOLDER';
   131→        el.replaceWith(doc.createTextNode(`[${label}]`));
   132→    });
   133→    doc.querySelectorAll('span[data-type="legal-context"]').forEach(el => {
   134→        const label = el.getAttribute('data-label') || '';
   135→        el.replaceWith(doc.createTextNode(label));
   136→    });
   137→
   138→    // 2. Walk every element — strip classes, data-attrs, and non-essential styles
   139→    doc.body.querySelectorAll('*').forEach(el => {
   140→        // Remove Tailwind class attrs (meaningless outside the app)
   141→        el.removeAttribute('class');
   142→
   143→        // Remove DraftPro data-* attributes
   144→        Array.from(el.attributes).forEach(attr => {
   145→            if (attr.name.startsWith('data-')) el.removeAttribute(attr.name);
   146→        });
   147→
   148→        // Keep only formatting styles; drop background-color, background, color: white, etc.
   149→        if (el.hasAttribute('style')) {
   150→            const raw = el.getAttribute('style') || '';
   151→            const kept: string[] = [];
   152→
   153→            raw.split(';').forEach(decl => {
   154→                const trimmed = decl.trim();
   155→                if (!trimmed) return;
   156→                const colonIdx = trimmed.indexOf(':');
   157→                if (colonIdx === -1) return;
   158→                const prop = trimmed.substring(0, colonIdx).trim().toLowerCase();
   159→                const val   = trimmed.substring(colonIdx + 1).trim().toLowerCase();
   160→
   161→                // Always drop background and background-color
   162→                if (prop === 'background' || prop === 'background-color') return;
   163→
   164→                // Drop colour styles that would be invisible on a white page
   165→                if (prop === 'color') {
   166→                    const invisible = ['#fff', '#ffffff', 'white', 'rgb(255, 255, 255)', 'rgba(255,255,255'];
   167→                    if (invisible.some(c => val.includes(c))) return;
   168→                }
   169→
   170→                if (CLEAN_PRESERVE_STYLES.has(prop)) kept.push(trimmed);
   171→            });
   172→
   173→            if (kept.length > 0) {
   174→                el.setAttribute('style', kept.join('; ') + ';');
   175→            } else {
   176→                el.removeAttribute('style');
   177→            }
   178→        }
   179→    });
   180→
   181→    // 3. Remove empty <span> wrappers left after attribute stripping
   182→    doc.body.querySelectorAll('span').forEach(el => {
   183→        if (!el.attributes.length && !el.children.length && el.textContent === '') {
   184→            el.remove();
   185→        }
   186→    });
   187→
   188→    return doc.body.innerHTML;
   189→}

   251→const AutoPagination = Extension.create({
   252→    name: 'autoPagination',
   253→
   254→    addProseMirrorPlugins() {
   255→        const pluginKey = new PluginKey('autoPagination');
   256→
   257→        return [
   258→            new Plugin({
   259→                key: pluginKey,
   260→                state: {
   261→                    init() {
   262→                        return DecorationSet.empty;
   263→                    },
   264→                    apply(tr, set) {
   265→                        const meta = tr.getMeta(pluginKey);
   266→                        if (meta) {
   267→                            return meta;
   268→                        }
   269→                        return set.map(tr.mapping, tr.doc);
   270→                    },
   271→                },
   272→                props: {
   273→                    decorations(state) {
   274→                        return pluginKey.getState(state);
   275→                    },
   276→                },
   277→                view() {
   278→                    return {
   279→                        update(view, prevState) {
   280→                            if (prevState.doc.eq(view.state.doc) && !view.state.tr.getMeta('forcePagination')) return;
   281→
   282→                            requestAnimationFrame(() => {
   283→                                const TOP_RESERVED = PAGE_MARGIN_PX + HEADER_HEIGHT_PX; // 176
   284→                                const BOTTOM_RESERVED = PAGE_MARGIN_PX + FOOTER_HEIGHT_PX; // 136
   285→
   286→                                let spaceUsed = 0;
   287→                                const decos: Decoration[] = [];
   288→
   289→                                view.state.doc.descendants((node, pos) => {
   290→                                    if (node.isBlock && view.state.doc.resolve(pos).depth === 0) {
   291→                                        const dom = view.nodeDOM(pos) as HTMLElement;
   292→                                        if (dom) {
   293→                                            const style = window.getComputedStyle(dom);
   294→                                            const marginBottom = parseFloat(style.marginBottom) || 0;
   295→                                            const childHeight = dom.offsetHeight + marginBottom;
   296→
   297→                                            if (spaceUsed + childHeight > USABLE_CONTENT_HEIGHT) {
   298→                                                const spaceLeft = USABLE_CONTENT_HEIGHT - spaceUsed;
   299→                                                const jumpMargin = spaceLeft + BOTTOM_RESERVED + PAGE_GAP_PX + TOP_RESERVED;
   300→
   301→                                                decos.push(
   302→                                                    Decoration.node(pos, pos + node.nodeSize, {
   303→                                                        style: `margin-top: ${jumpMargin}px !important;`,
   304→                                                        class: 'auto-paginated-node'
   305→                                                    })
   306→                                                );
   307→
   308→                                                spaceUsed = childHeight;
   309→                                            } else {
   310→                                                spaceUsed += childHeight;
   311→                                            }
   312→                                        }
   313→                                    }
   314→                                    return false; // Don't descend
   315→                                });
   316→
   317→                                const currentSet = pluginKey.getState(view.state) as DecorationSet;
   318→                                const currentArray = currentSet.find();
   319→
   320→                                let isDifferent = decos.length !== currentArray.length;
   321→                                if (!isDifferent) {
   322→                                    for (let i = 0; i < decos.length; i++) {
   323→                                        const d1 = decos[i] as any;
   324→                                        const d2 = currentArray[i] as any;
   325→                                        if (
   326→                                            d1.from !== d2.from ||
   327→                                            d1.to !== d2.to ||
   328→                                            d1.type.attrs.style !== d2.type.attrs.style
   329→                                        ) {
   330→                                            isDifferent = true;
   331→                                            break;
   332→                                        }
   333→                                    }
   334→                                }
   335→
   336→                                if (isDifferent) {
   337→                                    const newSet = DecorationSet.create(view.state.doc, decos);
   338→                                    view.dispatch(view.state.tr.setMeta(pluginKey, newSet));
   339→                                }
   340→                            });
   341→                        },
   342→                    };
   343→                },
   344→            }),
   345→        ];
   346→    },
   347→});
