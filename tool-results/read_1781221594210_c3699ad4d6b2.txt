   532→        if (currentUser && visualsComplete) {
   533→            // Portal users skip the data loading wait — their data comes from dedicated portal queries
   534→            const isPortal = currentUser.role === UserRole.Client || currentUser.role === UserRole.Tenant;
   535→            if (isPortal) {
   536→                setFlowState('app');
   537→                setHasInitialized(true);
   538→                setHasInitialSplashFinished(true);
   539→                return;
   540→            }
   541→            if (isDataLoaded || forceEntry) {
   542→                setLoadingMessage("Ready");
   543→                const t = setTimeout(() => {
   544→                    if (!currentUser.firmId) {
   545→                        setFlowState('setup');
   546→                    } else {
   547→                        setFlowState('app');
   548→                        if (appMode === AppMode.Solo) {
   549→                            setIsToolkitOpen(true);
   550→                        }
   551→                    }
   552→                    setHasInitialized(true);
   553→                    if (!hasInitialized && (modal === 'login' || modal === 'signup')) {
   554→                        closeModal();
   555→                    }
   556→                }, 500);
   557→                return () => clearTimeout(t);
   558→            } else {
   559→                setLoadingMessage("Finalizing Data Synchronization...");
   560→            }
   561→        }
   562→    }, [currentUser, visualsComplete, isDataLoaded, forceEntry, modal, closeModal]);
   563→
   564→    useEffect(() => {
   565→        if (flowState === 'app' && currentUser) {
   566→            if (forceEntry && !isDataLoaded) return;
   567→            // Don't auto-start the tour for portal users (Client/Tenant)
   568→            const isPortal = currentUser.role === UserRole.Client || currentUser.role === UserRole.Tenant;
   569→            if (isPortal) return;
   570→            const hasCompletedTour = localStorage.getItem('practicepro_tour_completed');
   571→            if (hasCompletedTour !== 'true') setTimeout(() => startTour(), 500);
   572→        }
   573→    }, [flowState, currentUser, startTour]);
   574→
   575→    const isPortalUserRole = currentUser?.role === UserRole.Client || currentUser?.role === UserRole.Tenant;
   576→
   577→    const renderAppContent = () => {
   578→        // ── Portal login routes ──
   579→        // If user is already authenticated as a portal user, redirect them to their dashboard
   580→        // instead of showing the login page again. This handles the case where a portal user
   581→        // navigates to the login page while already logged in.
   582→        if (location.pathname === '/portal/client/login') {
   583→            if (currentUser && currentUser.role === UserRole.Client) {
   584→                // Already logged in as client — redirect to portal dashboard via React Router
   585→                navigate('/', { replace: true });
   586→                return null;
   587→            }
   588→            return <ClientPortalLogin />;
   589→        }
   590→        if (location.pathname === '/portal/tenant/login') {
   591→            if (currentUser && currentUser.role === UserRole.Tenant) {
   592→                // Already logged in as tenant — redirect to portal dashboard via React Router
   593→                navigate('/', { replace: true });
   594→                return null;
   595→            }
   596→            return <TenantPortalLogin />;
   597→        }
   598→        if (location.pathname === '/setup-password') return <SetupPassword />;
   599→
   600→        if (!currentUser && !isLoadingSession) {
   601→            // Check if a portal user session is being restored — if so, show a loading
   602→            // state instead of the LandingPage to prevent the jarring flash
   603→            const hasRememberedPortal = typeof sessionStorage !== 'undefined' && (
   604→                sessionStorage.getItem('practicepro_portal_type') !== null ||
   605→                sessionStorage.getItem('practicepro_portal_session') !== null ||
   606→                localStorage.getItem('practicepro_portal_session') !== null
   607→            );
   608→            if (hasRememberedPortal && !portalRememberTimedOut) {
   609→                // Session might still be loading (e.g. slow Convex query). Show a minimal
   610→                // portal-themed loading screen rather than the public LandingPage.
   611→                // Auto-timeout after 15s: if the session truly isn't valid, clear the
   612→                // remembered portal type and fall through to the LandingPage.
   613→                setTimeout(() => setPortalRememberTimedOut(true), 15000);
   614→                return (
   615→                    <div className="min-h-screen bg-gradient-to-br from-slate-900 via-slate-950 to-slate-900 flex items-center justify-center">
   616→                        <div className="text-center">
   617→                            <div className="w-8 h-8 border-2 border-emerald-500 border-t-transparent rounded-full animate-spin mx-auto mb-4" />
   618→                            <p className="text-slate-400 text-sm">Loading your portal...</p>
   619→                        </div>
   620→                    </div>
   621→                );
   622→            }
   623→            if (view === 'termsOfService') return <TermsOfService onBack={() => navigateTo('dashboard')} activeProduct={product === 'property' ? 'atrium' : product === 'legal' ? 'vega' : undefined} />;
   624→            if (view === 'portalTermsOfUse') return <PortalTermsOfUse onBack={() => navigateTo('dashboard')} activeProduct={product === 'property' ? 'atrium' : product === 'legal' ? 'vega' : undefined} />;
   625→            if (view === 'privacyPolicy') return <PrivacyPolicy onBack={() => navigateTo('dashboard')} />;
   626→            if (view === 'dataProcessingAgreement') return <DataProcessingAgreement onBack={() => navigateTo('dashboard')} />;
   627→            if (view === 'cookiePolicy') return <CookiePolicy onBack={() => navigateTo('dashboard')} />;
   628→            return <LandingPage onDemo={(product) => openModal('leadCapture', null, { demoProduct: product })} />;
   629→        }
   630→
   631→        // New User Flow: Go straight to setup if no firm exists
   632→        if (currentUser && !currentUser.firmId) {
   633→            return <OnboardingWizard onComplete={() => setFlowState('app')} />;
   634→        }
   635→
   636→        // Only block on SplashScreen if the user has a saved session being restored.
   637→        // New/unauthenticated visitors must NEVER see the splash — they go straight to the Landing Page.
   638→        if (isLoadingSession) {
   639→            return null; // Top-level SplashScreen handles the branding
   640→        }
   641→
   642→        if (currentUser?.role === 'Pending') {
   643→            return (
   644→                <div className="flex flex-col items-center justify-center min-h-screen bg-slate-50 dark:bg-zinc-950 w-full p-8 text-center font-sans">
   645→                    <div className="max-w-md w-full bg-white dark:bg-zinc-900 p-10 rounded-[32px] shadow-2xl shadow-slate-200 dark:shadow-black/50 border border-slate-100 dark:border-zinc-800 flex flex-col items-center animate-in fade-in zoom-in duration-500">
   646→                        <div className="w-20 h-20 bg-amber-50 dark:bg-amber-900/20 rounded-full flex items-center justify-center mb-8 ring-8 ring-amber-50/50 dark:ring-amber-900/10">
   647→                            <OfficeBuildingIcon className="w-10 h-10 text-amber-600 dark:text-amber-400" />
   648→                        </div>
   649→                        <h2 className="text-3xl font-black text-slate-900 dark:text-white mb-3 tracking-tight">Access Pending</h2>
   650→                        <p className="text-slate-500 dark:text-zinc-400 mb-10 leading-relaxed text-lg">
   651→                            Your request to join <span className="font-bold text-slate-900 dark:text-white">{coreState?.firmDetails?.name || 'this workspace'}</span> is awaiting approval from {product === 'atrium' ? 'the portfolio manager' : 'a firm administrator'}.
   652→                        </p>
   653→                        
   654→                        <div className="flex flex-col gap-4 w-full">
   655→                            <button 
   656→                                onClick={() => window.location.reload()} 
   657→                                className="w-full py-4 bg-green-600 hover:bg-green-700 text-white rounded-2xl font-black text-sm uppercase tracking-widest transition-all shadow-xl shadow-green-600/20 active:scale-[0.98]"
   658→                            >
   659→                                Check Approval Status
   660→                            </button>
   661→                            <button 
   662→                                onClick={() => logout()} 
   663→                                className="w-full py-3 text-slate-400 dark:text-zinc-500 hover:text-slate-900 dark:hover:text-white font-bold text-sm transition-colors"
   664→                            >
   665→                                Sign Out & Switch Account
   666→                            </button>
   667→                        </div>
   668→
   669→                        <div className="mt-10 pt-8 border-t border-slate-50 dark:border-zinc-800 w-full">
   670→                            <p className="text-[10px] font-bold text-slate-400 dark:text-zinc-600 uppercase tracking-widest mb-1">Account Reference</p>
   671→                            <p className="text-xs font-mono text-slate-300 dark:text-zinc-700">{currentUser.id}</p>
   672→                        </div>
   673→                    </div>
   674→                </div>
   675→            );
   676→        }
   677→
   678→        return (
   679→            <>
   680→                {(flowState === 'app' || flowState === 'splash') && (
   681→                    <>
   682→                        <MainContent
   683→                            onToggleToolkit={() => setIsToolkitOpen(prev => !prev)}
   684→                            isToolkitOpen={isToolkitOpen}
   685→                            onCloseToolkit={() => setIsToolkitOpen(false)}
   686→                            onEnableDevMode={() => setIsToolkitOpen(true)}
   687→                            flowState={flowState}
   688→                        />
   689→                        {import.meta.env.DEV && <FloatingTestControls isOpen={isToolkitOpen} onClose={() => setIsToolkitOpen(false)} />}
   690→                    </>
   691→                )}
   692→            </>
   693→        );
   694→    };
   695→
   696→    useEffect(() => {
   697→        const timer = setTimeout(() => {
   698→            setHasInitialSplashFinished(true);
   699→        }, 12000);
   700→        return () => clearTimeout(timer);
   701→    }, []);
   702→
   703→    useEffect(() => {
   704→        if (splashAnimationComplete && isDataLoaded) {
   705→            setHasInitialSplashFinished(true);
   706→        }
   707→    }, [splashAnimationComplete, isDataLoaded]);
   708→
   709→    return (
   710→        <div className={`app-container font-sans text-base ${theme} h-[100dvh] bg-[rgb(var(--bg-main))] text-[rgb(var(--text-main))]`}>
   711→            <SplashScreen 
   712→                isVisible={!hasInitialSplashFinished && (hasSavedSession || flowState === 'splash')} 
   713→                statusMessage={loadingMessage} 
   714→                onReset={handleReset} 
   715→                onForceEnter={() => { setForceEntry(true); setHasInitialSplashFinished(true); }} 
   716→                onComplete={() => setSplashAnimationComplete(true)}
   717→                product={(!isDataLoaded || !currentUser?.firmId) ? 'practicepro' : (product === 'property' ? 'atrium' : 'vega')} 
   718→            />
   719→            {isSessionLocked ? (
   720→                <LockScreen onUnlock={() => setIsSessionLocked(false)} />
   721→            ) : (
   722→                renderAppContent()
   723→            )}
   724→            {((!currentUser) || (currentUser && flowState !== 'splash')) && (
   725→                <>
   726→                    <ModalManager />
   727→                    <ToastContainer />
   728→                </>
   729→            )}
   730→            {/* Onboarding Tour only shown for non-portal users; portal users get a simpler experience */}
   731→            {currentUser && currentUser.role !== UserRole.Client && currentUser.role !== UserRole.Tenant && <OnboardingTour />}
   732→            {/* What's New only for admin/firm users, not portal users */}
   733→            {flowState === 'app' && currentUser && currentUser.role !== UserRole.Client && currentUser.role !== UserRole.Tenant && <WhatsNew />}
   734→            <CookieConsent />
   735→        </div>
   736→    );
   737→};
   738→
   739→export default App;
