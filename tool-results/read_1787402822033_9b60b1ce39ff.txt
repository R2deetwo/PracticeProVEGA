    39→    39→export const getActivePeers = query({
    40→    40→  args: { firmId: v.optional(v.string()), userEmail: v.optional(v.string()) },
    41→    41→  handler: async (ctx, args) => {
    42→    42→    // SECURITY: Verify the caller belongs to the firm they're querying
    43→    43→    if (args.userEmail) {
    44→    44→      try {
    45→    45→        const auth = await requireFirmUser(ctx, args.userEmail);
    46→    46→        if (auth.firmId && auth.firmId !== args.firmId) return [];
    47→    47→      } catch { return []; }
    48→    48→    }
    49→    49→    if (!args.firmId) return [];
    50→    50→
    51→    51→    // Fetch ALL presence records for the firm (not just active ones).
    52→    52→    // We return both active and inactive users with their lastSeen timestamp
    53→    53→    // so the frontend can show an inactivity indicator (greyed out) and
    54→    54→    // last-seen time on hover.
    55→    55→    const allPresence = await ctx.db.query("presence").withIndex("by_firm", (q) => q.eq("firmId", args.firmId)).take(100);
    56→    56→
    57→    57→    // Fetch users to check their visibility preferences
    58→    58→    const users: any[] = [];
    59→    59→    for (const p of allPresence) {
    60→    60→      try {
    61→    61→        const user = await ctx.db.get(p.userId as any);
    62→    62→        if (user) users.push(user);
    63→    63→      } catch (e) { /* skip invalid IDs */ }
    64→    64→    }
    65→    65→
    66→    66→    // Build a set of user IDs that should be hidden from presence
    67→    67→    const hiddenUserIds = new Set<string>();
    68→    68→    for (const user of users) {
    69→    69→      if (user.role === 'Tenant' && user.portalPresenceHidden !== false) {
    70→    70→        hiddenUserIds.add(user._id.toString());
    71→    71→      }
    72→    72→      if (user.portalPresenceHidden === true) {
    73→    73→        hiddenUserIds.add(user._id.toString());
    74→    74→      }
    75→    75→    }
    76→    76→
    77→    77→    // Return rich presence data: userId, updatedAt (last seen), and isOnline
    78→    78→    // Active = heartbeat within the last 60 seconds
    79→    79→    const now = Date.now();
    80→    80→    const ACTIVE_THRESHOLD = 60 * 1000;
    81→    81→    return allPresence
    82→    82→      .filter((p: any) => !hiddenUserIds.has(p.userId))
    83→    83→      .map((p: any) => ({
    84→    84→        userId: p.userId,
    85→    85→        updatedAt: p.updatedAt,
    86→    86→        isOnline: (now - p.updatedAt) < ACTIVE_THRESHOLD,
    87→    87→      }));
    88→    88→  },
    89→    89→});

   100→   100→      if (identity) authenticatedEmail = identity.email || undefined;
   101→   101→    } catch {}
   102→   102→
   103→   103→    // If the caller is authenticated, they can only diagnose their own email
   104→   104→    // (unless no auth context exists — legacy fallback for repair flows)
   105→   105→    if (authenticatedEmail && args.userEmail) {
   106→   106→      const auth = await requireFirmUser(ctx, args.userEmail);
   107→   107→      // Verify the email being diagnosed belongs to the authenticated user
   108→   108→      if (auth.user?.tokenIdentifier?.toLowerCase() !== args.email.toLowerCase().trim()) {
   109→   109→        throw new Error("Unauthorized. You can only diagnose your own account.");
   110→   110→      }
   111→   111→    }
   112→   112→
   113→   113→    const emailInput = args.email.toLowerCase().trim();
   114→   114→
   115→   115→    const allUsers = await ctx.db.query("users").take(500);
   116→   116→    const user = allUsers.find((u: any) => u.tokenIdentifier && u.tokenIdentifier.toLowerCase() === emailInput);
   117→   117→
   118→   118→    const diagnosis = {
   119→   119→      emailSearched: emailInput,
   120→   120→      userFound: !!user,
   121→   121→      userId: user?._id || null,
   122→   122→      currentFirmId: user?.firmId || null,
   123→   123→      firmFound: false,
   124→   124→      availableFirms: [] as any[]
   125→   125→    };
   126→   126→
   127→   127→    if (user && user.firmId) {
   128→   128→      const firm = await ctx.db.get(user.firmId as any) as any;
   129→   129→      if (firm) {
   130→   130→        diagnosis.firmFound = true;
   131→   131→        diagnosis.availableFirms.push({
   132→   132→          id: firm._id,
   133→   133→          name: firm.name,
   134→   134→          status: 'Linked',
   135→   135→          createdAt: firm._creationTime
   136→   136→        });
   137→   137→      }
   138→   138→    }
   139→   139→
   140→   140→    const allFirms = await ctx.db.query("firms").take(500);
   141→   141→    const joinedIds = user?.joinedFirmIds || [];
   142→   142→    
   143→   143→    const availableFirmsFiltered = allFirms.filter((f: any) => {
   144→   144→      // 1. Created by this user
   145→   145→      const creatorIdMatch = f.createdBy === user?._id || f.createdBy === user?.tokenIdentifier;
   146→   146→      const settingsEmailMatch = f.settings && f.settings.creatorEmail && f.settings.creatorEmail.toLowerCase() === emailInput;
   147→   147→      
   148→   148→      // 2. Already joined - check the persistence list
   149→   149→      const previouslyJoined = joinedIds.includes(f._id);
   150→   150→      
   151→   151→      // 3. Current firm
   152→   152→      const isCurrentlyIn = user && user.firmId === f._id;
   153→   153→      
   154→   154→      // We want to show firms NOT currently active but where user has access
   155→   155→      return (creatorIdMatch || settingsEmailMatch || previouslyJoined || isCurrentlyIn) && f._id !== diagnosis.currentFirmId;
   156→   156→    });
   157→   157→
   158→   158→    availableFirmsFiltered.forEach(f => {
   159→   159→      diagnosis.availableFirms.push({
   160→   160→        id: f._id,
   161→   161→        name: f.name,
   162→   162→        status: 'Available',
   163→   163→        createdAt: f._creationTime
   164→   164→      });
   165→   165→    });
   166→   166→
   167→   167→    if (!diagnosis.firmFound && diagnosis.availableFirms.length > 0) {
   168→   168→      diagnosis.firmFound = true;
   169→   169→    }
   170→   170→
   171→   171→    return diagnosis;
   172→   172→  }
   173→   173→});

   814→   814→/**
   815→   815→ * PHASE 2+3: Chat Messages — loaded per conversation, NOT firm-wide.
   816→   816→ * Uses the existing by_conversation index which was always deployed.
   817→   817→ * For the firm-wide unread count query, uses .filter() safely.
   818→   818→ */
   819→   819→export const getChatMessages = query({
   820→   820→  args: {
   821→   821→    conversationId: v.optional(v.string()),
   822→   822→    firmId: v.optional(v.string()),
   823→   823→  },
   824→   824→  handler: async (ctx, args) => {
   825→   825→    try {
   826→   826→      if (!args.conversationId && !args.firmId) return [];
   827→   827→
   828→   828→      if (args.conversationId) {
   829→   829→        // Most efficient: use the by_conversation index
   830→   830→        // CRITICAL: Sort by _creationTime ASCENDING so messages render in
   831→   831→        // chronological order (oldest first, newest at bottom). Without
   832→   832→        // this, messages load out of order on refresh (e.g. Aug 14 messages
   833→   833→        // appearing above Aug 13 messages).
   834→   834→        return await ctx.db
   835→   835→          .query("chatMessages")
   836→   836→          .withIndex("by_conversation", (q) => q.eq("conversationId", args.conversationId as any))
   837→   837→          .order("asc")
   838→   838→          .collect();
   839→   839→      }
   840→   840→
   841→   841→      const firmId = args.firmId;
   842→   842→      if (!firmId) return [];
   843→   843→
   844→   844→      // 1. Try fetching by firmId index directly (fastest for new data)
   845→   845→      const messagesByFirm = await ctx.db
   846→   846→        .query("chatMessages")
   847→   847→        .withIndex("by_firm", (q) => q.eq("firmId", firmId))
   848→   848→        .order("desc")
   849→   849→        .take(1000);
   850→   850→
   851→   851→      // 2. Fetch firm conversations to find legacy messages without firmId
   852→   852→      const firmConvos = await ctx.db
   853→   853→          .query("chatConversations")
   854→   854→          .withIndex("by_firm", (q) => q.eq("firmId", firmId))
   855→   855→          .take(300);
   856→   856→          
   857→   857→      const convoIds = new Set(firmConvos.map(c => c._id.toString()));
   858→   858→
   859→   859→      // 3. Pull recent messages and filter for legacy ones belonging to these conversations
   860→   860→      const recentMessages = await ctx.db
   861→   861→        .query("chatMessages")
   862→   862→        .order("desc")
   863→   863→        .take(1000);
   864→   864→
   865→   865→      const legacyMessages = recentMessages.filter(m => !m.firmId && m.conversationId && convoIds.has(m.conversationId.toString()));
   866→   866→
   867→   867→      // 4. Combine and return unique, sorted by creation time ascending
   868→   868→      const combined = [...messagesByFirm, ...legacyMessages];
   869→   869→      const unique = Array.from(new Map(combined.map(m => [m._id, m])).values());
   870→   870→      // Sort by _creationTime ascending (oldest first) so the chat renders
   871→   871→      // in chronological order with the newest message at the bottom.
   872→   872→      return unique.sort((a: any, b: any) => (a._creationTime || 0) - (b._creationTime || 0));
   873→   873→
   874→   874→    } catch (e) {
   875→   875→      console.error("getChatMessages error:", e);

  1003→  1003→      const preferPortalRole = args.preferPortalRole === true;
  1004→  1004→

  1006→  1006→      //    so we can disambiguate when the same email exists as both an admin
  1007→  1007→      //    record AND a portal record (the "residents see admin dashboard" bug).
  1008→  1008→      //    Indexed lookups return rows in index order; .first() was returning
  1009→  1009→      //    whichever came first — usually the older Admin record.
  1010→  1010→      const directMatches = await ctx.db
  1011→  1011→        .query("users")
  1012→  1012→        .withIndex("by_token", (q) => q.eq("tokenIdentifier", token))
  1013→  1013→        .collect();
  1014→  1014→
  1015→  1015→      // 2. Case-insensitive indexed lookup
  1016→  1016→      const lowerMatches = directMatches.length === 0
  1017→  1017→        ? await ctx.db
  1018→  1018→            .query("users")
  1019→  1019→            .withIndex("by_token", (q) => q.eq("tokenIdentifier", token.toLowerCase()))
  1020→  1020→            .collect()
  1021→  1021→        : [];
  1022→  1022→
  1023→  1023→      // 3. Fallback: bounded scan — take only 500 to prevent timeout
  1024→  1024→      const allMatches = directMatches.length > 0 || lowerMatches.length > 0
  1025→  1025→        ? [...directMatches, ...lowerMatches]
  1026→  1026→        : (await ctx.db.query("users").take(500))
  1027→  1027→            .filter((u: any) =>
  1028→  1028→              u.tokenIdentifier &&
  1029→  1029→              u.tokenIdentifier.toLowerCase() === token.toLowerCase()
  1030→  1030→            );

  1033→  1033→

  1035→  1035→      // dashboard" bug occurred because the same email existed as BOTH an Admin
  1036→  1036→      // user record AND a Tenant/Client record, and .first() was returning the
  1037→  1037→      // Admin record when the user logged in via the portal.
  1038→  1038→      //
  1039→  1039→      // Resolution strategy:
  1040→  1040→      //   - If preferPortalRole is true (login via /portal/* route) AND a
  1041→  1041→      //     portal-role record exists, prefer it.
  1042→  1042→      //   - Otherwise, prefer the first record (preserves existing behavior
  1043→  1043→      //     for admin-side logins).

  1045→  1045→      //     that should never be the resolved user.
  1046→  1046→      const PORTAL_ROLES = new Set(["Client", "Tenant"]);

  1050→  1050→      // revoked-user handling in AuthContext take over.

  1052→  1052→
  1053→  1053→      let user: any;
  1054→  1054→      if (preferPortalRole) {

  1064→  1064→      // to the frontend. These fields MUST NEVER leave the server.

  1066→  1066→      const { 
  1067→  1067→        password, 
  1068→  1068→        mfaCode, 
  1069→  1069→        verificationCode, 
  1070→  1070→        failedLoginAttempts, 
  1071→  1071→        lockedUntil,
  1072→  1072→        // @ts-ignore - destructure to exclude, even if not always present
  1073→  1073→        passwordHash,
  1074→  1074→        ...safeUser 
  1075→  1075→      } = user as any;
  1076→  1076→
  1077→  1077→      return safeUser;
  1078→  1078→    } catch (e) {

  1108→  1108→ *   count: number,

  1112→  1112→ * }
  1113→  1113→ *
  1114→  1114→ * NOTE: This query is intentionally bounded by .take(2000) — sufficient for
  1115→  1115→ * thousands of users. If the user count exceeds this, increase the limit or

  1122→  1122→    // forgets to pass the role, return empty (defensive).
  1123→  1123→    if (args.requesterRole && args.requesterRole !== "Admin") return [];
  1124→  1124→

  1126→  1126→
  1127→  1127→    // Group by lowercased tokenIdentifier (email)
  1128→  1128→    const groups = new Map<string, any[]>();
  1129→  1129→    for (const u of allUsers as any[]) {
  1130→  1130→      const email = (u.tokenIdentifier || "").toLowerCase().trim();
  1131→  1131→      if (!email) continue;

  1133→  1133→      groups.get(email)!.push(u);
  1134→  1134→    }

  1137→  1137→      email: string;
  1138→  1138→      count: number;

  1142→  1142→    }> = [];

  1163→  1163→    // Sort: biggest conflict groups first, then alphabetically by email
  1164→  1164→    duplicates.sort((a, b) => b.count - a.count || a.email.localeCompare(b.email));

  1400→  1400→    email: v.string(),
  1401→  1401→    passwordHash: v.string(),
  1402→  1402→    rawPassword: v.optional(v.string()),

  1415→  1415→    //    record (usually the older Admin record), the password check passes against
  1416→  1416→    //    the Admin record (whose password was overwritten by setupPortalPassword),

  1444→  1444→      return {
  1445→  1445→        success: false,
  1446→  1446→        message: `This email is a portal account. Please log in through the portal: ${portalUrl}`,
  1447→  1447→        redirect: portalUrl,
  1448→  1448→      };
  1449→  1449→    }
  1450→  1450→
  1451→  1451→    // Distinguish between "email not confirmed" (never verified) and "account revoked/deactivated"

  1457→  1457→        // "Email not confirmed" error.

  1464→  1464→    // even if isVerified was somehow still true.
  1465→  1465→    if (user.role === "Pending") {
  1466→  1466→      return { success: false, message: "Your portal access has been revoked. Please contact your manager to request a new invitation.", isRevoked: true };
  1467→  1467→    }
  1468→  1468→

  1470→  1470→    // If a firm admin has deactivated this user (deactivatedAt is set),
  1471→  1471→    // block the login with a clear message. The user record still exists
  1472→  1472→    // (for audit/authorship) but they cannot access the app.

  1474→  1474→      return {
  1475→  1475→        success: false,
  1476→  1476→        message: "Your account has been deactivated. Please contact your firm administrator to request reactivation.",
  1477→  1477→        isDeactivated: true,
  1478→  1478→      };
  1479→  1479→    }

  1482→  1482→    const LOCKOUT_DURATION_MS = 15 * 60 * 1000;

  1488→  1488→      return {

  1490→  1490→        isLocked: true,

  1492→  1492→      };

  1494→  1494→      await ctx.runMutation(internal.myFunctions.updateUserSecurityFields, {
  1495→  1495→        userId: user._id,
  1496→  1496→        fields: { lockedUntil: null, failedLoginAttempts: 0 },
  1497→  1497→      });
  1498→  1498→      user.lockedUntil = null;
  1499→  1499→      user.failedLoginAttempts = 0;
  1500→  1500→    }
  1501→  1501→

  1504→  1504→    let needsMigration = false;
  1505→  1505→    const rawPw = args.rawPassword || "";

  1509→  1509→        password: rawPw,
  1510→  1510→        storedHash: user.password,

  1515→  1515→      // No password stored — trust on first use, migrate them
  1516→  1516→      isPasswordCorrect = true;
  1517→  1517→      needsMigration = true;
  1518→  1518→    }

  1522→  1522→      if (user.isMfaEnabled) {
  1523→  1523→        if (!args.mfaCode) {
  1524→  1524→          const code = Math.floor(100000 + Math.random() * 900000).toString();
  1525→  1525→          await ctx.runMutation(internal.myFunctions.updateUserSecurityFields, {
  1526→  1526→            userId: user._id,
  1527→  1527→            fields: { mfaCode: code },
  1528→  1528→          });
  1529→  1529→          try {
  1530→  1530→            await ctx.scheduler.runAfter(0, (internal as any).myFunctions.sendVerificationEmail, { email: args.email, code, product: user.product || 'legal' });
  1531→  1531→          } catch (e) {
  1532→  1532→            console.error("Failed to send MFA email", e);
  1533→  1533→          }
  1534→  1534→          return { success: false, requiresMfa: true, mfaType: 'email' };
  1535→  1535→        } else {
  1536→  1536→          if (user.mfaCode !== args.mfaCode) {
  1537→  1537→            return { success: false, message: "Incorrect security code. Please try again." };
  1538→  1538→          }
  1539→  1539→        }
  1540→  1540→      }
  1541→  1541→
  1542→  1542→      // Success! Build patch updates.
  1543→  1543→      const updates: Record<string, any> = {};
  1544→  1544→      if (user.failedLoginAttempts > 0) updates.failedLoginAttempts = 0;
  1545→  1545→      if (user.lockedUntil) updates.lockedUntil = null;
  1546→  1546→      if (user.mfaCode) updates.mfaCode = null;
  1547→  1547→
  1548→  1548→      // Upgrade legacy hash to PBKDF2 on successful login
  1549→  1549→      if (needsMigration && rawPw) {
  1550→  1550→        const newHash: any = await ctx.runAction(internal.authUtils.hashPassword, { password: rawPw });
  1551→  1551→        updates.password = newHash;
  1552→  1552→      }
  1553→  1553→
  1554→  1554→      if (Object.keys(updates).length > 0) {
  1555→  1555→        await ctx.runMutation(internal.myFunctions.updateUserSecurityFields, {
  1556→  1556→          userId: user._id,
  1557→  1557→          fields: updates,
  1558→  1558→        });
  1559→  1559→      }
  1560→  1560→
  1561→  1561→      // AUTO-REPAIR: If a portal user (Client/Tenant) has no firmId, try to repair it
  1562→  1562→      // on login. This handles the case where a previous version of
  1563→  1563→      // deletePortalInviteAndCleanup cleared firmId and deleted invite records,
  1564→  1564→      // leaving the user unable to load any portal data. We search multiple sources
  1565→  1565→      // to find the correct firmId and patch the user record.
  1566→  1566→      if ((user.role === "Client" || user.role === "Tenant") && !user.firmId) {
  1567→  1567→        try {
  1568→  1568→          const repairResult: any = await ctx.runMutation(api.portals.repairPortalUserFirmId, {
  1569→  1569→            email: token,
  1570→  1570→          });
  1571→  1571→          if (repairResult.success) {
  1572→  1572→            // Update the user object we're about to return so the client
  1573→  1573→            // gets the firmId immediately without needing a refresh
  1574→  1574→            (user as any).firmId = repairResult.firmId;
  1575→  1575→            (user as any).product = repairResult.firmId ? undefined : (user as any).product;
  1576→  1576→          }
  1577→  1577→        } catch (e) {
  1578→  1578→          // Non-blocking: if repair fails, the user can still log in
  1579→  1579→          // and will see the "Repair My Account" UI in their portal

  1581→  1581→        }
  1582→  1582→      }
  1583→  1583→
  1584→  1584→      // Strip sensitive fields before returning user to client
  1585→  1585→      const { password: _pw, mfaCode: _mfa, verificationCode: _vc, recoveryCode: _rc, ...safeUser } = user;
  1586→  1586→      return { success: true, user: safeUser };
  1587→  1587→    } else {
  1588→  1588→      const currentAttempts = (user.failedLoginAttempts || 0) + 1;
  1589→  1589→      const failureUpdates: any = { failedLoginAttempts: currentAttempts };
  1590→  1590→      if (currentAttempts >= MAX_ATTEMPTS) {
  1591→  1591→        failureUpdates.lockedUntil = now + LOCKOUT_DURATION_MS;
  1592→  1592→      }
  1593→  1593→      await ctx.runMutation(internal.myFunctions.updateUserSecurityFields, {
  1594→  1594→        userId: user._id,
  1595→  1595→        fields: failureUpdates,
  1596→  1596→      });
  1597→  1597→      return {
  1598→  1598→        success: false,
  1599→  1599→        message: failureUpdates.lockedUntil
  1600→  1600→          ? "Account locked due to too many failed attempts. Please try again in 15 minutes."
  1601→  1601→          : `Incorrect password. You have ${MAX_ATTEMPTS - currentAttempts} attempts remaining.`
  1602→  1602→      };
  1603→  1603→    }
  1604→  1604→  }
