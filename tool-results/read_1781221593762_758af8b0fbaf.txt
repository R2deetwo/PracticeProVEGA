   102→// ─── File Upload Helper ──────────────────────────────────────────────────────
   103→async function uploadFilesToConvex(files: File[]): Promise<string[]> {
   104→  const storageIds: string[] = [];
   105→  for (const file of files) {
   106→    const postUrl = await fetch('/api/storage/upload-url');
   107→    // Fallback: use the Convex action for upload URL
   108→    // We'll use the Convex generateUploadUrl mutation via the hook instead
   109→    // For now, we'll use the standard Convex file upload flow
   110→    const { url, storageId } = await (async () => {
   111→      // Use Convex's built-in file upload
   112→      const uploadUrlRes = await fetch(`${window.location.origin}/api/storage/upload-url`);
   113→      if (uploadUrlRes.ok) {
   114→        const { url: u, storageId: s } = await uploadUrlRes.json();
   115→        return { url: u, storageId: s };
   116→      }
   117→      throw new Error('Failed to get upload URL');
   118→    })();
   119→
   120→    const uploadRes = await fetch(url, {
   121→      method: 'POST',
   122→      body: file,
   123→    });
   124→    if (uploadRes.ok) {
   125→      const { storageId: sid } = await uploadRes.json();
   126→      storageIds.push(sid);
   127→    }
   128→  }
   129→  return storageIds;
   130→}

   321→  const { currentUser } = useAuth();

   323→  const userId = currentUser?.id || '';

   327→  // Fetch ledger entries from Convex using the resolved tenant ID

   329→    api.portals.getTenantLedger,
   330→    firmId && resolvedTenantId ? { firmId, tenantId: resolvedTenantId } : 'skip'
   331→  );
   332→
   333→  // Fetch service charges from Convex to compute current SC/MV
   334→  const serviceCharges = useQuery(
   335→    api.sentry.getServiceChargesByFirm,
   336→    firmId ? { firmId } : 'skip'
   337→  );
   338→
   339→  const isLoading = ledgerEntries === undefined || serviceCharges === undefined;
   340→
   341→  // Compute tenant-specific service charges
   342→  const tenantServiceCharges = useMemo(() => {
   343→    if (!serviceCharges || !resolvedTenantId) return [];
   344→    return serviceCharges.filter((sc: any) => sc.tenantId === resolvedTenantId);
   345→  }, [serviceCharges, resolvedTenantId]);
   346→
   347→  // Summary calculations
   348→  const currentMonthSC = useMemo(() => {
   349→    return tenantServiceCharges
   350→      .filter((sc: any) => !sc.isMinimumVend)
   351→      .reduce((sum: number, sc: any) => sum + (sc.outstandingBalance ?? sc.amount), 0);
   352→  }, [tenantServiceCharges]);
   353→
   354→  const currentMonthMV = useMemo(() => {
   355→    return tenantServiceCharges
   356→      .filter((sc: any) => sc.isMinimumVend)
   357→      .reduce((sum: number, sc: any) => sum + (sc.outstandingBalance ?? sc.amount), 0);
   358→  }, [tenantServiceCharges]);
   359→
   360→  const outstandingBalance = useMemo(() => {
   361→    if (!ledgerEntries) return 0;
   362→    return ledgerEntries
   363→      .filter((e: any) => e.status === 'pending' || e.status === 'defaulted')
   364→      .reduce((sum: number, e: any) => sum + e.amount, 0);
   365→  }, [ledgerEntries]);
   366→
   367→  const scPaid = tenantServiceCharges.filter((sc: any) => !sc.isMinimumVend && sc.serviceChargeStatus === 'PAID_FULLY').length > 0;

   374→      <div>
   375→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 mb-6">
   376→          {[1, 2, 3].map(i => (

   379→              <div className="h-7 bg-slate-200 dark:bg-zinc-700 rounded w-32 mb-1" />
   380→              <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-16" />
   381→            </div>
   382→          ))}
   383→        </div>
   384→        <div className="bg-white dark:bg-zinc-800 rounded-xl border border-slate-200 dark:border-zinc-700 p-6 animate-pulse">
   385→          <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-full mb-3" />
   386→          <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-3/4 mb-3" />
   387→          <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-1/2" />
   388→        </div>
   389→      </div>
   390→    );
   391→  }
   392→
   393→  const hasLedgerData = ledgerEntries && ledgerEntries.length > 0;
   394→  const hasServiceCharges = tenantServiceCharges.length > 0;
   395→
   396→  return (
   397→    <div>
   398→      <h3 className="text-lg font-bold text-slate-900 dark:text-white mb-1">Payment Ledger</h3>
   399→      <p className="text-sm text-slate-500 dark:text-zinc-400 mb-6">
   400→        View your rent, Service Charge (SC), and Minimum Vend (MV) obligations and payment status.
   401→      </p>
   402→
   403→      {/* Summary Cards */}
   404→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 mb-6">
   405→        <div className="bg-white dark:bg-zinc-800 rounded-xl p-4 border border-slate-200 dark:border-zinc-700">
   406→          <p className="text-[10px] font-bold uppercase tracking-widest text-slate-400 dark:text-zinc-500 mb-1">Current Month SC</p>
   407→          <p className="text-2xl font-bold text-emerald-600 dark:text-emerald-400">
   408→            {formatNaira(currentMonthSC)}
   409→          </p>
