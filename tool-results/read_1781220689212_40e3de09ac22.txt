   102→// ─── File Upload Helper ──────────────────────────────────────────────────────
   103→async function uploadFilesToConvex(files: File[]): Promise<string[]> {
   104→  const storageIds: string[] = [];
   105→  for (const file of files) {
   106→    const postUrl = await fetch('/api/storage/upload-url');
   107→    // Fallback: use the Convex action for upload URL
   108→    // We'll use the Convex generateUploadUrl mutation via the hook instead
   109→    // For now, we'll use the standard Convex file upload flow
   110→    const { url, storageId } = await (async () => {
   111→      // Use Convex's built-in file upload
   112→      const uploadUrlRes = await fetch(`${window.location.origin}/api/storage/upload-url`);
   113→      if (uploadUrlRes.ok) {
   114→        const { url: u, storageId: s } = await uploadUrlRes.json();
   115→        return { url: u, storageId: s };
   116→      }
   117→      throw new Error('Failed to get upload URL');
   118→    })();
   119→
   120→    const uploadRes = await fetch(url, {
   121→      method: 'POST',
   122→      body: file,
   123→    });
   124→    if (uploadRes.ok) {
   125→      const { storageId: sid } = await uploadRes.json();
   126→      storageIds.push(sid);
   127→    }
   128→  }
   129→  return storageIds;
   130→}

   318→const LedgerTab: React.FC<{ tenantInfo: any }> = ({ tenantInfo }) => {
   319→  const { currentUser } = useAuth();

   321→  const userId = currentUser?.id || '';

   323→  const resolvedTenantId = tenantInfo?.tenantId || userId;

   325→  // Fetch ledger entries from Convex using the resolved tenant ID
   326→  const ledgerEntries = useQuery(
   327→    api.portals.getTenantLedger,
   328→    firmId && resolvedTenantId ? { firmId, tenantId: resolvedTenantId } : 'skip'
   329→  );
   330→
   331→  // Fetch service charges from Convex to compute current SC/MV
   332→  const serviceCharges = useQuery(
   333→    api.sentry.getServiceChargesByFirm,
   334→    firmId ? { firmId } : 'skip'
   335→  );
   336→
   337→  const isLoading = ledgerEntries === undefined || serviceCharges === undefined;
   338→
   339→  // Compute tenant-specific service charges
   340→  const tenantServiceCharges = useMemo(() => {
   341→    if (!serviceCharges || !resolvedTenantId) return [];
   342→    return serviceCharges.filter((sc: any) => sc.tenantId === resolvedTenantId);
   343→  }, [serviceCharges, resolvedTenantId]);
   344→
   345→  // Summary calculations
   346→  const currentMonthSC = useMemo(() => {
   347→    return tenantServiceCharges
   348→      .filter((sc: any) => !sc.isMinimumVend)
   349→      .reduce((sum: number, sc: any) => sum + (sc.outstandingBalance ?? sc.amount), 0);
   350→  }, [tenantServiceCharges]);
   351→
   352→  const currentMonthMV = useMemo(() => {
   353→    return tenantServiceCharges
   354→      .filter((sc: any) => sc.isMinimumVend)
   355→      .reduce((sum: number, sc: any) => sum + (sc.outstandingBalance ?? sc.amount), 0);
   356→  }, [tenantServiceCharges]);
   357→
   358→  const outstandingBalance = useMemo(() => {
   359→    if (!ledgerEntries) return 0;

   361→      .filter((e: any) => e.status === 'pending' || e.status === 'defaulted')
   362→      .reduce((sum: number, e: any) => sum + e.amount, 0);
   363→  }, [ledgerEntries]);

   365→  const scPaid = tenantServiceCharges.filter((sc: any) => !sc.isMinimumVend && sc.serviceChargeStatus === 'PAID_FULLY').length > 0;

   372→      <div>
   373→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 mb-6">
   374→          {[1, 2, 3].map(i => (
   375→            <div key={i} className="bg-white dark:bg-zinc-800 rounded-xl p-4 border border-slate-200 dark:border-zinc-700 animate-pulse">
   376→              <div className="h-3 bg-slate-200 dark:bg-zinc-700 rounded w-24 mb-2" />
   377→              <div className="h-7 bg-slate-200 dark:bg-zinc-700 rounded w-32 mb-1" />
   378→              <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-16" />
   379→            </div>
   380→          ))}
   381→        </div>
   382→        <div className="bg-white dark:bg-zinc-800 rounded-xl border border-slate-200 dark:border-zinc-700 p-6 animate-pulse">
   383→          <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-full mb-3" />
   384→          <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-3/4 mb-3" />
   385→          <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-1/2" />
   386→        </div>
   387→      </div>
   388→    );
   389→  }
   390→
   391→  const hasLedgerData = ledgerEntries && ledgerEntries.length > 0;
   392→  const hasServiceCharges = tenantServiceCharges.length > 0;
   393→
   394→  return (
   395→    <div>
   396→      <h3 className="text-lg font-bold text-slate-900 dark:text-white mb-1">Payment Ledger</h3>
   397→      <p className="text-sm text-slate-500 dark:text-zinc-400 mb-6">
   398→        View your rent, Service Charge (SC), and Minimum Vend (MV) obligations and payment status.
   399→      </p>
   400→
   401→      {/* Summary Cards */}
   402→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 mb-6">
   403→        <div className="bg-white dark:bg-zinc-800 rounded-xl p-4 border border-slate-200 dark:border-zinc-700">
   404→          <p className="text-[10px] font-bold uppercase tracking-widest text-slate-400 dark:text-zinc-500 mb-1">Current Month SC</p>
   405→          <p className="text-2xl font-bold text-emerald-600 dark:text-emerald-400">
   406→            {formatNaira(currentMonthSC)}
   407→          </p>
   408→          {hasServiceCharges && !currentMonthSC ? (
   409→            <span className="inline-flex items-center gap-1 mt-1 text-[10px] font-bold text-emerald-600 dark:text-emerald-400 bg-emerald-50 dark:bg-emerald-900/20 px-2 py-0.5 rounded-full">
