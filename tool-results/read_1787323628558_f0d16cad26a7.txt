    93→// Wrapper that gives AIConsentModal access to React hooks (recordConsent mutation)
    94→const AIConsentModalWrapper: React.FC<{ modalContext: any; closeModal: () => void }> = ({ modalContext, closeModal }) => {
    95→  const recordConsent = useMutation(api.myFunctions.recordConsent);
    96→  const { currentUser } = useAuth();
    97→
    98→  const handleAccept = async () => {
    99→    localStorage.setItem('practicepro_ai_consent', 'true');
   100→    // Persist consent to DB for audit trail
   101→    if (currentUser?.email) {
   102→      try {
   103→        await recordConsent({
   104→          email: currentUser.email,
   105→          consentType: 'ai_processing',
   106→          granted: true,
   107→          ipHint: Intl.DateTimeFormat().resolvedOptions().timeZone,
   108→        });
   109→      } catch (e) {
   110→        console.warn('[Compliance] Failed to persist AI consent to DB:', e);
   111→      }
   112→    }
   113→    closeModal();
   114→    if (modalContext?.onConsent) modalContext.onConsent();
   115→  };
   116→
   117→  const handleDecline = async () => {
   118→    localStorage.setItem('practicepro_ai_consent', 'false');
   119→    if (currentUser?.email) {
   120→      try {
   121→        await recordConsent({
   122→          email: currentUser.email,
   123→          consentType: 'ai_processing',
   124→          granted: false,
   125→          ipHint: Intl.DateTimeFormat().resolvedOptions().timeZone,
   126→        });
   127→      } catch (e) {
   128→        console.warn('[Compliance] Failed to persist AI decline to DB:', e);
   129→      }
   130→    }
   131→    closeModal();
   132→    if (modalContext?.onDecline) modalContext.onDecline();
   133→  };
   134→
   135→  return <AIConsentModal onAccept={handleAccept} onDecline={handleDecline} />;
   136→};

   138→// Record Rent Payment — calls the Atrium ledger directly from the cog menu
   139→const RecordRentPaymentModalWrapper: React.FC<{ modalContext: any; closeModal: () => void }> = ({ modalContext, closeModal }) => {
   140→  const addEntry = useMutation(api.sentry.addLedgerEntry);
   141→  const { coreState } = useCoreState();
   142→  const { updateItem } = useDataActions();
   143→  const { addToast } = useUI();
   144→  const isProperty = useIsProperty();
   145→  const firmId = coreState.firmDetails?.id || '';
   146→  const [amount, setAmount] = React.useState(modalContext?.rentAmount ? String(modalContext.rentAmount) : '');
   147→  const [status, setStatus] = React.useState<'cleared' | 'pending' | 'defaulted'>('cleared');
   148→  const [loading, setLoading] = React.useState(false);
   149→  const unitId = modalContext?.unitId || '';
   150→  const unitName = modalContext?.unitName || 'Unit';
   151→  const tenantName = modalContext?.tenantName || '';
   152→
   153→  const handleSubmit = async (e: React.FormEvent) => {
   154→    e.preventDefault();
   155→    if (!unitId || !amount) return;
   156→    const amountNum = parseFloat(amount);
   157→    if (Number.isNaN(amountNum) || amountNum <= 0) return;
   158→    setLoading(true);
   159→    try {
   160→      const propertyRecord = (coreState.properties || []).find(p => p.id === unitId);
   161→      const today = new Date().toISOString().split('T')[0];
   162→
   163→      await addEntry({
   164→        firmId,
   165→        propertyId: propertyRecord?.id || unitId,
   166→        unitId,
   167→        amount: amountNum,
   168→        type: 'rent',
   169→        status,
   170→        channel: 'Bank Transfer',
   171→        description: `Rent payment${tenantName ? ' from ' + tenantName : ''}`,
   172→        paymentRef: '',
   173→      });
   174→
   175→      if (propertyRecord) {
   176→        const newPayment = {
   177→          id: `pay_${Date.now()}`,
   178→          dueDate: today,
   179→          paidDate: today,
   180→          amount: amountNum,
   181→          status: 'paid' as const,
   182→          paymentMethod: 'Bank Transfer',
   183→          receiptNumber: `REC-${Date.now().toString().slice(-6)}-${Math.floor(1000 + Math.random() * 9000)}`,
   184→        };
   185→        await updateItem('properties', {
   186→          ...propertyRecord,
   187→          id: propertyRecord.id,
   188→          rentPaymentHistory: [newPayment, ...(propertyRecord.rentPaymentHistory || [])],
   189→          status: propertyRecord.status === 'Vacant' ? 'Occupied' : propertyRecord.status,
   190→        }, 'Property Payment');
   191→      }
   192→
   193→      addToast(`Rent payment recorded for ${unitName}`, { type: 'success' });
   194→      closeModal();
   195→    } catch (err) {
   196→      addToast('Failed to record payment', { type: 'error' });
   197→    } finally { setLoading(false); }
   198→  };
   199→
   200→  return (
   201→    <div className="p-6 space-y-5">
   202→      <div>
   203→        <h3 className="text-lg font-bold text-slate-900 dark:text-white">Record Rent Payment</h3>
   204→        <p className="text-xs text-slate-500 mt-0.5">
   205→          {unitName}{tenantName ? ` • Tenant: ${tenantName}` : ''}
   206→        </p>
   207→        {/* Ledger vs Invoice explainer */}
   208→        <div className="mt-3 p-3 rounded-lg bg-emerald-50 dark:bg-emerald-950/40 border border-emerald-100">
   209→          <p className="text-2xs text-emerald-700 dark:text-emerald-300 leading-relaxed">
   210→            <strong>Atrium Ledger</strong> tracks rent & service charges for this property.
   211→            Formal <strong>Invoices</strong> (for {isProperty ? 'professional fees' : 'professional/legal fees'}) are managed separately in Billing.
   212→          </p>
   213→        </div>
   214→      </div>
   215→      <form onSubmit={handleSubmit} className="space-y-4">
   216→        <div>
   217→          <label className="block text-xs font-semibold text-slate-600 dark:text-zinc-400 mb-1.5">Amount</label>
   218→          <div className="relative">
   219→            <span className="absolute left-3 top-1/2 -translate-y-1/2 font-bold text-slate-400">₦</span>
   220→            <input type="number" value={amount} onChange={e => setAmount(e.target.value)} required min="0"
   221→              className="w-full pl-8 pr-3 py-2.5 border border-slate-200 dark:border-zinc-700 rounded-lg bg-white dark:bg-zinc-900 text-sm text-slate-900 dark:text-white focus:ring-2 focus:ring-emerald-500 focus:border-transparent"
   222→              placeholder="0.00" />
   223→          </div>
   224→        </div>
   225→        <div>
   226→          <label className="block text-xs font-semibold text-slate-600 dark:text-zinc-400 mb-1.5">Did they pay?</label>
   227→          <div className="grid grid-cols-3 gap-2">
   228→            {([
   229→              ['cleared', 'Yes, paid', 'border-emerald-500 bg-emerald-50 text-emerald-700', <CheckCircle2 className="w-3.5 h-3.5" />], 
   230→              ['pending', 'Pending', 'border-amber-500 bg-amber-50 text-amber-700', <Clock className="w-3.5 h-3.5" />], 
   231→              ['defaulted', 'Defaulted', 'border-rose-500 bg-rose-50 text-rose-700', <XCircle className="w-3.5 h-3.5" />]
   232→            ] as const).map(([val, label, activeClass, icon]) => (
   233→              <button key={val as string} type="button" onClick={() => setStatus(val as any)}
   234→                className={`py-2 rounded-lg border text-2xs font-bold transition-all flex items-center justify-center gap-1.5 ${
   235→                  status === val ? activeClass : 'bg-slate-50 dark:bg-zinc-900 border-slate-200 dark:border-zinc-700 text-slate-500 hover:border-slate-400'
   236→                }`}>{icon} {label as string}</button>
   237→            ))}
   238→          </div>
   239→        </div>
   240→        <div className="flex gap-3 pt-2">
   241→          <button type="button" onClick={closeModal} className="flex-1 py-2.5 border border-slate-200 dark:border-zinc-700 text-slate-600 dark:text-zinc-400 rounded-lg text-sm font-semibold hover:bg-slate-50 dark:hover:bg-zinc-800 dark:bg-zinc-900 transition-colors">Cancel</button>
   242→          <button type="submit" disabled={loading} className="flex-1 py-2.5 bg-emerald-600 text-white rounded-lg text-sm font-bold hover:bg-emerald-500 transition-colors disabled:opacity-50">
   243→            {loading ? 'Saving...' : 'Record Payment'}
   244→          </button>
   245→        </div>
   246→      </form>
   247→    </div>
   248→  );
   249→};

   868→      content = <NewDirectMessageForm users={coreState.users} onClose={closeModal} />;
   869→      break;
   870→    case 'deleteConfirmation':
   871→      content = <ConfirmationModal title={modalContext.title} message={modalContext.message} onConfirm={modalContext.onConfirm} onCancel={closeModal} confirmText={modalContext.confirmText} confirmButtonClass={modalContext.confirmButtonClass} onConfirmArchive={modalContext.onConfirmArchive} archiveText={modalContext.archiveText} verificationText={modalContext.verificationText} requiresPassword={modalContext.requiresPassword} />;
   872→      break;
   873→    case 'noTeamMembers':
   874→      content = <NoTeamMembersModal onNavigate={() => { closeModal(); navigateTo('settings', null, { settingsTargetId: 'user-management' }); }} onClose={closeModal} />;
   875→      break;
   876→    case 'aloaHelp':
   877→      content = <AloaHelpModal />;
   878→      break;
   879→    case 'feedback':
   880→      content = <FeedbackForm />;
   881→      break;
   882→    case 'matterIngestion':
   883→      content = <MatterIngestionWizard />;
   884→      break;
   885→    case 'joinFirm':
   886→      content = <JoinFirmModal onClose={closeModal} />;
   887→      break;
   888→    case 'newNotebook':
   889→    case 'editNotebook': {
   890→      const nb = coreState.noteNotebooks.find(n => n.id === editingId);
   891→      content = <NotebookForm notebookToEdit={nb} onAdd={(n) => dataHandlers.addItem('noteNotebooks', n, n.name)} onUpdate={(n) => dataHandlers.updateItem('noteNotebooks', n, n.name)} onClose={closeModal} currentUser={currentUser!} appMode={appMode} />;
   892→      break;
   893→    }
   894→    case 'newPage': {
   895→      content = (
   896→        <NotePageForm
   897→          onAdd={(p) => dataHandlers.addItem('notePages', p, p.title)}
   898→          onClose={closeModal}
   899→          initialContext={modalContext}
   900→        />
   901→      );
   902→      break;
   903→    }
   904→    case 'requestFinancialDocument': {
   905→      const m = matterState.matters.find(x => x.id === editingId);
   906→      if (m) content = <RequestFinancialDocumentForm matter={m} />;
   907→      break;
   908→    }
   909→    case 'stageChecklist': {
   910→      const m = matterState.matters.find(x => x.id === editingId);
   911→      if (m && modalContext) content = <StageChecklistForm matter={m} stage={modalContext.stage} workflow={modalContext.workflow} checklistTemplates={coreState.checklistTemplates} onClose={closeModal} />;
   912→      break;
   913→    }
   914→    case 'newExternalCounsel': {
   915→      content = <ExternalCounselInviteForm matterId={editingId!} onInvite={dataHandlers.handleInviteExternalCounsel} onClose={closeModal} />;
   916→      break;
   917→    }
   918→    case 'upgradePlan': {
   919→      content = <UpgradeModal
   920→        featureName={modalContext?.featureName}
   921→        targetPlan={modalContext?.targetPlan} // Pass target plan to modal
   922→        onUpgrade={() => { closeModal(); navigateTo('settings', null, { settingsTargetId: 'subscription-management' }); }}
   923→        onClose={closeModal}
   924→      />;
   925→      break;
   926→    }
   927→    case 'keyboardShortcuts':
   928→      content = <KeyboardShortcutsModal />;
   929→      break;
   930→    case 'quickLook':
   931→      content = <QuickLookModal />;
   932→      break;
   933→    case 'saveToNote':
   934→      content = <SaveToNoteForm initialContent={modalContext?.content || ''} onClose={closeModal} />;
   935→      break;
   936→    case 'newResearchNotebook':
   937→      content = (
   938→        <NewResearchNotebookForm
   939→          matters={matterState.matters}
   940→          onSubmit={async (data) => {
   941→            const { name, matterId, files } = data;
   942→
   943→            const notebook = dataHandlers.handleAddResearchNotebook({
   944→              firmId: coreState.firmDetails.id,
   945→              userId: currentUser!.id,
   946→              name,
   947→              matterId
   948→            });
   949→
   950→            if (files.length > 0) {
   951→              for (const file of files) {
   952→                const reader = new FileReader();
   953→                await new Promise<void>((resolve) => {
   954→                  reader.onload = () => {
   955→                    const fileDetails: FileDetails = {
   956→                      name: file.name,
   957→                      type: file.type,
   958→                      size: file.size,
   959→                      filePath: `research/${notebook.id}/${file.name}`,
   960→                      dataUrl: reader.result as string
   961→                    };
   962→
   963→                    dataHandlers.handleAddResearchSource(notebook.id, {
   964→                      name: file.name,
   965→                      type: file.type === 'application/pdf' ? 'pdf' : 'text',
   966→                      content: '',
   967→                      file: fileDetails
   968→                    });
   969→                    resolve();
   970→                  };
   971→                  reader.readAsDataURL(file);
   972→                });
   973→              }
   974→            }
   975→
   976→            if (updateCurrentHistoryEntry) {
   977→              updateCurrentHistoryEntry({ selectedResearchNotebookId: notebook.id });
   978→            }
   979→            closeModal();
   980→          }}
   981→          onClose={closeModal}
   982→        />
   983→      );
   984→      break;
   985→    case 'addResearchSource':
   986→      content = <AddResearchSourceModal notebookId={modalContext?.notebookId} onAdd={dataHandlers.handleAddResearchSource} onClose={closeModal} />;
   987→      break;
   988→    case 'addCaseToNotebook':
   989→      content = <AddCaseToNotebookModal caseData={modalContext?.caseData} notebooks={documentState.researchNotebooks.filter(nb => nb.userId === currentUser?.id)} matters={matterState.matters} onAdd={(nbId, cData) => dataHandlers.handleAddResearchSource(nbId, { name: cData.parties, type: 'text', content: JSON.stringify(cData) })} onCreateNotebook={dataHandlers.handleAddResearchNotebook} onClose={closeModal} />;
   990→      break;
   991→    case 'workspaceSetup':
   992→      content = <WorkspaceSetupModal onSuccess={() => { }} pendingAction={modalContext?.pendingAction} onClose={closeModal} />;
   993→      break;
   994→    case 'newLead':
   995→      content = <LeadForm onClose={closeModal} initialContext={modalContext?.fields || modalContext} />;
   996→      break;
   997→    case 'sendIntakeLink': {
   998→      content = <SendIntakeLinkModal />;
   999→      break;
  1000→    }
  1001→    case 'activateLead': {
  1002→      const lead = coreState.leads?.find(l => l.id === editingId);
  1003→      if (lead) {
  1004→        content = <ConfirmationModal
  1005→          title="Convert Lead to Contact?"
  1006→          message={`Are you sure you want to convert "${lead.name}" from a lead to a full contact? This will create a new Contact record and remove them from your Lead pipeline.`}
  1007→          confirmText="Convert to Contact"
  1008→          confirmButtonClass="bg-primary-600 hover:bg-primary-700"
  1009→          onConfirm={() => {
  1010→            dataHandlers.handleAddLead({ name: lead.name, email: lead.email || '' }, false);
  1011→            closeModal();
  1012→          }}
  1013→          onCancel={closeModal}
  1014→        />;
  1015→      }
  1016→      break;
  1017→    }
  1018→    case 'sendPostActivationEmail':
  1019→      content = <SendPostActivationEmailModal />;
  1020→      break;
  1021→    case 'composeEmail':
  1022→      content = <ComposeEmailModal onClose={closeModal} initialContext={modalContext} />;
  1023→      break;
  1024→    case 'demoUpsell':
  1025→      content = <DemoUpsellModal context={modalContext?.context} onSignup={() => openModal('signup')} onClose={closeModal} />;
  1026→      break;
  1027→    case 'leadCapture':
  1028→      content = <LeadCaptureModal />;
  1029→      break;
  1030→    case 'aiConsent':
  1031→      content = <AIConsentModalWrapper modalContext={modalContext} closeModal={closeModal} />;
  1032→      break;
  1033→
  1034→    default:
  1035→      content = <div className="p-4">Modal content for {modal} not found.</div>;
  1036→  }
  1037→
  1038→  if (!content) return null;
  1039→
  1040→  let modalTitle = getModalTitle(modal);
  1041→  if ((modal === 'newTask' || modal === 'editTask') && editingId) {
  1042→    modalTitle = 'Edit Task';
  1043→  }
  1044→  if (modal === 'newDraft' || modal === 'newDocument' || modal === 'editDocument') {
  1045→    const doc = documentState.documents.find(d => d.id === editingId);
  1046→    const ctx = modalContext?.fields || modalContext;
  1047→    const name = doc?.title || ctx?.draftTitle || ctx?.title || 'New Document';
  1048→    modalTitle = modal === 'editDocument' ? `Edit Document: ${name}` : (modal === 'newDraft' ? `Draft: ${name}` : 'New Document');
  1049→  }
  1050→  if (modal === 'batchUpload') {
  1051→    modalTitle = 'Uploading Files';
  1052→  }
  1053→  if (modal === 'composeEmail') {
  1054→    modalTitle = 'Compose Email';
  1055→  }
  1056→  // Product-aware title: 'Link Matter to Contact' → 'Link Property to Contact' for Atrium
  1057→  if (modal === 'linkMatterToContact') {
  1058→    modalTitle = `Link ${terminology.matter} to ${terminology.client}`;
  1059→  }
  1060→  // Product-aware title: 'ARIA Help' → 'ALOA Help' for Vega/Komplete
  1061→  if (modal === 'aloaHelp') {
  1062→    modalTitle = 'AI Assistant Help';
  1063→  }
  1064→  // Missing titles in getModalTitle switch
  1065→  if (modal === 'workspaceSetup') {
  1066→    modalTitle = 'Workspace Setup';
  1067→  }
  1068→  if (modal === 'demoUpsell') {
  1069→    modalTitle = 'Upgrade to PracticePro';
  1070→  }
  1071→
  1072→  // Size mapping based on form complexity:
  1073→  //  lg → multi-section forms (contacts, properties, invoices, events, matters, matter ingest, email)
  1074→  //  md → default (documents, tasks, most forms)
  1075→  //  sm → simple confirmations and tiny forms
  1076→  const lgModals = ['matterIngestion', 'editMatter', 'newMatter', 'composeEmail',
  1077→           'newContact', 'editContact', 'newProperty', 'editProperty',
  1078→           'newInvoice', 'editInvoice', 'viewInvoice', 'generateInvoice',
  1079→           'newEvent', 'editEvent', 'bulkEditProperty'];
  1080→  const smModals = ['leadCapture', 'deleteConfirmation', 'aiConsent', 'keyboardShortcuts',
  1081→           'noTeamMembers', 'aloaHelp', 'upgradePlan', 'sendIntakeLink',
  1082→           'sendPostActivationEmail', 'newPage', 'newNotebook', 'newResearchNotebook',
  1083→           'addResearchSource', 'addCaseToNotebook', 'mergeContact', 'linkContact',
  1084→           'linkContactToMatter', 'linkMatterToContact', 'closeMatter', 'archiveMatter',
  1085→           'workspaceSetup', 'joinFirm', 'folderPermissions', 'signDocument',
  1086→           'demoUpsell', 'googleDrivePicker'];
  1087→  const modalSize = lgModals.includes(modal) ? 'lg'
  1088→    : smModals.includes(modal) ? 'sm'
  1089→    : 'md';
  1090→
  1091→  return (
  1092→    <Modal isOpen={!!modal} onClose={() => closeModal()} title={modalTitle} size={modalSize} hideHeader={modal === 'deleteConfirmation'}>
  1093→      {content}
  1094→    </Modal>
  1095→  );
  1096→};
  1097→
  1098→export default ModalManager;
