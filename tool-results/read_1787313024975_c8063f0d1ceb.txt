    16→export const sendHeartbeat = mutation({
    17→  args: { firmId: v.string(), userId: v.string(), userName: v.string(), userEmail: v.optional(v.string()) },
    18→  handler: async (ctx, args) => {
    19→    // SECURITY: Verify the caller belongs to the firm they're sending presence for.
    20→    // Skip the firmId match check when auth.firmId is empty (legacy call without userEmail).
    21→    const auth = await requireFirmUser(ctx, args.userEmail);
    22→    if (auth.firmId && auth.firmId !== args.firmId) {
    23→      throw new Error("Unauthorized. firmId does not match your session.");
    24→    }
    25→
    26→    const existing = await ctx.db
    27→      .query("presence")
    28→      .withIndex("by_user", (q) => q.eq("userId", args.userId))
    29→      .first();
    30→
    31→    if (existing) {
    32→      await ctx.db.patch(existing._id, { updatedAt: Date.now(), userName: args.userName, firmId: args.firmId });
    33→    } else {
    34→      await ctx.db.insert("presence", { firmId: args.firmId, userId: args.userId, userName: args.userName, updatedAt: Date.now() });
    35→    }
    36→  },
    37→});

    39→export const getActivePeers = query({
    40→  args: { firmId: v.optional(v.string()), userEmail: v.optional(v.string()) },
    41→  handler: async (ctx, args) => {
    42→    // SECURITY: Verify the caller belongs to the firm they're querying
    43→    if (args.userEmail) {
    44→      try {
    45→        const auth = await requireFirmUser(ctx, args.userEmail);
    46→        if (auth.firmId && auth.firmId !== args.firmId) return [];
    47→      } catch { return []; }
    48→    }
    49→    if (!args.firmId) return [];
    50→
    51→    // Fetch ALL presence records for the firm (not just active ones).
    52→    // We return both active and inactive users with their lastSeen timestamp
    53→    // so the frontend can show an inactivity indicator (greyed out) and
    54→    // last-seen time on hover.
    55→    const allPresence = await ctx.db.query("presence").withIndex("by_firm", (q) => q.eq("firmId", args.firmId)).take(100);
    56→
    57→    // Fetch users to check their visibility preferences
    58→    const users: any[] = [];
    59→    for (const p of allPresence) {
    60→      try {
    61→        const user = await ctx.db.get(p.userId as any);
    62→        if (user) users.push(user);
    63→      } catch (e) { /* skip invalid IDs */ }
    64→    }
    65→
    66→    // Build a set of user IDs that should be hidden from presence
    67→    const hiddenUserIds = new Set<string>();
    68→    for (const user of users) {
    69→      if (user.role === 'Tenant' && user.portalPresenceHidden !== false) {
    70→        hiddenUserIds.add(user._id.toString());
    71→      }
    72→      if (user.portalPresenceHidden === true) {
    73→        hiddenUserIds.add(user._id.toString());
    74→      }
    75→    }
    76→
    77→    // Return rich presence data: userId, updatedAt (last seen), and isOnline
    78→    // Active = heartbeat within the last 60 seconds
    79→    const now = Date.now();
    80→    const ACTIVE_THRESHOLD = 60 * 1000;
    81→    return allPresence
    82→      .filter((p: any) => !hiddenUserIds.has(p.userId))
    83→      .map((p: any) => ({
    84→        userId: p.userId,
    85→        updatedAt: p.updatedAt,
    86→        isOnline: (now - p.updatedAt) < ACTIVE_THRESHOLD,
    87→      }));
    88→  },
    89→});

    93→export const diagnoseConnectivity = mutation({
    94→  args: { email: v.string(), userEmail: v.optional(v.string()) },
    95→  handler: async (ctx, args) => {
    96→    // SECURITY: Only authenticated users can diagnose their own account
    97→    let authenticatedEmail: string | undefined;
    98→    try {
    99→      const identity = await ctx.auth.getUserIdentity();
   100→      if (identity) authenticatedEmail = identity.email || undefined;
   101→    } catch {}
   102→
   103→    // If the caller is authenticated, they can only diagnose their own email
   104→    // (unless no auth context exists — legacy fallback for repair flows)
   105→    if (authenticatedEmail && args.userEmail) {
   106→      const auth = await requireFirmUser(ctx, args.userEmail);
   107→      // Verify the email being diagnosed belongs to the authenticated user
   108→      if (auth.user?.tokenIdentifier?.toLowerCase() !== args.email.toLowerCase().trim()) {
   109→        throw new Error("Unauthorized. You can only diagnose your own account.");
   110→      }
   111→    }
   112→
   113→    const emailInput = args.email.toLowerCase().trim();
   114→
   115→    const allUsers = await ctx.db.query("users").take(500);
   116→    const user = allUsers.find((u: any) => u.tokenIdentifier && u.tokenIdentifier.toLowerCase() === emailInput);
   117→
   118→    const diagnosis = {
   119→      emailSearched: emailInput,
   120→      userFound: !!user,
   121→      userId: user?._id || null,
   122→      currentFirmId: user?.firmId || null,
   123→      firmFound: false,
   124→      availableFirms: [] as any[]
   125→    };
   126→
   127→    if (user && user.firmId) {
   128→      const firm = await ctx.db.get(user.firmId as any) as any;
   129→      if (firm) {
   130→        diagnosis.firmFound = true;
   131→        diagnosis.availableFirms.push({
   132→          id: firm._id,
   133→          name: firm.name,
   134→          status: 'Linked',
   135→          createdAt: firm._creationTime
   136→        });
   137→      }
   138→    }
   139→
   140→    const allFirms = await ctx.db.query("firms").take(500);
   141→    const joinedIds = user?.joinedFirmIds || [];
   142→    
   143→    const availableFirmsFiltered = allFirms.filter((f: any) => {
   144→      // 1. Created by this user
   145→      const creatorIdMatch = f.createdBy === user?._id || f.createdBy === user?.tokenIdentifier;
   146→      const settingsEmailMatch = f.settings && f.settings.creatorEmail && f.settings.creatorEmail.toLowerCase() === emailInput;
   147→      
   148→      // 2. Already joined - check the persistence list
   149→      const previouslyJoined = joinedIds.includes(f._id);
   150→      
   151→      // 3. Current firm
   152→      const isCurrentlyIn = user && user.firmId === f._id;
   153→      
   154→      // We want to show firms NOT currently active but where user has access
   155→      return (creatorIdMatch || settingsEmailMatch || previouslyJoined || isCurrentlyIn) && f._id !== diagnosis.currentFirmId;
   156→    });
   157→
   158→    availableFirmsFiltered.forEach(f => {
   159→      diagnosis.availableFirms.push({
   160→        id: f._id,
   161→        name: f.name,
   162→        status: 'Available',
   163→        createdAt: f._creationTime
   164→      });
   165→    });
   166→
   167→    if (!diagnosis.firmFound && diagnosis.availableFirms.length > 0) {
   168→      diagnosis.firmFound = true;
   169→    }
   170→
   171→    return diagnosis;
   172→  }
   173→});

   176→export const repairAccountConnection = mutation({
   177→  args: { email: v.string(), targetFirmId: v.optional(v.string()), userEmail: v.optional(v.string()) },
   178→  handler: async (ctx, args) => {
   179→    // SECURITY: Only the account owner can repair their own connection
   180→    let authenticatedEmail: string | undefined;
   181→    try {
   182→      const identity = await ctx.auth.getUserIdentity();
   183→      if (identity) authenticatedEmail = identity.email || undefined;
   184→    } catch {}
   185→
   186→    if (authenticatedEmail && args.userEmail) {
   187→      try {
   188→        const auth = await requireFirmUser(ctx, args.userEmail);
   189→        if (auth.user?.tokenIdentifier?.toLowerCase() !== args.email.toLowerCase().trim()) {
   190→          throw new Error("Unauthorized. You can only repair your own account.");
   191→        }
   192→      } catch (e: any) {
   193→        if (e.message?.includes('Unauthorized')) throw e;
   194→        // If requireFirmUser fails (e.g. user has no firm yet), still allow self-repair
   195→        // but only if the authenticated email matches
   196→        if (authenticatedEmail.toLowerCase() !== args.email.toLowerCase().trim()) {
   197→          throw new Error("Unauthorized. You can only repair your own account.");
   198→        }
   199→      }
   200→    }
   201→
   202→    const emailInput = args.email.toLowerCase().trim();
   203→
   204→    const allUsers = await ctx.db.query("users").take(500);
   205→    const user = allUsers.find((u: any) => u.tokenIdentifier && u.tokenIdentifier.toLowerCase() === emailInput);
   206→
   207→    if (!user) {
   208→      return { success: false, code: 'USER_NOT_FOUND', message: "User account not found." };
   209→    }
   210→
   211→    if (args.targetFirmId) {
   212→      const firm = await ctx.db.get(args.targetFirmId as any) as any;
   213→      if (!firm) return { success: false, message: "Target firm does not exist." };
   214→
   215→      const joinedIds = user.joinedFirmIds || [];
   216→      const updatedJoinedIds = joinedIds.includes(args.targetFirmId) ? joinedIds : [...joinedIds, args.targetFirmId];
   217→
   218→      await ctx.db.patch(user._id, { 
   219→        firmId: args.targetFirmId, 
   220→        role: 'Admin', 
   221→        onboardingCompleted: true,
   222→        joinedFirmIds: updatedJoinedIds 
   223→      });
   224→      return { success: true, code: 'LINK_RESTORED', firmId: args.targetFirmId, message: `Successfully connected to ${firm.name}` };
   225→    }
   226→
   227→    return { success: false, code: 'MANUAL_SELECTION_REQUIRED', message: "Please select a specific firm to connect." };
   228→  }
   229→});

   237→export const getFirmData = query({
   238→  args: {
   239→    firmId: v.optional(v.string()),
   240→    userEmail: v.optional(v.string())
   241→  },
   242→  handler: async (ctx, args) => {
   243→    // RLS: Require firm user authentication. This prevents portal users
   244→    // and unauthenticated callers from reading the entire firm dataset.
   245→    const { firmId: authFirmId } = await requireFirmUser(ctx, args.userEmail);
   246→    let targetFirmId = args.firmId || authFirmId;
   247→    const userEmail = args.userEmail;
   248→
   249→    // Recovery logic: find firm from email if firmId is missing
   250→    if (!targetFirmId && userEmail) {
   251→      const allUsers = await ctx.db.query("users").collect();
   252→      const user = allUsers.find((u: any) => u.tokenIdentifier && u.tokenIdentifier.toLowerCase() === userEmail.toLowerCase());
   253→
   254→      if (user && user.firmId) {
   255→        targetFirmId = user.firmId;
   256→      } else {
   257→        const allFirms = await ctx.db.query("firms").take(500);
   258→        const foundFirm = allFirms.find((f: any) => {
   259→          const creator = f.createdBy || (f.settings && f.settings.creatorEmail);
   260→          return creator && creator.toLowerCase() === userEmail.toLowerCase();
   261→        });
   262→        if (foundFirm) targetFirmId = foundFirm._id;
   263→      }
   264→    }
   265→
   266→    if (!targetFirmId) {
   267→        console.warn(`[getFirmData] No firmId found for the given email. Returning null.`);
   268→        return null;
   269→    }
   270→
   271→    // CRITICAL: Normalize targetFirmId to a string to prevent type-mismatch in index queries
   272→    targetFirmId = targetFirmId.toString();
   273→
   274→    let firmDetails = null;
   275→    try { 
   276→        firmDetails = await ctx.db.get(targetFirmId as any); 
   277→    } catch (e) { 
   278→        console.error(`[getFirmData] Failed to fetch firmDetails for ${targetFirmId}:`, e);
   279→    }
   280→
   281→    if (!firmDetails) {
   282→        if (targetFirmId === 'demo_firm_id' || targetFirmId === 'atrium-demo-firm-id') {
   283→            firmDetails = {
   284→                _id: targetFirmId as any,
   285→                name: targetFirmId === 'atrium-demo-firm-id' ? "Atrium Demo Firm" : "Vega Demo Firm",
   286→                subscriptionPlan: "Enterprise",
   287→                product: targetFirmId === 'atrium-demo-firm-id' ? "atrium" : "vega",
   288→                aiSettings: { enableAllAiFeatures: true, enableJurisdictionalAnalysis: true }
   289→            } as any;
   290→        } else {
   291→            console.warn(`[getFirmData] Firm record missing for ID: ${targetFirmId}.`);
   292→            return null;
   293→        }
   294→    }
   295→
   296→    // Helper: fetch by firmId index with robust recovery for legacy/untagged data
   297→    const fetchByFirm = async (tableName: string) => {
   298→      try {
   299→        // 1. Primary Attempt: Use by_firm index for speed
   300→        const firmItems = await ctx.db.query(tableName as any)
   301→          .withIndex("by_firm", (q: any) => q.eq("firmId", targetFirmId))
   302→          .take(1000);
   303→
   304→        // 2. Secondary Attempt: Fetch 'system' tagged items
   305→        const systemItems = await ctx.db.query(tableName as any)
   306→          .withIndex("by_firm", (q: any) => q.eq("firmId", "system"))
   307→          .take(500);
   308→
   309→        let combined = [...firmItems, ...systemItems];
   310→
   311→        // 3. RECOVERY LOGIC: For specific tables, perform a scan to catch legacy/untagged data.
   312→        const RECOVERY_TABLES = ["eventTypes", "contactCategories", "documentCategories", "checklistTemplates", "notePages", "researchSources", "properties"];
   313→        if (RECOVERY_TABLES.includes(tableName)) {
   314→          const all = await ctx.db.query(tableName as any).take(1000);
   315→          const matches = all.filter((i: any) => {
   316→            const iFirmId = i.firmId?.toString();
   317→            const tFirmId = targetFirmId.toString();
   318→            return iFirmId === tFirmId || iFirmId === "system" || (tableName === "notePages" && i.matterId);
   319→          });
   320→          combined = [...combined, ...matches];
   321→        }
   322→
   323→        const uniqueResults = Array.from(new Map(combined.map(item => [item._id, item])).values());
   324→        if (tableName === "properties") {
   325→        }
   326→        return uniqueResults;
   327→      } catch (e) {
   328→        // Ultimate fallback: Full scan (safe for small firms)
   329→        try {
   330→          console.warn(`[getFirmData] Index failure for ${tableName}, falling back to scan.`);
   331→          const all = await ctx.db.query(tableName as any).take(1000);
   332→           return all.filter((i: any) => {
   333→             const iFirmId = i.firmId?.toString();
   334→             return iFirmId === targetFirmId || iFirmId === "system";
   335→           });
   336→        } catch (innerError) {
   337→          console.error(`FATAL fetch for ${tableName}:`, innerError);
   338→          return [];
   339→        }
   340→      }
   341→    };
   342→
   343→    // chatMessages: by conversation index (phase 2 - excluded from main query)
   344→    // firmActivity: excluded (phase 2, dedicated paginated query)
   345→    // researchMessages: excluded (phase 2, dedicated query)
   346→    // researchAnalysisResults: excluded (phase 2, low-usage)
   347→
   348→    // 1. Fetch primary tables (Firm-indexed)
   349→    const [
   350→      matters, contacts, tasks, documents, users, workflows, leads, notifications,
   351→      chatConversations, clientMessages, events, invoices, timeEntries, expenses,
   352→      noteNotebooks, researchNotebooks, eventTypes, contactCategories,

   369→      fetchByFirm("workflows"),
   370→      fetchByFirm("leads"),
   371→      fetchByFirm("notifications"),
   372→      fetchByFirm("chatConversations"),
   373→      fetchByFirm("clientMessages"),
   374→      fetchByFirm("events"),
   375→      fetchByFirm("invoices"),

   811→/**
   812→ * PHASE 2+3: Chat Messages — loaded per conversation, NOT firm-wide.
   813→ * Uses the existing by_conversation index which was always deployed.
   814→ * For the firm-wide unread count query, uses .filter() safely.
   815→ */
   816→export const getChatMessages = query({
   817→  args: {
   818→    conversationId: v.optional(v.string()),
   819→    firmId: v.optional(v.string()),
   820→  },
   821→  handler: async (ctx, args) => {
   822→    try {
   823→      if (!args.conversationId && !args.firmId) return [];
   824→
   825→      if (args.conversationId) {
   826→        // Most efficient: use the by_conversation index
   827→        // CRITICAL: Sort by _creationTime ASCENDING so messages render in
   828→        // chronological order (oldest first, newest at bottom). Without
   829→        // this, messages load out of order on refresh (e.g. Aug 14 messages
   830→        // appearing above Aug 13 messages).
   831→        return await ctx.db
   832→          .query("chatMessages")
   833→          .withIndex("by_conversation", (q) => q.eq("conversationId", args.conversationId as any))
   834→          .order("asc")
   835→          .collect();
   836→      }
   837→
   838→      const firmId = args.firmId;
   839→      if (!firmId) return [];
   840→
   841→      // 1. Try fetching by firmId index directly (fastest for new data)
   842→      const messagesByFirm = await ctx.db
   843→        .query("chatMessages")
   844→        .withIndex("by_firm", (q) => q.eq("firmId", firmId))
   845→        .order("desc")
   846→        .take(1000);
   847→
   848→      // 2. Fetch firm conversations to find legacy messages without firmId
   849→      const firmConvos = await ctx.db
   850→          .query("chatConversations")
   851→          .withIndex("by_firm", (q) => q.eq("firmId", firmId))
   852→          .take(300);
   853→          
   854→      const convoIds = new Set(firmConvos.map(c => c._id.toString()));
   855→
   856→      // 3. Pull recent messages and filter for legacy ones belonging to these conversations
   857→      const recentMessages = await ctx.db
   858→        .query("chatMessages")
   859→        .order("desc")
   860→        .take(1000);
   861→
   862→      const legacyMessages = recentMessages.filter(m => !m.firmId && m.conversationId && convoIds.has(m.conversationId.toString()));
   863→
   864→      // 4. Combine and return unique, sorted by creation time ascending
   865→      const combined = [...messagesByFirm, ...legacyMessages];
   866→      const unique = Array.from(new Map(combined.map(m => [m._id, m])).values());
   867→      // Sort by _creationTime ascending (oldest first) so the chat renders
   868→      // in chronological order with the newest message at the bottom.
   869→      return unique.sort((a: any, b: any) => (a._creationTime || 0) - (b._creationTime || 0));
   870→
   871→    } catch (e) {
   872→      console.error("getChatMessages error:", e);
   873→      return [];
   874→    }
   875→  },
   876→});

   995→export const getUser = query({
   996→  args: { tokenIdentifier: v.string(), preferPortalRole: v.optional(v.boolean()) },
   997→  handler: async (ctx, args) => {
   998→    try {
   999→      const token = args.tokenIdentifier;
  1000→      const preferPortalRole = args.preferPortalRole === true;
  1001→
  1002→      // 1. Primary: indexed lookup — collect ALL matching records, not just the first,
  1003→      //    so we can disambiguate when the same email exists as both an admin
  1004→      //    record AND a portal record (the "residents see admin dashboard" bug).
  1005→      //    Indexed lookups return rows in index order; .first() was returning
  1006→      //    whichever came first — usually the older Admin record.
  1007→      const directMatches = await ctx.db
  1008→        .query("users")
  1009→        .withIndex("by_token", (q) => q.eq("tokenIdentifier", token))
  1010→        .collect();
  1011→
  1012→      // 2. Case-insensitive indexed lookup
  1013→      const lowerMatches = directMatches.length === 0
  1014→        ? await ctx.db
  1015→            .query("users")
  1016→            .withIndex("by_token", (q) => q.eq("tokenIdentifier", token.toLowerCase()))
  1017→            .collect()
  1018→        : [];
  1019→
  1020→      // 3. Fallback: bounded scan — take only 500 to prevent timeout
  1021→      const allMatches = directMatches.length > 0 || lowerMatches.length > 0
  1022→        ? [...directMatches, ...lowerMatches]
  1023→        : (await ctx.db.query("users").take(500))
  1024→            .filter((u: any) =>
  1025→              u.tokenIdentifier &&
  1026→              u.tokenIdentifier.toLowerCase() === token.toLowerCase()
  1027→            );
  1028→
  1029→      if (allMatches.length === 0) return null;
  1030→
  1031→      // Pick the right record when duplicates exist. The "residents see admin
  1032→      // dashboard" bug occurred because the same email existed as BOTH an Admin
  1033→      // user record AND a Tenant/Client record, and .first() was returning the
  1034→      // Admin record when the user logged in via the portal.
  1035→      //
  1036→      // Resolution strategy:
  1037→      //   - If preferPortalRole is true (login via /portal/* route) AND a
  1038→      //     portal-role record exists, prefer it.
  1039→      //   - Otherwise, prefer the first record (preserves existing behavior
  1040→      //     for admin-side logins).
  1041→      //   - We also filter out 'Pending' records — those are revoked accounts
  1042→      //     that should never be the resolved user.
  1043→      const PORTAL_ROLES = new Set(["Client", "Tenant"]);
  1044→      const nonPending = allMatches.filter((u: any) => u.role !== "Pending");
  1045→
  1046→      // If everything is Pending, fall through and let the existing
  1047→      // revoked-user handling in AuthContext take over.
  1048→      const pool = nonPending.length > 0 ? nonPending : allMatches;
  1049→
  1050→      let user: any;
  1051→      if (preferPortalRole) {
  1052→        const portalRecord = pool.find((u: any) => PORTAL_ROLES.has(u.role));
  1053→        user = portalRecord || pool[0];
  1054→      } else {
  1055→        user = pool[0];
  1056→      }
  1057→
  1058→      // ============================================================
  1059→      // NDPA COMPLIANCE: Server-side privacy projection.
  1060→      // Strip ALL sensitive authentication fields before transmitting
  1061→      // to the frontend. These fields MUST NEVER leave the server.

  1091→/**
  1092→ * findDuplicateEmails — DIAGNOSTIC QUERY (admin only)
  1093→ *
  1094→ * Returns every email (tokenIdentifier) that has MORE THAN ONE user record.
  1095→ * Used to identify the data corruption that caused the "residents see admin
  1096→ * dashboard" bug: the same email existed as BOTH an Admin record AND a Tenant
  1097→ * record, and getUser() was returning whichever came first in the index.
  1098→ *
  1099→ * The query also flags each duplicate group with a `conflict` field describing
  1100→ * the combination of roles present (e.g. "Admin+Tenant", "Admin+Client",
  1101→ * "Lawyer+Tenant") so the admin can prioritize cleanup.
  1102→ *
  1103→ * Returns: Array of {
  1104→ *   email: string,
  1105→ *   count: number,
  1106→ *   roles: string[],                    // unique roles across all records
  1107→ *   conflict: string,                   // joined role names for quick scan
  1108→ *   records: Array<{ id, name, role, firmId, isVerified, product }>
  1109→ * }
  1110→ *
  1111→ * NOTE: This query is intentionally bounded by .take(2000) — sufficient for
  1112→ * thousands of users. If the user count exceeds this, increase the limit or
  1113→ * paginate. Disabled for non-admin callers (returns empty array).
  1114→ */
  1115→export const findDuplicateEmails = query({
  1116→  args: { requesterRole: v.optional(v.string()) },
  1117→  handler: async (ctx, args) => {
  1118→    // Only admins should be able to run this diagnostic. If the frontend
  1119→    // forgets to pass the role, return empty (defensive).
  1120→    if (args.requesterRole && args.requesterRole !== "Admin") return [];
  1121→
  1122→    const allUsers = await ctx.db.query("users").take(2000);
  1123→
  1124→    // Group by lowercased tokenIdentifier (email)
  1125→    const groups = new Map<string, any[]>();
  1126→    for (const u of allUsers as any[]) {
  1127→      const email = (u.tokenIdentifier || "").toLowerCase().trim();
  1128→      if (!email) continue;
  1129→      if (!groups.has(email)) groups.set(email, []);
  1130→      groups.get(email)!.push(u);
  1131→    }
  1132→
  1133→    const duplicates: Array<{
  1134→      email: string;
  1135→      count: number;
  1136→      roles: string[];
  1137→      conflict: string;
  1138→      records: Array<{ id: string; name: string; role: string; firmId: string | null; isVerified: boolean; product: string | null }>;
  1139→    }> = [];
  1140→
  1141→    for (const [email, records] of groups.entries()) {
  1142→      if (records.length < 2) continue;
  1143→      const roles = Array.from(new Set(records.map((r) => r.role || "(none)")));
  1144→      duplicates.push({
  1145→        email,
  1146→        count: records.length,
  1147→        roles,
  1148→        conflict: roles.join("+"),
  1149→        records: records.map((r) => ({
  1150→          id: String(r._id),
  1151→          name: r.name || "",
  1152→          role: r.role || "(none)",
  1153→          firmId: r.firmId || null,
  1154→          isVerified: !!r.isVerified,
  1155→          product: r.product || null,
  1156→        })),
  1157→      });
  1158→    }
  1159→
  1160→    // Sort: biggest conflict groups first, then alphabetically by email
  1161→    duplicates.sort((a, b) => b.count - a.count || a.email.localeCompare(b.email));
  1162→    return duplicates;
  1163→  }
  1164→});

  1376→/**
  1377→ * ISO 27001 / NDPA 2023 COMPLIANCE
  1378→ * Backend Authentication & Brute Force Prevention
  1379→ *
  1380→ * Receives the raw password over TLS, hashes server-side with PBKDF2-SHA512.
  1381→ * Enforces a 15-minute lockout after 5 consecutive failed attempts.
  1382→ * Automatically migrates legacy SHA-256 and plaintext hashes to PBKDF2 on successful login.
  1383→ */
  1384→export const verifyLogin = action({
  1385→  args: {
  1386→    email: v.string(),
  1387→    passwordHash: v.string(),
  1388→    rawPassword: v.optional(v.string()),
  1389→    mfaCode: v.optional(v.string()),
  1390→    // When the login is initiated from a portal route (/portal/tenant/login or
  1391→    // /portal/client/login), the frontend passes portalType so we can resolve
  1392→    // the correct user record when the same email exists as BOTH an admin
  1393→    // record AND a portal record (the "residents see admin dashboard" bug).
  1394→    portalType: v.optional(v.union(v.literal("tenant"), v.literal("client"))),
  1395→  },
  1396→  handler: async (ctx, args) => {
  1397→    const token = args.email.toLowerCase().trim();
  1398→
  1399→    // 1. Find User — when portalType is set, prefer the matching portal-role record.
  1400→    //    This is the critical fix: without it, getUser() returns the first matching
  1401→    //    record (usually the older Admin record), the password check passes against
  1402→    //    the Admin record (whose password was overwritten by setupPortalPassword),
  1403→    //    and the user ends up logged in as Admin — seeing the admin dashboard.
  1404→    const user: any = await ctx.runQuery(api.myFunctions.getUser, {
  1405→      tokenIdentifier: token,
  1406→      preferPortalRole: args.portalType !== undefined,
  1407→    });
  1408→
  1409→    if (!user) return { success: false, message: "Account not found. Please sign up." };
  1410→
  1411→    // PORTAL-ROUTE GUARD: If the user explicitly logged in via a portal route
  1412→    // (portalType is set), but the resolved user is NOT a portal user, refuse
  1413→    // the login. This is defense-in-depth: even if getUser somehow returned
  1414→    // the Admin record (e.g. no portal-role record exists for this email),
  1415→    // we don't let them through to the portal UI with admin privileges.
  1416→    if (args.portalType && user.role !== "Client" && user.role !== "Tenant") {
  1417→      return {
  1418→        success: false,
  1419→        message: "This email is registered as an admin account, not a portal account. Please log in from the main app, or ask your manager to send a portal invitation to a different email address.",
  1420→      };
  1421→    }
  1422→
  1423→    // REVERSE GUARD: If the user logged in via the MAIN APP (no portalType)
  1424→    // but their role is Tenant or Client, block them. Portal users must
  1425→    // log in through the portal login page, not the main app login.
  1426→    // This prevents portal users from accessing firm-wide data through
  1427→    // the main app's Convex subscriptions.
  1428→    if (!args.portalType && (user.role === "Tenant" || user.role === "Client")) {
  1429→      const portalUrl = user.role === "Client" ? "/portal/client/login" : "/portal/tenant/login";
  1430→      return {
  1431→        success: false,
  1432→        message: `This email is a portal account. Please log in through the portal: ${portalUrl}`,
  1433→        redirect: portalUrl,
  1434→      };
  1435→    }
  1436→
  1437→    // Distinguish between "email not confirmed" (never verified) and "account revoked/deactivated"
  1438→    // Portal users who were deleted via deletePortalInviteAndCleanup get role="Pending" + isVerified=false
  1439→    if (!user.isVerified) {
  1440→      if (user.role === "Pending" && !user.password) {
  1441→        // This user's portal access was revoked/deleted — they previously had an account
  1442→        // but it was cleaned up. Give them a clear message instead of the misleading
  1443→        // "Email not confirmed" error.
  1444→        return { success: false, message: "Your portal access has been revoked. Please contact your manager to request a new invitation.", isRevoked: true };
  1445→      }
  1446→      return { success: false, message: "Email not confirmed. Please check your inbox or resend code." };
  1447→    }
  1448→
  1449→    // Defense-in-depth: A user with role="Pending" should not be able to log in
  1450→    // even if isVerified was somehow still true.
  1451→    if (user.role === "Pending") {
  1452→      return { success: false, message: "Your portal access has been revoked. Please contact your manager to request a new invitation.", isRevoked: true };
  1453→    }
  1454→
  1455→    // ─── DEACTIVATION CHECK ────────────────────────────────────────────
  1456→    // If a firm admin has deactivated this user (deactivatedAt is set),
  1457→    // block the login with a clear message. The user record still exists
  1458→    // (for audit/authorship) but they cannot access the app.
  1459→    if (user.deactivatedAt) {
  1460→      return {
  1461→        success: false,
  1462→        message: "Your account has been deactivated. Please contact your firm administrator to request reactivation.",
  1463→        isDeactivated: true,
  1464→      };
  1465→    }
  1466→
  1467→    const MAX_ATTEMPTS = 5;
  1468→    const LOCKOUT_DURATION_MS = 15 * 60 * 1000;
  1469→    const now = Date.now();
  1470→
  1471→    // 2. Check lockout
  1472→    if (user.lockedUntil && user.lockedUntil > now) {
  1473→      const remainingMinutes = Math.ceil((user.lockedUntil - now) / 60000);
  1474→      return {
  1475→        success: false,
  1476→        isLocked: true,
  1477→        message: `Account locked due to too many failed attempts. Please try again in ${remainingMinutes} minutes.`
  1478→      };
  1479→    } else if (user.lockedUntil && user.lockedUntil <= now) {
  1480→      await ctx.runMutation(internal.myFunctions.updateUserSecurityFields, {
  1481→        userId: user._id,
  1482→        fields: { lockedUntil: null, failedLoginAttempts: 0 },
  1483→      });
  1484→      user.lockedUntil = null;
  1485→      user.failedLoginAttempts = 0;
  1486→    }
  1487→
  1488→    // 3. Verify Password via PBKDF2-aware internal action
  1489→    let isPasswordCorrect = false;
  1490→    let needsMigration = false;
  1491→    const rawPw = args.rawPassword || "";
  1492→
  1493→    if (user.password && user.password !== 'admin') {
  1494→      const result: any = await ctx.runAction(internal.authUtils.verifyPassword, {
  1495→        password: rawPw,
  1496→        storedHash: user.password,
  1497→      });
  1498→      isPasswordCorrect = result.valid;
  1499→      needsMigration = result.needsMigration;
  1500→    } else if (!user.password && rawPw && rawPw !== 'admin') {
  1501→      // No password stored — trust on first use, migrate them
  1502→      isPasswordCorrect = true;
  1503→      needsMigration = true;
  1504→    }
  1505→    // user.password === 'admin' → intentionally blocked
  1506→
  1507→    if (isPasswordCorrect) {
  1508→      if (user.isMfaEnabled) {
  1509→        if (!args.mfaCode) {
  1510→          const code = Math.floor(100000 + Math.random() * 900000).toString();
  1511→          await ctx.runMutation(internal.myFunctions.updateUserSecurityFields, {
  1512→            userId: user._id,
  1513→            fields: { mfaCode: code },
  1514→          });
  1515→          try {
  1516→            await ctx.scheduler.runAfter(0, (internal as any).myFunctions.sendVerificationEmail, { email: args.email, code, product: user.product || 'legal' });
  1517→          } catch (e) {
  1518→            console.error("Failed to send MFA email", e);
  1519→          }
  1520→          return { success: false, requiresMfa: true, mfaType: 'email' };
  1521→        } else {
  1522→          if (user.mfaCode !== args.mfaCode) {
  1523→            return { success: false, message: "Incorrect security code. Please try again." };
  1524→          }
  1525→        }
  1526→      }
  1527→
  1528→      // Success! Build patch updates.
  1529→      const updates: Record<string, any> = {};
  1530→      if (user.failedLoginAttempts > 0) updates.failedLoginAttempts = 0;
  1531→      if (user.lockedUntil) updates.lockedUntil = null;
  1532→      if (user.mfaCode) updates.mfaCode = null;
  1533→
  1534→      // Upgrade legacy hash to PBKDF2 on successful login
  1535→      if (needsMigration && rawPw) {
  1536→        const newHash: any = await ctx.runAction(internal.authUtils.hashPassword, { password: rawPw });
  1537→        updates.password = newHash;
  1538→      }
  1539→
  1540→      if (Object.keys(updates).length > 0) {
  1541→        await ctx.runMutation(internal.myFunctions.updateUserSecurityFields, {
  1542→          userId: user._id,
  1543→          fields: updates,
  1544→        });
  1545→      }
  1546→
  1547→      // AUTO-REPAIR: If a portal user (Client/Tenant) has no firmId, try to repair it
  1548→      // on login. This handles the case where a previous version of
  1549→      // deletePortalInviteAndCleanup cleared firmId and deleted invite records,
  1550→      // leaving the user unable to load any portal data. We search multiple sources
  1551→      // to find the correct firmId and patch the user record.
  1552→      if ((user.role === "Client" || user.role === "Tenant") && !user.firmId) {
  1553→        try {
  1554→          const repairResult: any = await ctx.runMutation(api.portals.repairPortalUserFirmId, {
  1555→            email: token,
