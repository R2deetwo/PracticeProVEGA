     6→     6→/** Simple deterministic hash for append-only enforcement */
     7→     7→function makeTxHash(firmId: string, unitId: string, amount: number, timestamp: number, type: string): string {
     8→     8→  const raw = `${firmId}|${unitId}|${amount}|${timestamp}|${type}`;
     9→     9→  let h = 0;
    10→    10→  for (let i = 0; i < raw.length; i++) {
    11→    11→    h = Math.imul(31, h) + raw.charCodeAt(i) | 0;
    12→    12→  }
    13→    13→  return `TX-${Math.abs(h).toString(16).toUpperCase().padStart(8, "0")}`;
    14→    14→}

    77→    77→export const getCashFlowSummary = query({
    78→    78→  args: { firmId: v.string() },
    79→    79→  handler: async (ctx, { firmId }) => {
    80→    80→    const entries = await ctx.db
    81→    81→      .query("ledger_entries")
    82→    82→      .withIndex("by_firm", q => q.eq("firmId", firmId))
    83→    83→      .collect();
    84→    84→
    85→    85→    const cleared = entries.filter(e => e.status === "cleared");
    86→    86→    const defaulted = entries.filter(e => e.status === "defaulted");
    87→    87→    const pending = entries.filter(e => e.status === "pending");
    88→    88→
    89→    89→    const totalIncome = cleared.reduce((s, e) => s + e.amount, 0);
    90→    90→    const revenueAtRisk = defaulted.reduce((s, e) => s + e.amount, 0) + pending.reduce((s, e) => s + e.amount, 0);
    91→    91→
    92→    92→    // Monthly buckets for the last 6 months
    93→    93→    const now = new Date();
    94→    94→    const monthlyData: Record<string, { income: number; risk: number }> = {};
    95→    95→    for (let i = 5; i >= 0; i--) {
    96→    96→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
    97→    97→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
    98→    98→      monthlyData[key] = { income: 0, risk: 0 };
    99→    99→    }
   100→   100→
   101→   101→    for (const e of entries) {
   102→   102→      const d = new Date(e.timestamp);
   103→   103→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   104→   104→      if (monthlyData[key]) {
   105→   105→        if (e.status === "cleared") monthlyData[key].income += e.amount;
   106→   106→        else monthlyData[key].risk += e.amount;
   107→   107→      }
   108→   108→    }
   109→   109→
   110→   110→    return { totalIncome, revenueAtRisk, totalTransactions: entries.length, monthlyData };
   111→   111→  },
   112→   112→});

   193→   193→export const markChargeAsPaid = mutation({
   194→   194→  args: {
   195→   195→    serviceChargeId: v.id("service_charges"),
   196→   196→    paidAmount: v.number(),
   197→   197→    firmId: v.string(),
   198→   198→    channel: v.optional(v.string()),
   199→   199→    isPartialPayment: v.optional(v.boolean()),
   200→   200→  },
   201→   201→  handler: async (ctx, { serviceChargeId, paidAmount, firmId, channel, isPartialPayment }) => {
   202→   202→    const sc = await ctx.db.get(serviceChargeId);
   203→   203→    if (!sc) throw new Error("Service charge not found");
   204→   204→
   205→   205→    const totalAmount = sc.amount;
   206→   206→    const previouslyPaid = sc.amountPaidThisCycle ?? 0;
   207→   207→    const newTotalPaid = previouslyPaid + paidAmount;
   208→   208→    const outstandingBalance = Math.max(0, totalAmount - newTotalPaid);
   209→   209→
   210→   210→    // Determine new status
   211→   211→    let newStatus: "PAID_FULLY" | "PARTIALLY_PAID" | "UNPAID";
   212→   212→    if (outstandingBalance <= 0) {
   213→   213→      newStatus = "PAID_FULLY";
   214→   214→    } else if (newTotalPaid > 0) {
   215→   215→      newStatus = "PARTIALLY_PAID";
   216→   216→    } else {
   217→   217→      newStatus = "UNPAID";
   218→   218→    }
   219→   219→
   220→   220→    // If fully paid, advance nextDueDate
   221→   221→    let nextDueDate = sc.nextDueDate;
   222→   222→    let isDefaulter = sc.isDefaulter;
   223→   223→    let daysOverdue = sc.daysOverdue;
   224→   224→    let penaltyApplied = sc.penaltyApplied;
   225→   225→
   226→   226→    if (newStatus === "PAID_FULLY") {
   227→   227→      const cycleMs = sc.cycle === "Monthly" ? 30 * 86400000 : sc.cycle === "Quarterly" ? 90 * 86400000 : 365 * 86400000;
   228→   228→      nextDueDate = Date.now() + cycleMs;
   229→   229→      isDefaulter = false;
   230→   230→      daysOverdue = 0;
   231→   231→      penaltyApplied = false;
   232→   232→    }
   233→   233→
   234→   234→    await ctx.db.patch(serviceChargeId, {
   235→   235→      isDefaulter,
   236→   236→      lastPaidDate: Date.now(),
   237→   237→      nextDueDate,
   238→   238→      penaltyApplied,
   239→   239→      daysOverdue,
   240→   240→      serviceChargeStatus: newStatus,
   241→   241→      outstandingBalance: newStatus === "PAID_FULLY" ? 0 : outstandingBalance,
   242→   242→      amountPaidThisCycle: newStatus === "PAID_FULLY" ? 0 : newTotalPaid,
   243→   243→      // Reset reminder tracking on any payment
   244→   244→      consecutiveReminderCount: 0,
   245→   245→      remindersPaused: false,
   246→   246→    });
   247→   247→
   248→   248→    // Ledger entry
   249→   249→    const timestamp = Date.now();
   250→   250→    const txHash = makeTxHash(firmId, sc.unitId, paidAmount, timestamp, "service_charge");
   251→   251→    await ctx.db.insert("ledger_entries", {
   252→   252→      firmId,
   253→   253→      unitId: sc.unitId,
   254→   254→      tenantId: sc.tenantId,
   255→   255→      amount: paidAmount,
   256→   256→      type: "service_charge",
   257→   257→      status: newStatus === "PAID_FULLY" ? "cleared" : "pending",
   258→   258→      timestamp,
   259→   259→      txHash,
   260→   260→      channel,
   261→   261→      description: newStatus === "PAID_FULLY"
   262→   262→        ? `${sc.category} service charge — full payment`
   263→   263→        : `${sc.category} service charge — partial payment (₦${newTotalPaid.toLocaleString()} of ₦${totalAmount.toLocaleString()})`,
   264→   264→    });
   265→   265→
   266→   266→    // ── PAID Confirmation WhatsApp Message ──
   267→   267→    // When a tenant is marked as fully paid by the administrator,
   268→   268→    // send a confirmation message acknowledging the payment.
   269→   269→    if (newStatus === "PAID_FULLY") {
   270→   270→      try {
   271→   271→        // Resolve tenant contact from the property/unit
   272→   272→        const unitDoc = await ctx.db
   273→   273→          .query("properties")
   274→   274→          .withIndex("by_custom_id", q => q.eq("id", sc.unitId))
   275→   275→          .first();
   276→   276→        const altDoc = !unitDoc ? await ctx.db
   277→   277→          .query("properties")
   278→   278→          .filter(q => q.eq(q.field("_id"), sc.unitId))
   279→   279→          .first() : null;
   280→   280→        const effectiveDoc = unitDoc || altDoc;
   281→   281→
   282→   282→        if (effectiveDoc) {
   283→   283→          const rd = (effectiveDoc as any).rentalDetails || {};
   284→   284→          const tenantPhone: string = rd.tenantPhone || '';
   285→   285→          const tenantEmail: string = rd.tenantEmail || '';
   286→   286→          const tenantName: string = rd.tenantName || 'Tenant';
   287→   287→          const unitName: string = rd.unitName || 'Unit';
   288→   288→          const chargeLabel = sc.isMinimumVend ? ((effectiveDoc as any).minimumVendLabel || 'Minimum Vend') : sc.category;
   289→   289→          const recipient = tenantPhone || tenantEmail;
   290→   290→
   291→   291→          if (recipient) {
   292→   292→            const confirmMessage = `Dear ${tenantName}, we confirm receipt of your ${chargeLabel} service charge payment of ₦${totalAmount.toLocaleString()} for ${unitName}. Your account is now fully settled. Thank you for your prompt payment.`;
   293→   293→            await ctx.db.insert("automation_logs", {
   294→   294→              firmId,
   295→   295→              unitId: sc.unitId,
   296→   296→              tenantId: sc.tenantId,
   297→   297→              messageType: "payment_receipt",
   298→   298→              channel: tenantPhone ? "whatsapp" : "email",
   299→   299→              recipient,
   300→   300→              messagePreview: confirmMessage,
   301→   301→              sentAt: Date.now(),
   302→   302→              status: "simulated",
   303→   303→              triggeredBy: "admin_mark_paid",
   304→   304→            });
   305→   305→          }
   306→   306→        }
   307→   307→      } catch (e) {
   308→   308→        // Non-blocking — don't fail the payment recording if confirmation fails
   309→   309→        console.warn("Failed to send PAID confirmation:", e);
   310→   310→      }
   311→   311→    }
   312→   312→  },
   313→   313→});

   315→   315→// Internal mutation called by cron job
   316→   316→export const flagOverdueCharges = internalMutation({
   317→   317→  args: {},
   318→   318→  handler: async (ctx) => {
   319→   319→    const now = Date.now();
   320→   320→    const allCharges = await ctx.db.query("service_charges").collect();
   321→   321→    let flagged = 0;
   322→   322→    for (const charge of allCharges) {
   323→   323→      if (charge.nextDueDate < now && charge.serviceChargeStatus !== "PAID_FULLY") {
   324→   324→        const daysOverdue = Math.floor((now - charge.nextDueDate) / 86400000);
   325→   325→        // If no partial payment has been made, status stays UNPAID (or becomes UNPAID if not set)
   326→   326→        const currentStatus = charge.serviceChargeStatus || (charge.isDefaulter ? "UNPAID" : "UNPAID");
   327→   327→        await ctx.db.patch(charge._id, {
   328→   328→          isDefaulter: true,
   329→   329→          daysOverdue,
   330→   330→          // Preserve PARTIALLY_PAID if tenant has made a partial payment, otherwise UNPAID
   331→   331→          serviceChargeStatus: currentStatus === "PARTIALLY_PAID" ? "PARTIALLY_PAID" : "UNPAID",
   332→   332→          outstandingBalance: currentStatus === "PARTIALLY_PAID"
   333→   333→            ? (charge.outstandingBalance ?? charge.amount - (charge.amountPaidThisCycle ?? 0))
   334→   334→            : charge.amount,
   335→   335→        });
   336→   336→        flagged++;
   337→   337→      }
   338→   338→    }
   339→   339→    return { flagged };
   340→   340→  },
   341→   341→});

   524→   524→      return sc.isDefaulter || sc.serviceChargeStatus === "PARTIALLY_PAID" || daysUntilDue <= 7;

   534→   534→      const units: any[] = (p as any).units || [];

   536→   536→        if (u.id) propertyMap.set(u.id, p);

   544→   544→        continue;
   545→   545→      }

   552→   552→      }

   558→   558→        continue;
   559→   559→      }

   569→   569→          remindersPaused: true,
   570→   570→        });

   572→   572→        remindersSkipped++;
   573→   573→        continue;

   584→   584→        const altDoc = await ctx.db
   585→   585→          .query("properties")
   586→   586→          .filter(q => q.eq(q.field("_id"), charge.unitId))
   587→   587→          .first();

   589→   589→          remindersSkipped++;
   590→   590→          continue;
   591→   591→        }
   592→   592→      }

   595→   595→      if (!effectiveDoc) {
   596→   596→        remindersSkipped++;
   597→   597→        continue;
   598→   598→      }

   601→   601→      const rd = (effectiveDoc as any).rentalDetails || {};
   602→   602→      const tenantPhone: string = rd.tenantPhone || '';
   603→   603→      const tenantEmail: string = rd.tenantEmail || '';
   604→   604→      const tenantName: string = rd.tenantName || 'Tenant';

   606→   606→      // Check if property has minimum vend enabled
   607→   607→      const minimumVendEnabled = (effectiveDoc as any).minimumVendEnabled || false;
   608→   608→      const vendLabel = charge.isMinimumVend ? ((effectiveDoc as any).minimumVendLabel || 'Minimum Vend') : null;
   609→   609→
   610→   610→      if (!tenantPhone && !tenantEmail) {
   611→   611→        remindersSkipped++;
   612→   612→        continue;
   613→   613→      }
   614→   614→
   615→   615→      // 3. Spam guard: check if we already sent a service_charge_alert in the last 24 hours
   616→   616→      const cutoff = now - 24 * 3600000;
   617→   617→      const recentAlert = await ctx.db
   618→   618→        .query("automation_logs")
   619→   619→        .withIndex("by_firm", q => q.eq("firmId", charge.firmId))
   620→   620→        .filter(q =>
   621→   621→          q.and(
