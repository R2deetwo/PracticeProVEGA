    37→const calculateRentReviewDate = (leaseEnd?: string, frequency?: string) => {
    38→    if (!leaseEnd) return null;
    39→    const endDate = new Date(leaseEnd);
    40→    let noticeMonths = 1;
    41→    const f = frequency?.toLowerCase() || '';
    42→    if (f.includes('year')) noticeMonths = 6;
    43→    else if (f.includes('6-month')) noticeMonths = 3;
    44→    else if (f.includes('quarter')) noticeMonths = 1;
    45→    else if (f.includes('month')) noticeMonths = 1;
    46→
    47→    const noticeDate = new Date(endDate);
    48→    noticeDate.setMonth(noticeDate.getMonth() - noticeMonths);
    49→
    50→    const reviewDate = new Date(noticeDate);
    51→    reviewDate.setDate(reviewDate.getDate() - 14);
    52→    
    53→    return reviewDate;
