    24→/**
    25→ * useScrollReveal — subtle fade-in-up when element enters viewport.
    26→ * Returns a ref to attach to the element. Professional, not bouncy.
    27→ * Respects prefers-reduced-motion (CSS handles the actual disable).
    28→ */
    29→function useScrollReveal<T extends HTMLElement = HTMLDivElement>(options?: { threshold?: number; rootMargin?: string }) {
    30→    const ref = useRef<T>(null);
    31→    useEffect(() => {
    32→        const el = ref.current;
    33→        if (!el) return;
    34→        // If IntersectionObserver isn't supported, just show the element.
    35→        if (!('IntersectionObserver' in window)) {
    36→            el.classList.add('is-visible');
    37→            return;
    38→        }
    39→        const observer = new IntersectionObserver((entries) => {
    40→            entries.forEach((entry) => {
    41→                if (entry.isIntersecting) {
    42→                    entry.target.classList.add('is-visible');
    43→                    observer.unobserve(entry.target);
    44→                }
    45→            });
    46→        }, {
    47→            threshold: options?.threshold ?? 0.15,
    48→            rootMargin: options?.rootMargin ?? '0px 0px -60px 0px',
    49→        });
    50→        observer.observe(el);
    51→        return () => observer.disconnect();
    52→    }, []);
    53→    return ref;
    54→}

    56→/**
    57→ * useScrollParallax — subtle vertical parallax on scroll.
    58→ * Returns a ref to attach to the image element. The image translates
    59→ * slowly as the user scrolls, creating depth. Max offset ±20px.
    60→ */
    61→function useScrollParallax<T extends HTMLElement = HTMLDivElement>() {
    62→    const ref = useRef<T>(null);
    63→    useEffect(() => {
    64→        const el = ref.current;
    65→        if (!el) return;
    66→        // Skip parallax if user prefers reduced motion
    67→        if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    68→
    69→        let rafId: number | null = null;
    70→        const update = () => {
    71→            const rect = el.getBoundingClientRect();
    72→            const viewportH = window.innerHeight;
    73→            // How far is the element's center from the viewport center?
    74→            const elementCenter = rect.top + rect.height / 2;
    75→            const viewportCenter = viewportH / 2;
    76→            const distance = elementCenter - viewportCenter;
    77→            // Normalize to -1..1 (element fully in viewport)
    78→            const normalized = Math.max(-1, Math.min(1, distance / viewportCenter));
    79→            // Apply parallax: max ±16px translate
    80→            const offset = normalized * -16;
    81→            el.style.transform = `translateY(${offset}px) scale(1.08)`;
    82→            rafId = null;
    83→        };
    84→        const onScroll = () => {
    85→            if (rafId === null) rafId = requestAnimationFrame(update);
    86→        };
    87→        update(); // Initial position
    88→        window.addEventListener('scroll', onScroll, { passive: true });
    89→        window.addEventListener('resize', onScroll);
    90→        return () => {
    91→            window.removeEventListener('scroll', onScroll);
    92→            window.removeEventListener('resize', onScroll);
    93→            if (rafId !== null) cancelAnimationFrame(rafId);
    94→        };
    95→    }, []);
    96→    return ref;
    97→}

  1002→//   1. Drag a slider to set their unit count (1-500)

  1009→// The previous version asked for average rent and showed "SCE as % of rent".
  1010→// However, average rent does NOT change the SCE calculation at all — SCE is
  1011→// simply (annual subscription ÷ 12 ÷ units). The percentage was purely
  1012→// informational and confused users ("does this change anything?"). The
  1013→// absorbability benchmark (₦2,000-5,000/mo based on typical Lagos service
  1014→// charges) is a more useful comparison point and doesn't require the user

  1024→    // Unit count drives the entire calculation. Default to 50 (a typical
  1025→    // small-to-medium Lagos portfolio). Slider range: 1-500, with smart
  1026→    // step scaling (1-unit steps for small portfolios, 5-unit steps for
  1027→    // larger ones) so the slider feels precise at low values and fast at
  1028→    // high values.

  1034→    // ── Calculations ────────────────────────────────────────────────────
  1035→    // For each tier, compute:
  1036→    //   - SCE per tenant per month = (price ÷ 12 ÷ units) for annual,
  1037→    //     or (monthlyPrice ÷ units) for monthly billing
  1038→    //   - Total cost (monthly or annual, matching the toggle)
  1039→    //   - Absorbability: based on Lagos SC benchmark
  1040→    //     · Easy:   SCE < ₦2,000/mo — residents barely notice
  1041→    //     · Moderate: SCE ₦2,000-5,000/mo — noticeable but reasonable
  1042→    //     · Tight:  SCE > ₦5,000/mo — consider a higher tier
