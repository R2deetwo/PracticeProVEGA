     6→/** Simple deterministic hash for append-only enforcement */
     7→function makeTxHash(firmId: string, unitId: string, amount: number, timestamp: number, type: string): string {
     8→  const raw = `${firmId}|${unitId}|${amount}|${timestamp}|${type}`;
     9→  let h = 0;
    10→  for (let i = 0; i < raw.length; i++) {
    11→    h = Math.imul(31, h) + raw.charCodeAt(i) | 0;
    12→  }
    13→  return `TX-${Math.abs(h).toString(16).toUpperCase().padStart(8, "0")}`;
    14→}

    77→export const getCashFlowSummary = query({
    78→  args: { firmId: v.string() },
    79→  handler: async (ctx, { firmId }) => {
    80→    const entries = await ctx.db
    81→      .query("ledger_entries")
    82→      .withIndex("by_firm", q => q.eq("firmId", firmId))
    83→      .collect();
    84→
    85→    const cleared = entries.filter(e => e.status === "cleared");
    86→    const defaulted = entries.filter(e => e.status === "defaulted");
    87→    const pending = entries.filter(e => e.status === "pending");
    88→
    89→    const totalIncome = cleared.reduce((s, e) => s + e.amount, 0);
    90→    const revenueAtRisk = defaulted.reduce((s, e) => s + e.amount, 0) + pending.reduce((s, e) => s + e.amount, 0);
    91→
    92→    // Monthly buckets for the last 6 months
    93→    const now = new Date();
    94→    const monthlyData: Record<string, { income: number; risk: number }> = {};
    95→    for (let i = 5; i >= 0; i--) {
    96→      const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
    97→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
    98→      monthlyData[key] = { income: 0, risk: 0 };
    99→    }
   100→
   101→    for (const e of entries) {
   102→      const d = new Date(e.timestamp);
   103→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   104→      if (monthlyData[key]) {
   105→        if (e.status === "cleared") monthlyData[key].income += e.amount;
   106→        else monthlyData[key].risk += e.amount;
   107→      }
   108→    }
   109→
   110→    return { totalIncome, revenueAtRisk, totalTransactions: entries.length, monthlyData };
   111→  },
   112→});

   193→export const markChargeAsPaid = mutation({
   194→  args: {
   195→    serviceChargeId: v.id("service_charges"),
   196→    paidAmount: v.number(),
   197→    firmId: v.string(),
   198→    channel: v.optional(v.string()),
   199→    isPartialPayment: v.optional(v.boolean()),
   200→  },
   201→  handler: async (ctx, { serviceChargeId, paidAmount, firmId, channel, isPartialPayment }) => {
   202→    const sc = await ctx.db.get(serviceChargeId);
   203→    if (!sc) throw new Error("Service charge not found");
   204→
   205→    const totalAmount = sc.amount;
   206→    const previouslyPaid = sc.amountPaidThisCycle ?? 0;
   207→    const newTotalPaid = previouslyPaid + paidAmount;
   208→    const outstandingBalance = Math.max(0, totalAmount - newTotalPaid);
   209→
   210→    // Determine new status
   211→    let newStatus: "PAID_FULLY" | "PARTIALLY_PAID" | "UNPAID";
   212→    if (outstandingBalance <= 0) {
   213→      newStatus = "PAID_FULLY";
   214→    } else if (newTotalPaid > 0) {
   215→      newStatus = "PARTIALLY_PAID";
   216→    } else {
   217→      newStatus = "UNPAID";
   218→    }
   219→
   220→    // If fully paid, advance nextDueDate
   221→    let nextDueDate = sc.nextDueDate;
   222→    let isDefaulter = sc.isDefaulter;
   223→    let daysOverdue = sc.daysOverdue;
   224→    let penaltyApplied = sc.penaltyApplied;
   225→
   226→    if (newStatus === "PAID_FULLY") {
   227→      const cycleMs = sc.cycle === "Monthly" ? 30 * 86400000 : sc.cycle === "Quarterly" ? 90 * 86400000 : 365 * 86400000;
   228→      nextDueDate = Date.now() + cycleMs;
   229→      isDefaulter = false;
   230→      daysOverdue = 0;
   231→      penaltyApplied = false;
   232→    }
   233→
   234→    await ctx.db.patch(serviceChargeId, {
   235→      isDefaulter,
   236→      lastPaidDate: Date.now(),
   237→      nextDueDate,
   238→      penaltyApplied,
   239→      daysOverdue,
   240→      serviceChargeStatus: newStatus,
   241→      outstandingBalance: newStatus === "PAID_FULLY" ? 0 : outstandingBalance,
   242→      amountPaidThisCycle: newStatus === "PAID_FULLY" ? 0 : newTotalPaid,
   243→      // Reset reminder tracking on any payment
   244→      consecutiveReminderCount: 0,
   245→      remindersPaused: false,
   246→    });
   247→
   248→    // Ledger entry
   249→    const timestamp = Date.now();
   250→    const txHash = makeTxHash(firmId, sc.unitId, paidAmount, timestamp, "service_charge");
   251→    await ctx.db.insert("ledger_entries", {
   252→      firmId,
   253→      unitId: sc.unitId,
   254→      tenantId: sc.tenantId,
   255→      amount: paidAmount,
   256→      type: "service_charge",
   257→      status: newStatus === "PAID_FULLY" ? "cleared" : "pending",
   258→      timestamp,
   259→      txHash,
   260→      channel,
   261→      description: newStatus === "PAID_FULLY"
   262→        ? `${sc.category} service charge — full payment`
   263→        : `${sc.category} service charge — partial payment (₦${newTotalPaid.toLocaleString()} of ₦${totalAmount.toLocaleString()})`,
   264→    });
   265→
   266→    // ── PAID Confirmation WhatsApp Message ──
   267→    // When a tenant is marked as fully paid by the administrator,
   268→    // send a confirmation message acknowledging the payment.
   269→    if (newStatus === "PAID_FULLY") {
   270→      try {
   271→        // Resolve tenant contact from the property/unit
   272→        const unitDoc = await ctx.db
   273→          .query("properties")
   274→          .withIndex("by_custom_id", q => q.eq("id", sc.unitId))
   275→          .first();
   276→        const altDoc = !unitDoc ? await ctx.db
   277→          .query("properties")
   278→          .filter(q => q.eq(q.field("_id"), sc.unitId))
   279→          .first() : null;
   280→        const effectiveDoc = unitDoc || altDoc;
   281→
   282→        if (effectiveDoc) {
   283→          const rd = (effectiveDoc as any).rentalDetails || {};
   284→          const tenantPhone: string = rd.tenantPhone || '';
   285→          const tenantEmail: string = rd.tenantEmail || '';
   286→          const tenantName: string = rd.tenantName || 'Tenant';
   287→          const unitName: string = rd.unitName || 'Unit';
   288→          const chargeLabel = sc.isMinimumVend ? ((effectiveDoc as any).minimumVendLabel || 'Minimum Vend') : sc.category;
   289→          const recipient = tenantPhone || tenantEmail;
   290→
   291→          if (recipient) {
   292→            const confirmMessage = `Dear ${tenantName}, we confirm receipt of your ${chargeLabel} service charge payment of ₦${totalAmount.toLocaleString()} for ${unitName}. Your account is now fully settled. Thank you for your prompt payment.`;
   293→            await ctx.db.insert("automation_logs", {
   294→              firmId,
   295→              unitId: sc.unitId,
   296→              tenantId: sc.tenantId,
   297→              messageType: "payment_receipt",
   298→              channel: tenantPhone ? "whatsapp" : "email",
   299→              recipient,
   300→              messagePreview: confirmMessage,
   301→              sentAt: Date.now(),
   302→              status: "simulated",
   303→              triggeredBy: "admin_mark_paid",
   304→            });
   305→          }
   306→        }
   307→      } catch (e) {
   308→        // Non-blocking — don't fail the payment recording if confirmation fails
   309→        console.warn("Failed to send PAID confirmation:", e);
   310→      }
   311→    }
   312→  },
   313→});

   315→// Internal mutation called by cron job
   316→export const flagOverdueCharges = internalMutation({
   317→  args: {},
   318→  handler: async (ctx) => {
   319→    const now = Date.now();
   320→    const allCharges = await ctx.db.query("service_charges").collect();
   321→    let flagged = 0;
   322→    for (const charge of allCharges) {
   323→      if (charge.nextDueDate < now && charge.serviceChargeStatus !== "PAID_FULLY") {
   324→        const daysOverdue = Math.floor((now - charge.nextDueDate) / 86400000);
   325→        // If no partial payment has been made, status stays UNPAID (or becomes UNPAID if not set)
   326→        const currentStatus = charge.serviceChargeStatus || (charge.isDefaulter ? "UNPAID" : "UNPAID");
   327→        await ctx.db.patch(charge._id, {
   328→          isDefaulter: true,
   329→          daysOverdue,
   330→          // Preserve PARTIALLY_PAID if tenant has made a partial payment, otherwise UNPAID
   331→          serviceChargeStatus: currentStatus === "PARTIALLY_PAID" ? "PARTIALLY_PAID" : "UNPAID",
   332→          outstandingBalance: currentStatus === "PARTIALLY_PAID"
   333→            ? (charge.outstandingBalance ?? charge.amount - (charge.amountPaidThisCycle ?? 0))
   334→            : charge.amount,
   335→        });
   336→        flagged++;
   337→      }
   338→    }
   339→    return { flagged };
   340→  },

   503→// ─── SERVICE CHARGE WHATSAPP REMINDERS ─────────────────────────────────────
   504→// Scans all active tenancies with unpaid or partially-paid service charges
   505→// and triggers automated WhatsApp reminder notifications via the integration gateway.
   506→// Called daily by cron job "serviceChargeWhatsAppReminder" at 6:30 AM UTC.
   507→// Dynamically includes outstanding balance for PARTIALLY_PAID tenants.
   508→export const sendServiceChargeReminders = internalMutation({
   509→  args: {},
   510→  handler: async (ctx) => {
   511→    console.log("Running Service Charge WhatsApp Reminder scan...");
   512→    const now = Date.now();
   513→    let remindersSent = 0;
   514→    let remindersSkipped = 0;
   515→    let remindersPausedCount = 0;
   516→
   517→    // 1. Find all service_charges that are overdue or upcoming (nextDue within 7 days)
   518→    const allCharges = await ctx.db.query("service_charges").collect();
   519→    const actionableCharges = allCharges.filter(sc => {
   520→      // Skip fully paid charges
   521→      if (sc.serviceChargeStatus === "PAID_FULLY") return false;
   522→      const daysUntilDue = Math.floor((sc.nextDueDate - now) / 86400000);
   523→      // Include: currently unpaid/defaulted OR partially paid OR due within the next 7 days
   524→      return sc.isDefaulter || sc.serviceChargeStatus === "PARTIALLY_PAID" || daysUntilDue <= 7;
   525→    });
   526→
   527→    // Pre-load all properties for property-level toggle checks
   528→    const allProperties = await ctx.db.query("properties").collect();
   529→    const propertyMap = new Map<string, any>();
   530→    for (const p of allProperties) {
   531→      if (p.id) propertyMap.set(p.id, p);
   532→      if ((p as any)._id) propertyMap.set((p as any)._id, p);
   533→      // Also index by unit IDs embedded in the property
   534→      const units: any[] = (p as any).units || [];
   535→      for (const u of units) {
   536→        if (u.id) propertyMap.set(u.id, p);
   537→      }
   538→    }
   539→
   540→    for (const charge of actionableCharges) {
   541→      // ── GUARD 0: Per-unit mute toggle ──
   542→      if (charge.remindersMuted) {
   543→        remindersSkipped++;
   544→        continue;
   545→      }
   546→
   547→      // ── GUARD 1: Cool-off — if remindersPaused, skip entirely ──
   548→      if (charge.remindersPaused) {
   549→        remindersPausedCount++;
   550→        remindersSkipped++;
   551→        continue;
   552→      }
   553→
   554→      // ── GUARD 2: Property-level reminders toggle ──
   555→      const parentProperty = propertyMap.get(charge.unitId);
   556→      if (parentProperty && parentProperty.remindersEnabled === false) {
   557→        remindersSkipped++;
   558→        continue;
   559→      }
   560→
   561→      // Resolve cool-off threshold (property-level override or default 7)
   562→      const coolOffThreshold = parentProperty?.reminderCoolOffDays || 7;
   563→
   564→      // ── GUARD 3: Cool-off — check consecutive reminder count ──
   565→      const consecutiveCount = charge.consecutiveReminderCount ?? 0;
   566→      if (consecutiveCount >= coolOffThreshold) {
   567→        // Auto-pause: flag the unit so admin sees REMINDERS_PAUSED_MAX_EFFORT
   568→        await ctx.db.patch(charge._id, {
   569→          remindersPaused: true,
   570→        });
   571→        remindersPausedCount++;
   572→        remindersSkipped++;
   573→        continue;
   574→      }
   575→
   576→      // 2. Resolve tenant phone from the properties table
   577→      const unitDoc = parentProperty || await ctx.db
   578→        .query("properties")
   579→        .withIndex("by_custom_id", q => q.eq("id", charge.unitId))
   580→        .first();
   581→
   582→      if (!unitDoc) {
   583→        // Also try by Convex _id match
   584→        const altDoc = await ctx.db
   585→          .query("properties")
   586→          .filter(q => q.eq(q.field("_id"), charge.unitId))
   587→          .first();
   588→        if (!altDoc) {
   589→          remindersSkipped++;
   590→          continue;
   591→        }
   592→      }
   593→
   594→      const effectiveDoc = unitDoc || propertyMap.get(charge.unitId);
   595→      if (!effectiveDoc) {
   596→        remindersSkipped++;
   597→        continue;
   598→      }
   599→
   600→      // Extract tenant contact info from rentalDetails
   601→      const rd = (effectiveDoc as any).rentalDetails || {};
   602→      const tenantPhone: string = rd.tenantPhone || '';
   603→      const tenantEmail: string = rd.tenantEmail || '';
   604→      const tenantName: string = rd.tenantName || 'Tenant';
   605→      const unitName: string = rd.unitName || 'Unit';
   606→      // Check if property has minimum vend enabled
   607→      const minimumVendEnabled = (effectiveDoc as any).minimumVendEnabled || false;
   608→      const vendLabel = charge.isMinimumVend ? ((effectiveDoc as any).minimumVendLabel || 'Minimum Vend') : null;
   609→
   610→      if (!tenantPhone && !tenantEmail) {
   611→        remindersSkipped++;
   612→        continue;
   613→      }
   614→
   615→      // 3. Spam guard: check if we already sent a service_charge_alert in the last 24 hours
   616→      const cutoff = now - 24 * 3600000;
   617→      const recentAlert = await ctx.db
   618→        .query("automation_logs")
   619→        .withIndex("by_firm", q => q.eq("firmId", charge.firmId))
   620→        .filter(q =>
   621→          q.and(
   622→            q.eq(q.field("unitId"), charge.unitId),
   623→            q.eq(q.field("messageType"), "service_charge_alert"),
   624→            q.gte(q.field("sentAt"), cutoff)
   625→          )
   626→        )
   627→        .first();
   628→
   629→      if (recentAlert) {
   630→        remindersSkipped++;
   631→        continue;
   632→      }
   633→
   634→      // 4. Build dynamic reminder message based on payment status
   635→      const daysOverdue = charge.isDefaulter && charge.daysOverdue
   636→        ? charge.daysOverdue
   637→        : Math.max(0, Math.floor((now - charge.nextDueDate) / 86400000));
   638→      const isUpcoming = !charge.isDefaulter && charge.nextDueDate > now;
   639→      const dueDateStr = new Date(charge.nextDueDate).toLocaleDateString('en-NG');
   640→      const chargeLabel = charge.isMinimumVend ? (vendLabel || 'Minimum Vend') : charge.category;
   641→
   642→      let messagePreview: string;
   643→      if (charge.serviceChargeStatus === "PARTIALLY_PAID") {
   644→        // Dynamic message for partial payments — include exact outstanding balance
   645→        const outstanding = charge.outstandingBalance ?? (charge.amount - (charge.amountPaidThisCycle ?? 0));
   646→        messagePreview = `Dear ${tenantName}, your ${chargeLabel} service charge for ${unitName} has an outstanding balance of ₦${outstanding.toLocaleString()}. You have paid ₦${(charge.amountPaidThisCycle ?? 0).toLocaleString()} of ₦${charge.amount.toLocaleString()}. Kindly complete the payment to avoid penalties.`;
   647→      } else if (isUpcoming) {
   648→        messagePreview = `Dear ${tenantName}, your ${chargeLabel} service charge of ₦${charge.amount.toLocaleString()} for ${unitName} is due on ${dueDateStr}. Please ensure timely payment.`;
   649→      } else {
   650→        messagePreview = `Dear ${tenantName}, your ${chargeLabel} service charge of ₦${charge.amount.toLocaleString()} for ${unitName} is ${daysOverdue} day(s) overdue. Kindly make payment to avoid penalties.`;
   651→      }
   652→
   653→      // 5. Determine channel — prefer WhatsApp if phone available
   654→      const channel = tenantPhone ? "whatsapp" as const : "email" as const;
   655→      const recipient = tenantPhone || tenantEmail;
   656→
   657→      // 6. Log the automation (simulated for now — actual WhatsApp/email dispatch
   658→      // would integrate with Twilio/Resend here)
   659→      await ctx.db.insert("automation_logs", {
   660→        firmId: charge.firmId,
   661→        unitId: charge.unitId,
   662→        tenantId: charge.tenantId,
   663→        messageType: "service_charge_alert",
   664→        channel,
   665→        recipient,
   666→        messagePreview,
   667→        sentAt: now,
   668→        status: "simulated",
   669→        triggeredBy: "cron_service_charge_reminder",
   670→      });
   671→
   672→      // 7. Increment consecutive reminder counter + record last reminder timestamp
   673→      await ctx.db.patch(charge._id, {
   674→        consecutiveReminderCount: (charge.consecutiveReminderCount ?? 0) + 1,
   675→        lastReminderSentAt: now,
   676→      });
   677→
   678→      remindersSent++;
   679→    }
   680→
   681→    console.log(`Service Charge Reminders: ${remindersSent} sent, ${remindersSkipped} skipped, ${remindersPausedCount} auto-paused.`);
   682→    return { remindersSent, remindersSkipped, remindersPausedCount };
   683→  },
   684→});

   747→// ─── MONTHLY SERVICE CHARGE RESET ─────────────────────────────────────────
   748→// Resets all active service charges back to UNPAID on the 1st of every month.
   749→// Called by cron job "monthlyServiceChargeReset" at 00:30 UTC on the 1st.
   750→// Handles both regular service charges and minimum vend charges.
   751→export const resetMonthlyServiceCharges = internalMutation({
   752→  args: {},
   753→  handler: async (ctx) => {
   754→    console.log("Running Monthly Service Charge Reset...");
   755→    const now = Date.now();
   756→    let resetCount = 0;
   757→    let skippedCount = 0;
   758→
   759→    // Get all service charges
   760→    const allCharges = await ctx.db.query("service_charges").collect();
   761→
   762→    for (const charge of allCharges) {
   763→      // Only reset Monthly cycle charges (Quarterly/Annual have their own cycle)
   764→      // Also reset minimum vend charges regardless of cycle since they're monthly
   765→      if (charge.cycle !== "Monthly" && !charge.isMinimumVend) {
   766→        skippedCount++;
   767→        continue;
   768→      }
   769→
   770→      // Skip if already reset this month (idempotency guard)
   771→      if (charge.lastResetAt) {
   772→        const lastReset = new Date(charge.lastResetAt);
   773→        const currentMonth = new Date(now);
   774→        if (lastReset.getMonth() === currentMonth.getMonth() && lastReset.getFullYear() === currentMonth.getFullYear()) {
   775→          skippedCount++;
   776→          continue;
   777→        }
   778→      }
   779→
   780→      // Reset to UNPAID, clear cycle payment tracking
   781→      await ctx.db.patch(charge._id, {
   782→        serviceChargeStatus: "UNPAID",
   783→        outstandingBalance: charge.amount,
   784→        amountPaidThisCycle: 0,
   785→        lastResetAt: now,
   786→        // Recalculate nextDueDate for the new month
   787→        nextDueDate: now + 30 * 86400000,
   788→        // Clear defaulter status for the new cycle
   789→        isDefaulter: false,
   790→        daysOverdue: 0,
   791→        penaltyApplied: false,
   792→      });
   793→      resetCount++;
   794→    }
   795→
   796→    // Also check for properties with minimumVendEnabled that might not have a service_charges record yet
   797→    const allProperties = await ctx.db.query("properties").collect();
   798→    const propertiesWithVend = allProperties.filter(p => p.minimumVendEnabled);
   799→
   800→    for (const prop of propertiesWithVend) {
   801→      // Check if a minimum vend charge already exists for each unit
   802→      const units: any[] = (prop as any).units || [];
   803→      const unitIds = units.length > 0
   804→        ? units.map((u: any) => u.id || u._id)
   805→        : [prop.id];
   806→
   807→      for (const unitId of unitIds) {
   808→        if (!unitId) continue;
   809→
   810→        const existingVend = allCharges.find(c =>
   811→          c.unitId === unitId && c.isMinimumVend
   812→        );
   813→
   814→        if (!existingVend) {
   815→          // Create the minimum vend charge record
   816→          const vendAmount = (prop as any).minimumVendAmount || 0;
   817→          if (vendAmount > 0) {
   818→            await ctx.db.insert("service_charges", {
   819→              firmId: prop.firmId!,
   820→              unitId,
   821→              tenantId: prop.tenantId || prop.currentTenantId || undefined,
   822→              category: "Other",
   823→              amount: vendAmount,
   824→              cycle: "Monthly",
   825→              nextDueDate: now + 30 * 86400000,
   826→              isDefaulter: false,
   827→              isMinimumVend: true,
   828→              serviceChargeStatus: "UNPAID",
   829→              outstandingBalance: vendAmount,
   830→              amountPaidThisCycle: 0,
   831→              lastResetAt: now,
   832→            });
   833→            resetCount++;
   834→          }
   835→        }
   836→      }
   837→    }
   838→
   839→    console.log(`Monthly SC Reset: ${resetCount} reset, ${skippedCount} skipped.`);
   840→    return { resetCount, skippedCount };
   841→  },
   842→});
