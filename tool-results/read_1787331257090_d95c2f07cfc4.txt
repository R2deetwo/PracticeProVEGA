   419→    //   - They don't fire when the device loses internet but stays on Wi-Fi
   420→    //     (no network interface state change).
   421→    //   - They don't fire when the WebView caches a stale connection state.
   422→    //
   423→    // This new implementation adds THREE layers of detection:
   424→    //   1. window 'online'/'offline' events (instant, when they fire)
   425→    //   2. Convex connection status (subscription error = offline)
   426→    //   3. Active polling every 15s using fetch() to a tiny endpoint
   427→    //
   428→    // The polling catches the cases where events don't fire — e.g. the user
   429→    // turns off mobile data but the WebView still thinks it's online.
   430→    React.useEffect(() => {
   431→        const handleOnline = () => setIsOnline(true);

   433→        window.addEventListener('online', handleOnline);
   434→        window.addEventListener('offline', handleOffline);

   436→        // Active polling — every 15 seconds, try a HEAD request to Convex.
   437→        // If it fails, we're offline. If it succeeds, we're online.
   438→        // This catches the "Wi-Fi connected but no internet" case that
   439→        // browser events miss.
   440→        let pollInterval: ReturnType<typeof setInterval> | null = null;
   441→        let lastReportedState = navigator.onLine;
   442→        const pollNetwork = async () => {
   443→            try {
   444→                // Use a no-cors fetch to Convex's health endpoint. no-cors
   445→                // means we can't read the response, but we don't care — we
   446→                // just want to know if the request succeeds or throws.
   447→                // A timeout of 5s ensures we don't hang forever on a slow
   448→                // connection that's actually working.
   449→                const controller = new AbortController();
   450→                const timeoutId = setTimeout(() => controller.abort(), 5000);
   451→                // Use Convex's query endpoint (valid HTTP path) instead of /health
   452→                // which doesn't exist. This hits a real Convex API endpoint.
   453→                await fetch('https://gregarious-malamute-537.convex.cloud/api/query', {
   454→                    method: 'POST',
   455→                    mode: 'no-cors',
   456→                    signal: controller.signal,
   457→                    cache: 'no-store',
   458→                    headers: { 'Content-Type': 'application/json' },
   459→                    body: JSON.stringify({ path: 'myFunctions:getServerTime', args: {} }),
   460→                });
   461→                clearTimeout(timeoutId);
   462→                // If we got here, the request succeeded (or opaque response)
   463→                if (!lastReportedState) {
   464→                    lastReportedState = true;
   465→                    setIsOnline(true);
   466→                }
   467→            } catch {
   468→                // Request failed — we're offline
   469→                if (lastReportedState) {
   470→                    lastReportedState = false;
   471→                    setIsOnline(false);
   472→                }
   473→            }
   474→        };
   475→
   476→        // Start polling after a short delay (don't fire immediately on mount)
   477→        setTimeout(() => {
   478→            pollNetwork();
   479→            pollInterval = setInterval(pollNetwork, 15000);
   480→        }, 2000);
   481→

   483→        try {
   484→            const capacitor = (window as any).Capacitor;
   485→            if (capacitor?.Plugins?.Network) {
   486→                const Network = capacitor.Plugins.Network;
   487→                Network.addListener('networkStatusChange', (status: any) => {
   488→                    const connected = status.connected && status.connectionType !== 'none';
   489→                    lastReportedState = connected;
   490→                    setIsOnline(connected);
   491→                });
   492→            }
   493→        } catch {
   494→            // Capacitor Network plugin not available — polling covers it
   495→        }
   496→
   497→        return () => {
   498→            window.removeEventListener('online', handleOnline);
   499→            window.removeEventListener('offline', handleOffline);
   500→            if (pollInterval) clearInterval(pollInterval);
   501→        };
   502→    }, []);
   503→
   504→    const view = React.useMemo(() => {
   505→        const pathParts = location.pathname.split('/').filter(Boolean);
   506→        const currentView = (pathParts[0] || 'dashboard') as View;
   507→        const currentId = pathParts[1] || null;

   509→        let mappedView = currentView;
   510→        if (location.pathname === '/') mappedView = 'dashboard';
   511→        if (location.pathname === '/cookie-policy') mappedView = 'cookiePolicy' as any;
   512→        if (location.pathname === '/privacy-policy') mappedView = 'privacyPolicy' as any;
   513→        if (location.pathname === '/terms-of-service') mappedView = 'termsOfService' as any;
   514→        if (location.pathname === '/data-processing-agreement') mappedView = 'dataProcessingAgreement' as any;
   515→        if (location.pathname === '/usage-policy') mappedView = 'usagePolicy' as any;
   516→        if (location.pathname === '/resources') mappedView = 'resources' as any;
   517→        if (currentView === 'matters' && currentId) mappedView = 'matterDetail';
   518→        if (currentView === 'contacts' && currentId) mappedView = 'contactDetail';
   519→        if (currentView === 'documents' && currentId) mappedView = 'documentDetail';
   520→        if (currentView === 'properties' && currentId) mappedView = 'propertyDetail';
   521→        
   522→        return mappedView;
   523→    }, [location.pathname]);
   524→
   525→    const selectedId = React.useMemo(() => {
   526→        const pathParts = location.pathname.split('/').filter(Boolean);
   527→        const id = pathParts[1] || null;
   528→        return id === 'null' ? null : id; // Sanitize string "null"
   529→    }, [location.pathname]);

   531→    const currentHistoryEntry = history[historyIndex] || { view: 'dashboard', selectedId: null };

   533→    const addToast = React.useCallback((message: React.ReactNode, options?: { type?: Toast['type']; link?: Toast['link']; duration?: number }) => {
   534→        const id = Date.now();
   535→        // DEBOUNCING — deduplicate identical toast messages within a 2-second
   536→        // window to prevent UI spam (e.g. when a button is double-clicked or
   537→        // a Convex subscription fires multiple rapid updates).
   538→        const messageStr = typeof message === 'string' ? message : String(message);
   539→        setToasts(prev => {
   540→            const now = id;
   541→            const isDuplicate = prev.some(t =>
   542→                t.id > now - 2000 && // within 2 seconds
   543→                (typeof t.message === 'string' ? t.message : String(t.message)) === messageStr &&
   544→                t.type === (options?.type || 'info')
   545→            );
   546→            if (isDuplicate) return prev; // Skip — duplicate within debounce window
   547→
   548→            // CAP: maximum 3 visible toasts at any time. Older toasts
   549→            // are dropped so the screen doesn't fill up with a cascade
   550→            // of alerts (e.g. when deleting a multi-unit property).
   551→            const newToasts = [...prev, { id, message, type: options?.type || 'info', link: options?.link }];
   552→            if (newToasts.length > 3) {
   553→                return newToasts.slice(-3);
   554→            }
   555→            return newToasts;
   556→        });
   557→        try {
   558→            const type = options?.type || 'info';
   559→            if (type === 'success') {
   560→                import('../utils/haptics').then(m => m.haptics.success());
   561→            } else if (type === 'error') {
   562→                import('../utils/haptics').then(m => m.haptics.error());
   563→            } else if (type === 'warning' || type === 'info') {
   564→                import('../utils/haptics').then(m => m.haptics.light());
   565→            }
   566→        } catch {}
   567→        const duration = options?.duration || 3500;
   568→        setTimeout(() => {
   569→            setToasts(prev => prev.filter(t => t.id !== id));
   570→        }, duration);
   571→    }, []);
   572→
   573→    React.useEffect(() => {
   574→        const handleExternalToast = (e: any) => {
   575→            const { message, options } = e.detail;
   576→            addToast(message, options);
   577→        };
   578→        window.addEventListener('practicepro-toast', handleExternalToast);
   579→
   580→        return () => {
   581→            window.removeEventListener('practicepro-toast', handleExternalToast);
   582→        };
   583→    }, [addToast]);

   585→    const navigateTo = React.useCallback((newView: View, newSelectedId: SelectedId = null, context?: any) => {
   586→        // Sanitize string "null" or undefined IDs
   587→        const safeId = (newSelectedId === 'null' || !newSelectedId) ? '' : newSelectedId;
   588→        
   589→        let path = `/${newView}`;
   590→        
   591→        // Map legacy view names to routes
   592→        if (newView === 'dashboard') path = '/';
   593→        else if (newView === 'cookiePolicy') path = '/cookie-policy';
   594→        else if (newView === 'privacyPolicy') path = '/privacy-policy';
   595→        else if (newView === 'termsOfService') path = '/terms-of-service';
   596→        else if (newView === 'portalTermsOfUse') path = '/portal-terms-of-use';
   597→        else if (newView === 'dataProcessingAgreement') path = '/data-processing-agreement';
   598→        else if (newView === 'usagePolicy') path = '/usage-policy';
   599→        else if (newView === 'resources') path = '/resources';
   600→        else if (newView === 'matterDetail') path = `/matters/${safeId}`;
   601→        else if (newView === 'contactDetail') path = `/contacts/${safeId}`;
   602→        else if (newView === 'documentDetail') path = `/documents/${safeId}`;
   603→        else if (newView === 'propertyDetail') path = `/properties/${safeId}`;
   604→        else if (newView === 'properties') path = `/properties`;
   605→        else if (safeId) {
   606→            path = `/${newView}/${safeId}`;
   607→        }
   608→        
   609→        // Remove trailing slashes resulting from empty IDs
   610→        if (path.endsWith('/')) path = path.slice(0, -1);
   611→        
   612→        navigate(path, { state: context });
   613→        
   614→        // Sync custom history state
   615→        // CRO AUDIT FIX — populate previousView so detail views know where
   616→        // the user came from (e.g. matterDetail opened from matters list
   617→        // → previousView = 'matters'). This was declared in the type but
   618→        // never actually set, causing back buttons to go to the wrong place.
   619→        const currentView = history[historyIndex]?.view || 'dashboard';
   620→        setHistory(prev => {
   621→            const newHistory = prev.slice(0, historyIndex + 1);
   622→            newHistory.push({ view: newView, selectedId: newSelectedId, context, previousView: currentView });
   623→            return newHistory;
   624→        });
   625→        setHistoryIndex(prev => prev + 1);
   626→
   627→        setMobileSearchOpen(false);
   628→        // CRO AUDIT FIX — only close modal if there's no dirty form.
   629→        // If a form has unsaved changes, we skip the modal close — the
   630→        // navigation still happens, but the modal stays mounted so the
   631→        // user can see their draft. This is a simpler approach than a
   632→        // confirmation dialog and prevents silent data loss.
   633→        if (!formInteractionState.isReadyForSubmit || Object.keys(formInteractionState.fieldValues || {}).length === 0) {
   634→            setModal(null);
   635→        }
   636→    }, [navigate, historyIndex, formInteractionState]);
   637→
   638→    const goBack = React.useCallback(() => {
   639→        if (historyIndex > 0) {
   640→            setHistoryIndex(prev => prev - 1);
   641→            navigate(-1);
   642→        } else {
   643→            // CRO AUDIT FIX — No in-app history to go back to. This happens
   644→            // when a route was opened directly (e.g. deep link from a
   645→            // notification). Previously fell through to window.history.back()
   646→            // which could EJECT the user from the app entirely on a fresh tab.
   647→            // Now: if we're on a detail view or legal doc, go to the parent
   648→            // view. Otherwise, go to the dashboard.
   649→            //
   650→            // BUG FIX (this session): The previous implementation read
   651→            // `history[historyIndex].view`, but `history` is initialised to
   652→            // `[{view: 'dashboard'}]` on first mount — so for a deep-linked
   653→            // URL like /privacy-policy, `currentEntry.view` was always
   654→            // 'dashboard', never the actual current view. This caused legal
   655→            // docs to fall through to the default 'dashboard' branch and call
   656→            // `navigate('/dashboard')`, which is not a valid route (the real
   657→            // dashboard lives at `/`). We now use the URL-derived `view`
   658→            // (which IS the actual current view) and also map legal docs to
   659→            // their parent ('resources') so deep-link visitors land on a
   660→            // sensible page instead of being routed to a non-existent URL.
   661→            const currentView = view;
   662→            const parentMap: Record<string, string> = {
   663→                'matterDetail': 'matters',
   664→                'propertyDetail': 'properties',
   665→                'invoiceDetail': 'billing',
   666→                'documentDetail': 'documents',
   667→                'contactDetail': 'contacts',
   668→                'editor': 'matters',
   669→                'atriumEngine': 'billing',
   670→                // Public legal/docs pages — fall back to Resources, which is
   671→                // the page that links to all of them. Better than 'dashboard'
   672→                // for unauthenticated visitors (who can't access dashboard).
   673→                'privacyPolicy': 'resources',
   674→                'termsOfService': 'resources',
   675→                'portalTermsOfUse': 'resources',
   676→                'dataProcessingAgreement': 'resources',
   677→                'cookiePolicy': 'resources',
   678→                'usagePolicy': 'resources',
   679→            };
   680→            const parentView = parentMap[currentView] || 'dashboard';
   681→            // BUG FIX: previously called `navigate(`/${parentView}`)` which
   682→            // produces `/dashboard` for the default case — but `/dashboard` is
   683→            // NOT a valid route (dashboard lives at `/`). Use the same
   684→            // view→path mapping that `navigateTo` uses.
   685→            const parentPath = parentView === 'dashboard' ? '/' : `/${parentView}`;
   686→            navigate(parentPath);
   687→        }
   688→    }, [navigate, historyIndex, history, view]);

   720→    const openModal = React.useCallback((modalType: ModalType, id: string | null = null, context: any = null) => {
   721→        // ─── Terms Gate ────────────────────────────────────────────────
