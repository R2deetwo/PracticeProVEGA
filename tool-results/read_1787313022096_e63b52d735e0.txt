    26→// ─── AccordionSection (MODULE-LEVEL — outside PropertyForm) ──────────
    27→// CRITICAL: This component MUST be defined outside the PropertyForm render
    28→// function. When defined inside (as a closure), React treats it as a new
    29→// component type on every render, causing all children to unmount/remount
    30→// on every keystroke — which causes input focus loss.
    31→// By defining it at module level with React.memo, the component identity
    32→// is stable across re-renders, preserving DOM focus.
    33→interface AccordionSectionProps {
    34→    id: string;
    35→    title: string;
    36→    subtitle: string;
    37→    icon: React.ReactNode;
    38→    iconBg: string;
    39→    children: React.ReactNode;
    40→    isOpen: boolean;
    41→    onToggle: (id: string) => void;
    42→    /** Optional: called when any input inside this section receives focus
    43→     *  (via Tab key). Enables keyboard focus-driven accordion switching. */
    44→    onFocusSection?: (id: string) => void;
    45→}
    46→const AccordionSectionInner: React.FC<AccordionSectionProps> = ({ id, title, subtitle, icon, iconBg, children, isOpen, onToggle, onFocusSection }) => {
    47→    const headerRef = useRef<HTMLButtonElement>(null);
    48→    return (
    49→        <div
    50→            className={`rounded-lg border shadow-sm overflow-hidden ${isOpen ? 'bg-white dark:bg-zinc-800 border-slate-200 dark:border-zinc-700' : 'bg-slate-50/50 dark:bg-zinc-800/30 border-slate-100 dark:border-zinc-700/50'}`}
    51→            style={{ willChange: 'height', contain: 'layout style' }}
    52→            // Focus capture: when any child input receives focus (via Tab key),
    53→            // notify the parent so it can auto-expand this section + collapse others.
    54→            // This enables keyboard-driven accordion switching without disrupting
    55→            // manual mouse-click toggles.
    56→            onFocusCapture={onFocusSection ? () => onFocusSection(id) : undefined}
    57→        >
    58→            <button
    59→                ref={headerRef}
    60→                type="button"
    61→                tabIndex={0}
    62→                onClick={() => {
    63→                    onToggle(id);
    64→                    if (!isOpen) {
    65→                        setTimeout(() => {
    66→                            headerRef.current?.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
    67→                        }, 50);
    68→                    }
    69→                }}
    70→                onKeyDown={(e) => {
    71→                    if (e.key === 'Enter' || e.key === ' ') {
    72→                        e.preventDefault();
    73→                        onToggle(id);
    74→                    }
    75→                }}
    76→                className="w-full flex items-center gap-4 p-3 sm:p-4 hover:bg-slate-100/50 dark:hover:bg-zinc-700/30 transition-colors focus:outline-none focus:ring-2 focus:ring-primary-500/30"
    77→            >
    78→                <div className={`p-1.5 ${iconBg} text-white rounded-lg shadow-sm flex-shrink-0`}>
    79→                    {icon}
    80→                </div>
    81→                <div className="text-left flex-1 min-w-0">
    82→                    <p className="text-2xs font-bold text-slate-600/70 dark:text-zinc-400 uppercase tracking-widest leading-none mb-0.5">{subtitle}</p>
    83→                    <h3 className="text-base font-black text-slate-800 dark:text-white tracking-tight">{title}</h3>
    84→                </div>
    85→                <svg className={`w-4 h-4 text-slate-400 transition-transform duration-200 flex-shrink-0 ${isOpen ? 'rotate-180' : ''}`} fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}>
    86→                    <path strokeLinecap="round" strokeLinejoin="round" d="M19 9l-7 7-7-7" />
    87→                </svg>
    88→            </button>
    89→            {isOpen && (
    90→                <div className="p-3 sm:p-4 pt-0 space-y-2 sm:space-y-3">
    91→                    {children}
    92→                </div>
    93→            )}
    94→        </div>
    95→    );
    96→};
