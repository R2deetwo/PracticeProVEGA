    18→// ─── Stage Badge ────────────────────────────────────────────────────────
    19→const StageBadge: React.FC<{ stage: string }> = ({ stage }) => {
    20→    const s = stage?.toLowerCase() || '';
    21→    let bg = 'bg-slate-100 text-slate-700 dark:bg-zinc-700 dark:text-zinc-300';
    22→    if (s.includes('file') || s.includes('plead') || s.includes('initi')) {
    23→        bg = 'bg-amber-50 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400';
    24→    } else if (s.includes('discover') || s.includes('evid') || s.includes('research')) {
    25→        bg = 'bg-sky-50 text-sky-700 dark:bg-sky-900/30 dark:text-sky-400';
    26→    } else if (s.includes('trial') || s.includes('hear') || s.includes('argu')) {
    27→        bg = 'bg-violet-50 text-violet-700 dark:bg-violet-900/30 dark:text-violet-400';
    28→    } else if (s.includes('judg') || s.includes('conclu') || s.includes('final')) {
    29→        bg = 'bg-emerald-50 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400';
    30→    } else if (s.includes('appeal')) {
    31→        bg = 'bg-rose-50 text-rose-700 dark:bg-rose-900/30 dark:text-rose-400';
    32→    } else if (s.includes('settlem') || s.includes('negoti') || s.includes('medi')) {
    33→        bg = 'bg-teal-50 text-teal-700 dark:bg-teal-900/30 dark:text-teal-400';
    34→    } else if (s.includes('compli') || s.includes('exec') || s.includes('enforc')) {
    35→        bg = 'bg-indigo-50 text-indigo-700 dark:bg-indigo-900/30 dark:text-indigo-400';
    36→    }
    37→    return (
    38→        <span className={`inline-flex items-center px-2.5 py-0.5 rounded-full text-xs font-semibold ${bg}`}>
    39→            {stage || 'Unknown'}
    40→        </span>
    41→    );
    42→};

   121→// ─── Main Component ─────────────────────────────────────────────────────

   124→    const { currentUser, originalUser, revertToOriginalUser, logout } = useAuth();

   180→    // an already-authenticated portal user from accessing their own dashboard.
   181→    // If the firm data hasn't loaded yet, we skip the gate entirely rather than
   182→    // showing a misleading "Portal Unavailable" screen.
   183→    // Only show the upgrade prompt if the user is NOT a Client role (e.g. admin previewing).
   184→    if (!canUseClientPortal && currentUser.role !== 'Client') {
   185→        return (
   186→            <div className="flex flex-col items-center justify-center h-full p-8 text-center">
   187→                <div className="max-w-md bg-white dark:bg-zinc-900 p-10 rounded-3xl shadow-xl border border-slate-100 dark:border-zinc-800 flex flex-col items-center">

   206→            d.isSharedWithClient && !d.isSignatureRequested && d.clientReviewStatus !== 'reviewed'

   210→    const outstandingInvoicesCount = useMemo(() => {

   218→        const user = coreState.users?.find(u => u.id === userId);
   219→        return user?.name || 'Unknown';
   220→    };
   221→
   222→    const getLawyerNames = (matter: any): string[] => {
   223→        if (!matter.assignedUsers || matter.assignedUsers.length === 0) return [];

   225→    };

   240→    }, [clientDocs, docFilter]);
   241→
   242→    const unreadMessagesCount = useMemo(() => {
   243→        if (!clientMessages) return 0;
   244→        return clientMessages.filter((m: any) => m.authorId !== currentUser.id && !m.isRead).length;
   245→    }, [clientMessages, currentUser.id]);
   246→
   247→    const sharedDocsCount = useMemo(() => {
   248→        if (!clientDocs) return 0;
   249→        return clientDocs.filter((d: any) => d.isSharedWithClient).length;

   260→            addToast('Please select a matter first', { type: 'error' });
   261→            return;
   262→        }
   263→        try {
   264→            handleSendClientMessage(selectedMatterForMessage, messageText.trim());

   266→            setIsComposing(false);
   267→            setSelectedMatterForMessage('');
   268→            addToast('Message sent', { type: 'success' });

   297→            {/* Recent Activity */}

   318→                                    <span className="text-xs font-bold text-emerald-700 dark:text-emerald-400">
   319→                                        {getInitials(activity.userName)}
   320→                                    </span>
   321→                                </div>
