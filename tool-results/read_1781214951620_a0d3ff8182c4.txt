   194→   194→const LedgerTab: React.FC = () => {
   195→   195→  const { currentUser } = useAuth();
   196→   196→  const { coreState } = useCoreState();
   197→   197→  const firmId = currentUser?.firmId || '';
   198→   198→  const userId = currentUser?.id || '';
   199→   199→  const email = currentUser?.email || '';
   200→   200→
   201→   201→  // First, resolve the correct tenant ID using getTenantInfo
   202→   202→  // This handles the case where properties store tenantId as email instead of Convex _id
   203→   203→  const tenantInfo = useQuery(
   204→   204→    api.portals.getTenantInfo,
   205→   205→    firmId && userId ? { firmId, userId, email } : 'skip'
   206→   206→  );
   207→   207→
   208→   208→  // Use the resolved tenantId from getTenantInfo, or fall back to currentUser.id
   209→   209→  const resolvedTenantId = tenantInfo?.tenantId || userId;
   210→   210→
   211→   211→  // Fetch ledger entries from Convex using the resolved tenant ID
   212→   212→  const ledgerEntries = useQuery(
   213→   213→    api.portals.getTenantLedger,
   214→   214→    firmId && resolvedTenantId ? { firmId, tenantId: resolvedTenantId } : 'skip'
   215→   215→  );
   216→   216→
   217→   217→  // Fetch service charges from Convex to compute current SC/MV
   218→   218→  const serviceCharges = useQuery(
   219→   219→    api.sentry.getServiceChargesByFirm,
   220→   220→    firmId ? { firmId } : 'skip'
   221→   221→  );
   222→   222→
   223→   223→  const isLoading = ledgerEntries === undefined || serviceCharges === undefined;
   224→   224→
   225→   225→  // Compute tenant-specific service charges
   226→   226→  const tenantServiceCharges = useMemo(() => {
   227→   227→    if (!serviceCharges || !resolvedTenantId) return [];
   228→   228→    return serviceCharges.filter((sc: any) => sc.tenantId === resolvedTenantId);
   229→   229→  }, [serviceCharges, resolvedTenantId]);
   230→   230→
   231→   231→  // Summary calculations
   232→   232→  const currentMonthSC = useMemo(() => {
   233→   233→    return tenantServiceCharges
   234→   234→      .filter((sc: any) => !sc.isMinimumVend)
   235→   235→      .reduce((sum: number, sc: any) => sum + (sc.outstandingBalance ?? sc.amount), 0);
   236→   236→  }, [tenantServiceCharges]);
   237→   237→
   238→   238→  const currentMonthMV = useMemo(() => {
   239→   239→    return tenantServiceCharges
   240→   240→      .filter((sc: any) => sc.isMinimumVend)
   241→   241→      .reduce((sum: number, sc: any) => sum + (sc.outstandingBalance ?? sc.amount), 0);
   242→   242→  }, [tenantServiceCharges]);
   243→   243→
   244→   244→  const outstandingBalance = useMemo(() => {
   245→   245→    if (!ledgerEntries) return 0;
   246→   246→    return ledgerEntries
   247→   247→      .filter((e: any) => e.status === 'pending' || e.status === 'defaulted')
   248→   248→      .reduce((sum: number, e: any) => sum + e.amount, 0);
   249→   249→  }, [ledgerEntries]);
   250→   250→
   251→   251→  const scPaid = tenantServiceCharges.filter((sc: any) => !sc.isMinimumVend && sc.serviceChargeStatus === 'PAID_FULLY').length > 0;
   252→   252→  const mvStatus = tenantServiceCharges.filter((sc: any) => sc.isMinimumVend).length > 0
   253→   253→    ? (tenantServiceCharges.filter((sc: any) => sc.isMinimumVend && sc.serviceChargeStatus === 'PAID_FULLY').length > 0 ? 'paid' : 'pending')
   254→   254→    : null;
   255→   255→
   256→   256→  if (isLoading) {
   257→   257→    return (
   258→   258→      <div>
   259→   259→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 mb-6">
   260→   260→          {[1, 2, 3].map(i => (
   261→   261→            <div key={i} className="bg-white dark:bg-zinc-800 rounded-xl p-4 border border-slate-200 dark:border-zinc-700 animate-pulse">
   262→   262→              <div className="h-3 bg-slate-200 dark:bg-zinc-700 rounded w-24 mb-2" />
   263→   263→              <div className="h-7 bg-slate-200 dark:bg-zinc-700 rounded w-32 mb-1" />
   264→   264→              <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-16" />
   265→   265→            </div>
   266→   266→          ))}
   267→   267→        </div>
   268→   268→        <div className="bg-white dark:bg-zinc-800 rounded-xl border border-slate-200 dark:border-zinc-700 p-6 animate-pulse">
   269→   269→          <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-full mb-3" />
   270→   270→          <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-3/4 mb-3" />
   271→   271→          <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-1/2" />
   272→   272→        </div>
   273→   273→      </div>
   274→   274→    );
   275→   275→  }
   276→   276→
   277→   277→  const hasLedgerData = ledgerEntries && ledgerEntries.length > 0;
   278→   278→  const hasServiceCharges = tenantServiceCharges.length > 0;
   279→   279→
   280→   280→  return (
   281→   281→    <div>
   282→   282→      <h3 className="text-lg font-bold text-slate-900 dark:text-white mb-1">Payment Ledger</h3>

   291→   291→          <p className="text-2xl font-bold text-emerald-600 dark:text-emerald-400">
   292→   292→            {formatNaira(currentMonthSC)}
   293→   293→          </p>
   294→   294→          {hasServiceCharges && !currentMonthSC ? (
   295→   295→            <span className="inline-flex items-center gap-1 mt-1 text-[10px] font-bold text-emerald-600 dark:text-emerald-400 bg-emerald-50 dark:bg-emerald-900/20 px-2 py-0.5 rounded-full">
   296→   296→              <CheckIcon className="w-3 h-3" /> Paid
   297→   297→            </span>
   298→   298→          ) : currentMonthSC > 0 ? (
   299→   299→            <span className="inline-flex items-center gap-1 mt-1 text-[10px] font-bold text-amber-600 dark:text-amber-400 bg-amber-50 dark:bg-amber-900/20 px-2 py-0.5 rounded-full">
   300→   300→              <ExclamationTriangleIcon className="w-3 h-3" /> Due
   301→   301→            </span>
   302→   302→          ) : null}
   303→   303→        </div>
   304→   304→        <div className="bg-white dark:bg-zinc-800 rounded-xl p-4 border border-slate-200 dark:border-zinc-700">
   305→   305→          <p className="text-[10px] font-bold uppercase tracking-widest text-slate-400 dark:text-zinc-500 mb-1">Current Month MV</p>

   309→   309→          {mvStatus === 'paid' ? (
   310→   310→            <span className="inline-flex items-center gap-1 mt-1 text-[10px] font-bold text-emerald-600 dark:text-emerald-400 bg-emerald-50 dark:bg-emerald-900/20 px-2 py-0.5 rounded-full">

   327→   327→            </p>
   328→   328→          )}
   329→   329→          {outstandingBalance === 0 && hasLedgerData && (
