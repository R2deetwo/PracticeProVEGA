     1→/**
     2→ * SetupPassword — Dedicated page for portal invitees to create their password
     3→ *
     4→ * Flow:
     5→ * 1. Extract ?token= from URL
     6→ * 2. Call verifyInviteToken to validate the token is real, unexpired, and pending
     7→ * 3. Show a polished "Create Your Password" form with pre-filled, disabled email
     8→ * 4. On submit, call setupPortalPassword which:
     9→ *    - Creates or updates the user account with the hashed password
    10→ *    - Marks the invite as accepted
    11→ *    - Clears the token to prevent replay attacks
    12→ * 5. Redirect to the appropriate portal login page
    13→ */
    14→import React, { useState, useEffect, useCallback } from 'react';
    15→import { useQuery, useAction } from 'convex/react';
    16→import { useNavigate } from 'react-router-dom';
    17→import { api } from '../../../convex/_generated/api';
    18→import { useAuth } from '../../contexts/AuthContext';
    19→import { useUI } from '../../contexts/UIContext';
    20→import { Logo, LockClosedIcon, ShieldCheckIcon, EyeIcon, EyeOffIcon, CheckCircleIcon, XIcon } from '../../constants';
    21→
    22→// Inline icons that aren't in constants
    23→const AlertCircleIcon: React.FC<{ className?: string }> = ({ className }) => (
    24→    <svg xmlns="http://www.w3.org/2000/svg" className={className} fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={1.5}>
    25→        <path strokeLinecap="round" strokeLinejoin="round" d="M12 9v3.75m9-.75a9 9 0 11-18 0 9 9 0 0118 0zm-9 3.75h.008v.008H12v-.008z" />
    26→    </svg>
    27→);
    28→
    29→type Step = 'loading' | 'invalid' | 'form' | 'submitting' | 'success';
    30→
    31→const SetupPassword: React.FC = () => {
    32→    const { login } = useAuth();
    33→    const { addToast } = useUI();
    34→    const navigate = useNavigate();
    35→
    36→    // URL token
    37→    const [token, setToken] = useState<string | null>(null);
    38→    const [step, setStep] = useState<Step>('loading');
    39→    const [invalidReason, setInvalidReason] = useState<string>('');
    40→    const [inviteData, setInviteData] = useState<any>(null);
    41→
    42→    // Form state
    43→    const [password, setPassword] = useState('');
    44→    const [confirmPassword, setConfirmPassword] = useState('');
    45→    const [name, setName] = useState('');
    46→    const [showPassword, setShowPassword] = useState(false);
    47→    const [showConfirm, setShowConfirm] = useState(false);
    48→    const [error, setError] = useState('');
    49→    const [loginError, setLoginError] = useState('');
    50→    const [agreedToTerms, setAgreedToTerms] = useState(false);
    51→    const [showTerms, setShowTerms] = useState(false);
    52→
    53→    // Extract token from URL on mount
    54→    useEffect(() => {
    55→        const params = new URLSearchParams(window.location.search);
    56→        const t = params.get('token');
    57→        if (t) {
    58→            setToken(t);
    59→        } else {
    60→            setStep('invalid');
    61→            setInvalidReason('no_token');
    62→        }
    63→    }, []);
    64→
    65→    // Verify the token
    66→    const verification = useQuery(
    67→        api.portals.verifyInviteToken,
    68→        token ? { token } : 'skip'
    69→    );
    70→
    71→    // Setup password action
    72→    const setupPassword = useAction(api.portals.setupPortalPassword);
    73→
    74→    // Handle verification result
    75→    useEffect(() => {
    76→        if (verification === undefined) return; // still loading
    77→
    78→        if (verification && 'valid' in verification) {
    79→            if (verification.valid) {
    80→                setInviteData(verification);
    81→                setName(verification.invite?.inviteeName || '');
    82→                setStep('form');
    83→            } else {
    84→                setStep('invalid');
    85→                setInvalidReason(verification.reason || 'unknown');
    86→            }
    87→        }
    88→    }, [verification]);
    89→
    90→    // Password strength indicator
    91→    const getPasswordStrength = useCallback((pw: string) => {
    92→        if (!pw) return { label: '', score: 0, color: '' };
    93→        let score = 0;
    94→        if (pw.length >= 8) score++;
    95→        if (pw.length >= 12) score++;
    96→        if (/[A-Z]/.test(pw)) score++;
    97→        if (/[0-9]/.test(pw)) score++;
    98→        if (/[^A-Za-z0-9]/.test(pw)) score++;
    99→
   100→        if (score <= 1) return { label: 'Weak', score: 1, color: 'bg-red-500' };
   101→        if (score <= 2) return { label: 'Fair', score: 2, color: 'bg-orange-500' };
   102→        if (score <= 3) return { label: 'Good', score: 3, color: 'bg-yellow-500' };
   103→        if (score <= 4) return { label: 'Strong', score: 4, color: 'bg-emerald-400' };
   104→        return { label: 'Excellent', score: 5, color: 'bg-emerald-500' };
   105→    }, []);
   106→
   107→    const strength = getPasswordStrength(password);
   108→
   109→    const handleSubmit = async () => {
   110→        setError('');
   111→
   112→        // Validation
   113→        if (!password) {
   114→            setError('Please enter a password.');
   115→            return;
   116→        }
   117→        if (password.length < 8) {
   118→            setError('Password must be at least 8 characters.');
   119→            return;
   120→        }
   121→        if (password !== confirmPassword) {
   122→            setError('Passwords do not match.');
   123→            return;
   124→        }
   125→        if (!token) {
   126→            setError('Invalid invitation link.');
   127→            return;
   128→        }
   129→
   130→        setStep('submitting');
   131→
   132→        try {
   133→            const result = await setupPassword({
   134→                token,
   135→                password,
   136→                name: name || undefined,
   137→            });
   138→
   139→            if (result.success) {
   140→                setStep('success');
   141→
   142→                // Store portal type immediately so App.tsx knows this is a portal user
   143→                // during the auth loading phase (prevents flash of LandingPage)
   144→                const portalType = inviteData?.invite?.portalType;
   145→                if (portalType === 'client') {
   146→                    sessionStorage.setItem('practicepro_portal_type', 'client');
   147→                } else if (portalType === 'resident') {
   148→                    sessionStorage.setItem('practicepro_portal_type', 'tenant');
   149→                }
   150→
   151→                // Auto-login after a short delay
   152→                setTimeout(async () => {
   153→                    try {
   154→                        const loginResult = await login(result.email!, password);
   155→                        if (loginResult.success) {
   156→                            // Redirect to the main app — the router will detect the user's
   157→                            // Client/Tenant role and show the correct portal view automatically.
   158→                            navigate('/', { replace: true });
   159→                        } else {
   160→                            setLoginError('Your password has been set! Please sign in using the button below.');
   161→                            setStep('success');
   162→                        }
   163→                    } catch {
   164→                        setLoginError('Your password has been set! Please sign in manually.');
   165→                        setStep('success');
   166→                    }
   167→                }, 1500);
   168→            } else {
   169→                setError(result.message || 'Something went wrong. Please try again.');
   170→                setStep('form');
   171→            }
   172→        } catch (e: any) {
   173→            setError(e.message || 'An error occurred. Please try again.');
   174→            setStep('form');
   175→        }
   176→    };
   177→
   178→    const handleKeyDown = (e: React.KeyboardEvent) => {
   179→        if (e.key === 'Enter' && step === 'form') handleSubmit();
   180→    };
   181→
   182→    // Determine product branding from invite data
   183→    const isClientPortal = inviteData?.invite?.portalType === 'client';
   184→    const productName = isClientPortal ? 'VEGA' : 'ATRIUM';
   185→    const portalLabel = isClientPortal ? 'Client Portal' : "Residents' Portal";
   186→    // Use concrete classes instead of dynamic template strings — Tailwind JIT can't compile bg-${color}-500
   187→    const glowClass = isClientPortal ? 'bg-violet-500/10' : 'bg-amber-500/10';
   188→    const brandTextClass = isClientPortal ? 'text-violet-400' : 'text-amber-400';
   189→
   190→    // ─── Invalid token screen ─────────────────────────────────────────────
   191→    if (step === 'invalid') {
   192→        const messages: Record<string, { title: string; body: string }> = {
   193→            no_token: {
   194→                title: 'Missing Invitation',
   195→                body: 'No invitation token was found. Please use the link from your invitation email.',
   196→            },
   197→            not_found: {
   198→                title: 'Invalid Invitation Link',
   199→                body: 'This invitation link does not exist or has already been used. Please request a new invitation.',
   200→            },
   201→            expired: {
   202→                title: 'Invitation Expired',
   203→                body: 'This invitation has expired. Please contact your manager to request a new one.',
   204→            },
   205→            revoked: {
   206→                title: 'Invitation Revoked',
   207→                body: 'This invitation has been revoked by the sender. Please contact them for more information.',
   208→            },
   209→            already_accepted: {
   210→                title: 'Already Accepted',
   211→                body: 'Your invitation has already been used and your account is set up. You can sign in directly using your email and password.',
   212→            },
   213→        };
   214→        const msg = messages[invalidReason] || messages.not_found;
   215→
   216→        return (
   217→            <div className="min-h-screen bg-gradient-to-br from-slate-900 via-slate-950 to-slate-900 flex flex-col overflow-x-hidden overflow-y-auto">
   218→                <div className="absolute top-0 left-1/2 -translate-x-1/2 w-[600px] h-[400px] rounded-full blur-[120px] bg-rose-500/10 pointer-events-none" />
   219→                <div className="relative z-10 flex-shrink-0">
   220→                    <div className="container mx-auto px-4 sm:px-6 h-16 flex items-center">
   221→                        <div className="flex items-center gap-2.5">
   222→                            <Logo className="h-7 w-7 text-primary-500" />
   223→                            <span className="text-[19px] font-bold tracking-tight text-white">Practice<span className="text-primary-500">Pro</span></span>
   224→                        </div>
   225→                    </div>
   226→                </div>
   227→                <div className="flex-1 flex items-center justify-center p-4">
   228→                    <div className="relative z-10 w-full max-w-md text-center">
   229→                        <div className="w-16 h-16 rounded-full bg-rose-500/10 border border-rose-500/20 flex items-center justify-center mx-auto mb-6">
   230→                            <AlertCircleIcon className="w-8 h-8 text-rose-400" />
   231→                        </div>
   232→                        <h1 className="text-xl font-bold text-white mb-3">{msg.title}</h1>
   233→                        <p className="text-sm text-slate-400 leading-relaxed mb-6">{msg.body}</p>
   234→                        <button
   235→                            onClick={() => { navigate('/'); }}
   236→                            className="px-6 py-3 rounded-xl text-slate-300 text-sm font-medium hover:text-white hover:bg-white/[0.04] border border-white/[0.08] transition-all"
   237→                        >
   238→                            Back to PracticePro
   239→                        </button>
   240→                        {invalidReason === 'already_accepted' && (
   241→                            <button
   242→                                onClick={() => {
   243→                                    const isClient = inviteData?.invite?.portalType === 'client';
   244→                                    navigate(isClient ? '/portal/client/login' : '/portal/tenant/login');
   245→                                }}
   246→                                className="ml-3 px-6 py-3 rounded-xl bg-amber-500 text-white text-sm font-bold hover:bg-amber-400 transition-all"
   247→                            >
   248→                                Sign In
   249→                            </button>
   250→                        )}
   251→                    </div>
   252→                </div>
   253→            </div>
   254→        );
   255→    }
   256→
   257→    // ─── Loading screen ───────────────────────────────────────────────────
   258→    if (step === 'loading') {
   259→        return (
   260→            <div className="min-h-screen bg-gradient-to-br from-slate-900 via-slate-950 to-slate-900 flex items-center justify-center">
   261→                <div className="text-center">
   262→                    <div className="w-8 h-8 border-2 border-amber-500/30 border-t-amber-500 rounded-full animate-spin mx-auto mb-4" />
   263→                    <p className="text-sm text-slate-400">Verifying your invitation…</p>
   264→                </div>
   265→            </div>
   266→        );
   267→    }
   268→
   269→    // ─── Success screen ───────────────────────────────────────────────────
   270→    if (step === 'success') {
   271→        return (
   272→            <div className="min-h-screen bg-gradient-to-br from-slate-900 via-slate-950 to-slate-900 flex flex-col overflow-x-hidden overflow-y-auto">
   273→                <div className="absolute top-0 left-1/2 -translate-x-1/2 w-[600px] h-[400px] rounded-full blur-[120px] bg-emerald-500/10 pointer-events-none" />
   274→                <div className="relative z-10 flex-shrink-0">
   275→                    <div className="container mx-auto px-4 sm:px-6 h-16 flex items-center">
   276→                        <div className="flex items-center gap-2.5">
   277→                            <Logo className="h-7 w-7 text-primary-500" />
   278→                            <span className="text-[19px] font-bold tracking-tight text-white">Practice<span className="text-primary-500">Pro</span></span>
   279→                        </div>
   280→                    </div>
   281→                </div>
   282→                <div className="flex-1 flex items-center justify-center p-4">
   283→                    <div className="relative z-10 w-full max-w-md text-center">
   284→                        <div className="w-16 h-16 rounded-full bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center mx-auto mb-6">
   285→                            <CheckCircleIcon className="w-8 h-8 text-emerald-400" />
   286→                        </div>
   287→                        <h1 className="text-xl font-bold text-white mb-3">You're All Set!</h1>
   288→                        <p className="text-sm text-slate-400 leading-relaxed mb-2">
   289→                            Your password has been created and your account is now active.
   290→                        </p>
   291→                        <p className="text-sm text-slate-400 mb-6">
   292→                            Signing you in automatically…
   293→                        </p>
   294→                        <div className="w-6 h-6 border-2 border-emerald-500/30 border-t-emerald-500 rounded-full animate-spin mx-auto mb-6" />
   295→                        {loginError && (
   296→                            <div className="mt-4">
   297→                                <p className="text-sm text-amber-400 mb-4">{loginError}</p>
   298→                                <button
   299→                                    onClick={() => {
   300→                                        // Go to the portal login page to sign in manually
   301→                                        const isClient = inviteData?.invite?.portalType === 'client';
   302→                                        navigate(isClient ? '/portal/client/login' : '/portal/tenant/login');
   303→                                    }}
   304→                                    className="px-8 py-3 rounded-xl bg-gradient-to-r from-amber-500 to-amber-600 text-white font-bold text-sm hover:from-amber-400 hover:to-amber-500 shadow-lg shadow-amber-500/20 transition-all"
   305→                                >
   306→                                    Sign In to {portalLabel}
   307→                                </button>
   308→                            </div>
   309→                        )}
   310→                    </div>
   311→                </div>
   312→            </div>
   313→        );
   314→    }
   315→
   316→    // ─── Main form ────────────────────────────────────────────────────────
   317→    return (
   318→        <div className="min-h-screen bg-gradient-to-br from-slate-900 via-slate-950 to-slate-900 flex flex-col overflow-x-hidden overflow-y-auto">
   319→            {/* Ambient glow */}
   320→            <div className={`absolute top-0 left-1/2 -translate-x-1/2 w-[600px] sm:w-[800px] h-[400px] rounded-full blur-[120px] ${glowClass} pointer-events-none`} />
   321→
   322→            {/* Brand header */}
   323→            <div className="relative z-10 flex-shrink-0">
   324→                <div className="container mx-auto px-4 sm:px-6 h-16 flex items-center">
   325→                    <div className="flex items-center gap-2.5">
   326→                        <Logo className="h-7 w-7 text-primary-500" />
   327→                        <span className="text-[19px] font-bold tracking-tight text-white flex items-center">
   328→                            Practice<span className="text-primary-500">Pro</span>
   329→                            <span className={`ml-2 text-[15px] font-black uppercase tracking-tight ${brandTextClass}`}>
   330→                                {productName}
   331→                            </span>
   332→                        </span>
   333→                    </div>
   334→                </div>
   335→            </div>
   336→
   337→            {/* Form card */}
   338→            <div className="flex-1 flex items-center justify-center p-4 sm:p-6">
   339→                <div className="relative z-10 w-full max-w-md">
   340→                    {/* Header */}
   341→                    <div className="text-center mb-6 sm:mb-8">
   342→                        <div className="w-14 h-14 rounded-2xl bg-gradient-to-br from-amber-500/20 to-amber-600/10 border border-amber-500/20 flex items-center justify-center mx-auto mb-4">
   343→                            <LockClosedIcon className="w-6 h-6 text-amber-400" />
   344→                        </div>
   345→                        <h1 className="text-xl sm:text-2xl font-bold text-white mb-2">
   346→                            Create Your Password
   347→                        </h1>
   348→                        <p className="text-sm text-slate-400 leading-relaxed">
   349→                            Set up your secure password to access the {portalLabel}
   350→                        </p>
   351→                    </div>
   352→
   353→                    {/* Invite info */}
   354→                    {inviteData?.invite && (
   355→                        <div className="mb-4 px-4 py-3 rounded-xl bg-violet-500/10 border border-violet-500/20 text-violet-300 text-sm">
   356→                            <div className="flex items-start gap-2">
   357→                                <ShieldCheckIcon className="w-4 h-4 flex-shrink-0 mt-0.5" />
   358→                                <div>
   359→                                    <p className="font-semibold">Invitation Verified</p>
   360→                                    <p className="text-violet-400/80 text-xs mt-0.5">
   361→                                        {inviteData.invite.inviteeName
   362→                                            ? `Welcome, ${inviteData.invite.inviteeName}. `
   363→                                            : ''}
   364→                                        {inviteData.hasAccount
   365→                                            ? 'Your account has been located — setting your password will activate portal access.'
   366→                                            : 'Your account will be created automatically.'}
   367→                                    </p>
   368→                                </div>
   369→                            </div>
   370→                        </div>
   371→                    )}
   372→
   373→                    {/* Form */}
   374→                    <div className="bg-white/[0.04] border border-white/[0.08] backdrop-blur-xl rounded-2xl p-5 sm:p-8 shadow-2xl">
   375→                        <div className="space-y-4">
   376→                            {error && (
   377→                                <div className="px-4 py-3 rounded-xl bg-red-500/10 border border-red-500/20 text-red-400 text-sm font-medium">
   378→                                    {error}
   379→                                </div>
   380→                            )}
   381→
   382→                            {/* Name (optional, for new accounts) */}
   383→                            {inviteData && !inviteData.hasAccount && !inviteData.invite?.inviteeName && (
   384→                                <div>
   385→                                    <label className="block text-xs font-bold text-slate-400 uppercase tracking-wider mb-1.5">Your Name</label>
   386→                                    <input
   387→                                        type="text"
   388→                                        value={name}
   389→                                        onChange={e => setName(e.target.value)}
   390→                                        onKeyDown={handleKeyDown}
   391→                                        placeholder="Full name"
   392→                                        className="w-full px-4 py-3 rounded-xl bg-white/[0.06] border border-white/[0.08] text-white placeholder-slate-500 text-sm focus:ring-2 focus:ring-amber-500/50 focus:border-amber-500/30 transition-all"
   393→                                    />
   394→                                </div>
   395→                            )}
   396→
   397→                            {/* Email (disabled, pre-filled) */}
   398→                            <div>
   399→                                <label className="block text-xs font-bold text-slate-400 uppercase tracking-wider mb-1.5">Email Address</label>
   400→                                <input
   401→                                    type="email"
   402→                                    value={inviteData?.invite?.inviteeEmail || ''}
   403→                                    disabled
   404→                                    className="w-full px-4 py-3 rounded-xl bg-white/[0.03] border border-white/[0.06] text-slate-400 text-sm cursor-not-allowed"
   405→                                />
   406→                                <p className="text-[10px] text-slate-600 mt-1">This is locked to your invitation email</p>
   407→                            </div>
   408→
   409→                            {/* Password */}
   410→                            <div>
   411→                                <label className="block text-xs font-bold text-slate-400 uppercase tracking-wider mb-1.5">Create Password</label>
   412→                                <div className="relative">
   413→                                    <input
   414→                                        type={showPassword ? 'text' : 'password'}
   415→                                        value={password}
   416→                                        onChange={e => setPassword(e.target.value)}
   417→                                        onKeyDown={handleKeyDown}
   418→                                        placeholder="At least 8 characters"
   419→                                        className="w-full px-4 py-3 pr-10 rounded-xl bg-white/[0.06] border border-white/[0.08] text-white placeholder-slate-500 text-sm focus:ring-2 focus:ring-amber-500/50 focus:border-amber-500/30 transition-all"
   420→                                    />
   421→                                    <button
   422→                                        type="button"
   423→                                        onClick={() => setShowPassword(!showPassword)}
   424→                                        className="absolute right-3 top-1/2 -translate-y-1/2 text-slate-500 hover:text-slate-300 transition-colors"
   425→                                    >
   426→                                        {showPassword ? <EyeOffIcon className="w-4 h-4" /> : <EyeIcon className="w-4 h-4" />}
   427→                                    </button>
   428→                                </div>
   429→                                {/* Password strength bar */}
   430→                                {password && (
   431→                                    <div className="mt-2">
   432→                                        <div className="flex gap-1 mb-1">
   433→                                            {[1, 2, 3, 4, 5].map(i => (
   434→                                                <div key={i} className={`h-1 flex-1 rounded-full ${i <= strength.score ? strength.color : 'bg-slate-700'} transition-all`} />
   435→                                            ))}
   436→                                        </div>
   437→                                        <p className="text-[10px] text-slate-500">{strength.label}</p>
   438→                                    </div>
   439→                                )}
   440→                            </div>
   441→
   442→                            {/* Confirm Password */}
   443→                            <div>
   444→                                <label className="block text-xs font-bold text-slate-400 uppercase tracking-wider mb-1.5">Confirm Password</label>
   445→                                <div className="relative">
   446→                                    <input
   447→                                        type={showConfirm ? 'text' : 'password'}
   448→                                        value={confirmPassword}
   449→                                        onChange={e => setConfirmPassword(e.target.value)}
   450→                                        onKeyDown={handleKeyDown}
   451→                                        placeholder="Re-enter your password"
   452→                                        className={`w-full px-4 py-3 pr-10 rounded-xl bg-white/[0.06] border text-white placeholder-slate-500 text-sm focus:ring-2 focus:ring-amber-500/50 transition-all ${
   453→                                            confirmPassword && confirmPassword !== password
   454→                                                ? 'border-rose-500/50 focus:border-rose-500/50'
   455→                                                : confirmPassword && confirmPassword === password
   456→                                                ? 'border-emerald-500/50 focus:border-emerald-500/50'
   457→                                                : 'border-white/[0.08] focus:border-amber-500/30'
   458→                                        }`}
   459→                                    />
   460→                                    <button
   461→                                        type="button"
   462→                                        onClick={() => setShowConfirm(!showConfirm)}
   463→                                        className="absolute right-3 top-1/2 -translate-y-1/2 text-slate-500 hover:text-slate-300 transition-colors"
   464→                                    >
   465→                                        {showConfirm ? <EyeOffIcon className="w-4 h-4" /> : <EyeIcon className="w-4 h-4" />}
   466→                                    </button>
   467→                                </div>
