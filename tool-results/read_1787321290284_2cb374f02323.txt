   225→/** Per-period tracking for Service Charge (SC) and Minimum Vend (MV).
   226→ *  One entry per elapsed billing period from leaseStart to current date,
   227→ *  plus optional advance pre-paid future periods.
   228→ *
   229→ *  Status semantics:
   230→ *  - 'paid'          → balance settled (green pill)
   231→ *  - 'late'          → paid after due date OR currently past due & unpaid (orange pill)
   232→ *  - 'outstanding'   → unpaid, past due (red pill — default for historical periods)
   233→ *  - 'advance_paid'  → settled ahead of future billing date (blue pill)
   234→ *
   235→ *  The `paidOnTime` flag retains the historical audit distinction:
   236→ *  - paidOnTime: true  → "PAID ON TIME" (settled on or before due date)
   237→ *  - paidOnTime: false → "PAID LATE" (settled after due date — still orange
   238→ *                        in the historical timeline even though balance is ₦0)
   239→ *  - undefined         → no payment logged yet
   240→ *
   241→ *  The `isAdvance` flag marks future pre-paid periods that haven't elapsed yet. */
   242→export interface ServiceChargePeriod {
   243→    index: number;       // 1-based period number
   244→    dueDate: string;     // ISO date when this period's charge was due
   245→    status: 'paid' | 'late' | 'outstanding' | 'advance_paid';
   246→    paidDate?: string;   // ISO date when payment was recorded (if paid/late/advance_paid)
   247→    amount: number;      // per-period charge amount
   248→    /** Whether the payment was settled on time (≤ due date) vs late (> due date).
   249→     *  Retained permanently for historical audit even after balance settles. */
   250→    paidOnTime?: boolean;
   251→    /** True for future pre-paid periods (advance cycles not yet elapsed). */
   252→    isAdvance?: boolean;
   253→    /** Receipt number if a receipt has been issued for this period.
   254→     *  When set, the UI shows [View Issued Receipt] instead of [Generate Receipt]. */
   255→    receiptNumber?: string;
   256→}
