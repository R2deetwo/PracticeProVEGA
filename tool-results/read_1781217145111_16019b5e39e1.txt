    33→export const getMaintenanceTicketsByTenant = query({
    34→  args: { tenantId: v.string() },
    35→  handler: async (ctx, args) => {
    36→    const tickets = await ctx.db
    37→      .query("maintenance_tickets")
    38→      .withIndex("by_tenant", (q) => q.eq("tenantId", args.tenantId))
    39→      .order("desc")
    40→      .collect();
    41→
    42→    // Also try matching by email in case the tenantId stored is the user's email
    43→    // This handles the case where the invite was created with the email as the tenant ID
    44→    if (tickets.length === 0 && args.tenantId.includes('@')) {
    45→      // The tenantId looks like an email, also search by the user's Convex _id
    46→      const user: any = await ctx.db
    47→        .query("users")
    48→        .withIndex("by_token", (q) => q.eq("tokenIdentifier", args.tenantId.toLowerCase()))
    49→        .first();
    50→      if (user) {
    51→        const byUserId = await ctx.db
    52→          .query("maintenance_tickets")
    53→          .withIndex("by_tenant", (q) => q.eq("tenantId", String(user._id)))
    54→          .order("desc")
    55→          .collect();
    56→        return byUserId;
    57→      }
    58→    }
    59→
    60→    return tickets;
    61→  },
    62→});

    91→/** Generate a crypto-random invite token (12 chars, URL-safe) */
    92→function generateToken(): string {
    93→  const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
    94→  const arr = new Uint8Array(12);
    95→  // @ts-ignore — crypto available in Convex runtime
    96→  crypto.getRandomValues(arr);
    97→  return Array.from(arr, b => chars[b % chars.length]).join("");
    98→}

   275→/**
   276→ * Internal mutation — only used by createPortalInvite action to write the DB record.
   277→ * Not exported for direct frontend use.
   278→ */
   279→export const insertInviteRecord = mutation({
   280→  args: {
   281→    firmId: v.string(),
   282→    inviterId: v.string(),
   283→    inviteeEmail: v.optional(v.string()),
   284→    inviteeName: v.optional(v.string()),
   285→    inviteePhone: v.optional(v.string()),
   286→    portalType: v.union(v.literal("client"), v.literal("resident")),
   287→    relatedId: v.optional(v.string()),
   288→    token: v.optional(v.string()),
   289→    channel: v.optional(v.string()),
   290→    message: v.optional(v.string()),
   291→    expiresAt: v.number(),
   292→  },
   293→  handler: async (ctx, args) => {
   294→    const now = Date.now();
   295→
   296→    // Clean up any existing invites for the same email + firm + portal type.
   297→    // This prevents "invitation already accepted" errors when re-inviting
   298→    // after a previous invite was deleted via the simple deletePortalInvite
   299→    // (which doesn't clean up). We mark old invites as "superseded" rather
   300→    // than deleting them, so there's an audit trail.
   301→    const email = (args.inviteeEmail || "").toLowerCase().trim();
   302→    if (email) {
   303→      const existingInvites = await ctx.db
   304→        .query("portal_invites")
   305→        .withIndex("by_email", (q) => q.eq("inviteeEmail", email))
   306→        .collect();
   307→
   308→      for (const inv of existingInvites) {
   309→        // Only supersede invites for the same firm and portal type
   310→        if (inv.firmId === args.firmId && inv.portalType === args.portalType) {
   311→          if (inv.status === "pending" || inv.status === "accepted") {
   312→            await ctx.db.patch(inv._id, {
   313→              status: "superseded",
   314→              updatedAt: now,
   315→            });
   316→          }
   317→        }
   318→      }
   319→    }
   320→
   321→    return await ctx.db.insert("portal_invites", {
   322→      firmId: args.firmId,
   323→      inviterId: args.inviterId,
   324→      inviteeEmail: args.inviteeEmail,
   325→      inviteeName: args.inviteeName,
   326→      inviteePhone: args.inviteePhone,
   327→      portalType: args.portalType,
   328→      relatedId: args.relatedId,
   329→      token: args.token ?? undefined,
   330→      channel: args.channel,
   331→      message: args.message,
   332→      status: "pending",
   333→      expiresAt: args.expiresAt,
   334→      createdAt: now,
   335→      updatedAt: now,
   336→    });
   337→  },
   338→});

   520→/**
   521→ * deletePortalInviteAndCleanup — Permanently deletes a portal invite AND resets
   522→ * the associated portal user account so the same email can be re-invited cleanly.
   523→ *
   524→ * This prevents the "invitation already accepted" error when an admin:
   525→ * 1. Deletes a portal access record
   526→ * 2. Creates a new invite for the same email
   527→ *
   528→ * Steps:
   529→ * 1. Delete the portal_invites record
   530→ * 2. Find the user with the matching email + Client/Tenant role
   531→ * 3. Reset their role, verification, and password so they're treated as a brand-new invitee
   532→ */
   533→export const deletePortalInviteAndCleanup = mutation({
   534→  args: {
   535→    inviteId: v.id("portal_invites"),
   536→    inviteeEmail: v.optional(v.string()),
   537→  },
   538→  handler: async (ctx, args) => {
   539→    // 1. Delete the specified invite record
   540→    await ctx.db.delete(args.inviteId);
   541→
   542→    // 2. Also delete ANY other invite records for the same email
   543→    //    to prevent residual "already accepted" errors on re-invite
   544→    const email = (args.inviteeEmail || "").toLowerCase().trim();
   545→    if (!email) return;
   546→
   547→    const otherInvites = await ctx.db
   548→      .query("portal_invites")
   549→      .withIndex("by_email", (q) => q.eq("inviteeEmail", email))
   550→      .collect();
   551→
   552→    for (const inv of otherInvites) {
   553→      if (String(inv._id) === String(args.inviteId)) continue; // Already deleted above
   554→      try { await ctx.db.delete(inv._id); } catch (e) { /* ignore */ }
   555→    }
   556→
   557→    // 3. Find and reset the associated portal user
   558→    const existingUser = await ctx.db
   559→      .query("users")
   560→      .withIndex("by_token", (q) => q.eq("tokenIdentifier", email))
   561→      .first();
   562→
   563→    if (!existingUser) return;
   564→
   565→    // Only reset if the user has a Client or Tenant role (portal user)
   566→    const role = (existingUser as any).role;
   567→    if (role === "Client" || role === "Tenant" || role === "Portal User") {
   568→      // Reset the user so they can be re-invited cleanly
   569→      // Set isVerified to false so they can't log in with old credentials
   570→      // Remove the password so setup-password creates a fresh one
   571→      // Clear firmId so it will be set fresh on the next invite acceptance
   572→      await ctx.db.patch(existingUser._id, {
   573→        isVerified: false,
   574→        emailVerified: false,
   575→        password: undefined,
   576→        role: "Pending",
   577→        firmId: undefined,
   578→      } as any);
   579→    }
   580→  },
   581→});

   630→/**
   631→ * verifyInviteToken — validates a portal invite token and returns
   632→ * the invite details + whether the user already has an account.
   633→ * Used by the /setup-password page to determine what to show.
   634→ *
   635→ * ROBUSTNESS FIX: If the invite is "accepted" but the user's role is
   636→ * "Pending" (meaning their access was deleted via deletePortalInviteAndCleanup
   637→ * and they need to re-setup), we allow them to proceed. This prevents the
   638→ * "invitation already accepted" dead-end when a user is re-invited after
   639→ * their portal access was removed.
   640→ */
   641→export const verifyInviteToken = query({
   642→  args: { token: v.string() },
   643→  handler: async (ctx, args) => {
   644→    const results = await ctx.db
   645→      .query("portal_invites")
   646→      .withIndex("by_token", (q) => q.eq("token", args.token))
   647→      .collect();
   648→    const invite = results[0] || null;
   649→
   650→    if (!invite) {
   651→      return { valid: false, reason: "not_found" } as const;
   652→    }
   653→
   654→    // Check expiry (note: we don't patch status here since this is a query/read-only)
   655→    if (invite.expiresAt < Date.now()) {
   656→      return { valid: false, reason: "expired" } as const;
   657→    }
   658→
   659→    if (invite.status === "revoked") {
   660→      return { valid: false, reason: "revoked" } as const;
   661→    }
   662→
   663→    // Check if the user exists and their current state
   664→    const email = invite.inviteeEmail?.toLowerCase().trim();
   665→    let existingUser: any = null;
   666→    if (email) {
   667→      existingUser = await ctx.db
   668→        .query("users")
   669→        .withIndex("by_token", (q) => q.eq("tokenIdentifier", email))
   670→        .first();
   671→      // Case-insensitive fallback
   672→      if (!existingUser) {
   673→        existingUser = await ctx.db
   674→          .query("users")
   675→          .withIndex("by_token", (q) => q.eq("tokenIdentifier", email.toLowerCase()))
   676→          .first();
   677→      }
   678→    }
   679→
   680→    if (invite.status === "accepted") {
   681→      // ROBUSTNESS: If the user's role is Pending (their access was reset by
   682→      // deletePortalInviteAndCleanup), allow them to re-accept. This invite
   683→      // might be a stale record from before the cleanup ran. The user needs
   684→      // to set up their password again, so we should let them through.
   685→      const userRole = existingUser?.role;
   686→      const isUserUnverified = existingUser && !existingUser.isVerified;
   687→      if (userRole === "Pending" || isUserUnverified) {
   688→        // User was reset — allow re-accepting this invite
   689→        // Fall through to return valid: true
   690→      } else {
   691→        return { valid: false, reason: "already_accepted" } as const;
   692→      }
   693→    }
   694→
   695→    // Token is valid and pending — check if user already has an account
   696→    return {
   697→      valid: true,
   698→      invite: {
   699→        _id: invite._id,
   700→        inviteeEmail: invite.inviteeEmail,
   701→        inviteeName: invite.inviteeName,
   702→        portalType: invite.portalType,
   703→        firmId: invite.firmId,
   704→        relatedId: invite.relatedId,
   705→        expiresAt: invite.expiresAt,
   706→      },
   707→      hasAccount: !!existingUser,
   708→      hasPassword: !!(existingUser as any)?.password,
   709→    } as const;
   710→  },
   711→});
   712→
   713→/**
   714→ * setupPortalPassword — ACTION that completes the invite flow:
   715→ * 1. Validates the invite token (must be pending + unexpired)
   716→ * 2. If user exists → sets their password + marks verified
   717→ * 3. If user doesn't exist → creates a new user record with the password
   718→ * 4. Marks the invite as accepted
   719→ * 5. Clears the token to prevent replay attacks
   720→ */
   721→export const setupPortalPassword = action({
   722→  args: {
   723→    token: v.string(),
   724→    password: v.string(),
   725→    name: v.optional(v.string()),
   726→  },
   727→  handler: async (ctx, args): Promise<{ success: boolean; message?: string; email?: string }> => {
   728→    // 1. Validate the invite
   729→    const invite = await ctx.runQuery(api.portals.getInviteByToken, { token: args.token });
   730→    if (!invite) return { success: false, message: "Invalid invitation link." };
   731→    if (invite.status === "revoked") return { success: false, message: "This invitation has been revoked." };
   732→    if (invite.status === "expired" || invite.expiresAt < Date.now()) {
   733→      await ctx.runMutation(api.portals.updateInviteRecord, {
   734→        inviteId: invite._id,
   735→        updates: { status: "expired", updatedAt: Date.now() },
   736→      });
   737→      return { success: false, message: "This invitation has expired. Please request a new one." };
   738→    }
   739→
   740→    // ROBUSTNESS: If the invite was already accepted but the user's account was
   741→    // reset (role=Pending, isVerified=false), allow them to re-accept. This can
   742→    // happen when deletePortalInviteAndCleanup resets the user but a stale invite
   743→    // record still exists with status "accepted" and a valid token.
   744→    if (invite.status === "accepted") {
   745→      const emailCheck = (invite.inviteeEmail || "").toLowerCase().trim();
   746→      const existingUserCheck: any = await ctx.runQuery(api.myFunctions.getUser, { tokenIdentifier: emailCheck });
   747→      const wasReset = existingUserCheck?.role === "Pending" || (existingUserCheck && !existingUserCheck.isVerified);
   748→      if (!wasReset) {
   749→        return { success: false, message: "This invitation has already been used." };
   750→      }
   751→      // User was reset — allow re-accepting this invite (fall through)
   752→    }
   753→
   754→    const email = (invite.inviteeEmail || "").toLowerCase().trim();
   755→    if (!email) return { success: false, message: "This invitation has no email address associated." };
   756→
   757→    // 2. Hash the password
   758→    const hashedPassword = await ctx.runAction(internal.authUtils.hashPassword, { password: args.password });
   759→
   760→    // 3. Check if user exists
   761→    const existingUser: any = await ctx.runQuery(api.myFunctions.getUser, { tokenIdentifier: email });
   762→
   763→    if (existingUser) {
   764→      // Update existing user: set password + mark verified
   765→      // Also upgrade their role to Client/Tenant if they were a "Portal User"
   766→      // or had no role assigned, so the frontend can route them correctly.
   767→      const portalRole = invite.portalType === "client" ? "Client" : "Tenant";
   768→      const needsRoleUpdate = !existingUser.role
   769→        || existingUser.role === "Portal User"
   770→        || existingUser.role === "Pending";
   771→      // Ensure firmId is always set — portal users need it for the ProductContext
   772→      // to derive the correct product and for data loading to work properly.
   773→      // When an existing user is re-invited after portal access deletion, their
   774→      // firmId might have been cleared or might never have been set.
   775→      const needsFirmId = !existingUser.firmId || existingUser.firmId !== invite.firmId;
   776→      await ctx.runMutation(internal.myFunctions.updateUserSecurityFields, {
   777→        userId: existingUser._id,
   778→        fields: {
   779→          password: hashedPassword,
   780→          isVerified: true,
   781→          emailVerified: true,
   782→          verificationCode: null,
   783→          ...(args.name && !existingUser.name ? { name: args.name } : {}),
   784→          ...(needsRoleUpdate ? { role: portalRole } : {}),
   785→          ...(needsFirmId ? { firmId: invite.firmId } : {}),
   786→          // Online presence privacy: property/resident portals default to hidden
   787→          portalPresenceHidden: invite.portalType === "resident" ? true : (existingUser as any).portalPresenceHidden ?? false,
   788→        },
   789→      });
   790→    } else {
   791→      // Create new user record for this portal user
   792→      // Use the proper UserRole (Client / Tenant) so the frontend can route
   793→      // them to the correct portal view automatically.
   794→      const portalProduct = invite.portalType === "client" ? "legal" : "property";
   795→      const portalRole = invite.portalType === "client" ? "Client" : "Tenant";
   796→      const userName = args.name || invite.inviteeName || email.split("@")[0];
   797→      await ctx.runMutation(internal.myFunctions.createUser, {
   798→        tokenIdentifier: email,
   799→        name: userName,
   800→        email: email,
   801→        password: hashedPassword,
   802→        role: portalRole,
   803→        product: portalProduct,
   804→        onboardingCompleted: false,
   805→        isVerified: true,
   806→        emailVerified: true,
   807→        firmId: invite.firmId,
   808→        avatarUrl: `https://ui-avatars.com/api/?name=${encodeURIComponent(userName)}&background=random`,
   809→        // Online presence privacy: property/resident portals default to hidden
   810→        // (portal users can't see if the manager is online). Legal/client portals
   811→        // default to visible (clients can see if their lawyer is online).
   812→        portalPresenceHidden: invite.portalType === "resident",
   813→      });
   814→    }
   815→
   816→    // 4. Mark invite as accepted
   817→    await ctx.runMutation(api.portals.updateInviteRecord, {
   818→      inviteId: invite._id,
   819→      updates: {
   820→        status: "accepted",
   821→        acceptedAt: Date.now(),
   822→        updatedAt: Date.now(),
   823→        // Clear token to prevent replay attacks — the invite is now consumed
   824→        token: undefined,
   825→      },
   826→    });
   827→
   828→    return { success: true, email };
   829→  },
   830→});

   893→/**
   894→ * getTenantInfo — Resolves the tenant's property/unit assignments using their
   895→ * user ID and email. This is critical because the tenantId stored in
   896→ * properties/units may be the user's Convex _id OR their email, depending
   897→ * on how the invite was created. This query normalizes both cases.
   898→ *
   899→ * Returns: { tenantId, properties, units } where tenantId is the canonical
   900→ * ID that should be used for all other portal queries.
   901→ */
   902→export const getTenantInfo = query({
   903→  args: { firmId: v.string(), userId: v.string(), email: v.string() },
   904→  handler: async (ctx, args) => {
   905→    const properties = await ctx.db
   906→      .query("properties")
   907→      .withIndex("by_firm", (q) => q.eq("firmId", args.firmId))
   908→      .collect();
   909→
   910→    // Search for tenant in properties using both userId and email
   911→    const tenantProperties: any[] = [];
   912→    const tenantUnits: any[] = [];
   913→    let resolvedTenantId = args.userId;
   914→
   915→    for (const prop of properties) {
   916→      // Check property-level tenant (legacy single-tenant model)
   917→      const propTenantId = (prop as any).currentTenantId || (prop as any).tenantId;
   918→      if (propTenantId === args.userId || propTenantId === args.email ||
   919→          String(propTenantId).toLowerCase() === args.email.toLowerCase()) {
   920→        tenantProperties.push({
   921→          id: String(prop._id),
   922→          name: (prop as any).name || prop.address || 'Unnamed Property',
   923→          address: prop.address,
   924→        });
   925→        if (propTenantId && propTenantId !== args.userId) {
   926→          resolvedTenantId = propTenantId;
   927→        }
   928→      }
   929→
   930→      // Check unit-level tenants (multi-unit model)
   931→      const units = (prop as any).units || [];
   932→      for (const unit of units) {
   933→        const unitTenantId = unit.currentTenantId || unit.tenantId;
   934→        if (unitTenantId === args.userId || unitTenantId === args.email ||
   935→            String(unitTenantId).toLowerCase() === args.email.toLowerCase()) {
   936→          tenantUnits.push({

  1239→/**

  1241→ * using their user ID. This is needed because the DataProvider skips loading
  1242→ * firm data for portal users, so matterState.contacts is empty. The
  1243→ * ClientDashboard uses this to find the contactId needed for its portal queries.
  1244→ *
  1245→ * Also searches by email as a fallback — the contact's userId field might
  1246→ * not always match the user's Convex _id (e.g. if the contact was created
  1247→ * before the portal user account existed).
  1248→ */
  1249→export const getClientContactByUserId = query({
  1250→  args: { firmId: v.string(), userId: v.string() },
  1251→  handler: async (ctx, args) => {
  1252→    const contacts = await ctx.db
  1253→      .query("contacts")
  1254→      .withIndex("by_firm", (q) => q.eq("firmId", args.firmId))
  1255→      .collect();
  1256→
  1257→    // First try matching by userId field
  1258→    let contact = contacts.find(c => c.userId === args.userId) || null;
  1259→

  1261→    // the contact's email field. This handles the case where the contact was
  1262→    // created before the portal user account existed.
  1263→    if (!contact) {
  1264→      // Try to look up the user by their Convex _id to get their email
  1265→      let user: any = null;

  1269→      if (!user) {
  1270→        user = await ctx.db
  1271→          .query("users")
  1272→          .withIndex("by_token", (q) => q.eq("tokenIdentifier", args.userId.toString()))
  1273→          .first();
  1274→      }
  1275→
  1276→      const userEmail = user?.email?.toLowerCase();
  1277→      if (userEmail) {
  1278→        contact = contacts.find(c =>
  1279→          (c as any).email?.toLowerCase() === userEmail
  1280→        ) || null;
  1281→      }
  1282→
  1283→      // Also try matching contact by userId against the user's _id as a string
  1284→      if (!contact && user) {
  1285→        const userDocId = String(user._id);
  1286→        contact = contacts.find(c => c.userId === userDocId) || null;
