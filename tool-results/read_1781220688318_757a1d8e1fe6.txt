   134→// ─── Main Component ─────────────────────────────────────────────────────
   135→const ClientDashboard: React.FC = () => {
   136→    // ── ALL HOOKS CALLED FIRST (before any conditional returns) ──────────
   137→    const { currentUser, originalUser, revertToOriginalUser, logout } = useAuth();
   138→    const { coreState } = useCoreState();
   139→    const { navigateTo, openModal, addToast, theme, setTheme } = useUI();
   140→    const { canUseClientPortal } = useFeatures();
   141→    const { handleSendClientMessage } = useDataActions();
   142→
   143→    const [activeTab, setActiveTab] = useState<PortalTab>('overview');
   144→    const [docFilter, setDocFilter] = useState<string>('all');
   145→    const [messageText, setMessageText] = useState('');
   146→    const [selectedMatterForMessage, setSelectedMatterForMessage] = useState<string>('');
   147→    const [isComposing, setIsComposing] = useState(false);
   148→
   149→    const isDark = theme === 'dark' || theme === 'midnight' || theme === 'oled' ||
   150→        theme === 'neon-cyber' || theme === 'midnight-emerald' || theme === 'army-dark' ||
   151→        (theme === 'system' && typeof window !== 'undefined' && window.matchMedia('(prefers-color-scheme: dark)').matches);

   158→    // Portal users don't have matterState (DataProvider skips firm data),
   159→    // so we use dedicated portal queries that look up the contact by userId.
   160→
   161→    // 1. Look up the client's contact record by userId
   162→    const clientContactResult = useQuery(
   163→        api.portals.getClientContactByUserId,
   164→        (currentUser?.id && currentUser?.firmId)
   165→            ? { firmId: currentUser.firmId, userId: currentUser.id }
   166→            : 'skip'
   167→    );
   168→
   169→    // 2. Get the client's matters directly from Convex
   170→    const clientMattersResult = useQuery(
   171→        api.portals.getClientMattersByUserId,
   172→        (currentUser?.id && currentUser?.firmId)
   173→            ? { firmId: currentUser.firmId, userId: currentUser.id }
   174→            : 'skip'
   175→    );
   176→
   177→    // Derive contactId for the other portal queries
   178→    const clientContactId = clientContactResult?._id ? String(clientContactResult._id) : null;
   179→
   180→    // Convex queries — always called, use "skip" when args not ready
   181→    const portalQueryArgs = (clientContactId && currentUser?.firmId)
   182→        ? { firmId: currentUser.firmId, contactId: clientContactId }
   183→        : 'skip';
   184→
   185→    const clientDocs = useQuery(api.portals.getClientDocuments, portalQueryArgs);
   186→    const clientMessages = useQuery(api.portals.getClientMessages, portalQueryArgs);
   187→    const clientActivity = useQuery(api.portals.getClientActivity, portalQueryArgs);
   188→    const clientInvoices = useQuery(api.portals.getClientInvoices, portalQueryArgs);
   189→

   191→    const clientMatters = (clientMattersResult || []) as any[];
   192→
   193→    // ── Access Control (after all hooks) ─────────────────────────────────
   194→    if (!currentUser || currentUser.role !== 'Client') {
   195→        return <div>Access Denied.</div>;
   196→    }
   197→
   198→    // SAFETY NET: Portal users with a valid Client role should ALWAYS be able
   199→    // to access their portal. The canUseClientPortal feature gate is meant to
   200→    // control whether ADMINS can create portal invites — it should never block
   201→    // an already-authenticated portal user from accessing their own dashboard.
   202→    // If the firm data hasn't loaded yet, we skip the gate entirely rather than
   203→    // showing a misleading "Portal Unavailable" screen.
   204→    // Only show the upgrade prompt if the user is NOT a Client role (e.g. admin previewing).
   205→    if (!canUseClientPortal && currentUser.role !== 'Client') {
   206→        return (
   207→            <div className="flex flex-col items-center justify-center h-full p-8 text-center">
   208→                <div className="max-w-md bg-white dark:bg-zinc-900 p-10 rounded-3xl shadow-xl border border-slate-100 dark:border-zinc-800 flex flex-col items-center">
   209→                    <div className="w-16 h-16 bg-amber-50 dark:bg-amber-900/20 rounded-full flex items-center justify-center mb-6">
   210→                        <LockClosedIcon className="w-8 h-8 text-amber-500" />
   211→                    </div>
   212→                    <h2 className="text-2xl font-black text-slate-900 dark:text-white mb-3">Portal Unavailable</h2>
   213→                    <p className="text-slate-500 dark:text-zinc-400 leading-relaxed">
   214→                        The Client Portal is not available on your firm's current plan. Please contact your firm administrator.
   215→                    </p>
   216→                </div>
   217→            </div>
   218→        );
   219→    }
   220→
   221→    // ── Computed Values ──────────────────────────────────────────────────
   222→    const activeMattersCount = clientMatters.filter(m => m.status !== 'Closed' && m.status !== 'Archived').length;
   223→
   224→    const pendingDocsCount = useMemo(() => {
   225→        if (!clientDocs) return 0;
   226→        return clientDocs.filter((d: any) =>
   227→            d.isSharedWithClient && !d.isSignatureRequested && d.clientReviewStatus !== 'reviewed'
   228→        ).length;
   229→    }, [clientDocs]);
   230→
   231→    const outstandingInvoicesCount = useMemo(() => {
   232→        if (!clientInvoices) return 0;
   233→        return clientInvoices.filter((inv: any) =>
   234→            inv.status === 'Overdue' || inv.status === 'Unpaid' || inv.status === 'Sent'
   235→        ).length;
   236→    }, [clientInvoices]);
   237→
   238→    const getUserName = (userId: string): string => {
   239→        const user = coreState.users?.find(u => u.id === userId);
   240→        return user?.name || 'Unknown';
   241→    };
   242→
   243→    const getLawyerNames = (matter: any): string[] => {
   244→        if (!matter.assignedUsers || matter.assignedUsers.length === 0) return [];
   245→        return matter.assignedUsers.map((uid: string) => getUserName(uid)).filter((n: string) => n !== 'Unknown');
   246→    };
   247→
   248→    const getUpcomingDeadlines = (matterId: string): string[] => {
   249→        const matter = clientMatters.find(m => m.id === matterId);
   250→        const deadlines: string[] = [];
   251→        if (matter?.nextAdjournedDate) {
   252→            deadlines.push(`Adjourned: ${matter.nextAdjournedDate}`);
   253→        }
   254→        return deadlines;
   255→    };
   256→
   257→    const filteredDocs = useMemo(() => {
   258→        if (!clientDocs) return [];
   259→        if (docFilter === 'all') return clientDocs;
   260→        return clientDocs.filter((d: any) => String(d.matterId) === docFilter);
   261→    }, [clientDocs, docFilter]);
   262→
   263→    const unreadMessagesCount = useMemo(() => {
   264→        if (!clientMessages) return 0;
   265→        return clientMessages.filter((m: any) => m.authorId !== currentUser.id && !m.isRead).length;
   266→    }, [clientMessages, currentUser.id]);
   267→
   268→    const sharedDocsCount = useMemo(() => {
   269→        if (!clientDocs) return 0;
   270→        return clientDocs.filter((d: any) => d.isSharedWithClient).length;
   271→    }, [clientDocs]);
   272→
   273→    // ── Handlers ─────────────────────────────────────────────────────────
   274→    const handleUploadClick = () => {
   275→        addToast('Upload coming soon', { type: 'info' });
   276→    };
   277→
   278→    const handleSendMessage = () => {
   279→        if (!messageText.trim()) return;
   280→        if (!selectedMatterForMessage) {
   281→            addToast('Please select a matter first', { type: 'error' });
   282→            return;
   283→        }
   284→        try {
   285→            handleSendClientMessage(selectedMatterForMessage, messageText.trim());
   286→            setMessageText('');
   287→            setIsComposing(false);
   288→            setSelectedMatterForMessage('');
   289→            addToast('Message sent', { type: 'success' });
   290→        } catch {
   291→            addToast('Failed to send message', { type: 'error' });
   292→        }
   293→    };
   294→
   295→    // ── Render: Overview Tab ─────────────────────────────────────────────

   303→                    value={activeMattersCount}
   304→                />

   308→                    value={pendingDocsCount}

   314→                    accent={outstandingInvoicesCount > 0 ? 'text-amber-600 dark:text-amber-400' : undefined}
   315→                />
   316→            </div>
   317→
   318→            {/* Recent Activity */}
   319→            <div className="bg-white dark:bg-zinc-800 rounded-xl border border-slate-200 dark:border-zinc-700">
   320→                <div className="px-6 py-4 border-b border-slate-100 dark:border-zinc-700">
   321→                    <h3 className="text-lg font-bold text-slate-900 dark:text-white">Recent Activity</h3>
   322→                </div>
   323→                <div className="divide-y divide-slate-100 dark:divide-zinc-700 max-h-96 overflow-y-auto">

   325→                        <div className="px-6 py-8 text-center">
   326→                            <div className="w-6 h-6 border-2 border-emerald-500 border-t-transparent rounded-full animate-spin mx-auto" />
   327→                            <p className="mt-2 text-sm text-slate-500">Loading activity...</p>

   329→                    ) : clientActivity.length === 0 ? (
   330→                        <EmptyState
   331→                            icon={<ClockIcon className="w-6 h-6" />}
   332→                            title="No Recent Activity"
   333→                            description="Activity related to your matters will appear here."
   334→                        />
   335→                    ) : (
   336→                        clientActivity.slice(0, 5).map((activity: any) => (

   339→                                    <span className="text-xs font-bold text-emerald-700 dark:text-emerald-400">
   340→                                        {getInitials(activity.userName)}
   341→                                    </span>
   342→                                </div>
   343→                                <div className="flex-1 min-w-0">
   344→                                    <p className="text-sm text-slate-800 dark:text-zinc-200">
   345→                                        <span className="font-semibold">{activity.userName}</span>{' '}

   347→                                        {activity.targetName && (
   348→                                            <> &middot; <span className="text-emerald-600 dark:text-emerald-400">{activity.targetName}</span></>
   349→                                        )}
   350→                                    </p>
   351→                                    <p className="text-xs text-slate-400 dark:text-zinc-500 mt-0.5">
   352→                                        {activity.timestamp ? timeAgo(activity.timestamp) : ''}
   353→                                    </p>
   354→                                </div>
   355→                            </div>
   356→                        ))
   357→                    )}
   358→                </div>
   359→            </div>
   360→
   361→            {/* Quick Links */}
   362→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   363→                <button
   364→                    onClick={() => setActiveTab('matters')}
   365→                    className="bg-white dark:bg-zinc-800 rounded-xl border border-slate-200 dark:border-zinc-700 p-5 text-left hover:border-emerald-300 dark:hover:border-emerald-700 transition-colors group"
   366→                >
   367→                    <div className="flex items-center justify-between">
   368→                        <div className="flex items-center gap-3">

   379→                <button

   395→                <button

   398→                >
   399→                    <div className="flex items-center justify-between">
