   171→        api.portals.getClientMattersByUserId,

   174→            : 'skip'

   179→

   191→        currentUser?.email ? { email: currentUser.email } : 'skip'
   192→    );
   193→

   195→    const clientMatters = (clientMattersResult || []) as any[];
   196→

   198→    if (!currentUser || currentUser.role !== 'Client') {
   199→        return <div>Access Denied.</div>;

   204→    // control whether ADMINS can create portal invites — it should never block
   205→    // an already-authenticated portal user from accessing their own dashboard.
   206→    // If the firm data hasn't loaded yet, we skip the gate entirely rather than
   207→    // showing a misleading "Portal Unavailable" screen.
   208→    // Only show the upgrade prompt if the user is NOT a Client role (e.g. admin previewing).
   209→    if (!canUseClientPortal && currentUser.role !== 'Client') {

   212→                <div className="max-w-md bg-white dark:bg-zinc-900 p-10 rounded-3xl shadow-xl border border-slate-100 dark:border-zinc-800 flex flex-col items-center">

   227→

   231→            d.isSharedWithClient && !d.isSignatureRequested && d.clientReviewStatus !== 'reviewed'

   233→    }, [clientDocs]);
   234→

   243→        const user = coreState.users?.find(u => u.id === userId);
   244→        return user?.name || 'Unknown';
   245→    };
   246→

   248→        if (!matter.assignedUsers || matter.assignedUsers.length === 0) return [];
   249→        return matter.assignedUsers.map((uid: string) => getUserName(uid)).filter((n: string) => n !== 'Unknown');
   250→    };
   251→

   253→        const matter = clientMatters.find(m => m.id === matterId);
   254→        const deadlines: string[] = [];
   255→        if (matter?.nextAdjournedDate) {
   256→            deadlines.push(`Adjourned: ${matter.nextAdjournedDate}`);
   257→        }
   258→        return deadlines;
   259→    };
   260→
   261→    const filteredDocs = useMemo(() => {
   262→        if (!clientDocs) return [];

   264→        return clientDocs.filter((d: any) => String(d.matterId) === docFilter);
   265→    }, [clientDocs, docFilter]);

   268→        if (!clientMessages) return 0;
   269→        return clientMessages.filter((m: any) => m.authorId !== currentUser.id && !m.isRead).length;

   272→    const sharedDocsCount = useMemo(() => {
   273→        if (!clientDocs) return 0;

   292→            setSelectedMatterForMessage('');

   322→            {/* Recent Activity */}

   361→                    )}
   362→                </div>

   364→
   365→            {/* Quick Links */}

   798→                                className="w-full text-left px-4 py-3 rounded-lg bg-slate-50 dark:bg-zinc-700 text-slate-400 dark:text-zinc-500 text-sm hover:bg-slate-100 dark:hover:bg-zinc-600 transition-colors"
   799→                            >
   800→                                Send a message to your legal team...
   801→                            </button>
   802→                        </div>
   803→                    ) : (
   804→                        <div className="p-4 space-y-3">
   805→                            <div className="flex items-center gap-3">
   806→                                <label className="text-sm font-medium text-slate-700 dark:text-zinc-300 flex-shrink-0">Matter:</label>
   807→                                <select
   808→                                    value={selectedMatterForMessage}
   809→                                    onChange={(e) => setSelectedMatterForMessage(e.target.value)}
   810→                                    className="flex-1 text-sm border border-slate-200 dark:border-zinc-600 rounded-lg px-3 py-1.5 bg-white dark:bg-zinc-700 text-slate-700 dark:text-zinc-200 focus:ring-2 focus:ring-emerald-500 focus:border-emerald-500 outline-none"
   811→                                >
   812→                                    <option value="">Select a matter...</option>
   813→                                    {clientMatters.map(m => (
   814→                                        <option key={m.id} value={m.id}>{m.title}</option>
   815→                                    ))}
   816→                                </select>
   817→                            </div>
   818→                            <textarea
   819→                                value={messageText}
   820→                                onChange={(e) => setMessageText(e.target.value)}
   821→                                placeholder="Type your message here..."
   822→                                rows={3}
   823→                                className="w-full text-sm border border-slate-200 dark:border-zinc-600 rounded-lg px-4 py-3 bg-slate-50 dark:bg-zinc-700 text-slate-800 dark:text-zinc-200 placeholder:text-slate-400 dark:placeholder:text-zinc-500 focus:ring-2 focus:ring-emerald-500 focus:border-emerald-500 outline-none resize-none"
   824→                            />
   825→                            <div className="flex items-center justify-end gap-2">
   826→                                <button
   827→                                    onClick={() => { setIsComposing(false); setMessageText(''); setSelectedMatterForMessage(''); }}
   828→                                    className="px-4 py-2 text-sm font-medium text-slate-600 dark:text-zinc-300 hover:text-slate-800 dark:hover:text-zinc-100 transition-colors"
   829→                                >
   830→                                    Cancel
   831→                                </button>
   832→                                <button
   833→                                    onClick={handleSendMessage}
   834→                                    disabled={!messageText.trim() || !selectedMatterForMessage}
   835→                                    className="inline-flex items-center gap-2 px-4 py-2 bg-emerald-600 text-white rounded-lg font-semibold text-sm hover:bg-emerald-700 transition-colors disabled:opacity-50 disabled:cursor-not-allowed"
   836→                                >
   837→                                    <SendIcon className="w-4 h-4" />
   838→                                    Send Message
   839→                                </button>
   840→                            </div>
   841→                        </div>
   842→                    )}
   843→                </div>
   844→
   845→                {/* Messages List */}
   846→                {isLoading ? (
   847→                    <div className="space-y-3">
   848→                        {[1, 2, 3].map(i => (
   849→                            <div key={i} className="bg-white dark:bg-zinc-800 rounded-xl border border-slate-200 dark:border-zinc-700 p-5 animate-pulse">
   850→                                <div className="flex items-start gap-3">
   851→                                    <div className="w-8 h-8 rounded-full bg-slate-200 dark:bg-zinc-700" />
   852→                                    <div className="flex-1 space-y-2">
   853→                                        <div className="h-3 bg-slate-200 dark:bg-zinc-700 rounded w-1/4" />
   854→                                        <div className="h-4 bg-slate-200 dark:bg-zinc-700 rounded w-3/4" />
   855→                                    </div>
   856→                                </div>
   857→                            </div>
   858→                        ))}
   859→                    </div>
   860→                ) : messages.length === 0 ? (
   861→                    <EmptyState
   862→                        icon={<ChatAltIcon className="w-7 h-7" />}
   863→                        title="No Messages Yet"
   864→                        description="Start a conversation with your legal team by sending a message above."
   865→                    />
   866→                ) : (
   867→                    <div className="space-y-2">
   868→                        {messages.map((msg: any) => {
   869→                            const isCurrentUser = msg.authorId === currentUser.id;
   870→                            const authorName = isCurrentUser
   871→                                ? currentUser.name
   872→                                : getUserName(msg.authorId);
   873→
   874→                            return (
   875→                                <div
   876→                                    key={String(msg._id)}
   877→                                    className={`bg-white dark:bg-zinc-800 rounded-xl border border-slate-200 dark:border-zinc-700 p-4 ${
   878→                                        !msg.isRead && !isCurrentUser ? 'border-l-4 border-l-emerald-400' : ''
   879→                                    }`}
   880→                                >
   881→                                    <div className="flex items-start gap-3">
   882→                                        <div className={`w-8 h-8 rounded-full flex items-center justify-center flex-shrink-0 text-xs font-bold ${
   883→                                            isCurrentUser
   884→                                                ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400'
   885→                                                : 'bg-slate-100 text-slate-600 dark:bg-zinc-700 dark:text-zinc-300'
   886→                                        }`}>
   887→                                            {getInitials(authorName)}
   888→                                        </div>
   889→                                        <div className="flex-1 min-w-0">
   890→                                            <div className="flex items-center gap-2 flex-wrap">
   891→                                                <span className="font-semibold text-sm text-slate-900 dark:text-white">
   892→                                                    {authorName}
   893→                                                </span>
   894→                                                {msg.matterTitle && (
   895→                                                    <span className="text-xs px-1.5 py-0.5 rounded bg-slate-100 dark:bg-zinc-700 text-slate-500 dark:text-zinc-400">
   896→                                                        {msg.matterTitle}
   897→                                                    </span>
   898→                                                )}
   899→                                                {!msg.isRead && !isCurrentUser && (
   900→                                                    <span className="w-2 h-2 rounded-full bg-emerald-500" />
   901→                                                )}
   902→                                            </div>
   903→                                            <p className="mt-1 text-sm text-slate-700 dark:text-zinc-300 leading-relaxed">
   904→                                                {msg.content}
   905→                                            </p>
   906→                                            <p className="mt-1 text-xs text-slate-400 dark:text-zinc-500">
   907→                                                {msg.timestamp ? timeAgo(msg.timestamp) : ''}
   908→                                            </p>
   909→                                        </div>
   910→                                    </div>
   911→                                </div>
   912→                            );
   913→                        })}
   914→                    </div>
   915→                )}
   916→            </div>
   917→        );
   918→    };
   919→
   920→    // ── Main Render ──────────────────────────────────────────────────────
   921→    const renderTabContent = () => {
   922→        switch (activeTab) {
   923→            case 'overview': return renderOverview();
   924→            case 'matters': return renderMatters();
   925→            case 'documents': return renderDocuments();
   926→            case 'messages': return renderMessages();
   927→        }
   928→    };
   929→
   930→    return (
   931→        <div className="min-h-full">
   932→            {/* Impersonation Banner — shown when admin is viewing as this client */}
   933→            {originalUser && (
   934→                <div className="mb-4 px-4 py-3 rounded-xl bg-amber-50 dark:bg-amber-900/20 border border-amber-200 dark:border-amber-800/50 flex items-center justify-between gap-3">
   935→                    <div className="flex items-center gap-2">
   936→                        <EyeIcon className="w-4 h-4 text-amber-600 dark:text-amber-400 flex-shrink-0" />
   937→                        <p className="text-sm text-amber-800 dark:text-amber-300 font-medium">
   938→                            You are previewing the portal as <strong>{currentUser.name || currentUser.email}</strong>
   939→                        </p>
   940→                    </div>
   941→                    <button
   942→                        onClick={revertToOriginalUser}
   943→                        className="px-3 py-1.5 text-xs font-bold bg-amber-600 text-white rounded-lg hover:bg-amber-700 transition-colors flex-shrink-0"
   944→                    >
   945→                        Return to Admin
   946→                    </button>
   947→                </div>
   948→            )}
   949→            {/* Header */}
   950→            <div className="mb-6">
   951→                <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 mb-2">
   952→                    <div>
   953→                        <h1 className="text-2xl sm:text-3xl font-bold text-slate-900 dark:text-white">
   954→                            Welcome, {currentUser.name?.split(' ')[0] || 'Client'}
   955→                        </h1>
   956→                        <p className="text-sm text-slate-500 dark:text-zinc-400 mt-1">
   957→                            Your legal matters portal &mdash; stay informed and connected
   958→                        </p>
   959→                    </div>
   960→                    <div className="flex items-center gap-2 flex-shrink-0">
   961→                        <button
   962→                            onClick={() => openModal('newLead', null, { name: currentUser.name, email: currentUser.email, isClientRequest: true })}
   963→                            className="inline-flex items-center gap-2 px-4 py-2.5 bg-emerald-600 text-white rounded-lg font-semibold hover:bg-emerald-700 text-sm transition-colors shadow-sm"
   964→                        >
   965→                            <PlusIcon className="w-4 h-4" />
   966→                            Request Service
   967→                        </button>
   968→                        {/* Theme Toggle */}
   969→                        <button
   970→                            onClick={toggleTheme}
   971→                            className="p-2 rounded-lg text-slate-500 dark:text-zinc-400 hover:bg-slate-100 dark:hover:bg-zinc-800 transition-colors"
   972→                            title={isDark ? 'Switch to light mode' : 'Switch to dark mode'}
   973→                        >
   974→                            {isDark ? <SunIcon className="w-4 h-4" /> : <MoonIcon className="w-4 h-4" />}
   975→                        </button>
   976→                        <button
   977→                            onClick={() => logout()}
   978→                            className="px-3 py-2.5 text-xs font-bold text-slate-500 dark:text-zinc-400 hover:text-slate-700 dark:hover:text-zinc-200 hover:bg-slate-100 dark:hover:bg-zinc-800 rounded-lg transition-colors"
   979→                        >
   980→                            Sign Out
   981→                        </button>
   982→                    </div>
   983→                </div>
   984→            </div>
   985→
   986→            {/* Tabs */}
   987→            <div className="mb-6 border-b border-slate-200 dark:border-zinc-700 overflow-x-auto">
   988→                <nav className="-mb-px flex space-x-0 sm:space-x-1 md:space-x-4">
   989→                    <TabButton
   990→                        label="Overview"
   991→                        tab="overview"
   992→                        active={activeTab}
   993→                        onClick={() => setActiveTab('overview')}
   994→                        icon={<ScalesIcon className="w-4 h-4" />}
   995→                    />
   996→                    <TabButton
   997→                        label="Matters"
   998→                        tab="matters"
   999→                        active={activeTab}
  1000→                        onClick={() => setActiveTab('matters')}
  1001→                        icon={<MattersIcon className="w-4 h-4" />}
  1002→                        badge={clientMatters.length}
  1003→                    />
  1004→                    <TabButton
  1005→                        label="Documents"
  1006→                        tab="documents"
  1007→                        active={activeTab}
  1008→                        onClick={() => setActiveTab('documents')}
  1009→                        icon={<DocumentIcon className="w-4 h-4" />}
  1010→                        badge={sharedDocsCount}
  1011→                    />
  1012→                    <TabButton
  1013→                        label="Messages"
  1014→                        tab="messages"

  1016→                        onClick={() => setActiveTab('messages')}

  1019→                    />
  1020→                </nav>

  1023→            {/* Tab Content */}

  1025→        </div>
