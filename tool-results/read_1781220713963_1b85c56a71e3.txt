   233→export const sendMessage = async (
   234→    history: AloaMessage[],
   235→    context: {
   236→        appState: AppState;
   237→        currentUser: User;
   238→        currentHistoryEntry: HistoryEntry;
   239→        localFiles?: any[];
   240→        aloaXLibrary?: any[];
   241→        isFirmSearchEnabled?: boolean;
   242→        searchBrain?: (query: string) => Promise<string>;
   243→        injectedContext?: AriaChatContext | null;
   244→    },
   245→    modelPreference: 'auto' | 'flash' | 'pro' = 'auto'
   246→): Promise<{ text?: string; toolCalls?: any[]; modelUsed?: string }> => {
   247→
   248→    const { appState, currentUser, currentHistoryEntry, localFiles, aloaXLibrary, isFirmSearchEnabled } = context;
   249→    const firmKey = appState.firmDetails?.aiSettings?.firmGeminiApiKey || getGeminiApiKey();
   250→
   251→    // --- ALOA BRAIN: SEMANTIC RETRIEVAL ---
   252→    let semanticContext: string | undefined;
   253→    if (context.isFirmSearchEnabled && appState.firmDetails?.id) {
   254→        try {
   255→            const lastUserMsg = history.filter(m => m.role === 'user').pop()?.content;
   256→            if (typeof lastUserMsg === 'string' && context.searchBrain) {
   257→                semanticContext = await context.searchBrain(lastUserMsg);
   258→            }
   259→        } catch (e) {
   260→            console.warn('[Brain] Search skipped:', e);
   261→        }
   262→    }
   263→
   264→    const systemInstruction = getSystemInstruction(
   265→        appState,
   266→        currentUser,
   267→        currentHistoryEntry,
   268→        localFiles,
   269→        aloaXLibrary,
   270→        isFirmSearchEnabled,
   271→        semanticContext,
   272→        new Date().toISOString(),
   273→        context.injectedContext
   274→    );
   275→
   276→    const preferredModelName = 
   277→        modelPreference === 'pro' ? AI_CONFIG.gemini.proModel : 
   278→        modelPreference === 'flash' ? (AI_CONFIG.gemini as any).flashModel : 
   279→        AI_CONFIG.gemini.defaultModel;
   280→
   281→    const contents: Content[] = history.map((msg): Content | null => {
   282→        if (msg.role === 'tool' && msg.toolResult) {
   283→            return {
   284→                role: 'user', 
   285→                parts: [{
   286→                    functionResponse: {
   287→                        name: msg.toolResult.toolName,
   288→                        response: { result: msg.toolResult.output }
   289→                    }
   290→                }]
   291→            };
   292→        }
   293→
   294→        if (msg.role === 'model' && msg.toolCalls) {
   295→            return {
   296→                role: 'model',
   297→                parts: msg.toolCalls.map(tc => ({
   298→                    functionCall: { name: tc.name, args: tc.args }
   299→                }))
   300→            };
   301→        }
   302→
   303→        const text = typeof msg.content === 'string' ? stripPII(msg.content) : '';
   304→        if (!text) return null;
   305→        return { role: msg.role === 'user' ? 'user' : 'model', parts: [{ text }] };
   306→    }).filter((c): c is Content => c !== null);
   307→
   308→    const modelsToTry = [preferredModelName, ...AI_CONFIG.gemini.fallbackPlan.filter(m => m !== preferredModelName)];
   309→    let lastError = null;
   310→
   311→    // ─── STRATEGY: Direct-First when client key is available ───────────────────
   312→    // This bypasses Convex routing entirely, which avoids the issue where the
   313→    // frontend may be pointed at a different Convex deployment than expected.
   314→    const clientKey = firmKey || getGeminiApiKey();
   315→    if (clientKey) {
   316→        for (const modelName of modelsToTry) {
   317→            try {
   318→                const modelTag = modelName.includes('models/') ? modelName : `models/${modelName}`;
   319→                const directUrl = `https://generativelanguage.googleapis.com/v1beta/${modelTag}:generateContent?key=${clientKey}`;
   320→
   321→                const directResponse = await fetch(directUrl, {
   322→                    method: "POST",
   323→                    headers: { "Content-Type": "application/json" },
   324→                    body: JSON.stringify({
   325→                        contents,
   326→                        systemInstruction: { parts: [{ text: stripPII(systemInstruction) }] },
   327→                        tools: [{ functionDeclarations: tools }],
   328→                        generationConfig: { temperature: 0.2, topP: 0.1, topK: 40 },
   329→                        safetySettings: [
   330→                            { category: "HARM_CATEGORY_HARASSMENT", threshold: "BLOCK_NONE" },
   331→                            { category: "HARM_CATEGORY_HATE_SPEECH", threshold: "BLOCK_NONE" },
   332→                            { category: "HARM_CATEGORY_SEXUALLY_EXPLICIT", threshold: "BLOCK_NONE" },
   333→                            { category: "HARM_CATEGORY_DANGEROUS_CONTENT", threshold: "BLOCK_NONE" }
   334→                        ]
   335→                    })
   336→                });
   337→
   338→                if (directResponse.ok) {
   339→                    const data = await directResponse.json();
   340→                    const candidate = data?.candidates?.[0];
   341→                    const rawText = candidate?.content?.parts?.find((p: any) => p.text)?.text || '';
   342→                    const functionCalls = candidate?.content?.parts?.filter((p: any) => p.functionCall)?.map((p: any) => p.functionCall) || undefined;
   343→
   344→                    const agent = currentHistoryEntry?.view === 'atriumEngine' || ['properties', 'propertyDetail'].includes(currentHistoryEntry?.view) ? 'ARIA' : 'ALOA';
   345→                    const { sanitized } = validateAIResponse(rawText || '', agent as 'ARIA' | 'ALOA');
   346→                    return { text: sanitized, toolCalls: functionCalls, modelUsed: `${modelName}` };
   347→                }
   348→
   349→                // Non-OK response — parse error and try next model
   350→                const errData = await directResponse.json().catch(() => ({}));
   351→                lastError = new Error(`AI Request Failed (${directResponse.status}): ${errData?.error?.message || directResponse.statusText}`);
   352→                console.warn(`[AI Direct] ${modelName} failed (${directResponse.status}). Trying next...`);
   353→            } catch (directErr: any) {
   354→                lastError = directErr;
   355→                console.warn(`[AI Direct] ${modelName} threw:`, directErr.message);
   356→            }
   357→        }
   358→    }
   359→
   360→    // ─── FALLBACK: Convex Proxy ────────────────────────────────────────────────
   361→    for (const modelName of modelsToTry) {
   362→        try {
   363→            const response = await convex.action(api.ai.generateContent, {
   364→                modelName,
   365→                contents,
   366→                systemInstruction: { parts: [{ text: stripPII(systemInstruction) }] },
   367→                tools: [{ functionDeclarations: tools }],
   368→                generationConfig: { temperature: 0.2, topP: 0.1, topK: 40 },
   369→                firmGeminiApiKey: clientKey
   370→            });
   371→
   372→            const candidate = response?.candidates?.[0];
   373→            const rawText = candidate?.content?.parts?.find((p: any) => p.text)?.text || '';
   374→            const functionCalls = candidate?.content?.parts?.filter((p: any) => p.functionCall)?.map((p: any) => p.functionCall) || undefined;
   375→
   376→            const agent = currentHistoryEntry?.view === 'atriumEngine' || ['properties', 'propertyDetail'].includes(currentHistoryEntry?.view) ? 'ARIA' : 'ALOA';
   377→            const { sanitized } = validateAIResponse(rawText || '', agent as 'ARIA' | 'ALOA');
   378→            return { text: sanitized, toolCalls: functionCalls, modelUsed: `${modelName} (Proxy)` };
   379→        } catch (proxyErr: any) {
   380→            lastError = proxyErr;
   381→            console.warn(`[AI Proxy] ${modelName} failed via Convex:`, proxyErr.message);
   382→        }
   383→    }
   384→
   385→    console.error("All AI paths exhausted. Last error:", lastError);
   386→    throw lastError || new Error("All AI models are currently unavailable.");
   387→};

   389→/** Stream chat text (no tools) for responsive ALOA UI — tool flows still use sendMessage. */
   390→export const streamMessage = async (
   391→    history: AloaMessage[],
   392→    context: {
   393→        appState: AppState;
   394→        currentUser: User;
   395→        currentHistoryEntry: HistoryEntry;
   396→        localFiles?: any[];
   397→        aloaXLibrary?: any[];
   398→        isFirmSearchEnabled?: boolean;
   399→        searchBrain?: (query: string) => Promise<string>;
   400→        injectedContext?: AriaChatContext | null;
   401→    },
   402→    onChunk: (text: string) => void,
   403→    modelPreference: 'auto' | 'flash' | 'pro' = 'auto',
   404→    signal?: AbortSignal
   405→): Promise<{ text: string; modelUsed?: string }> => {
   406→    const { appState, currentUser, currentHistoryEntry, localFiles, aloaXLibrary, isFirmSearchEnabled } = context;
   407→    const firmKey = appState.firmDetails?.aiSettings?.firmGeminiApiKey || getGeminiApiKey();
   408→
   409→    let semanticContext: string | undefined;
   410→    if (context.isFirmSearchEnabled && appState.firmDetails?.id) {
   411→        try {
   412→            const lastUserMsg = history.filter(m => m.role === 'user').pop()?.content;
   413→            if (typeof lastUserMsg === 'string' && context.searchBrain) {
   414→                semanticContext = await context.searchBrain(lastUserMsg);
   415→            }
   416→        } catch (e) {
   417→            console.warn('[Brain] Search skipped:', e);
   418→        }
   419→    }
   420→
   421→    const systemInstruction = getSystemInstruction(
   422→        appState,
   423→        currentUser,
   424→        currentHistoryEntry,
   425→        localFiles,
   426→        aloaXLibrary,
   427→        isFirmSearchEnabled,
   428→        semanticContext,
   429→        new Date().toISOString(),
   430→        context.injectedContext
   431→    );
   432→
   433→    const preferredModelName =
   434→        modelPreference === 'pro' ? AI_CONFIG.gemini.proModel :
   435→        modelPreference === 'flash' ? (AI_CONFIG.gemini as any).flashModel :
   436→        (AI_CONFIG.gemini as any).flashModel || AI_CONFIG.gemini.defaultModel;
   437→
   438→    const contents: Content[] = history.map((msg): Content | null => {
   439→        const text = typeof msg.content === 'string' ? stripPII(msg.content) : '';
   440→        if (!text || msg.role === 'tool') return null;
   441→        return { role: msg.role === 'user' ? 'user' : 'model', parts: [{ text }] };
   442→    }).filter((c): c is Content => c !== null);
   443→
   444→    const clientKey = firmKey || getGeminiApiKey();
   445→    if (!clientKey) throw new Error('No Gemini API key configured.');
   446→
   447→    const modelTag = preferredModelName.includes('models/') ? preferredModelName : `models/${preferredModelName}`;
   448→    const url = `https://generativelanguage.googleapis.com/v1beta/${modelTag}:streamGenerateContent?alt=sse&key=${clientKey}`;
   449→    const response = await fetch(url, {
   450→        method: 'POST',
   451→        headers: { 'Content-Type': 'application/json' },
   452→        signal,
   453→        body: JSON.stringify({
   454→            contents,
   455→            systemInstruction: { parts: [{ text: stripPII(systemInstruction) }] },
   456→            generationConfig: { temperature: 0.2, topP: 0.1, topK: 40 },
   457→            safetySettings: [
   458→                { category: 'HARM_CATEGORY_HARASSMENT', threshold: 'BLOCK_NONE' },
   459→                { category: 'HARM_CATEGORY_HATE_SPEECH', threshold: 'BLOCK_NONE' },
   460→                { category: 'HARM_CATEGORY_SEXUALLY_EXPLICIT', threshold: 'BLOCK_NONE' },
   461→                { category: 'HARM_CATEGORY_DANGEROUS_CONTENT', threshold: 'BLOCK_NONE' },
   462→            ],
   463→        }),
   464→    });
   465→
   466→    if (!response.ok) {
   467→        const errData = await response.json().catch(() => ({}));
   468→        throw new Error(`Stream failed (${response.status}): ${errData?.error?.message || response.statusText}`);
   469→    }
   470→    if (!response.body) throw new Error('No stream body');
   471→
   472→    const agent = currentHistoryEntry?.view === 'atriumEngine' || ['properties', 'propertyDetail'].includes(currentHistoryEntry?.view || '') ? 'ARIA' : 'ALOA';
   473→    let fullText = '';
   474→    const reader = response.body.getReader();
   475→    const decoder = new TextDecoder('utf-8');
   476→    let buffer = '';
   477→
   478→    while (true) {
   479→        const { done, value } = await reader.read();
   480→        if (done) break;
   481→        buffer += decoder.decode(value, { stream: true });
   482→        const normalized = buffer.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
   483→        const parts = normalized.split('\n\n');
   484→        buffer = parts.pop() || '';
   485→        for (const part of parts) {
   486→            const line = part.trim();
   487→            if (!line.startsWith('data: ')) continue;
   488→            const dataStr = line.slice(6).trim();
   489→            if (!dataStr || dataStr === '[DONE]') continue;
   490→            try {
   491→                const data = JSON.parse(dataStr);
   492→                const text = data?.candidates?.[0]?.content?.parts?.[0]?.text;
   493→                if (text) {
   494→                    fullText += text;
   495→                    onChunk(text);
   496→                }
   497→            } catch { /* skip */ }
   498→        }
   499→    }
   500→
   501→    const { sanitized } = validateAIResponse(fullText, agent as 'ARIA' | 'ALOA');
   502→    return { text: sanitized, modelUsed: preferredModelName };
   503→};

   507→    context: { appState: AppState; currentUser: User; },

   509→    signal?: AbortSignal
   510→) => {
   511→    const contents: Content[] = history.map(msg => ({
   512→        role: msg.role === 'user' ? 'user' : 'model',
   513→        parts: [{ text: typeof msg.content === 'string' ? msg.content : '' }]
   514→    })).filter(c => c.parts[0].text);
   515→
   516→    const systemInstruction = `
   517→    ${ALOA_PRECISION_PROTOCOL}
   518→    
   519→    ${DRAFTPRO_HTML_FORMATTING_RULES}
   520→
   521→    CONTEXT: The user is a legal practitioner based in Asaba, Delta State, Nigeria. 
   522→    Unless explicitly instructed otherwise (e.g., "Lagos High Court"), ALL drafts must be tailored to the jurisdiction of the High Court of Delta State or relevant Delta State laws.
   523→    
   524→    TASK: Write a perfectly formatted, authoritative legal document.
   525→    
   526→    Adhere strictly to Nigerian Law, specifically Delta State Civil Procedure Rules where applicable for litigation, or general Nigerian statutes (CAMA 2020, Land Use Act).
   527→    `;
   528→
   529→    const modelsToTry = AI_CONFIG.gemini.fallbackPlan;
   530→    let lastError = null;
   531→
   532→    const clientKey = getGeminiApiKey();
   533→    if (!clientKey) {
   534→        throw new Error("No Gemini API key found. Please add VITE_GEMINI_API_KEY in your environment settings.");
   535→    }
   536→    
   537→    // ─── STRATEGY: Direct REST Stream (mirrors the working sendMessage pattern) ─
   538→    for (const modelToTest of modelsToTry) {
   539→        try {
   540→            const modelTag = modelToTest.includes('models/') ? modelToTest : `models/${modelToTest}`;
   541→            const url = `https://generativelanguage.googleapis.com/v1beta/${modelTag}:streamGenerateContent?alt=sse&key=${clientKey}`;
   542→
   543→            const response = await fetch(url, {
   544→                method: 'POST',
   545→                headers: { 'Content-Type': 'application/json' },
   546→                signal,
   547→                body: JSON.stringify({
   548→                    contents,
   549→                    systemInstruction: { parts: [{ text: systemInstruction }] },
   550→                    generationConfig: { temperature: 0.7, maxOutputTokens: 8192 },
   551→                    safetySettings: [
   552→                        { category: 'HARM_CATEGORY_HARASSMENT', threshold: 'BLOCK_NONE' },
   553→                        { category: 'HARM_CATEGORY_HATE_SPEECH', threshold: 'BLOCK_NONE' },
   554→                        { category: 'HARM_CATEGORY_SEXUALLY_EXPLICIT', threshold: 'BLOCK_NONE' },
   555→                        { category: 'HARM_CATEGORY_DANGEROUS_CONTENT', threshold: 'BLOCK_NONE' },
   556→                        { category: 'HARM_CATEGORY_CIVIC_INTEGRITY', threshold: 'BLOCK_NONE' }
   557→                    ]
   558→                })
   559→            });
   560→
   561→            if (!response.ok) {
   562→                const errData = await response.json().catch(() => ({}));
   563→                throw new Error(`API error ${response.status}: ${errData?.error?.message || response.statusText}`);
   564→            }
   565→
   566→            if (!response.body) throw new Error('No response body from stream.');
   567→
   568→            const reader = response.body.getReader();
   569→            const decoder = new TextDecoder('utf-8');
   570→            let buffer = '';
   571→            let chunkCount = 0;
   572→
   573→            while (true) {
   574→                const { done, value } = await reader.read();
   575→                if (done) break;
   576→
   577→                buffer += decoder.decode(value, { stream: true });
   578→                // Normalize line endings
   579→                const normalized = buffer.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
   580→                const parts = normalized.split('\n\n');
   581→                buffer = parts.pop() || '';
   582→
   583→                for (const part of parts) {
   584→                    const line = part.trim();
   585→                    if (!line.startsWith('data: ')) continue;
   586→                    const dataStr = line.slice(6).trim();
   587→                    if (!dataStr || dataStr === '[DONE]') continue;
   588→                    try {
   589→                        const data = JSON.parse(dataStr);
   590→                        const candidate = data?.candidates?.[0];
   591→                        if (candidate?.finishReason === 'SAFETY') continue;
   592→                        const text = candidate?.content?.parts?.[0]?.text;
   593→                        if (text) { onChunk(text); chunkCount++; }
   594→                    } catch (_) { /* skip malformed chunks */ }
   595→                }
   596→            }
   597→
   598→            // Flush any remaining buffer
   599→            if (buffer.trim()) {
   600→                const line = buffer.trim();
   601→                if (line.startsWith('data: ')) {
   602→                    const dataStr = line.slice(6).trim();
   603→                    if (dataStr && dataStr !== '[DONE]') {
   604→                        try {
   605→                            const data = JSON.parse(dataStr);
   606→                            const text = data?.candidates?.[0]?.content?.parts?.[0]?.text;
   607→                            if (text) { onChunk(text); chunkCount++; }
   608→                        } catch (_) {}
   609→                    }
   610→                }
   611→            }
   612→
   613→            console.log(`[Draft Stream] ${modelToTest} completed. Chunks: ${chunkCount}`);
   614→            return; // ✅ Success
   615→
   616→        } catch (err: any) {
   617→            if (err.name === 'AbortError' || signal?.aborted) throw err; // user cancelled
   618→            lastError = err;
   619→            console.warn(`[Draft Stream] ${modelToTest} failed:`, err.message);
   620→        }
   621→    }
   622→
   623→    // ─── FALLBACK: Convex Proxy ────────────────────────────────────────────────
   624→    for (const modelToTest of modelsToTry) {
   625→        try {
   626→            const streamUrl = CONVEX_URL.replace('.cloud', '.site') + '/ai/stream';
   627→            const response = await fetch(streamUrl, {
   628→                method: "POST",
   629→                headers: { "Content-Type": "application/json" },
   630→                signal,
   631→                body: JSON.stringify({
   632→                    model: modelToTest,
   633→                    contents,
   634→                    systemInstruction,
   635→                    apiKey: clientKey
   636→                })
   637→            });
   638→
   639→            if (!response.ok || !response.body) {
   640→                throw new Error("Stream connection failed.");
   641→            }
   642→
   643→            const reader = response.body.getReader();
   644→            const decoder = new TextDecoder("utf-8");
   645→            let buffer = "";
   646→
   647→            while (true) {
   648→                const { done, value } = await reader.read();
   649→                if (done) {
   650→                    if (buffer.trim()) {
   651→                        if (buffer.startsWith('data: ')) {
   652→                            const dataStr = buffer.replace('data: ', '').trim();
   653→                            if (dataStr !== '[DONE]') {
   654→                                try {
   655→                                    const data = JSON.parse(dataStr);
   656→                                    const chunkText = data?.candidates?.[0]?.content?.parts?.[0]?.text;
   657→                                    if (chunkText) onChunk(chunkText);
   658→                                } catch (e) {}
   659→                            }
   660→                        }
   661→                    }
   662→                    break;
   663→                }
   664→                
   665→                buffer += decoder.decode(value, { stream: true });
   666→                const parts = buffer.split('\n\n');
   667→                buffer = parts.pop() || ''; 
   668→                
   669→                for (const part of parts) {
   670→                    if (part.startsWith('data: ')) {
   671→                        const dataStr = part.replace('data: ', '').trim();
   672→                        if (dataStr === '[DONE]') continue;
   673→                        try {
   674→                            const data = JSON.parse(dataStr);
   675→                            const chunkText = data?.candidates?.[0]?.content?.parts?.[0]?.text;
   676→                            if (chunkText) onChunk(chunkText);
   677→                        } catch (e) {}
   678→                    }
   679→                }
   680→            }
   681→            return; // Success
   682→        } catch (error: any) {
   683→            lastError = error;
   684→            console.warn(`[Drafting Proxy] Model ${modelToTest} failed via Convex:`, error.message);
   685→        }
   686→    }
   687→
   688→    throw lastError || new Error("Drafting failed to initialize via proxy.");
   689→};
