    30→export const getMaintenanceTicketsByTenant = query({
    31→  args: { tenantId: v.string() },
    32→  handler: async (ctx, args) => {
    33→    const tickets = await ctx.db
    34→      .query("maintenance_tickets")
    35→      .withIndex("by_tenant", (q) => q.eq("tenantId", args.tenantId))
    36→      .order("desc")
    37→      .collect();
    38→
    39→    // Also try matching by email in case the tenantId stored is the user's email
    40→    // This handles the case where the invite was created with the email as the tenant ID
    41→    if (tickets.length === 0 && args.tenantId.includes('@')) {
    42→      // The tenantId looks like an email, also search by the user's Convex _id
    43→      const user: any = await ctx.db
    44→        .query("users")
    45→        .withIndex("by_token", (q) => q.eq("tokenIdentifier", args.tenantId.toLowerCase()))
    46→        .first();
    47→      if (user) {
    48→        const byUserId = await ctx.db
    49→          .query("maintenance_tickets")
    50→          .withIndex("by_tenant", (q) => q.eq("tenantId", String(user._id)))
    51→          .order("desc")
    52→          .collect();
    53→        return byUserId;
    54→      }
    55→    }
    56→
    57→    return tickets;
    58→  },
    59→});

    88→/** Generate a crypto-random invite token (12 chars, URL-safe) */
    89→function generateToken(): string {
    90→  const chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
    91→  const arr = new Uint8Array(12);
    92→  // @ts-ignore — crypto available in Convex runtime
    93→  crypto.getRandomValues(arr);
    94→  return Array.from(arr, b => chars[b % chars.length]).join("");
    95→}

   272→/**
   273→ * Internal mutation — only used by createPortalInvite action to write the DB record.
   274→ * Not exported for direct frontend use.
   275→ */
   276→export const insertInviteRecord = mutation({
   277→  args: {
   278→    firmId: v.string(),
   279→    inviterId: v.string(),
   280→    inviteeEmail: v.optional(v.string()),
   281→    inviteeName: v.optional(v.string()),
   282→    inviteePhone: v.optional(v.string()),
   283→    portalType: v.union(v.literal("client"), v.literal("resident")),
   284→    relatedId: v.optional(v.string()),
   285→    token: v.optional(v.string()),
   286→    channel: v.optional(v.string()),
   287→    message: v.optional(v.string()),
   288→    expiresAt: v.number(),
   289→  },
   290→  handler: async (ctx, args) => {
   291→    const now = Date.now();
   292→
   293→    // Clean up any existing invites for the same email + firm + portal type.
   294→    // This prevents "invitation already accepted" errors when re-inviting
   295→    // after a previous invite was deleted via the simple deletePortalInvite
   296→    // (which doesn't clean up). We mark old invites as "superseded" rather
   297→    // than deleting them, so there's an audit trail.
   298→    const email = (args.inviteeEmail || "").toLowerCase().trim();
   299→    if (email) {
   300→      const existingInvites = await ctx.db
   301→        .query("portal_invites")
   302→        .withIndex("by_email", (q) => q.eq("inviteeEmail", email))
   303→        .collect();
   304→
   305→      for (const inv of existingInvites) {
   306→        // Only supersede invites for the same firm and portal type
   307→        if (inv.firmId === args.firmId && inv.portalType === args.portalType) {
   308→          if (inv.status === "pending" || inv.status === "accepted") {
   309→            await ctx.db.patch(inv._id, {
   310→              status: "superseded",
   311→              updatedAt: now,
   312→            });
   313→          }
   314→        }
   315→      }
   316→    }
   317→
   318→    return await ctx.db.insert("portal_invites", {
   319→      firmId: args.firmId,
   320→      inviterId: args.inviterId,
   321→      inviteeEmail: args.inviteeEmail,
   322→      inviteeName: args.inviteeName,
   323→      inviteePhone: args.inviteePhone,
   324→      portalType: args.portalType,
   325→      relatedId: args.relatedId,
   326→      token: args.token ?? undefined,
   327→      channel: args.channel,
   328→      message: args.message,
   329→      status: "pending",
   330→      expiresAt: args.expiresAt,
   331→      createdAt: now,
   332→      updatedAt: now,
   333→    });
   334→  },
   335→});

   517→/**
   518→ * deletePortalInviteAndCleanup — Permanently deletes a portal invite AND resets
   519→ * the associated portal user account so the same email can be re-invited cleanly.
   520→ *
   521→ * This prevents the "invitation already accepted" error when an admin:
   522→ * 1. Deletes a portal access record
   523→ * 2. Creates a new invite for the same email
   524→ *
   525→ * Steps:
   526→ * 1. Delete the portal_invites record
   527→ * 2. Find the user with the matching email + Client/Tenant role
   528→ * 3. Reset their role, verification, and password so they're treated as a brand-new invitee
   529→ */
   530→export const deletePortalInviteAndCleanup = mutation({
   531→  args: {
   532→    inviteId: v.id("portal_invites"),
   533→    inviteeEmail: v.optional(v.string()),
   534→  },
   535→  handler: async (ctx, args) => {
   536→    // 1. Delete the specified invite record
   537→    await ctx.db.delete(args.inviteId);
   538→
   539→    // 2. Also delete ANY other invite records for the same email
   540→    //    to prevent residual "already accepted" errors on re-invite
   541→    const email = (args.inviteeEmail || "").toLowerCase().trim();
   542→    if (!email) return;
   543→
   544→    const otherInvites = await ctx.db
   545→      .query("portal_invites")
   546→      .withIndex("by_email", (q) => q.eq("inviteeEmail", email))
   547→      .collect();
   548→
   549→    for (const inv of otherInvites) {
   550→      if (String(inv._id) === String(args.inviteId)) continue; // Already deleted above
   551→      try { await ctx.db.delete(inv._id); } catch (e) { /* ignore */ }
   552→    }
   553→
   554→    // 3. Find and reset the associated portal user
   555→    const existingUser = await ctx.db
   556→      .query("users")
   557→      .withIndex("by_token", (q) => q.eq("tokenIdentifier", email))
   558→      .first();
   559→
   560→    if (!existingUser) return;
   561→
   562→    // Only reset if the user has a Client or Tenant role (portal user)
   563→    const role = (existingUser as any).role;
   564→    if (role === "Client" || role === "Tenant" || role === "Portal User") {
   565→      // Reset the user so they can be re-invited cleanly
   566→      // Set isVerified to false so they can't log in with old credentials
   567→      // Remove the password so setup-password creates a fresh one
   568→      // Clear firmId so it will be set fresh on the next invite acceptance
   569→      await ctx.db.patch(existingUser._id, {
   570→        isVerified: false,
   571→        emailVerified: false,
   572→        password: undefined,
   573→        role: "Pending",
   574→        firmId: undefined,
   575→      } as any);
   576→    }
   577→  },
   578→});

   627→/**
   628→ * verifyInviteToken — validates a portal invite token and returns
   629→ * the invite details + whether the user already has an account.
   630→ * Used by the /setup-password page to determine what to show.
   631→ *
   632→ * ROBUSTNESS FIX: If the invite is "accepted" but the user's role is
   633→ * "Pending" (meaning their access was deleted via deletePortalInviteAndCleanup
   634→ * and they need to re-setup), we allow them to proceed. This prevents the
   635→ * "invitation already accepted" dead-end when a user is re-invited after
   636→ * their portal access was removed.
   637→ */
   638→export const verifyInviteToken = query({
   639→  args: { token: v.string() },
   640→  handler: async (ctx, args) => {
   641→    const results = await ctx.db
   642→      .query("portal_invites")
   643→      .withIndex("by_token", (q) => q.eq("token", args.token))
   644→      .collect();
   645→    const invite = results[0] || null;
   646→
   647→    if (!invite) {
   648→      return { valid: false, reason: "not_found" } as const;
   649→    }
   650→
   651→    // Check expiry (note: we don't patch status here since this is a query/read-only)
   652→    if (invite.expiresAt < Date.now()) {
   653→      return { valid: false, reason: "expired" } as const;
   654→    }
   655→
   656→    if (invite.status === "revoked") {
   657→      return { valid: false, reason: "revoked" } as const;
   658→    }
   659→
   660→    // Check if the user exists and their current state
   661→    const email = invite.inviteeEmail?.toLowerCase().trim();
   662→    let existingUser: any = null;
   663→    if (email) {
   664→      existingUser = await ctx.db
   665→        .query("users")
   666→        .withIndex("by_token", (q) => q.eq("tokenIdentifier", email))
   667→        .first();
   668→      // Case-insensitive fallback
   669→      if (!existingUser) {
   670→        existingUser = await ctx.db
   671→          .query("users")
   672→          .withIndex("by_token", (q) => q.eq("tokenIdentifier", email.toLowerCase()))
   673→          .first();
   674→      }
   675→    }
   676→
   677→    if (invite.status === "accepted") {
   678→      // ROBUSTNESS: If the user's role is Pending (their access was reset by
   679→      // deletePortalInviteAndCleanup), allow them to re-accept. This invite
   680→      // might be a stale record from before the cleanup ran. The user needs
   681→      // to set up their password again, so we should let them through.
   682→      const userRole = existingUser?.role;
   683→      const isUserUnverified = existingUser && !existingUser.isVerified;
   684→      if (userRole === "Pending" || isUserUnverified) {
   685→        // User was reset — allow re-accepting this invite
   686→        // Fall through to return valid: true
   687→      } else {
   688→        return { valid: false, reason: "already_accepted" } as const;
   689→      }
   690→    }
   691→
   692→    // Token is valid and pending — check if user already has an account
   693→    return {
   694→      valid: true,
   695→      invite: {
   696→        _id: invite._id,
   697→        inviteeEmail: invite.inviteeEmail,
   698→        inviteeName: invite.inviteeName,
   699→        portalType: invite.portalType,
   700→        firmId: invite.firmId,
   701→        relatedId: invite.relatedId,
   702→        expiresAt: invite.expiresAt,
   703→      },
   704→      hasAccount: !!existingUser,
   705→      hasPassword: !!(existingUser as any)?.password,
   706→    } as const;
   707→  },
   708→});
   709→
   710→/**
   711→ * setupPortalPassword — ACTION that completes the invite flow:
   712→ * 1. Validates the invite token (must be pending + unexpired)
   713→ * 2. If user exists → sets their password + marks verified
   714→ * 3. If user doesn't exist → creates a new user record with the password
   715→ * 4. Marks the invite as accepted
   716→ * 5. Clears the token to prevent replay attacks
   717→ */
   718→export const setupPortalPassword = action({
   719→  args: {
   720→    token: v.string(),
   721→    password: v.string(),
   722→    name: v.optional(v.string()),
   723→  },
   724→  handler: async (ctx, args): Promise<{ success: boolean; message?: string; email?: string }> => {
   725→    // 1. Validate the invite
   726→    const invite = await ctx.runQuery(api.portals.getInviteByToken, { token: args.token });
   727→    if (!invite) return { success: false, message: "Invalid invitation link." };
   728→    if (invite.status === "revoked") return { success: false, message: "This invitation has been revoked." };
   729→    if (invite.status === "expired" || invite.expiresAt < Date.now()) {
   730→      await ctx.runMutation(api.portals.updateInviteRecord, {
   731→        inviteId: invite._id,
   732→        updates: { status: "expired", updatedAt: Date.now() },
   733→      });
   734→      return { success: false, message: "This invitation has expired. Please request a new one." };
   735→    }
   736→
   737→    // ROBUSTNESS: If the invite was already accepted but the user's account was
   738→    // reset (role=Pending, isVerified=false), allow them to re-accept. This can
   739→    // happen when deletePortalInviteAndCleanup resets the user but a stale invite
   740→    // record still exists with status "accepted" and a valid token.
   741→    if (invite.status === "accepted") {
   742→      const emailCheck = (invite.inviteeEmail || "").toLowerCase().trim();
   743→      const existingUserCheck: any = await ctx.runQuery(api.myFunctions.getUser, { tokenIdentifier: emailCheck });
   744→      const wasReset = existingUserCheck?.role === "Pending" || (existingUserCheck && !existingUserCheck.isVerified);
   745→      if (!wasReset) {
   746→        return { success: false, message: "This invitation has already been used." };
   747→      }
   748→      // User was reset — allow re-accepting this invite (fall through)
   749→    }
   750→
   751→    const email = (invite.inviteeEmail || "").toLowerCase().trim();
   752→    if (!email) return { success: false, message: "This invitation has no email address associated." };
   753→
   754→    // 2. Hash the password
   755→    const hashedPassword = await ctx.runAction(internal.authUtils.hashPassword, { password: args.password });
   756→
   757→    // 3. Check if user exists
   758→    const existingUser: any = await ctx.runQuery(api.myFunctions.getUser, { tokenIdentifier: email });
   759→
   760→    if (existingUser) {
   761→      // Update existing user: set password + mark verified
   762→      // Also upgrade their role to Client/Tenant if they were a "Portal User"
   763→      // or had no role assigned, so the frontend can route them correctly.
   764→      const portalRole = invite.portalType === "client" ? "Client" : "Tenant";
   765→      const needsRoleUpdate = !existingUser.role
   766→        || existingUser.role === "Portal User"
   767→        || existingUser.role === "Pending";
   768→      // Ensure firmId is always set — portal users need it for the ProductContext
   769→      // to derive the correct product and for data loading to work properly.
   770→      // When an existing user is re-invited after portal access deletion, their
   771→      // firmId might have been cleared or might never have been set.
   772→      const needsFirmId = !existingUser.firmId || existingUser.firmId !== invite.firmId;
   773→      await ctx.runMutation(internal.myFunctions.updateUserSecurityFields, {
   774→        userId: existingUser._id,
   775→        fields: {
   776→          password: hashedPassword,
   777→          isVerified: true,
   778→          emailVerified: true,
   779→          verificationCode: null,
   780→          ...(args.name && !existingUser.name ? { name: args.name } : {}),
   781→          ...(needsRoleUpdate ? { role: portalRole } : {}),
   782→          ...(needsFirmId ? { firmId: invite.firmId } : {}),
   783→          // Online presence privacy: property/resident portals default to hidden
   784→          portalPresenceHidden: invite.portalType === "resident" ? true : (existingUser as any).portalPresenceHidden ?? false,
   785→        },
   786→      });
   787→    } else {
   788→      // Create new user record for this portal user
   789→      // Use the proper UserRole (Client / Tenant) so the frontend can route
   790→      // them to the correct portal view automatically.
   791→      const portalProduct = invite.portalType === "client" ? "legal" : "property";
   792→      const portalRole = invite.portalType === "client" ? "Client" : "Tenant";
   793→      const userName = args.name || invite.inviteeName || email.split("@")[0];
   794→      await ctx.runMutation(internal.myFunctions.createUser, {
   795→        tokenIdentifier: email,
   796→        name: userName,
   797→        email: email,
   798→        password: hashedPassword,
   799→        role: portalRole,
   800→        product: portalProduct,
   801→        onboardingCompleted: false,
   802→        isVerified: true,
   803→        emailVerified: true,
   804→        firmId: invite.firmId,
   805→        avatarUrl: `https://ui-avatars.com/api/?name=${encodeURIComponent(userName)}&background=random`,
   806→        // Online presence privacy: property/resident portals default to hidden
   807→        // (portal users can't see if the manager is online). Legal/client portals
   808→        // default to visible (clients can see if their lawyer is online).
   809→        portalPresenceHidden: invite.portalType === "resident",
   810→      });
   811→    }
   812→
   813→    // 4. Mark invite as accepted
   814→    await ctx.runMutation(api.portals.updateInviteRecord, {
   815→      inviteId: invite._id,
   816→      updates: {
   817→        status: "accepted",
   818→        acceptedAt: Date.now(),
   819→        updatedAt: Date.now(),
   820→        // Clear token to prevent replay attacks — the invite is now consumed
   821→        token: undefined,
   822→      },
   823→    });
   824→
   825→    return { success: true, email };
   826→  },
   827→});

   890→/**
   891→ * getTenantInfo — Resolves the tenant's property/unit assignments using their
   892→ * user ID and email. This is critical because the tenantId stored in
   893→ * properties/units may be the user's Convex _id OR their email, depending
   894→ * on how the invite was created. This query normalizes both cases.
   895→ *
   896→ * Returns: { tenantId, properties, units } where tenantId is the canonical
   897→ * ID that should be used for all other portal queries.
   898→ */
   899→export const getTenantInfo = query({
   900→  args: { firmId: v.string(), userId: v.string(), email: v.string() },
   901→  handler: async (ctx, args) => {
   902→    const properties = await ctx.db
   903→      .query("properties")
   904→      .withIndex("by_firm", (q) => q.eq("firmId", args.firmId))
   905→      .collect();
   906→
   907→    // Search for tenant in properties using both userId and email
   908→    const tenantProperties: any[] = [];
   909→    const tenantUnits: any[] = [];
   910→    let resolvedTenantId = args.userId;
   911→
   912→    for (const prop of properties) {
   913→      // Check property-level tenant (legacy single-tenant model)
   914→      const propTenantId = (prop as any).currentTenantId || (prop as any).tenantId;
   915→      if (propTenantId === args.userId || propTenantId === args.email ||
   916→          String(propTenantId).toLowerCase() === args.email.toLowerCase()) {
   917→        tenantProperties.push({
   918→          id: String(prop._id),
   919→          name: (prop as any).name || prop.address || 'Unnamed Property',
   920→          address: prop.address,
   921→        });
   922→        if (propTenantId && propTenantId !== args.userId) {
   923→          resolvedTenantId = propTenantId;
   924→        }
   925→      }
   926→
   927→      // Check unit-level tenants (multi-unit model)
   928→      const units = (prop as any).units || [];
   929→      for (const unit of units) {
   930→        const unitTenantId = unit.currentTenantId || unit.tenantId;
   931→        if (unitTenantId === args.userId || unitTenantId === args.email ||
   932→            String(unitTenantId).toLowerCase() === args.email.toLowerCase()) {
   933→          tenantUnits.push({
   934→            id: unit.id || unit._id,

  1185→  args: {},
  1186→  handler: async (ctx) => {

  1209→        // the more common case for "Portal User" accounts

  1211→        migrated++;
  1212→      }

  1221→/**
  1222→ * getClientContactByUserId — Finds the contact record for a portal client
  1223→ * using their user ID. This is needed because the DataProvider skips loading
  1224→ * firm data for portal users, so matterState.contacts is empty. The
  1225→ * ClientDashboard uses this to find the contactId needed for its portal queries.
  1226→ *
  1227→ * Also searches by email as a fallback — the contact's userId field might
  1228→ * not always match the user's Convex _id (e.g. if the contact was created
  1229→ * before the portal user account existed).
  1230→ */
  1231→export const getClientContactByUserId = query({
  1232→  args: { firmId: v.string(), userId: v.string() },
  1233→  handler: async (ctx, args) => {
  1234→    const contacts = await ctx.db
  1235→      .query("contacts")
  1236→      .withIndex("by_firm", (q) => q.eq("firmId", args.firmId))
  1237→      .collect();
  1238→
  1239→    // First try matching by userId field
  1240→    let contact = contacts.find(c => c.userId === args.userId) || null;
  1241→
  1242→    // If not found, try to match by looking up the user's email and matching
  1243→    // the contact's email field. This handles the case where the contact was
  1244→    // created before the portal user account existed.
  1245→    if (!contact) {
  1246→      // Try to look up the user by their Convex _id to get their email
  1247→      let user: any = null;
  1248→      try { user = await ctx.db.get(args.userId as any); } catch {}
  1249→
  1250→      // If that didn't work, the userId might be an email-based tokenIdentifier
  1251→      if (!user) {
  1252→        user = await ctx.db
  1253→          .query("users")
  1254→          .withIndex("by_token", (q) => q.eq("tokenIdentifier", args.userId.toString()))
  1255→          .first();
  1256→      }
  1257→
  1258→      const userEmail = user?.email?.toLowerCase();
  1259→      if (userEmail) {
  1260→        contact = contacts.find(c =>
  1261→          (c as any).email?.toLowerCase() === userEmail
  1262→        ) || null;
  1263→      }
  1264→
  1265→      // Also try matching contact by userId against the user's _id as a string
  1266→      if (!contact && user) {
  1267→        const userDocId = String(user._id);
  1268→        contact = contacts.find(c => c.userId === userDocId) || null;
  1269→      }
  1270→    }
  1271→
  1272→    return contact;
  1273→  },
  1274→});
