    41→// ─── CSV Export ──────────────────────────────────────────────────────────────
    42→function exportMattersToCSV(matters: Matter[], contacts: any[], notePages: any[]) {
    43→    const headers = ['Reference', 'Title', 'Client', 'Type', 'Stage', 'Status', 'Court', 'Created', 'Latest Endorsements'];
    44→    const rows = matters.map(m => {
    45→        const client = contacts.find((c: any) => c.id === m.clientId);
    46→        
    47→        // Find endorsements for this matter
    48→        const matterNotes = notePages
    49→            .filter((n: any) => n.matterId === m.id && n.type === 'endorsement')
    50→            .sort((a: any, b: any) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
    51→            
    52→        // Combine up to 3 latest endorsements into a single string
    53→        const endorsementsText = matterNotes.slice(0, 3)
    54→            .map((n: any) => `[${new Date(n.createdAt).toLocaleDateString('en-GB')}] ${n.title}: ${n.content.replace(/<[^>]*>?/gm, '').substring(0, 100)}...`)
    55→            .join(' | ');
    56→
    57→        return [
    58→            m.referenceNumber || '',
    59→            `"${m.title.replace(/"/g, '""')}"`,
    60→            `"${(client?.name || 'Unknown').replace(/"/g, '""')}"`,
    61→            m.type || '',
    62→            m.stage || '',
    63→            m.status || '',
    64→            `"${(m.court || '').replace(/"/g, '""')}"`,
    65→            m.createdAt ? new Date(m.createdAt).toLocaleDateString('en-GB') : '',
    66→            `"${endorsementsText.replace(/"/g, '""')}"`
    67→        ].join(',');
    68→    });
    69→    const csv = [headers.join(','), ...rows].join('\n');
    70→    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
    71→    const url = URL.createObjectURL(blob);
    72→    const a = document.createElement('a');
    73→    a.href = url;
    74→    a.download = `matters_${new Date().toISOString().slice(0, 10)}.csv`;
    75→    a.click();
    76→    URL.revokeObjectURL(url);
    77→}

   220→    const mattersContainerRef = React.useRef<HTMLDivElement>(null);
   221→    useHighlight(mattersContainerRef, 'matters', 'ring');
   222→
   223→    const [searchTerm, setSearchTerm] = useState('');
   224→    const [preset, setPreset] = useState<Preset>('all');
   225→    const [typeFilter, setTypeFilter] = useState<string>('all');
   226→    const [localViewMode, setLocalViewMode] = useState<'list' | 'board'>(propViewMode || 'list');
   227→    const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   228→
   229→    const viewMode = propViewMode ?? localViewMode;
   230→
   231→    const handleViewModeChange = (mode: 'list' | 'board') => {
   232→        setLocalViewMode(mode);
   233→        onViewModeChange?.(mode);
   234→        setSelectedIds(new Set());

   236→
   237→    const validMatters = useMemo(() => matterState.matters.filter((m: any) => m.title && m.id), [matterState.matters]);
   238→

   240→
   241→    // Filter by preset
   242→    const presetFiltered = useMemo(() => {
   243→        const now = new Date();
   244→        const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1);
   245→        return validMatters.filter((m: any) => {
   246→            if (preset === 'my-active') return m.status === MatterStatus.Active && m.assignedUsers?.includes(currentUser?.id);
   247→            if (preset === 'overdue') {
   248→                return executionState.tasks.some(t => t.matterId === m.id && t.status !== TaskStatus.Done && t.dueDate && new Date(t.dueDate) < now);
   249→            }
   250→            if (preset === 'this-month') return new Date(m.createdAt) >= startOfMonth;
   251→            return true;
   252→        });
