   793→   793→                        if (key === 'firmDetails' && backendValue && typeof backendValue === 'object') {
   794→   794→                            const backendFirm = backendValue as any;
   795→   795→                            const prevFirm = (prev.firmDetails as any) || {};
   796→   796→                            (newState as any)[key] = {
   797→   797→                                ...backendFirm,
   798→   798→                                id: backendFirm.id || backendFirm._id || prevFirm.id || prevFirm._id,
   799→   799→                                // Preserve locally-edited bank accounts if the backend
   800→   800→                                // record doesn't echo them back (e.g. immediately after
   801→   801→                                // createFirm, before the next firmData sync).
   802→   802→                                bankAccounts: Array.isArray(backendFirm.bankAccounts)
   803→   803→                                    ? backendFirm.bankAccounts
   804→   804→                                    : (prevFirm.bankAccounts || []),
   805→   805→                                integrations: backendFirm.integrations || prevFirm.integrations,
   806→   806→                                aiSettings: backendFirm.aiSettings || prevFirm.aiSettings,
   807→   807→                            };
   808→   808→                        } else {
   809→   809→                            (newState as any)[key] = backendValue;
   810→   810→                        }
   811→   811→                    }
   812→   812→                }
   813→   813→                return newState;
   814→   814→            });
   815→   815→            if (!isFullyLoaded) setIsFullyLoaded(true);
   816→   816→
   817→   817→            // ─── Cache app data for offline use ────────────────────────────
   818→   818→            // After successfully loading from the backend, cache a snapshot
   819→   819→            // in localStorage. When the app loads offline, we'll use this
   820→   820→            // to show the user's data in read-only mode.
   821→   821→            // We only cache essential arrays (not chatMessages which can be huge).
   822→   822→            try {
   823→   823→                const toCache = { ...appStateRef.current };
   824→   824→                // Strip large/unnecessary fields to keep cache size manageable
   825→   825→                delete (toCache as any).chatMessages;
   826→   826→                delete (toCache as any).firmActivity;
   827→   827→                localStorage.setItem('practicepro_cached_appstate', JSON.stringify({
   828→   828→                    firmId: currentUser?.firmId,
   829→   829→                    state: toCache,
   830→   830→                    cachedAt: Date.now(),
   831→   831→                }));
   832→   832→            } catch {} // localStorage might be full — non-critical
   833→   833→        }
   834→   834→    }, [firmMetadata, firmData, currentUser, isDemo, isDataLoaded, isFullyLoaded]);
   835→   835→
   836→   836→    const dataStateValue = React.useMemo(() => ({
   837→   837→        appState, setAppState, isDataLoaded, isSaving: false, isOutdated: false, availableBackups: [] as string[]
   838→   838→    }), [appState, isDataLoaded]);
   839→   839→
   840→   840→    return (
   841→   841→        <DataStateContext.Provider value={dataStateValue}>
   842→   842→            <DataActionsContext.Provider value={contextActions as unknown as ExtendedDataActions}>
   843→   843→                {children}
   844→   844→            </DataActionsContext.Provider>
   845→   845→        </DataStateContext.Provider>
   846→   846→    );
   847→   847→};
