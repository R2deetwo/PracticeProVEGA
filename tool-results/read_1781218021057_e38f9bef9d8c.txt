    55→export function buildMessage(
    56→  type: AutomationMessageType, 
    57→  unitLabel: string, 
    58→  tenantName?: string, 
    59→  amount?: number, 
    60→  customText?: string, 
    61→  customTemplates?: Record<string, string>,
    62→  extraData?: {
    63→    serviceCharge?: number;
    64→    legalFee?: number;
    65→    agencyFee?: number;
    66→    cautionDeposit?: number;
    67→    dueDate?: string;
    68→    firmName?: string;
    69→  }
    70→): string {
    71→  if (customText) return customText;
    72→
    73→  const name = tenantName || 'Tenant';
    74→  const addr = unitLabel;
    75→  const baseRent = amount || 0;
    76→  
    77→  const sc = extraData?.serviceCharge || 0;
    78→  const lf = extraData?.legalFee || 0;
    79→  const af = extraData?.agencyFee || 0;
    80→  const cd = extraData?.cautionDeposit || 0;
    81→  const totalPayable = baseRent + sc + lf + af + cd;
    82→  
    83→  const amtStr = `₦${baseRent.toLocaleString('en-NG')}`;
    84→  const totalStr = `₦${totalPayable.toLocaleString('en-NG')}`;
    85→
    86→  let message = customTemplates?.[type];
    87→  
    88→  if (!message) {
    89→    switch (type) {
    90→      case 'rent_reminder': 
    91→        message = `OFFICIAL DEMAND NOTICE\n\nDear {{TENANT_NAME}},\n\nThis is a formal reminder that your payment for {{PROPERTY_ADDRESS}} is due on or before {{DUE_DATE}}.\n\nFinancial Breakdown:\n- Rent: {{AMOUNT}}\n{{SERVICE_CHARGE_LINE}}{{CAUTION_DEPOSIT_LINE}}{{FEES_LINE}}\nTotal Payable: {{TOTAL_PAYABLE}}\n\nPlease ensure payment is made promptly. Thank you.\n\n— {{FIRM_NAME}}`; 
    92→        break;
    93→      case 'late_notice': 
    94→        message = `URGENT: OVERDUE PAYMENT\n\nDear {{TENANT_NAME}},\n\nYour payment of {{TOTAL_PAYABLE}} for {{PROPERTY_ADDRESS}} is now OVERDUE.\n\nPlease make payment immediately to avoid late penalties or access restrictions. Contact management at {{FIRM_NAME}} to confirm your payment.\n\n— {{FIRM_NAME}}`; 
    95→        break;
    96→      case 'payment_receipt': 
    97→        message = `PAYMENT RECEIPT\n\nDear {{TENANT_NAME}},\n\nWe confirm receipt of {{TOTAL_PAYABLE}} for your unit at {{PROPERTY_ADDRESS}}.\n\nThank you for your prompt payment.\n\n— {{FIRM_NAME}}`; 
    98→        break;
    99→      case 'service_charge_alert': 
   100→        message = `SERVICE CHARGE ALERT\n\nDear {{TENANT_NAME}},\n\nYour service charge of ₦${sc.toLocaleString('en-NG')} for {{PROPERTY_ADDRESS}} is outstanding. Please settle this at your earliest convenience to avoid access restrictions.\n\nTotal Payable: {{TOTAL_PAYABLE}}`; 
   101→        break;
   102→      case 'access_restriction': 
   103→        message = `NOTICE OF ACCESS RESTRICTION\n\nAccess to {{PROPERTY_ADDRESS}} has been restricted due to non-payment of outstanding charges totaling {{TOTAL_PAYABLE}}. Please contact {{FIRM_NAME}} immediately to resolve.`; 
   104→        break;
   105→      case 'penalty_notice': 
   106→        message = `PENALTY NOTICE\n\nDear {{TENANT_NAME}},\n\nA late payment penalty has been applied to your account for {{PROPERTY_ADDRESS}}.\n\nRevised Total Payable: {{TOTAL_PAYABLE}}\n\nPlease settle the outstanding balance urgently. — {{FIRM_NAME}}`; 
   107→        break;
   108→      case 'lease_renewal': 
   109→        message = `LEASE RENEWAL\n\nDear {{TENANT_NAME}},\n\nYour lease for {{PROPERTY_ADDRESS}} is expiring soon. We invite you to renew your tenancy agreement. Please contact {{FIRM_NAME}} to discuss renewal terms.`; 
   110→        break;
   111→      case 'welcome_note': 
   112→        message = `Welcome to your new home at {{PROPERTY_ADDRESS}}, {{TENANT_NAME}}! We are excited to have you. Please find the tenant handbook in your portal. — {{FIRM_NAME}}`; 
   113→        break;
   114→      case 'promotion': 
   115→        message = `Hello {{TENANT_NAME}}, we have a special offer for our residents! Get 10% off professional cleaning services this month. Use code: CLEAN10. — {{FIRM_NAME}}`; 
   116→        break;
   117→      case 'vendor_update': 
   118→        message = `Dear {{TENANT_NAME}}, we have partnered with new verified maintenance vendors to serve you better. You can now request plumbing and electrical repairs directly from the app. — {{FIRM_NAME}}`; 
   119→        break;
   120→      case 'general_announcement': 
   121→        message = `Attention residents of {{PROPERTY_ADDRESS}}: Routine maintenance will be carried out on the central generators this Saturday. Expect intermittent power supply between 10am and 2pm. — {{FIRM_NAME}}`; 
   122→        break;
   123→      case 'maintenance_update': 
   124→        message = `Update on your maintenance request for {{PROPERTY_ADDRESS}}: The vendor has confirmed your appointment for tomorrow. Please ensure someone is available to grant access. — {{FIRM_NAME}}`; 
   125→        break;
   126→      default: 
   127→        message = ''; 
   128→        break;
   129→    }
   130→  }
   131→
   132→  if (!message) return '';
   133→
   134→  const scLine = sc > 0 ? `- Service Charge: ₦${sc.toLocaleString('en-NG')}\n` : '';
   135→  const cdLine = cd > 0 ? `- Caution Deposit: ₦${cd.toLocaleString('en-NG')}\n` : '';
   136→  const feesLine = (lf + af) > 0 ? `- Legal/Agency Fees: ₦${(lf + af).toLocaleString('en-NG')}\n` : '';
   137→
   138→  let result = message
   139→    .replace(/\{\{TENANT_NAME\}\}/g, name)
   140→    .replace(/\{\{AMOUNT\}\}/g, amtStr)
   141→    .replace(/\{\{TOTAL_PAYABLE\}\}/g, totalStr)
   142→    .replace(/\{\{SERVICE_CHARGE_LINE\}\}/g, scLine)
   143→    .replace(/\{\{CAUTION_DEPOSIT_LINE\}\}/g, cdLine)
   144→    .replace(/\{\{FEES_LINE\}\}/g, feesLine)
   145→    .replace(/\{\{FIRM_NAME\}\}/g, extraData?.firmName || 'Management');
   146→
   147→  // Handle Unit Context smartly
   148→  if (addr === 'General' || addr === 'All Residents') {
   149→      result = result.replace(/ for \{\{PROPERTY_ADDRESS\}\}/g, '');
   150→      result = result.replace(/ at \{\{PROPERTY_ADDRESS\}\}/g, '');
   151→      result = result.replace(/\{\{PROPERTY_ADDRESS\}\}/g, 'your unit');
   152→  } else {
   153→      result = result.replace(/\{\{PROPERTY_ADDRESS\}\}/g, addr);
   154→  }
   155→  
   156→  if (extraData) {
   157→    result = result.replace(/\{\{SERVICE_CHARGE\}\}/g, `₦${sc.toLocaleString('en-NG')}`);
   158→    result = result.replace(/\{\{LEGAL_FEE\}\}/g, `₦${lf.toLocaleString('en-NG')}`);
   159→    result = result.replace(/\{\{AGENCY_FEE\}\}/g, `₦${af.toLocaleString('en-NG')}`);
   160→    result = result.replace(/\{\{CAUTION_DEPOSIT\}\}/g, `₦${cd.toLocaleString('en-NG')}`);
   161→    result = result.replace(/\{\{DUE_DATE\}\}/g, extraData.dueDate || 'the due date');
   162→  }
   163→  
   164→  return result;
   165→}
