     1→/**
     2→ * SetupPassword — Dedicated page for portal invitees to create their password
     3→ *
     4→ * Flow:
     5→ * 1. Extract ?token= from URL
     6→ * 2. Call verifyInviteToken to validate the token is real, unexpired, and pending
     7→ * 3. Show a polished "Create Your Password" form with pre-filled, disabled email
     8→ * 4. On submit, call setupPortalPassword which:
     9→ *    - Creates or updates the user account with the hashed password
    10→ *    - Marks the invite as accepted
    11→ *    - Clears the token to prevent replay attacks
    12→ * 5. Redirect to the appropriate portal login page
    13→ */
    14→import React, { useState, useEffect, useCallback } from 'react';
    15→import { useQuery, useAction } from 'convex/react';
    16→import { useNavigate } from 'react-router-dom';
    17→import { api } from '../../../convex/_generated/api';
    18→import { useAuth } from '../../contexts/AuthContext';
    19→import { useUI } from '../../contexts/UIContext';
    20→import { Logo, LockClosedIcon, ShieldCheckIcon, EyeIcon, EyeOffIcon, CheckCircleIcon } from '../../constants';
    21→
    22→// Inline icons that aren't in constants
    23→const AlertCircleIcon: React.FC<{ className?: string }> = ({ className }) => (
    24→    <svg xmlns="http://www.w3.org/2000/svg" className={className} fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={1.5}>
    25→        <path strokeLinecap="round" strokeLinejoin="round" d="M12 9v3.75m9-.75a9 9 0 11-18 0 9 9 0 0118 0zm-9 3.75h.008v.008H12v-.008z" />
    26→    </svg>
    27→);
    28→
    29→type Step = 'loading' | 'invalid' | 'form' | 'submitting' | 'success';
    30→
    31→const SetupPassword: React.FC = () => {
    32→    const { login } = useAuth();
    33→    const { addToast } = useUI();
    34→    const navigate = useNavigate();
    35→
    36→    // URL token
    37→    const [token, setToken] = useState<string | null>(null);
    38→    const [step, setStep] = useState<Step>('loading');
    39→    const [invalidReason, setInvalidReason] = useState<string>('');
    40→    const [inviteData, setInviteData] = useState<any>(null);
    41→
    42→    // Form state
    43→    const [password, setPassword] = useState('');
    44→    const [confirmPassword, setConfirmPassword] = useState('');
    45→    const [name, setName] = useState('');
    46→    const [showPassword, setShowPassword] = useState(false);
    47→    const [showConfirm, setShowConfirm] = useState(false);
    48→    const [error, setError] = useState('');
    49→    const [loginError, setLoginError] = useState('');
    50→
    51→    // Extract token from URL on mount
    52→    useEffect(() => {
    53→        const params = new URLSearchParams(window.location.search);
    54→        const t = params.get('token');
    55→        if (t) {
    56→            setToken(t);
    57→        } else {
    58→            setStep('invalid');
    59→            setInvalidReason('no_token');
    60→        }
    61→    }, []);
    62→
    63→    // Verify the token
    64→    const verification = useQuery(
    65→        api.portals.verifyInviteToken,
    66→        token ? { token } : 'skip'
    67→    );
    68→
    69→    // Setup password action
    70→    const setupPassword = useAction(api.portals.setupPortalPassword);
    71→
    72→    // Handle verification result
    73→    useEffect(() => {
    74→        if (verification === undefined) return; // still loading
    75→
    76→        if (verification && 'valid' in verification) {
    77→            if (verification.valid) {
    78→                setInviteData(verification);
    79→                setName(verification.invite?.inviteeName || '');
    80→                setStep('form');
    81→            } else {
    82→                setStep('invalid');
    83→                setInvalidReason(verification.reason || 'unknown');
    84→            }
    85→        }
    86→    }, [verification]);
    87→
    88→    // Password strength indicator
    89→    const getPasswordStrength = useCallback((pw: string) => {
    90→        if (!pw) return { label: '', score: 0, color: '' };
    91→        let score = 0;
    92→        if (pw.length >= 8) score++;
    93→        if (pw.length >= 12) score++;
    94→        if (/[A-Z]/.test(pw)) score++;
    95→        if (/[0-9]/.test(pw)) score++;
    96→        if (/[^A-Za-z0-9]/.test(pw)) score++;
    97→
    98→        if (score <= 1) return { label: 'Weak', score: 1, color: 'bg-red-500' };
    99→        if (score <= 2) return { label: 'Fair', score: 2, color: 'bg-orange-500' };
   100→        if (score <= 3) return { label: 'Good', score: 3, color: 'bg-yellow-500' };
   101→        if (score <= 4) return { label: 'Strong', score: 4, color: 'bg-emerald-400' };
   102→        return { label: 'Excellent', score: 5, color: 'bg-emerald-500' };
   103→    }, []);
   104→
   105→    const strength = getPasswordStrength(password);
   106→
   107→    const handleSubmit = async () => {
   108→        setError('');
   109→
   110→        // Validation
   111→        if (!password) {
   112→            setError('Please enter a password.');
   113→            return;
   114→        }
   115→        if (password.length < 8) {
   116→            setError('Password must be at least 8 characters.');
   117→            return;
   118→        }
   119→        if (password !== confirmPassword) {
   120→            setError('Passwords do not match.');
   121→            return;
   122→        }
   123→        if (!token) {
   124→            setError('Invalid invitation link.');
   125→            return;
   126→        }
   127→
   128→        setStep('submitting');
   129→
   130→        try {
   131→            const result = await setupPassword({
   132→                token,
   133→                password,
   134→                name: name || undefined,
   135→            });
   136→
   137→            if (result.success) {
   138→                setStep('success');
   139→
   140→                // Store portal type immediately so App.tsx knows this is a portal user
   141→                // during the auth loading phase (prevents flash of LandingPage)
   142→                const portalType = inviteData?.invite?.portalType;
   143→                if (portalType === 'client') {
   144→                    sessionStorage.setItem('practicepro_portal_type', 'client');
   145→                } else if (portalType === 'resident') {
   146→                    sessionStorage.setItem('practicepro_portal_type', 'tenant');
   147→                }
   148→
   149→                // Auto-login after a short delay
   150→                setTimeout(async () => {
   151→                    try {
   152→                        const loginResult = await login(result.email!, password);
   153→                        if (loginResult.success) {
   154→                            // Redirect to the main app — the router will detect the user's
   155→                            // Client/Tenant role and show the correct portal view automatically.
   156→                            navigate('/', { replace: true });
   157→                        } else {
   158→                            setLoginError('Your password has been set! Please sign in using the button below.');
   159→                            setStep('success');
   160→                        }
   161→                    } catch {
   162→                        setLoginError('Your password has been set! Please sign in manually.');
   163→                        setStep('success');
   164→                    }
   165→                }, 1500);
   166→            } else {
   167→                setError(result.message || 'Something went wrong. Please try again.');
   168→                setStep('form');
   169→            }
   170→        } catch (e: any) {
   171→            setError(e.message || 'An error occurred. Please try again.');
   172→            setStep('form');
   173→        }
   174→    };
   175→
   176→    const handleKeyDown = (e: React.KeyboardEvent) => {
   177→        if (e.key === 'Enter' && step === 'form') handleSubmit();
   178→    };
   179→
   180→    // Determine product branding from invite data
   181→    const isClientPortal = inviteData?.invite?.portalType === 'client';
   182→    const productName = isClientPortal ? 'VEGA' : 'ATRIUM';
   183→    const portalLabel = isClientPortal ? 'Client Portal' : "Residents' Portal";
   184→    // Use concrete classes instead of dynamic template strings — Tailwind JIT can't compile bg-${color}-500
   185→    const glowClass = isClientPortal ? 'bg-violet-500/10' : 'bg-amber-500/10';
   186→    const brandTextClass = isClientPortal ? 'text-violet-400' : 'text-amber-400';
   187→
   188→    // ─── Invalid token screen ─────────────────────────────────────────────
   189→    if (step === 'invalid') {
   190→        const messages: Record<string, { title: string; body: string }> = {
   191→            no_token: {
   192→                title: 'Missing Invitation',
   193→                body: 'No invitation token was found. Please use the link from your invitation email.',
   194→            },
   195→            not_found: {
   196→                title: 'Invalid Invitation Link',
   197→                body: 'This invitation link does not exist or has already been used. Please request a new invitation.',
   198→            },
   199→            expired: {
   200→                title: 'Invitation Expired',
   201→                body: 'This invitation has expired. Please contact your manager to request a new one.',
   202→            },
   203→            revoked: {
   204→                title: 'Invitation Revoked',
   205→                body: 'This invitation has been revoked by the sender. Please contact them for more information.',
   206→            },
   207→            already_accepted: {
   208→                title: 'Already Accepted',
   209→                body: 'Your invitation has already been used and your account is set up. You can sign in directly using your email and password.',
   210→            },
   211→        };
   212→        const msg = messages[invalidReason] || messages.not_found;
   213→
   214→        return (
   215→            <div className="min-h-screen bg-gradient-to-br from-slate-900 via-slate-950 to-slate-900 flex flex-col overflow-x-hidden overflow-y-auto">
   216→                <div className="absolute top-0 left-1/2 -translate-x-1/2 w-[600px] h-[400px] rounded-full blur-[120px] bg-rose-500/10 pointer-events-none" />
   217→                <div className="relative z-10 flex-shrink-0">
   218→                    <div className="container mx-auto px-4 sm:px-6 h-16 flex items-center">
   219→                        <div className="flex items-center gap-2.5">
   220→                            <Logo className="h-7 w-7 text-primary-500" />
   221→                            <span className="text-[19px] font-bold tracking-tight text-white">Practice<span className="text-primary-500">Pro</span></span>
   222→                        </div>
   223→                    </div>
   224→                </div>
   225→                <div className="flex-1 flex items-center justify-center p-4">
   226→                    <div className="relative z-10 w-full max-w-md text-center">
   227→                        <div className="w-16 h-16 rounded-full bg-rose-500/10 border border-rose-500/20 flex items-center justify-center mx-auto mb-6">
   228→                            <AlertCircleIcon className="w-8 h-8 text-rose-400" />
   229→                        </div>
   230→                        <h1 className="text-xl font-bold text-white mb-3">{msg.title}</h1>
   231→                        <p className="text-sm text-slate-400 leading-relaxed mb-6">{msg.body}</p>
   232→                        <button
   233→                            onClick={() => { navigate('/'); }}
   234→                            className="px-6 py-3 rounded-xl text-slate-300 text-sm font-medium hover:text-white hover:bg-white/[0.04] border border-white/[0.08] transition-all"
   235→                        >
   236→                            Back to PracticePro
   237→                        </button>
   238→                        {invalidReason === 'already_accepted' && (
   239→                            <button
   240→                                onClick={() => {
   241→                                    const isClient = inviteData?.invite?.portalType === 'client';
   242→                                    navigate(isClient ? '/portal/client/login' : '/portal/tenant/login');
   243→                                }}
   244→                                className="ml-3 px-6 py-3 rounded-xl bg-amber-500 text-white text-sm font-bold hover:bg-amber-400 transition-all"
   245→                            >
   246→                                Sign In
   247→                            </button>
   248→                        )}
   249→                    </div>
   250→                </div>
   251→            </div>
   252→        );
   253→    }
   254→
   255→    // ─── Loading screen ───────────────────────────────────────────────────
   256→    if (step === 'loading') {
   257→        return (
   258→            <div className="min-h-screen bg-gradient-to-br from-slate-900 via-slate-950 to-slate-900 flex items-center justify-center">
   259→                <div className="text-center">
   260→                    <div className="w-8 h-8 border-2 border-amber-500/30 border-t-amber-500 rounded-full animate-spin mx-auto mb-4" />
   261→                    <p className="text-sm text-slate-400">Verifying your invitation…</p>
   262→                </div>
   263→            </div>
   264→        );
   265→    }
   266→
   267→    // ─── Success screen ───────────────────────────────────────────────────
   268→    if (step === 'success') {
   269→        return (
   270→            <div className="min-h-screen bg-gradient-to-br from-slate-900 via-slate-950 to-slate-900 flex flex-col overflow-x-hidden overflow-y-auto">
   271→                <div className="absolute top-0 left-1/2 -translate-x-1/2 w-[600px] h-[400px] rounded-full blur-[120px] bg-emerald-500/10 pointer-events-none" />
   272→                <div className="relative z-10 flex-shrink-0">
   273→                    <div className="container mx-auto px-4 sm:px-6 h-16 flex items-center">
   274→                        <div className="flex items-center gap-2.5">
   275→                            <Logo className="h-7 w-7 text-primary-500" />
   276→                            <span className="text-[19px] font-bold tracking-tight text-white">Practice<span className="text-primary-500">Pro</span></span>
   277→                        </div>
   278→                    </div>
   279→                </div>
   280→                <div className="flex-1 flex items-center justify-center p-4">
   281→                    <div className="relative z-10 w-full max-w-md text-center">
   282→                        <div className="w-16 h-16 rounded-full bg-emerald-500/10 border border-emerald-500/20 flex items-center justify-center mx-auto mb-6">
   283→                            <CheckCircleIcon className="w-8 h-8 text-emerald-400" />
   284→                        </div>
   285→                        <h1 className="text-xl font-bold text-white mb-3">You're All Set!</h1>
   286→                        <p className="text-sm text-slate-400 leading-relaxed mb-2">
   287→                            Your password has been created and your account is now active.
   288→                        </p>
   289→                        <p className="text-sm text-slate-400 mb-6">
   290→                            Signing you in automatically…
   291→                        </p>
   292→                        <div className="w-6 h-6 border-2 border-emerald-500/30 border-t-emerald-500 rounded-full animate-spin mx-auto mb-6" />
   293→                        {loginError && (
   294→                            <div className="mt-4">
   295→                                <p className="text-sm text-amber-400 mb-4">{loginError}</p>
   296→                                <button
   297→                                    onClick={() => {
   298→                                        // Go to the portal login page to sign in manually
   299→                                        const isClient = inviteData?.invite?.portalType === 'client';
   300→                                        navigate(isClient ? '/portal/client/login' : '/portal/tenant/login');
   301→                                    }}
   302→                                    className="px-8 py-3 rounded-xl bg-gradient-to-r from-amber-500 to-amber-600 text-white font-bold text-sm hover:from-amber-400 hover:to-amber-500 shadow-lg shadow-amber-500/20 transition-all"
   303→                                >
   304→                                    Sign In to {portalLabel}
   305→                                </button>
   306→                            </div>
   307→                        )}
   308→                    </div>
   309→                </div>
   310→            </div>
   311→        );
   312→    }
   313→
   314→    // ─── Main form ────────────────────────────────────────────────────────
   315→    return (
   316→        <div className="min-h-screen bg-gradient-to-br from-slate-900 via-slate-950 to-slate-900 flex flex-col overflow-x-hidden overflow-y-auto">
   317→            {/* Ambient glow */}
   318→            <div className={`absolute top-0 left-1/2 -translate-x-1/2 w-[600px] sm:w-[800px] h-[400px] rounded-full blur-[120px] ${glowClass} pointer-events-none`} />
   319→
   320→            {/* Brand header */}
   321→            <div className="relative z-10 flex-shrink-0">
   322→                <div className="container mx-auto px-4 sm:px-6 h-16 flex items-center">
   323→                    <div className="flex items-center gap-2.5">
   324→                        <Logo className="h-7 w-7 text-primary-500" />
   325→                        <span className="text-[19px] font-bold tracking-tight text-white flex items-center">
   326→                            Practice<span className="text-primary-500">Pro</span>
   327→                            <span className={`ml-2 text-[15px] font-black uppercase tracking-tight ${brandTextClass}`}>
   328→                                {productName}
   329→                            </span>
   330→                        </span>
   331→                    </div>
   332→                </div>
   333→            </div>
   334→
   335→            {/* Form card */}
   336→            <div className="flex-1 flex items-center justify-center p-4 sm:p-6">
   337→                <div className="relative z-10 w-full max-w-md">
   338→                    {/* Header */}
   339→                    <div className="text-center mb-6 sm:mb-8">
   340→                        <div className="w-14 h-14 rounded-2xl bg-gradient-to-br from-amber-500/20 to-amber-600/10 border border-amber-500/20 flex items-center justify-center mx-auto mb-4">
   341→                            <LockClosedIcon className="w-6 h-6 text-amber-400" />
   342→                        </div>
   343→                        <h1 className="text-xl sm:text-2xl font-bold text-white mb-2">
   344→                            Create Your Password
   345→                        </h1>
   346→                        <p className="text-sm text-slate-400 leading-relaxed">
   347→                            Set up your secure password to access the {portalLabel}
   348→                        </p>
   349→                    </div>
   350→
   351→                    {/* Invite info */}
   352→                    {inviteData?.invite && (
   353→                        <div className="mb-4 px-4 py-3 rounded-xl bg-violet-500/10 border border-violet-500/20 text-violet-300 text-sm">
   354→                            <div className="flex items-start gap-2">
   355→                                <ShieldCheckIcon className="w-4 h-4 flex-shrink-0 mt-0.5" />
   356→                                <div>
   357→                                    <p className="font-semibold">Invitation Verified</p>
   358→                                    <p className="text-violet-400/80 text-xs mt-0.5">
   359→                                        {inviteData.invite.inviteeName
   360→                                            ? `Welcome, ${inviteData.invite.inviteeName}. `
   361→                                            : ''}
   362→                                        {inviteData.hasAccount
   363→                                            ? 'Your account has been located — setting your password will activate portal access.'
   364→                                            : 'Your account will be created automatically.'}
   365→                                    </p>
   366→                                </div>
   367→                            </div>
   368→                        </div>
   369→                    )}
   370→
   371→                    {/* Form */}
   372→                    <div className="bg-white/[0.04] border border-white/[0.08] backdrop-blur-xl rounded-2xl p-5 sm:p-8 shadow-2xl">
   373→                        <div className="space-y-4">
   374→                            {error && (
   375→                                <div className="px-4 py-3 rounded-xl bg-red-500/10 border border-red-500/20 text-red-400 text-sm font-medium">
   376→                                    {error}
   377→                                </div>
   378→                            )}
   379→
   380→                            {/* Name (optional, for new accounts) */}
   381→                            {inviteData && !inviteData.hasAccount && !inviteData.invite?.inviteeName && (
   382→                                <div>
   383→                                    <label className="block text-xs font-bold text-slate-400 uppercase tracking-wider mb-1.5">Your Name</label>
   384→                                    <input
   385→                                        type="text"
   386→                                        value={name}
   387→                                        onChange={e => setName(e.target.value)}
   388→                                        onKeyDown={handleKeyDown}
   389→                                        placeholder="Full name"
   390→                                        className="w-full px-4 py-3 rounded-xl bg-white/[0.06] border border-white/[0.08] text-white placeholder-slate-500 text-sm focus:ring-2 focus:ring-amber-500/50 focus:border-amber-500/30 transition-all"
   391→                                    />
   392→                                </div>
   393→                            )}
   394→
   395→                            {/* Email (disabled, pre-filled) */}
   396→                            <div>
   397→                                <label className="block text-xs font-bold text-slate-400 uppercase tracking-wider mb-1.5">Email Address</label>
   398→                                <input
   399→                                    type="email"
   400→                                    value={inviteData?.invite?.inviteeEmail || ''}
   401→                                    disabled
   402→                                    className="w-full px-4 py-3 rounded-xl bg-white/[0.03] border border-white/[0.06] text-slate-400 text-sm cursor-not-allowed"
   403→                                />
   404→                                <p className="text-[10px] text-slate-600 mt-1">This is locked to your invitation email</p>
   405→                            </div>
   406→
   407→                            {/* Password */}
   408→                            <div>
   409→                                <label className="block text-xs font-bold text-slate-400 uppercase tracking-wider mb-1.5">Create Password</label>
   410→                                <div className="relative">
   411→                                    <input
   412→                                        type={showPassword ? 'text' : 'password'}
   413→                                        value={password}
   414→                                        onChange={e => setPassword(e.target.value)}
   415→                                        onKeyDown={handleKeyDown}
   416→                                        placeholder="At least 8 characters"
   417→                                        className="w-full px-4 py-3 pr-10 rounded-xl bg-white/[0.06] border border-white/[0.08] text-white placeholder-slate-500 text-sm focus:ring-2 focus:ring-amber-500/50 focus:border-amber-500/30 transition-all"
   418→                                    />
   419→                                    <button
   420→                                        type="button"
   421→                                        onClick={() => setShowPassword(!showPassword)}
   422→                                        className="absolute right-3 top-1/2 -translate-y-1/2 text-slate-500 hover:text-slate-300 transition-colors"
   423→                                    >
   424→                                        {showPassword ? <EyeOffIcon className="w-4 h-4" /> : <EyeIcon className="w-4 h-4" />}
   425→                                    </button>
   426→                                </div>
   427→                                {/* Password strength bar */}
   428→                                {password && (
   429→                                    <div className="mt-2">
   430→                                        <div className="flex gap-1 mb-1">
   431→                                            {[1, 2, 3, 4, 5].map(i => (
   432→                                                <div key={i} className={`h-1 flex-1 rounded-full ${i <= strength.score ? strength.color : 'bg-slate-700'} transition-all`} />
   433→                                            ))}
   434→                                        </div>
   435→                                        <p className="text-[10px] text-slate-500">{strength.label}</p>
   436→                                    </div>
   437→                                )}
   438→                            </div>
   439→
   440→                            {/* Confirm Password */}
   441→                            <div>
   442→                                <label className="block text-xs font-bold text-slate-400 uppercase tracking-wider mb-1.5">Confirm Password</label>
   443→                                <div className="relative">
   444→                                    <input
   445→                                        type={showConfirm ? 'text' : 'password'}
   446→                                        value={confirmPassword}
   447→                                        onChange={e => setConfirmPassword(e.target.value)}
   448→                                        onKeyDown={handleKeyDown}
   449→                                        placeholder="Re-enter your password"
   450→                                        className={`w-full px-4 py-3 pr-10 rounded-xl bg-white/[0.06] border text-white placeholder-slate-500 text-sm focus:ring-2 focus:ring-amber-500/50 transition-all ${
   451→                                            confirmPassword && confirmPassword !== password
   452→                                                ? 'border-rose-500/50 focus:border-rose-500/50'
   453→                                                : confirmPassword && confirmPassword === password
   454→                                                ? 'border-emerald-500/50 focus:border-emerald-500/50'
   455→                                                : 'border-white/[0.08] focus:border-amber-500/30'
   456→                                        }`}
   457→                                    />
   458→                                    <button
   459→                                        type="button"
   460→                                        onClick={() => setShowConfirm(!showConfirm)}
   461→                                        className="absolute right-3 top-1/2 -translate-y-1/2 text-slate-500 hover:text-slate-300 transition-colors"
   462→                                    >
   463→                                        {showConfirm ? <EyeOffIcon className="w-4 h-4" /> : <EyeIcon className="w-4 h-4" />}
   464→                                    </button>
   465→                                </div>
   466→                                {confirmPassword && confirmPassword === password && (
   467→                                    <p className="text-[10px] text-emerald-400 mt-1 flex items-center gap-1">
   468→                                        <CheckCircleIcon className="w-3 h-3" /> Passwords match
   469→                                    </p>
   470→                                )}
   471→                            </div>
