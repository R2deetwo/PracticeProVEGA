    65→const formatSettingsPrice = (
    66→    tier: TierDef,

    70→    if (tier.monthlyPrice === 0 && (tier.annualPrice ?? 0) === 0) return 'Free';

    72→    if (tier.monthlyPrice === null && tier.annualPrice === null) return 'Custom';
    73→    // Atrium tiers have monthlyPrice=null but valid annualPrice — show annual price
    74→    if (tier.monthlyPrice === null && tier.annualPrice !== null) {
    75→        if (viewAsMonthlyCost) {
    76→            const monthlyFromAnnual = Math.round(tier.annualPrice / 12);
    77→            return `₦${monthlyFromAnnual.toLocaleString('en-NG')}`;
    78→        }
    79→        return tier.annualPriceDisplay;
    80→    }
    81→    if (!isAnnual) return tier.monthlyPriceDisplay;
    82→    if (viewAsMonthlyCost) {
    83→        const monthlyFromAnnual = Math.round((tier.annualPrice ?? 0) / 12);
    84→        return `₦${monthlyFromAnnual.toLocaleString('en-NG')}`;
