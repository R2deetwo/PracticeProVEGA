   102→   102→// ─── File Upload Helper ──────────────────────────────────────────────────────
   103→   103→async function uploadFilesToConvex(files: File[]): Promise<string[]> {
   104→   104→  const storageIds: string[] = [];
   105→   105→  for (const file of files) {
   106→   106→    const postUrl = await fetch('/api/storage/upload-url');
   107→   107→    // Fallback: use the Convex action for upload URL
   108→   108→    // We'll use the Convex generateUploadUrl mutation via the hook instead
   109→   109→    // For now, we'll use the standard Convex file upload flow
   110→   110→    const { url, storageId } = await (async () => {
   111→   111→      // Use Convex's built-in file upload
   112→   112→      const uploadUrlRes = await fetch(`${window.location.origin}/api/storage/upload-url`);
   113→   113→      if (uploadUrlRes.ok) {
   114→   114→        const { url: u, storageId: s } = await uploadUrlRes.json();
   115→   115→        return { url: u, storageId: s };
   116→   116→      }
   117→   117→      throw new Error('Failed to get upload URL');
   118→   118→    })();
   119→   119→
   120→   120→    const uploadRes = await fetch(url, {
   121→   121→      method: 'POST',
   122→   122→      body: file,
   123→   123→    });
   124→   124→    if (uploadRes.ok) {
   125→   125→      const { storageId: sid } = await uploadRes.json();
   126→   126→      storageIds.push(sid);
   127→   127→    }
   128→   128→  }
   129→   129→  return storageIds;
   130→   130→}

   340→   340→
   341→   341→  // Compute tenant-specific service charges
   342→   342→  const tenantServiceCharges = useMemo(() => {
   343→   343→    if (!serviceCharges || !resolvedTenantId) return [];
   344→   344→    return serviceCharges.filter((sc: any) => sc.tenantId === resolvedTenantId);
   345→   345→  }, [serviceCharges, resolvedTenantId]);
   346→   346→
   347→   347→  // Summary calculations
   348→   348→  const currentMonthSC = useMemo(() => {
   349→   349→    return tenantServiceCharges
   350→   350→      .filter((sc: any) => !sc.isMinimumVend)
   351→   351→      .reduce((sum: number, sc: any) => sum + (sc.outstandingBalance ?? sc.amount), 0);
   352→   352→  }, [tenantServiceCharges]);
   353→   353→
   354→   354→  const currentMonthMV = useMemo(() => {
   355→   355→    return tenantServiceCharges
   356→   356→      .filter((sc: any) => sc.isMinimumVend)
   357→   357→      .reduce((sum: number, sc: any) => sum + (sc.outstandingBalance ?? sc.amount), 0);
   358→   358→  }, [tenantServiceCharges]);
   359→   359→
   360→   360→  const outstandingBalance = useMemo(() => {
   361→   361→    if (!ledgerEntries) return 0;
