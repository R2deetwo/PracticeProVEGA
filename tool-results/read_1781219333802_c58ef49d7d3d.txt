    26→const ActionCard: React.FC<{
    27→    actionName: string;
    28→    args: any;
    29→    onExecute: () => void;
    30→    executed?: boolean
    31→}> = ({ actionName, args, onExecute }) => {
    32→    let label = "Open Item";
    33→    const normalized = actionName.toLowerCase();
    34→    if (normalized.includes('matter')) label = "Open Matter Form";
    35→    else if (normalized.includes('contact')) label = "Open Contact Form";
    36→    else if (normalized.includes('task')) label = "Open Task Form";
    37→    else if (normalized.includes('drafting')) label = "Open Editor";
    38→    else if (normalized.includes('workflow')) label = "Review Workflow";
    39→    else if (normalized === 'note') label = "Open Note";
    40→
    41→    const isCompleted = args?.context?.isCompleted;
    42→
    43→    return (
    44→        <div className={`mt-2 p-2 bg-slate-50 dark:bg-zinc-800 rounded-xl border border-slate-100 dark:border-zinc-700 ${isCompleted ? 'opacity-50' : ''}`}>
    45→            <div className="text-[9px] text-primary-600 dark:text-primary-400 mb-1.5 font-bold uppercase tracking-widest flex justify-between items-center">
    46→                <span>Action Available</span>
    47→                {isCompleted && <span className="text-emerald-500 font-black">DONE</span>}
    48→            </div>
    49→            <button
    50→                onClick={onExecute}
    51→                disabled={isCompleted}
    52→                className={`w-full py-1.5 rounded-lg text-[10px] font-bold transition-all ${isCompleted ? 'bg-slate-100 dark:bg-zinc-800 text-slate-400' : 'bg-primary-600 hover:bg-primary-700 text-white shadow-sm'}`}
    53→            >
    54→                {isCompleted ? 'Completed' : label}
    55→            </button>
    56→        </div>
    57→    );

   104→    const mapToolToModal = (toolName: string, args: any): { modalType: ModalType, mappedArgs: any } | null => {
   105→        const normalizedName = toolName.toLowerCase();
   106→        const rawModalType = args.modalType || '';
   107→        if (normalizedName === 'navigate_to' || normalizedName === 'start_drafting' || normalizedName === 'draft_workflow') return null;
   108→        let targetModal: ModalType | null = null;
   109→        if (normalizedName === 'open_modal') {
   110→            if (rawModalType === 'create_matter' || rawModalType === 'newMatter') targetModal = 'newMatter';
   111→            else if (rawModalType === 'create_contact' || rawModalType === 'newContact') targetModal = 'newContact';
   112→            else if (rawModalType === 'create_task' || rawModalType === 'newTask') targetModal = 'newTask';
   113→            else targetModal = rawModalType as ModalType;
   114→        } else if (normalizedName === 'create_matter') targetModal = 'newMatter';
   115→        else if (normalizedName === 'create_contact') targetModal = 'newContact';
   116→        else if (normalizedName === 'create_task') targetModal = 'newTask';
   117→        if (targetModal) return { modalType: targetModal, mappedArgs: args };
   118→        return null;
   119→    };

   121→    const handleToolExecution = async (toolCalls: any[]): Promise<any[]> => {
   122→        if (!toolCalls || toolCalls.length === 0) return [];
   123→        const outputs: any[] = [];
   124→        const isAdmin = currentUser?.role === 'Admin';
   125→        for (const tool of toolCalls) {
   126→            const { name, args } = tool;
   127→            let feedback = "";
   128→            let actionData: any = null;
   129→            let toolOutput: any = { success: true };
   130→            try {
   131→                if (name === 'query_firm_data') {
   132→                    const query = (args.query || '').toLowerCase();
   133→                    const category = args.category || 'all';
   134→                    const results: any = { tasks: [], notes: [], matters: [], documents: [] };
   135→                    if (category === 'all' || category === 'tasks') results.tasks = (executionState.tasks || []).filter(t => t.title.toLowerCase().includes(query)).slice(0, 3);
   136→                    if (category === 'all' || category === 'notes') results.notes = (documentState.notePages || []).filter(n => n.title.toLowerCase().includes(query)).slice(0, 5);
   137→                    toolOutput = { results };
   138→                } else if (name === 'analyze_document') {
   139→                    const { sourceId, type } = args;
   140→                    feedback = `Analyzing ${type}...`;
   141→                    let analyzable: any = null;
   142→                    if (type === 'document') {
   143→                        const doc = documentState.documents?.find(d => d.id === sourceId);
   144→                        if (doc) analyzable = { title: doc.title, content: doc.content, file: doc.file };
   145→                    } else {
   146→                        const note = documentState.notePages?.find(p => p.id === sourceId);
   147→                        if (note) analyzable = { title: note.title, content: note.content };
   148→                    }
   149→                    if (analyzable) {
   150→                        try {
   151→                            const report = await analyzeDocument(analyzable);
   152→                            toolOutput = { report };
   153→                            feedback = `Analysis complete: ${report.summary}`;
   154→                            actionData = { type: 'analysis', report, label: 'View Analysis' };
   155→                        } catch (err: any) { toolOutput = { error: err.message }; feedback = `Analysis failed.`; }
   156→                    } else { toolOutput = { error: "Not found" }; }
   157→                } else {
   158→                    const mapped = mapToolToModal(name, args);
   159→                    if (mapped) {
   160→                        openModal(mapped.modalType, null, { ...mapped.mappedArgs.context, openedByAloa: true });
   161→                        feedback = `Opening form...`;
   162→                    } else if (name === 'navigate_to') {
   163→                        openModal(args.view, args.selectedId, args.context); // Simplified for mini
   164→                        feedback = `Navigating...`;
   165→                    }
   166→                }
   167→                if (feedback) {
   168→                    const modelMsg: AloaMessage = { id: uuidv4(), role: 'model' as const, content: feedback, toolAction: actionData };
   169→                    setMessages(prev => [...prev, modelMsg]);
   170→                }
   171→                outputs.push({ toolName: name, output: toolOutput });
   172→            } catch (e) { outputs.push({ toolName: name, output: { error: String(e) } }); }
   173→        }
   174→        return outputs;
   175→    };
   176→

   178→        if (!textInput.trim() || isLoading) return;
   179→

   181→        const newUserMsg: AloaMessage = { id: uuidv4(), role: 'user', content };
   182→        const streamId = uuidv4();
   183→
   184→        setMessages(prev => [...prev, newUserMsg, { id: streamId, role: 'model', content: '' }]);
   185→        setTextInput('');
   186→        setIsLoading(true);
   187→        setMiniStatus(isFirmSearchEnabled ? 'Searching…' : 'Thinking…');
   188→
   189→        const effectiveModel = preferredModel === 'auto' ? 'flash' : preferredModel;

   191→            appState: { ...coreState, ...matterState, ...executionState, ...financeState, ...documentState } as any,
   192→            currentUser: currentUser!,
   193→            currentHistoryEntry,
   194→            localFiles,
   195→            aloaXLibrary: loadAloaXLibrary(),
   196→            isFirmSearchEnabled,
   197→            searchBrain: undefined as ((query: string) => Promise<string>) | undefined,
   198→        };
   199→
   200→        try {
   201→            const { brain } = await import('../../services/brainService');
   202→            aiContext.searchBrain = async (query: string) => {
   203→                if (!isFirmSearchEnabled) return '';
   204→                setMiniStatus('Searching…');
   205→                return await brain.search({
   206→                    query,
   207→                    firmId: coreState.firmDetails?.id || '',
   208→                    convexQuery: (name: any, args: any) => convex.query(name, args),
   209→                });
   210→            };
   211→
   212→            const wantsToolAction = /\b(create|open|add|new|draft|navigate|show me|find my|schedule|invoice|task|matter|contact)\b/i.test(content);

   214→            if (!wantsToolAction) {
   215→                setMiniStatus('Writing…');

   218→                        [...messages, newUserMsg],
   219→                        aiContext,
   220→                        (chunk) => {
   221→                            setMessages(prev =>
   222→                                prev.map(m =>
   223→                                    m.id === streamId
   224→                                        ? { ...m, content: `${typeof m.content === 'string' ? m.content : ''}${chunk}` }
   225→                                        : m
   226→                                )
   227→                            );
   228→                        },
   229→                        effectiveModel
   230→                    );
   231→                    if (streamed.text?.trim()) {
   232→                        setMessages(prev =>
   233→                            prev.map(m =>
   234→                                m.id === streamId
   235→                                    ? { id: streamId, role: 'model', content: streamed.text!, modelUsed: streamed.modelUsed }
   236→                                    : m
   237→                            )
   238→                        );
   239→                        return;
   240→                    }
   241→                } catch {
   242→                    setMessages(prev => prev.map(m => (m.id === streamId ? { ...m, content: '' } : m)));
   243→                }
   244→            }
   245→
   246→            setMiniStatus('Thinking…');
   247→            const response = await aiService.sendMessage([...messages, newUserMsg], aiContext, effectiveModel);
   248→
   249→            let currentResponse = response;
   250→            let iterations = 0;
   251→            while (currentResponse.toolCalls && currentResponse.toolCalls.length > 0 && iterations < 3) {
   252→                iterations++;
   253→                setMiniStatus('Using tools…');
   254→                const toolOutputs = await handleToolExecution(currentResponse.toolCalls);
   255→                const assistantToolCallMsg: AloaMessage = { id: uuidv4(), role: 'model', toolCalls: currentResponse.toolCalls };
   256→                const toolResultsMsgs: AloaMessage[] = toolOutputs.map(o => ({ id: uuidv4(), role: 'tool' as const, toolResult: o }));
   257→                const history = [...messages, newUserMsg, assistantToolCallMsg, ...toolResultsMsgs];
   258→                setMiniStatus('Writing…');
   259→                currentResponse = await aiService.sendMessage(history, aiContext, effectiveModel);
   260→            }
   261→
   262→            if (currentResponse.text) {
   263→                setMessages(prev =>
   264→                    prev.map(m =>
   265→                        m.id === streamId
   266→                            ? { id: streamId, role: 'model', content: currentResponse.text!, modelUsed: currentResponse.modelUsed }
   267→                            : m
   268→                    )
   269→                );
   270→            } else {
   271→                setMessages(prev => prev.filter(m => m.id !== streamId));
   272→            }
   273→        } catch (error: any) {
   274→            console.error('MiniAloa Error:', error);
   275→            setMessages(prev => [
   276→                ...prev.filter(m => m.id !== streamId),
   277→                {
   278→                    id: uuidv4(),
   279→                    role: 'model' as const,
   280→                    content: `Error: ${error.message || 'Connection failed'}`,
   281→                    isError: true,
   282→                },
   283→            ]);
   284→        } finally {
   285→            setIsLoading(false);
   286→            setMiniStatus('');
   287→        }
   288→    };
   289→
   290→    const handleExpand = () => {
   291→        setIsMinimized(false);
   292→        if (!isPanelOpen) togglePanel();
   293→    };
   294→
   295→    const [position, setPosition] = useState({ x: 0, y: 0 });
   296→    const isDragging = useRef(false);
   297→    const prevPosition = useRef({ x: 0, y: 0 });
   298→
   299→    const handleDragStart = (e: React.MouseEvent | React.TouchEvent) => {
   300→        isDragging.current = true;
   301→        const clientX = 'touches' in e ? e.touches[0].clientX : e.clientX;
   302→        const clientY = 'touches' in e ? e.touches[0].clientY : e.clientY;
   303→        prevPosition.current = { x: clientX, y: clientY };
   304→    };
   305→
   306→    const handleDragMove = (e: React.MouseEvent | React.TouchEvent) => {
   307→        if (!isDragging.current) return;
   308→        const clientX = 'touches' in e ? e.touches[0].clientX : e.clientX;
   309→        const clientY = 'touches' in e ? e.touches[0].clientY : e.clientY;
   310→
   311→        const deltaX = clientX - prevPosition.current.x;
   312→        const deltaY = clientY - prevPosition.current.y;
   313→
   314→        setPosition(prev => ({ x: prev.x + deltaX, y: prev.y + deltaY }));
   315→        prevPosition.current = { x: clientX, y: clientY };
   316→    };
   317→
   318→    const handleDragEnd = () => {
   319→        isDragging.current = false;
   320→    };
   321→
   322→    useEffect(() => {
   323→        if (isDragging.current) {
   324→            window.addEventListener('mousemove', handleDragMove as any);
   325→            window.addEventListener('mouseup', handleDragEnd);
   326→            window.addEventListener('touchmove', handleDragMove as any);
   327→            window.addEventListener('touchend', handleDragEnd);
   328→        }
   329→        return () => {
   330→            window.removeEventListener('mousemove', handleDragMove as any);
   331→            window.removeEventListener('mouseup', handleDragEnd);
   332→            window.removeEventListener('touchmove', handleDragMove as any);
   333→            window.removeEventListener('touchend', handleDragEnd);
   334→        };
   335→    }, [isDragging.current]);
   336→
   337→    const { dockedModalType, view } = useUI();
   338→    const isActive = aloaState !== 'idle' || isLoading;
   339→
   340→    // HIDE MINI ALOA IF DOCKED MODAL IS OPEN (Side Panel)
   341→    if (dockedModalType) return null;
   342→
   343→    // FADE OUT IN CERTAIN VIEWS
   344→    const isHidden = view === 'messaging' || view === 'research';
   345→    if (isHidden) return null;
   346→
   347→    return (
   348→        <div
   349→            className="
   350→                fixed bottom-24 right-4 z-[2000] 
   351→                w-[calc(100vw-32px)] md:w-80 h-[420px] max-h-[70dvh] rounded-2xl shadow-[0_20px_60px_-15px_rgba(0,0,0,0.5)]
   352→                bg-white/80 dark:bg-zinc-900/80 backdrop-blur-2xl 
   353→                border border-white/20 dark:border-zinc-700/50
   354→                flex flex-col overflow-hidden transition-all duration-300
   355→            "
   356→            style={{
   357→                transform: `translate(${position.x}px, ${position.y}px)`,
   358→                transition: isDragging.current ? 'none' : 'transform 0.3s cubic-bezier(0.34, 1.56, 0.64, 1)'
   359→            }}
   360→        >
   361→            {/* Header - Premium Dark, Draggable */}
   362→            <div
   363→                onMouseDown={handleDragStart}
   364→                onTouchStart={handleDragStart}
   365→                className="h-10 bg-slate-900/90 dark:bg-zinc-950/90 backdrop-blur-md flex items-center justify-between px-3 cursor-move active:cursor-grabbing select-none"
   366→            >
   367→                <div className="flex items-center gap-2">
   368→                    <div className={`
   369→                        w-6 h-6 rounded-lg flex items-center justify-center transition-colors duration-500
   370→                        ${isActive ? (isFirmSearchEnabled ? 'bg-blue-600 shadow-lg shadow-blue-500/30' : 'bg-green-600 shadow-lg shadow-green-600/30') : 'bg-primary-500'}
   371→                        ${isActive ? 'animate-pulse' : ''}
   372→                    `}>
   373→                        <AloaIcon className="w-3.5 h-3.5 text-white" />
   374→                    </div>
   375→                    <span className="text-[10px] font-black uppercase tracking-[0.2em] text-white/90">
   376→                        {isLoading ? 'Processing' : 'ALOA Mini'}
   377→                    </span>
   378→                </div>
   379→
   380→                <div className="flex items-center gap-0.5">
   381→                    <button
   382→                        className="p-1 hover:bg-white/10 rounded-md text-white/70 hover:text-white transition-all active:scale-90"
   383→                        onClick={handleExpand}
   384→                        title="Open Full ALOA Assistant"
   385→                    >
   386→                        <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" strokeWidth={2.5} stroke="currentColor" className="w-3.5 h-3.5">
   387→                            <path strokeLinecap="round" strokeLinejoin="round" d="M13.5 6H5.25A2.25 2.25 0 003 8.25v10.5A2.25 2.25 0 005.25 21h10.5A2.25 2.25 0 0018 18.75V10.5m-10.5 6L21 3m0 0h-5.25M21 3v5.25" />
   388→                        </svg>
   389→                    </button>
   390→                    <button
   391→                        className="p-1 hover:bg-white/10 rounded-md text-white/70 hover:text-white transition-all active:scale-90"
   392→                        onClick={() => closePanel()}
   393→                        title="Dismiss"
   394→                    >
   395→                        <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" strokeWidth={2.5} stroke="currentColor" className="w-3.5 h-3.5">
   396→                            <path strokeLinecap="round" strokeLinejoin="round" d="M6 18L18 6M6 6l12 12" />
   397→                        </svg>
   398→                    </button>
   399→                </div>
   400→            </div>
   401→
   402→            {/* Messages */}
   403→            <div className="flex-1 overflow-y-auto p-3 space-y-4 bg-transparent custom-scrollbar">
   404→                {messages.length === 0 && (
   405→                    <div className="flex flex-col items-center justify-center h-full text-center opacity-30 mt-4">
   406→                        <AloaIcon className="w-10 h-10 text-slate-400 dark:text-zinc-600 mb-3" />
   407→                        <p className="text-[10px] uppercase font-black tracking-widest text-slate-500">Awaiting Query</p>

   409→                )}
   410→                {messages.map((msg) => (
   411→                    <div key={msg.id} className={`flex ${msg.role === 'user' ? 'justify-end' : 'justify-start'} animate-in fade-in slide-in-from-bottom-1 group/msg`}>
   412→                        <div className={`max-w-[90%] p-3 rounded-xl text-[12px] leading-relaxed shadow-sm relative ${msg.role === 'user'
   413→                            ? 'bg-primary-600 text-white rounded-tr-none'
   414→                            : msg.isError
   415→                                ? 'bg-red-500/10 text-red-600 dark:text-red-400 rounded-tl-none border border-red-500/20'
   416→                                : 'bg-white/80 dark:bg-zinc-800/80 text-slate-800 dark:text-zinc-200 rounded-tl-none border border-slate-100 dark:border-zinc-700/50'
   417→                            }`}>
   418→                            <div
   419→                                className="prose prose-xs dark:prose-invert max-w-none break-words"
   420→                                dangerouslySetInnerHTML={{ __html: parseAloaMarkdown(msg.content || '') }}
   421→                            />
   422→
   423→                            {msg.toolAction && (
   424→                                <ActionCard
   425→                                    actionName={msg.toolAction.modalType || msg.toolAction.type || 'action'}
   426→                                    args={{ context: msg.toolAction.context }}
   427→                                    onExecute={() => {
   428→                                        if (msg.toolAction.type === 'modal' || msg.toolAction.modalType) {
   429→                                            openModal(msg.toolAction.modalType || (msg.toolAction.type as any), null, { ...msg.toolAction.context, openedByAloa: true });

   431→                                    }}
   432→                                />
   433→                            )}
   434→
   435→                            {msg.role === 'model' && !msg.isError && (
   436→                                <button
   437→                                    onClick={() => handleSaveMessage(msg.content || '')}
   438→                                    className="absolute -right-2 -bottom-2 p-1 min-h-[44px] min-w-[44px] flex items-center justify-center bg-white dark:bg-zinc-700 border border-slate-200 dark:border-zinc-600 rounded-md text-slate-400 hover:text-emerald-500 shadow-md opacity-100 md:opacity-0 md:group-hover/msg:opacity-100 transition-all active:scale-90 z-20"
   439→                                    title="Save to Notes"
   440→                                >

   442→                                        <path strokeLinecap="round" strokeLinejoin="round" d="M17.593 3.322c1.1.128 1.907 1.077 1.907 2.185V21L12 17.5 4.5 21V5.507c0-1.108.806-2.057 1.907-2.185a48.507 48.507 0 0111.186 0z" />
   443→                                    </svg>
   444→                                </button>

   446→                        </div>
   447→                    </div>
   448→                ))}
   449→                {isLoading && miniStatus && (
   450→                    <p className="text-[9px] font-medium text-primary-600 dark:text-primary-400 px-1 animate-pulse">{miniStatus}</p>
   451→                )}
   452→                <div ref={messagesEndRef} />
   453→            </div>
   454→
   455→            {/* Input Overlay with RAG Toggle */}
   456→            <div className="p-3 bg-white/50 dark:bg-zinc-950/50 border-t border-slate-200 dark:border-zinc-800">
   457→                <div className="flex flex-col gap-2">
