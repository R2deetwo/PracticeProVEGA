    56→export function buildMessage(
    57→  type: AutomationMessageType, 
    58→  unitLabel: string, 
    59→  tenantName?: string, 
    60→  amount?: number, 
    61→  customText?: string, 
    62→  customTemplates?: Record<string, string>,
    63→  extraData?: {
    64→    serviceCharge?: number;
    65→    legalFee?: number;
    66→    agencyFee?: number;
    67→    cautionDeposit?: number;
    68→    dueDate?: string;
    69→    firmName?: string;
    70→  }
    71→): string {
    72→  if (customText) return customText;
    73→
    74→  const name = tenantName || 'Tenant';
    75→  const addr = unitLabel;
    76→  const baseRent = amount || 0;
    77→  
    78→  const sc = extraData?.serviceCharge || 0;
    79→  const lf = extraData?.legalFee || 0;
    80→  const af = extraData?.agencyFee || 0;
    81→  const cd = extraData?.cautionDeposit || 0;
    82→  const totalPayable = baseRent + sc + lf + af + cd;
    83→  
    84→  const amtStr = `₦${baseRent.toLocaleString('en-NG')}`;
    85→  const totalStr = `₦${totalPayable.toLocaleString('en-NG')}`;
    86→
    87→  let message = customTemplates?.[type];
    88→  
    89→  if (!message) {
    90→    switch (type) {
    91→      case 'rent_reminder': 
    92→        message = `OFFICIAL DEMAND NOTICE\n\nDear {{TENANT_NAME}},\n\nThis is a formal reminder that your payment for {{PROPERTY_ADDRESS}} is due on or before {{DUE_DATE}}.\n\nFinancial Breakdown:\n- Rent: {{AMOUNT}}\n{{SERVICE_CHARGE_LINE}}{{CAUTION_DEPOSIT_LINE}}{{FEES_LINE}}\nTotal Payable: {{TOTAL_PAYABLE}}\n\nPlease ensure payment is made promptly. Thank you.\n\n— {{FIRM_NAME}}`; 
    93→        break;
    94→      case 'late_notice': 
    95→        message = `URGENT: OVERDUE PAYMENT\n\nDear {{TENANT_NAME}},\n\nYour payment of {{TOTAL_PAYABLE}} for {{PROPERTY_ADDRESS}} is now OVERDUE.\n\nPlease make payment immediately to avoid late penalties or access restrictions. Contact management at {{FIRM_NAME}} to confirm your payment.\n\n— {{FIRM_NAME}}`; 
    96→        break;
    97→      case 'payment_receipt': 
    98→        message = `PAYMENT RECEIPT\n\nDear {{TENANT_NAME}},\n\nWe confirm receipt of {{TOTAL_PAYABLE}} for your unit at {{PROPERTY_ADDRESS}}.\n\nThank you for your prompt payment.\n\n— {{FIRM_NAME}}`; 
    99→        break;
   100→      case 'service_charge_alert': 
   101→        message = `SERVICE CHARGE ALERT\n\nDear {{TENANT_NAME}},\n\nYour service charge of ₦${sc.toLocaleString('en-NG')} for {{PROPERTY_ADDRESS}} is outstanding. Please settle this at your earliest convenience to avoid access restrictions.\n\nTotal Payable: {{TOTAL_PAYABLE}}`; 
   102→        break;
   103→      case 'access_restriction': 
   104→        message = `NOTICE OF ACCESS RESTRICTION\n\nAccess to {{PROPERTY_ADDRESS}} has been restricted due to non-payment of outstanding charges totaling {{TOTAL_PAYABLE}}. Please contact {{FIRM_NAME}} immediately to resolve.`; 
   105→        break;
   106→      case 'penalty_notice': 
   107→        message = `PENALTY NOTICE\n\nDear {{TENANT_NAME}},\n\nA late payment penalty has been applied to your account for {{PROPERTY_ADDRESS}}.\n\nRevised Total Payable: {{TOTAL_PAYABLE}}\n\nPlease settle the outstanding balance urgently. — {{FIRM_NAME}}`; 
   108→        break;
   109→      case 'lease_renewal': 
   110→        message = `LEASE RENEWAL\n\nDear {{TENANT_NAME}},\n\nYour lease for {{PROPERTY_ADDRESS}} is expiring soon. We invite you to renew your tenancy agreement. Please contact {{FIRM_NAME}} to discuss renewal terms.`; 
   111→        break;
   112→      case 'welcome_note': 
   113→        message = `Welcome to your new home at {{PROPERTY_ADDRESS}}, {{TENANT_NAME}}! We are excited to have you. Please find the tenant handbook in your portal. — {{FIRM_NAME}}`; 
   114→        break;
   115→      case 'promotion': 
   116→        message = `Hello {{TENANT_NAME}}, we have a special offer for our residents! Get 10% off professional cleaning services this month. Use code: CLEAN10. — {{FIRM_NAME}}`; 
   117→        break;
   118→      case 'vendor_update': 
   119→        message = `Dear {{TENANT_NAME}}, we have partnered with new verified maintenance vendors to serve you better. You can now request plumbing and electrical repairs directly from the app. — {{FIRM_NAME}}`; 
   120→        break;
   121→      case 'general_announcement': 
   122→        message = `Attention residents of {{PROPERTY_ADDRESS}}: Routine maintenance will be carried out on the central generators this Saturday. Expect intermittent power supply between 10am and 2pm. — {{FIRM_NAME}}`; 
   123→        break;
   124→      case 'maintenance_update': 
   125→        message = `Update on your maintenance request for {{PROPERTY_ADDRESS}}: The vendor has confirmed your appointment for tomorrow. Please ensure someone is available to grant access. — {{FIRM_NAME}}`; 
   126→        break;
   127→      default: 
   128→        message = ''; 
   129→        break;
   130→    }
   131→  }
   132→
   133→  if (!message) return '';
   134→
   135→  const scLine = sc > 0 ? `- Service Charge: ₦${sc.toLocaleString('en-NG')}\n` : '';
   136→  const cdLine = cd > 0 ? `- Caution Deposit: ₦${cd.toLocaleString('en-NG')}\n` : '';
   137→  const feesLine = (lf + af) > 0 ? `- Legal/Agency Fees: ₦${(lf + af).toLocaleString('en-NG')}\n` : '';
   138→
   139→  let result = message
   140→    .replace(/\{\{TENANT_NAME\}\}/g, name)
   141→    .replace(/\{\{AMOUNT\}\}/g, amtStr)
   142→    .replace(/\{\{TOTAL_PAYABLE\}\}/g, totalStr)
   143→    .replace(/\{\{SERVICE_CHARGE_LINE\}\}/g, scLine)
   144→    .replace(/\{\{CAUTION_DEPOSIT_LINE\}\}/g, cdLine)
   145→    .replace(/\{\{FEES_LINE\}\}/g, feesLine)
   146→    .replace(/\{\{FIRM_NAME\}\}/g, extraData?.firmName || 'Management');
   147→
   148→  // Handle Unit Context smartly
   149→  if (addr === 'General' || addr === 'All Residents') {
   150→      result = result.replace(/ for \{\{PROPERTY_ADDRESS\}\}/g, '');
   151→      result = result.replace(/ at \{\{PROPERTY_ADDRESS\}\}/g, '');
   152→      result = result.replace(/\{\{PROPERTY_ADDRESS\}\}/g, 'your unit');
   153→  } else {
   154→      result = result.replace(/\{\{PROPERTY_ADDRESS\}\}/g, addr);
   155→  }
   156→  
   157→  if (extraData) {
   158→    result = result.replace(/\{\{SERVICE_CHARGE\}\}/g, `₦${sc.toLocaleString('en-NG')}`);
   159→    result = result.replace(/\{\{LEGAL_FEE\}\}/g, `₦${lf.toLocaleString('en-NG')}`);
   160→    result = result.replace(/\{\{AGENCY_FEE\}\}/g, `₦${af.toLocaleString('en-NG')}`);
   161→    result = result.replace(/\{\{CAUTION_DEPOSIT\}\}/g, `₦${cd.toLocaleString('en-NG')}`);
   162→    result = result.replace(/\{\{DUE_DATE\}\}/g, extraData.dueDate || 'the due date');
   163→  }
   164→  
   165→  return result;
   166→}
